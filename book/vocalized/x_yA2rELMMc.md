## סֵפֶר הָעֲקֵדָה עַל פָּרָשַׁת שַׁלַח

[[0:02]] עַל הַפָּרָשָׁה הַזֹּאת, פָּרָשַׁת שֶׁלַח, יֵשׁ דְּבָרִים בְּסֵפֶר הָעֲקֵדָה לְרַבִּי יִצְחָק עֲרָאמָה עָלָיו הַשָּׁלוֹם, שֶׁחַי לִפְנֵי כְּשֵׁשׁ מֵאוֹת שָׁנָה. [?0:20] הוּא לוֹקֵחַ אֶת הַתּוֹרָה וְאֶת עִמְקוּתָהּ, אֶת הַשֵּׂכֶל, אֶת הָעִיּוּן הַפִילוֹסוֹפִי, אֶת הַהַשְׁקָפָה, אֶת הַמִּדְרָשׁ וְאֶת הַגְּמָרָא – אֶת כָּל מָה שֶׁנִּקְרָא "הַסַּמְמָנִים" – וּמַגִּישׁ לְךָ אֶת הַפָּרָשָׁה, אֶת הַנּוֹשֵׂא וְאֶת הַמְּאוֹרָע בַּצּוּרָה הֲכִי קְלָאסִית, הֲכִי עֲמוּקָּה, הֲכִי גְּבוֹהָה, הֲכִי חֲזָקָה וַהֲכִי אֲמִיתִּית.

[[0:58]] הָרֶקַע הוּא הַמִּדְרָשׁ: ⟦src|אָמַר רַבִּי יְהוֹשֻׁעַ בֶּן לֵוִי, לָמָּה יִשְׂרָאֵל דּוֹמִין כְּשֶׁאָמְרוּ נִשְׁלְחָה אֲנָשִׁים לְפָנֵינוּ⟧{{=במדבר רבה טז, הובא בעקדת יצחק, שער עז}}. כְּלוֹמַר, מָה בְּעֶצֶם רָצָה עַם יִשְׂרָאֵל כְּשֶׁבָּא לְמֹשֶׁה רַבֵּנוּ וְאָמַר לוֹ "נִשְׁלְחָה אֲנָשִׁים"? הֵם עָמְדוּ עַל גְּבוּל אֶרֶץ יִשְׂרָאֵל, אַחֲרֵי אַרְבָּעִים שָׁנָה שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא הָפַךְ אֶת כָּל הָעוֹלָם בִּשְׁבִילָם. כְּמוֹ שֶׁאָמַרְנוּ, כָּל יְצִיאַת מִצְרַיִם הָיְתָה בְּעֶצֶם בְּרִיאָה חֲדָשָׁה; וְאָמַרְנוּ שֶׁיְּצִיאַת מִצְרַיִם מוֹכִיחָה עַל בְּרִיאַת הָעוֹלָם.

## מַכַּת אַרְבֶּה – הַהוֹכָחָה לִשְׁלִיטַת הַבּוֹרֵא

[[1:50]] נַנִּיחַ שֶׁפִּתְאוֹם אֲנַחְנוּ רוֹאִים מִילְיוֹנֵי מִילְיוֹנִים שֶׁל אַרְבֶּה שֶׁמַּגִּיעִים לָאָרֶץ וּמְכַסִּים אֶת כָּל הַשֶּׁטַח. מִי יָכוֹל לַעֲשׂוֹת דָּבָר כָּזֶה? יֵשׁ מִישֶׁהוּ בָּעוֹלָם שֶׁיָּכוֹל לְהַמְצִיא אוֹתוֹ? טוֹב, יֹאמַר אָדָם: אַרְבֶּה בָּא לְבַד – תּוֹפַעַת טֶבַע. בְּסֵדֶר.

[[2:36]] אֲבָל אָדָם שֶׁקּוֹרְאִים לוֹ מֹשֶׁה אוֹמֵר לְפַרְעֹה: אָבִיא עָלֶיךָ אֶת הָאַרְבֶּה בְּדִיּוּק מָתַי שֶׁאֲנִי רוֹצֶה. יֵשׁ מִישֶׁהוּ שֶׁיָּכוֹל לוֹמַר "מָתַי שֶׁאֲנִי רוֹצֶה"? אֵין דָּבָר כָּזֶה. עַד כָּאן עוֹד אֶפְשָׁר [?2:55].

[[2:57]] אֲבָל לְפַרְעֹה נִמְאַס, וְהוּא אוֹמֵר לְמֹשֶׁה: רַחֵם עָלַי, אֲנִי מְבַקֵּשׁ, הָסֵר אֶת הָאַרְבֶּה מֵעָלַי. אָמַר לוֹ מֹשֶׁה: מָתַי אַתָּה רוֹצֶה שֶׁאָסִיר אֶת הָאַרְבֶּה? וְהַמְּשׁוּגָּע הַזֶּה – שֶׁהָיָה כּוֹפֵר קְלָאסִי, כּוֹפֵר מִדַּעַת, לֹא כּוֹפֵר מִבּוּרוֹת – כְּשֶׁאוֹמֵר לוֹ מֹשֶׁה "עַכְשָׁיו?", הוּא עוֹנֶה: לֹא עַכְשָׁיו – מָחָר. יֵשׁ מִי שֶׁסּוֹבֵל מַכּוֹת וְאוֹמֵר: תַּמְשִׁיךְ לְהַכּוֹת אוֹתִי, רַק מָחָר תַּפְסִיק? אֲבָל הוּא הָיָה רָשָׁע, כּוֹפֵר, פִילוֹסוֹף גָּדוֹל וְחוֹשֵׁב גָּדוֹל. אוֹמֵר לוֹ: לְמָחָר. אוֹמֵר לוֹ מֹשֶׁה: כִּדְבָרְךָ. בְּאֵיזוֹ שָׁעָה אַתָּה רוֹצֶה? בְּשֶׁבַע בַּבּוֹקֶר. בְּשֶׁבַע בַּבּוֹקֶר – אַרְבֶּה.

[[3:55]] אִם אָמַרְנוּ שֶׁהָאַרְבֶּה בָּא מֵעַצְמוֹ – בְּסֵדֶר. אֲבָל עַכְשָׁיו הִגִּיעַ הַזְּמַן שֶׁיֵּלֵךְ. אִם הוּא הוֹלֵךְ, יֵלֵךְ מֵעַצְמוֹ: אֵיךְ אַרְבֶּה הוֹלֵךְ מֵעַצְמוֹ? לְפִי הַחֵשֶׁק שֶׁלּוֹ, כְּמוֹ שֶׁבָּא? אִילּוּ הָיָה זֶה טֶבַע, הָיָה חֵלֶק הוֹלֵךְ, יוֹם, יוֹמַיִם. וּבְשֶׁבַע בַּבּוֹקֶר לֹא נִשְׁאַר אַרְבֶּה אֶחָד בְּכָל גְּבוּל מִצְרַיִם. יָכוֹל לִהְיוֹת שֶׁזֶּה קָרָה בְּמִקְרֶה? יֵשׁ לָהֶם אֵיזֶה מְפַקֵּד שֶׁאָמַר לָהֶם? וְאִם יֵשׁ מְפַקֵּד – יֵשׁ מְפַקֵּד שֶׁמוּרֶה לָהֶם לָלֶכֶת בְּשֶׁבַע. אִם כֵּן, מִי שֶׁיֵּשׁ לוֹ שְׁלִיטָה עַל הָאַרְבֶּה – אַתָּה רוֹצֶה לוֹמַר לִי שֶׁהוּא אֵינוֹ מַנְהִיג אֶת הָאַרְבֶּה בְּעַצְמוֹ? יִשְׁתַּבַּחַ שְׁמוֹ – יֵשׁ לוֹ שְׁלִיטָה מוּחְלֶטֶת עַל כָּל תְּנוּעָה וּתְנוּעָה. זוֹ יְצִיאַת מִצְרַיִם: הִיא הוֹכִיחָה שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא הוּא שֶׁבָּרָא אֶת הָעוֹלָם. כִּי אִם לֹא בָּרָא אֶת הָעוֹלָם – אֵיךְ פִּתְאוֹם הוּא שׁוֹלֵט בְּאַרְבָּה? הוּא שׁוֹלֵט.

## מָשַׁל הַמֶּלֶךְ שֶׁזִּימֵּן לִבְנוֹ אִישָּׁה

[[5:35]] וּבְנֵי יִשְׂרָאֵל, אַחֲרֵי כָּל הַמּוֹפְתִים הָאֵלֶּה, אָמְרוּ: מֹשֶׁה, אֲנַחְנוּ הוֹלְכִים לָאָרֶץ? צָרִיךְ לִשְׁלוֹחַ מְרַגְּלִים – כְּאִילּוּ שֶׁעַד עַכְשָׁיו [?5:50].

[[5:52]] וְהַמֶּשָׁל: ⟦src|לַמֶּלֶךְ שֶׁזִּימֵּן לִבְנוֹ אִשָּׁה נָאָה עֲשִׁירָה וּבַת טוֹבִים⟧{{=עקדת יצחק, שער עז}} – שִׁידּוּךְ לִבְנוֹ, אִישָּׁה יָפָה, עֲשִׁירָה וּבַת מִשְׁפָּחָה טוֹבָה. ⟦src|אָמַר לוֹ הַבֵּן אֵלֵךְ וְאֶרְאֶה אוֹתָהּ, שֶׁלֹּא הָיָה מַאֲמִין לְאָבִיו. מִיָּד הוּקְשֶׁה הַדָּבָר בְּעֵינֵי אָבִיו⟧{{=עקדת יצחק, שער עז}}. מָה יַעֲשֶׂה? ⟦src|אִם אוֹמַר לוֹ אֵינִי מַרְאֶה אוֹתָהּ לְךָ עַכְשָׁיו, יֹאמַר כְּעוֹרָהּ הִיא, וּלְפִיכָךְ אֵינוֹ רוֹצֶה לְהַרְאוֹתָהּ לוּ⟧{{=עקדת יצחק, שער עז}} – אוּלַי יֵשׁ לָהּ אֵיזוֹ גִּיבֶּנֶת? [?6:50]

[[7:00]] ⟦src|לַסּוֹף אָמַר לוֹ לְךָ וּרְאָה אוֹתָהּ וְתֵדַע אִם כָּזַבְתִּי לְךָ⟧{{=עקדת יצחק, שער עז}} – קוֹדֶם כּוֹל, תִּיוָּכַח שֶׁלֹּא שִׁיקַּרְתִּי לְךָ. ⟦src|וּבִשְׁבִיל שֶׁלֹּא הֶאֱמַנְתָּ בִּי, קוֹנָם שֶׁאֵין אַתָּה רוֹאֶה אוֹתָהּ בְּבֵיתְךָ, אֲבָל לִבְנְךָ אֲנִי נוֹתְנָה⟧{{=עקדת יצחק, שער עז}}. זֶה מָה שֶׁקָּרָה לִבְנֵי יִשְׂרָאֵל.

[[7:34]] ⟦src|וְכָךְ אָמַר הַקָּדוֹשׁ בָּרוּךְ הוּא לְיִשְׂרָאֵל: טוֹבָה הָאָרֶץ, וְלֹא הֶאֱמִינוּ, אֶלָּא אָמְרוּ נִשְׁלְחָה אֲנָשִׁים וְגוּ׳. אָמַר הַקָּדוֹשׁ בָּרוּךְ הוּא: אֵין אֲנִי מְעַכֵּב עַל יָדָם, אֶלָּא יִרְאוּ אוֹתָהּ, וּבִשְׁבוּעָה שֶׁאֵין אֶחָד מֵהֶם נִכְנָס בְּתוֹכָהּ, שֶׁנֶּאֱמַר⟧{{=עקדת יצחק, שער עז}} ⟦sefaria|אִם־יִרְאוּ אֶת־הָאָרֶץ⟧{{=במדבר יד, כג}} ⟦src|אֶלָּא לִבְנֵיהֶם אֲנִי נוֹתְנָה⟧{{=עקדת יצחק, שער עז}}. זֶה הַסִּיפּוּר שֶׁל הַמְּרַגְּלִים.

## הַקְדָּמַת בַּעַל הָעֲקֵדָה: הַטּוֹב, הַמּוֹעִיל וְהֶעָרֵב

[[8:04]] לָרַב יֵשׁ כָּאן שְׁתַיִם, שָׁלוֹשׁ, אַרְבַּע הַקְדָּמוֹת, וְהֵן הַקְדָּמוֹת שֶׁל מַחֲשָׁבָה וְשֶׁל פִילוֹסוֹפְיָה, כְּמוֹ שֶׁדִּיבַּרְנוּ עַל הָרַמְבָּ״ם [?8:20]. לֹא נוּכַל לְהִיכָּנֵס לְכוּלָּן; זֶה יֶאֱרַךְ יָמִים. נִיגַּשׁ לַהַקְדָּמָה הָעִיקָּרִית, אֲבָל חַיָּבִים לְהַקְדִּים הַקְדָּמָה קְצָרָה שֶׁתַּכְנִיס אוֹתָנוּ לָעִנְיָן.

[[8:40]] ⟦src|וְזֶה כִּי אַחַר שֶׁעִנְיַן הַהַגְבָּלָה הַשִּׂכְלִית הוּא הִמְצָא הַחַיּוֹת עַל דֶּרֶךְ הַטּוֹב וְהַמּוֹעִיל וְהָעֲרֵב עַל הַמָּצוֹעַ וּכְמוֹ שֶׁיִּתָּכֵן לֹא הַקְּצָווֹת בָּהֶם. כִּי כְּמוֹ שֶׁהַהִתְרַחֵק מֵהַטּוֹב אֶל הַקָּצֶה הָאֶחָד הוּא הָרַע הַמּוּחְלָט וְלֹא יִהְיֶה חַיּוּת מוּגְבָּל כֵּן הַהִתְרַחֵק מֵהָעֶרֵב וְהַמּוֹעִיל כְּמוֹ כֵן כִּי חַיֵּי הָעֶצֶב וְהָעוֹנִי שֶׁהֵם הִפְכָּם הֵם בִּלְתִּי מוּגְבָּלִים כְּמוֹ שֶׁזָּכַר הֶחָכָם בַּמָּקוֹם שֶׁזָּכַרְנוּ בַּמַּאֲמָר הַט׳. וְלָכֵן הַהַגְבָּלָה הַנְּכוֹנָה הִיא שֶׁיַּחְיֶה הָאָדָם בְּטוֹב בְּחַיִּים טוֹבִים מַסְפִּיקִים וּמוּגְבָּלִים⟧{{=עקדת יצחק, שער עז}}.

[[9:30]] אוֹמֵר הָרַב כָּךְ: יֵשׁ בַּחַיִּים שֶׁאָדָם חַי שְׁלוֹשָׁה דְּבָרִים שֶׁנּוֹתְנִים לוֹ אֶת טַעַם הַחַיִּים. אֶחָד – הַטּוֹב: דָּבָר שֶׁהוּא טוֹב בְּאוֹפֶן כְּלָלִי. הַשֵּׁנִי – הַמּוֹעִיל: שֶׁהַדָּבָר לֹא רַק יִהְיֶה טוֹב, אֶלָּא גַּם יוֹעִיל. הַשְּׁלִישִׁי – הֶעָרֵב: שֶׁיִּהְיֶה מָתוֹק. [?10:20] הַטּוֹב; דָּבָר שֵׁנִי – מָה שֶׁמּוֹעִיל לָאָדָם; דָּבָר שְׁלִישִׁי – [?10:34]. כְּמוֹ שֶׁאוֹכְלִים [?10:40] – וְגַם כְּשֶׁאוֹכְלִים, שָׂמִים מֶלַח וְלִימוֹן וְכַדּוֹמֶה. עַכְשָׁיו תַּקְשִׁיבוּ טוֹב.

## דֶּרֶךְ הַמִּיצּוּעַ

[[11:01]] שְׁלוֹשֶׁת הַדְּבָרִים הָאֵלֶּה – אִם אֲנַחְנוּ רוֹצִים לְהָפִיק מֵהֶם אֶת הַתּוֹעֶלֶת שֶׁלָּנוּ – צְרִיכִים לִהְיוֹת עַל דֶּרֶךְ הַמִּיצּוּעַ, בָּאֶמְצַע. מָה זֶה הָאֶמְצַע? יֵשׁ כָּאן קָצֶה, וְכָאן קָצֶה, וְכָאן הָאֶמְצַע – מָה שֶׁנִּקְרָא מִשְׁקָל אֲמִיתִּי, כַּאֲשֶׁר שְׁתֵי כַּפּוֹת הַמֹּאזְנַיִם שָׁווֹת. אִם כֵּן, אִם אֲנַחְנוּ רוֹצִים לִחְיוֹת נָכוֹן, עָלֵינוּ לְחַפֵּשׂ אֶת [?11:50].

[[12:01]] "כִּי כְּמוֹ שֶׁהִתְרַחֵק מֵהַטּוֹב אֶל הַקָּצֶה הָאֶחָד": כָּאן הַצַּד הַטּוֹב, וְכָאן הַצַּד הָרַע. אַתָּה הוֹלֵךְ וְרוֹצֶה לִחְיוֹת לְפִי [?12:31] – כָּל צַעַד שֶׁאֲנִי עוֹשֶׂה, אֲנִי מִתְרַחֵק מִן הַטּוֹב אֶל הָרַע הַמּוּחְלָט.

[[13:00]] עוֹד פַּעַם: הַטּוֹב הַמּוּחְלָט כָּאן, וְהָרַע הַמּוּחְלָט כָּאן. כָּל צַעַד שֶׁאֲנִי עוֹשֶׂה בַּכִּיוּוּן הָרַע, אֲנִי מִתְרַחֵק מִן הַטּוֹב וּמִתְקָרֵב לָרַע הַמּוּחְלָט. "וְלֹא יִהְיֶה חִיּוּב מוּגְבָּל" – מָה פֵּירוּשׁ? שֶׁלֹּא הִגְבַּלְתִּי אֶת עַצְמִי. כְּשֶׁאֲנִי מִתְרַחֵק מִן הַטּוֹב אֶל הָרַע הַמּוּחְלָט – פֵּירוּשׁ הַדָּבָר שֶׁלֹּא הִגְבַּלְתִּי אֶת עַצְמִי; לֹא אָמַרְתִּי לְעַצְמִי: הָלוֹ, הָלוֹ, עֲצוֹר! אַתָּה מִתְקָרֵב יוֹתֵר מִדַּי אֶל הָרַע. [?13:55] – מַתְחִיל לִשְׁתּוֹת בְּלִי הַגְבָּלָה. אַתָּה מִתְרַחֵק מִן הַטּוֹב אֶל הָרַע הַמּוּחְלָט.

[[14:15]] וְאוֹמֵר: כְּמוֹ כֵן מִי שֶׁהִתְרַחֵק מִן הֶעָרֵב וְהַמּוֹעִיל – אוֹתוֹ דָּבָר. אִם יֵשׁ דָּבָר שֶׁהוּא עָרֵב וּמוֹעִיל – כְּמוֹ שֶׁאָמַרְנוּ, הַתַּפּוּחַ: הוּא עָרֵב, וְגַם מוֹעִיל – וְאוֹמֵר אָדָם: אֲנִי רוֹצֶה לָגֶשֶׁת אֶל הַדָּבָר הֶעָרֵב וְהַמּוֹעִיל בְּלִי גְּבוּל. זֶה דָּבָר עָרֵב, זֶה דָּבָר מוֹעִיל – אַל תַּגְבִּיל אוֹתִי! אוֹמֵר הָרַב: כְּמוֹ שֶׁהַמִּתְרַחֵק מִן הַטּוֹב הַמּוּחְלָט אֶל הָרַע הַמּוּחְלָט בְּלִי הַגְבָּלָה, כָּךְ גַּם בַּדָבָר שֶׁמּוֹעִיל לְךָ וּבַדָבָר שֶׁעָרֵב לְךָ, כְּשֶׁאַתָּה מַחֲלִיט [?15:25].

[[15:25]] [?15:25] קְצָת יוֹתֵר עָמוֹק. טוֹב וְרַע – זֶה מוּבָן לָנוּ. אֲבָל עָרֵב וּמוֹעִיל – לֹא מוּבָן מָה פֵּירוּשׁ לְהִתְרַחֵק מֵהֶם. אִם זֶה עָרֵב לִי, לָמָּה שֶׁיַּגְבִּילוּ אוֹתִי? אוֹ דָּבָר שֶׁמּוֹעִיל לִי – לָמָּה שֶׁאֶתְרַחֵק מִמֶּנּוּ? הוּא אוֹמֵר שֶׁגַּם כָּאן צָרִיךְ לְהַגִּיעַ לָאֶמְצַע: אַל תִּתְרַחֵק אֶל הַקָּצֶה.

## חַיֵּי הָעֶצֶב וְהָעוֹנִי – גַּם הֵם קָצֶה

[[15:58]] וְאוֹמֵר: "כִּי חַיֵּי הָעֶצֶב וְהָעוֹנִי, שֶׁהֵם הִפְכָם, הֵם בִּלְתִּי מוּגְבָּלִים". כְּלוֹמַר, לִחְיוֹת חַיִּים בְּלִי טַעַם – גַּם זֶה לֹא נָכוֹן. כְּמוֹ שֶׁהָרַע הַמּוּחְלָט הוּא הִיפּוּכוֹ שֶׁל הַטּוֹב הַמּוּחְלָט, כָּךְ מִי שֶׁחַי חַיֵּי עוֹנִי – זֶה הַהֶפֶךְ [?16:35]. עֶצֶב וְעוֹנִי: נַנִּיחַ שֶׁאָדָם לוֹקֵחַ דָּבָר עָרֵב וְאוֹמֵר: אֵינֶנִּי רוֹצֶה לֶאֱכוֹל דְּבָרִים עַרְבִים וּמְתוּקִים, אֲנִי רוֹצֶה לֶאֱכוֹל רַק לְפַת וְצָנוֹן – גַּם זֶה לֹא נָכוֹן. [?17:15]

[[17:20]] "וְלָכֵן הַהַגְבָּלָה הַנְּכוֹנָה הִיא שֶׁיִּהְיֶה הָאָדָם בְּטוֹב, בְּחַיִּים טוֹבִים, מַסְפִּיקִים וּמוּגְבָּלִים". הַנָּכוֹן הוּא שֶׁאָדָם יִחְיֶה חַיִּים טוֹבִים – חַיִּים שֶׁהֵם מַסְפִּיקִים. מַסְפִּיקִים, כְּלוֹמַר: מַסְפִּיק לֶחֶם בַּבַּיִת כְּדֵי שֶׁלֹּא אֶהְיֶה רָעֵב. אֲבָל גַּם מוּגְבָּלִים: נָכוֹן, לֶחֶם צָרִיךְ לִהְיוֹת בַּבַּיִת, אֲבָל בְּמִידָּה – לֹא [?18:10] עֶשֶׂר כִּיכְּרוֹת לֶחֶם כָּל יוֹם. כְּלוֹמַר, אָדָם שֶׁרוֹצֶה לִחְיוֹת נָכוֹן צָרִיךְ לִהְיוֹת גַּם בְּטוֹב, גַּם בְּמוֹעִיל וְגַם בָּעֶרֶב – בָּאֶמְצַע. זוֹ הַדֶּרֶךְ: שֶׁהוּא מַגְבִּיל אֶת עַצְמוֹ. לִלְמוֹד אֶת הַנְּקוּדָּה הַזֹּאת: גַּם בִּדְבַר הֶעָרֵב וְגַם בַּדָּבָר הַמּוֹעִיל צָרִיךְ לִחְיוֹת בְּהַגְבָּלָה.

## שְׁלוֹשׁ הַנְּפָשׁוֹת: הַחִיּוּנִית, הָרַגְזָנִית וְהַמַּשְׂכֶּלֶת

[[19:00]] ⟦src|אִם אֶל צוֹרֶךְ הַפַּרְנָסָה וְהַמְּזוֹנוֹת וְיֶתֶר מִינֵי הֶעָרֵב אֲשֶׁר תִּדְרְשֵׁם הַנֶּפֶשׁ הַחִיּוּנִית.⟧{{=עקדת יצחק, שער עז}}. הַנֶּפֶשׁ הַחִיּוּנִית: לָאָדָם יֵשׁ חָמֵשׁ דַּרְגוֹת נְשָׁמָה, וְהַדַּרְגָּה הַשְּׁפֵלָה בְּיוֹתֵר הִיא הַחֵלֶק שֶׁנִּקְרָא "נֶפֶשׁ חִיּוּנִית". "חִיּוּנִי" בְּעִבְרִית – דָּבָר הֶכְרֵחִי. בַּדָּבָר הַזֶּה אֲנַחְנוּ כְּמוֹ הַבְּהֵמָה: לַבְּהֵמָה אֵין אֶת הַ"שִּׁדְרוּג" הַזֶּה; יֵשׁ לָהּ רַק אֶת הַנֶּפֶשׁ. נֶפֶשׁ – פֵּירוּשָׁהּ לֶאֱכוֹל, לִשְׁתּוֹת, לִישׁוֹן. זֶה יֵשׁ גַּם לַבְּהֵמָה וְגַם לָנוּ. בַּדַּרְגָּה הַזֹּאת – בַּגַּשְׁמִיּוּת, בְּחוֹמֶר – צָרִיךְ שֶׁנַּחְיֶה לְפִי הַגְבָּלָה.

[[20:20]] ⟦src|וְאִם אַל סְפוּק שְׁאַר הַטּוֹבוֹת מֵהָעוֹשֶׁר וְהַכָּבוֹד וְיֶתֶר מִינֵי הַמּוֹעִיל שֶׁהֵם מְבוּקַּשׁ הַנֶּפֶשׁ הָרַגְזָנִית.⟧{{=עקדת יצחק, שער עז}}. יֵשׁ נֶפֶשׁ שְׁנִיָּה, שֶׁהָרַב קוֹרֵא לָהּ "הַנֶּפֶשׁ הָרַגְזָנִית". [?20:40] מִכָּאן אַתָּה נִכְנָס לָעוֹלָם [?21:05]. ⟦src|אֲשֶׁר בְּהִמְצֵא לָהֶם שְׁתַים אֵלֶּה עַל הַשְׁעָרָתָם הַנְּאוֹתָה וְהַגְבָּלָתָם הַנְּכוֹנָה⟧{{=עקדת יצחק, שער עז}}. כְּלוֹמַר: גַּם בְּאֵלֶּה – זוֹ נֶפֶשׁ גְּבוֹהָה יוֹתֵר. הַבְּהֵמָה לְעוֹלָם לֹא תִּשְׁאַף לִהְיוֹת עֲשִׁירָה; אֵין לָהּ אֶת הַנֶּפֶשׁ הַזֹּאת, שֶׁיְּכַבְּדוּ אוֹתָהּ. הַיּוֹם מְכַבְּדִים גַּם אֶת הַכֶּלֶב, אֲבָל בְּטִבְעוֹ הוּא אֵינוֹ מְחַפֵּשׂ כָּבוֹד וְעוֹשֶׁר. לְאָדָם יֵשׁ אֶת הַנֶּפֶשׁ הַזֹּאת.

[[21:45]] הַדְּבָרִים הָאֵלֶּה חֲשׁוּבִים מְאוֹד בֶּאֱמֶת. הַיּוֹם אֵינֶנּוּ יוֹדְעִים אֲפִילּוּ מָה קוֹרֶה אִיתָּנוּ [?21:55]. הַכּוֹל מְעוּרְבָּב בַּמּוֹחַ שֶׁלָּנוּ; אֵינֶנּוּ יוֹדְעִים בֶּאֱמֶת מָה שַׁיָּךְ לְמַעְלָה.

[[22:15]] אָז הוּא אוֹמֵר שֶׁגַּם בָּזֶה – בְּסֵדֶר, אַתָּה רוֹצֶה לִהְיוֹת עָשִׁיר? אַל תִּשְׁתַּגֵּעַ; [?22:20] תַּגְבִּיל אֶת עַצְמְךָ. גַּם בָּזֶה תַּגְבִּיל אֶת עַצְמְךָ, בְּדִיּוּק [?22:30] – וְאָז אַתָּה נִמְצָא עַל הַדֶּרֶךְ הַטּוֹבָה. כְּלוֹמַר, יְכוֹלִים לָבוֹא וְלוֹמַר לָאָדָם: מָה, אַתָּה רוֹצֶה לִהְיוֹת עָשִׁיר? אוֹמֵר הָרַב: כֵּן, אֶפְשָׁר – אֲבָל בְּמִידָּה [?22:50]. עָשִׁיר אֲמִיתִּי – זֶה אֲמִיתִּי.

[[23:00]] וְזֶה "הַטּוֹב וְהַיָּשָׁר". זֶהוּ הַטּוֹב וְהַיָּשָׁר שֶׁדִּיבַּרְנוּ עָלָיו: אַתָּה רוֹצֶה לְהַעֲשִׁיר? אַל תִּשְׁתַּגֵּעַ. "הַטּוֹב וְהַיָּשָׁר" – שֶׁיִּהְיֶה לְךָ טוֹב בָּעוֹשֶׁר, וְגַם שֶׁתִּהְיֶה יָשָׁר. כִּי אִם אָדָם אֵינוֹ יָשָׁר, הוּא הוֹלֵךְ לַעֲשׂוֹת כֶּסֶף לֹא יָשָׁר. זֶה מָה שֶׁהָרַב אוֹמֵר: לִהְיוֹת עָשִׁיר – זֶה בְּסֵדֶר; לִרְצוֹת לִהְיוֹת עָשִׁיר וְלִהְיוֹת עָשִׁיר – זֶה בְּסֵדֶר. אֵין בְּעָיָה. אֵינֶנִּי צָרִיךְ לְצַמְצֵם אֶת עַצְמִי וְלוֹמַר "אֲנִי מִסְתַּפֵּק בְּמָה שֶׁיֵּשׁ לִי". מוּתָּר לִי לוֹמַר: אֲנִי בֵּן שֶׁל מֶלֶךְ, מוּתָּר לִי לִהְיוֹת עָשִׁיר, הַכּוֹל בְּסֵדֶר – שֶׁיִּהְיֶה לִי. נָכוֹן. אֲבָל בְּמִידָּה; לֹא לוֹמַר "הִגַּעְתִּי".

## מַעֲשֶׂה בִּיהוּדִי שֶׁבָּא לַיְּשִׁיבָה שֶׁל בְּנוֹ

[[24:05]] [?24:05] יְהוּדִי בָּא לַיְּשִׁיבָה שֶׁל בְּנוֹ וְאָמַר לָרַב: בְּנִי לוֹמֵד אֶצְלְךָ שָׁנִים רַבּוֹת, וְאֵינֶנִּי רוֹאֶה שֶׁהוּא מַצְלִיחַ בְּמַשֶּׁהוּ. אָמַר לוֹ: לָמָּה אַתָּה מִתְכַּוֵּן? אָמַר: [?24:40] מֹשֶׁה רַבֵּנוּ הוּא לֹא יִהְיֶה; הוּא סְתָם אָדָם מְמוּצָּע. טוֹב – שֶׁיָּבוֹא לַעֲבוֹד אִיתִּי ב[?24:50]. שָׁאַל אוֹתוֹ הָרַב: בְּמָה אַתָּה עוֹסֵק? אֲנִי מוֹכֵר טֶלֶפוֹנִים. וּמָה הַהַכְנָסוֹת שֶׁלְּךָ? בָּרוּךְ ה׳ [?25:15], קְצָת קָשֶׁה בְּפַרְנָסָה. [?25:20] אָמַר לוֹ: [?25:30] – אַתָּה לֹא תִּהְיֶה מֹשֶׁה רַבֵּנוּ, אֲבָל [?25:40].

[[25:43]] [?25:43] אוֹמְרִים לִי: הָרַב, כְּאִילּוּ אֲנִי הוֹלֵךְ לְבַקֵּשׁ כֶּסֶף בִּשְׁבִיל הַיְּשִׁיבָה. הָרַב, אִם אֲנִי מְמַלֵּא אוֹתוֹ, אִם אֲנִי זוֹכֶה – [?26:10].

## הַטּוֹב וְהַיָּשָׁר – מְבוּקָּשׁ הַנֶּפֶשׁ הַמַּשְׂכֶּלֶת

[[26:12]] "מָצָא לָהֶם הַטּוֹב וְהַיָּשָׁר". לָמָּה "יָשָׁר"? [?26:31] יָשָׁר – שֶׁאַתָּה נֶהֱנֶה מ[?] שֶׁלְּךָ. הָאָדָם צָרִיךְ שֶׁיִּהְיֶה לוֹ סִיפּוּק מָלֵא. אֲפִילּוּ אָדָם שׁ[?26:45] – הוּא מַרְגִּישׁ. אָדָם מַרְגִּישׁ: אִם עָבַד עַל מִישֶׁהוּ בָּעִסְקָה שֶׁלּוֹ, נָכוֹן שֶׁהִרְוִיחַ חֲמֵשֶׁת אֲלָפִים שֶׁקֶל, אֲבָל הוּא יוֹדֵעַ – הַמַּצְפּוּן יוֹדֵעַ. וְלָכֵן [?27:15] הֶעָשִׁיר יוֹדֵעַ; הוּא יוֹדֵעַ מָה הוּא עוֹשֶׂה.

[[27:33]] "הַטּוֹב וְהַיָּשָׁר, אֲשֶׁר הוּא מְבוּקָּשׁ הַנֶּפֶשׁ הַמַּשְׂכֶּלֶת". עַכְשָׁיו הוּא עוֹבֵר לַנֶּפֶשׁ [?27:40]. אֵיךְ זֶה הוֹלֵךְ: הַנֶּפֶשׁ הַחִיּוּנִית, הַנֶּפֶשׁ הָרַגְזָנִית, הַנֶּפֶשׁ הַמַּשְׂכֶּלֶת. הַנֶּפֶשׁ הַחִיּוּנִית אוֹמֶרֶת: תֵּן, תֵּן, תֵּן לָאָדָם אוֹכֵל, אַל תַּגְבִּיל אוֹתוֹ. אִם לֹא תֹּאמַר לוֹ "מַסְפִּיק" – [?28:20]. אַתָּה צָרִיךְ שֶׁהַנֶּפֶשׁ הַמַּשְׂכֶּלֶת תִּיתֵּן לְךָ אֶת הַקַּו הַיָּשָׁר בַּדָּבָר. הַנֶּפֶשׁ הַחִיּוּנִית וְהָרַגְזָנִית אֵינָן הַמּוֹדֵד, אֵינָן אֱמֶת הַמִּידָּה. אִם אַתָּה רוֹצֶה לָדַעַת אִם מַעֲשֶׂיךָ נְכוֹנִים – [?28:45] הַנֶּפֶשׁ שֶׁבַּדַּרְגָּה הַשְּׁלִישִׁית, הִיא תַּרְגִּישׁ בְּנוֹחַ. זוֹ הַנֶּפֶשׁ הַמַּשְׂכֶּלֶת; זֶה הַטּוֹב וְהַיָּשָׁר.

[[29:05]] וַאֲנַחְנוּ אוֹמְרִים אֲפִילּוּ אֶת הַהֶפֶךְ, אַחִים: מְדוּבָּר כָּאן בְּאָדָם שֶׁבָּא מִלְּמַטָּה. הָאָדָם שֶׁבָּא מִלְּמַעְלָה – אֵין לוֹ בִּכְלָל אֶת הַבְּעָיוֹת הָאֵלֶּה, כִּי מֵרֹאשׁ הוּא רוֹאֶה שֶׁהָעִסְקָה הַזֹּאת מְבִיאָה אוֹתוֹ לְמַשֶּׁהוּ שֶׁאֵינוֹ רָאוּי, וְהוּא אֵינוֹ נִכְנָס אֵלֶיהָ. הָרַב מְדַבֵּר אֲפִילּוּ עַל אָדָם רָגִיל, שֶׁנִּכְנַס לָעִסְקָה [?29:35], וּבָעִסְקָה יֵשׁ פִּקְפּוּקִים. הָעִסְקָה הַזֹּאת עוֹבֶרֶת אֶת בִּיקּוֹרֶת הַנֶּפֶשׁ הַחִיּוּנִית, עוֹבֶרֶת אֶת בִּיקּוֹרֶת הַנֶּפֶשׁ הָרַגְזָנִית, וּמַגִּיעָה אֶל הַנֶּפֶשׁ הַמַּשְׂכֶּלֶת. [?30:02] אִם הוּא רוֹצֶה לָדַעַת שֶׁהָעִסְקָה שֶׁלּוֹ נְכוֹנָה – [?30:15].

## הַיְּצִיאָה מִן הַגְּבוּל – פְּעוּלּוֹת הַיֵּצֶר הָרַע

[[30:15]] וּמָה קוֹרֶה אִם הֵן יוֹצְאוֹת מִגְּבוּלָן? אִם הָעִסְקָה יוֹצֵאת מִן הַגְּבוּל? ⟦src|אָמְנָם כְּשֶׁיָּצְאוּ מֵהַגְבָּלָתָם הִנֵּה הֵם פְּעוּלּוֹת הַיֵּצֶר הָרַע לְבַד שֶׁהִיא כַּת רְבִיעִית⟧{{=עקדת יצחק, שער עז}}. אִם הֵן יוֹצְאוֹת מִן הַגְּבוּל הַזֶּה שֶׁלָּהֶן – מִן הַטּוֹב וְהַיָּשָׁר – דַּע לְךָ שֶׁאֵלֶּה פְּעוּלּוֹת הַיֵּצֶר הָרַע לְבַדּוֹ.

[[31:00]] [?31:02] חָשַׁבְתִּי שֶׁנַּתְחִיל מִזֶּה, לְפִי הַהַקְדָּמָה שֶׁלּוֹ. אֲבָל הוּא מוֹנֶה אֶת כָּל הַדְּרָגוֹת שֶׁל סוּגֵי הָאֲנָשִׁים, וְאוֹמֵר: זוֹ הַכַּת הָרְבִיעִית, "שֶׁהִיא נִפְסֶדֶת וּמְשׁוּבֶּשֶׁת". [?31:40] "אַרְבַּעְתָּן אָמְרוּ זָ״ל: אַרְבַּע תְּשׁוּקוֹת" – שִׂימוּ לֵב לַבִּיטּוּי "תְּשׁוּקָה": מִשְׁאָלָה, מִשְׁאָלָה שֶׁעַל הַגְּבוּל שֶׁבֵּין הָרַע לְטוֹב. הוּא אוֹמֵר שֶׁכַּאֲשֶׁר עוֹבְרִים אֶת הַגְּבוּל – זוֹ הַנְּקוּדָּה.

[[32:05]] מָה זֶה "כַּת"? סוּג שֶׁל בְּנֵי אָדָם. עַכְשָׁיו, אָמַרְנוּ שֶׁבִּשְׁלוֹשׁ הַדְּרָגוֹת הָאֵלֶּה צְרִיכָה לִהְיוֹת הַגְבָּלָה. אָדָם אוֹמֵר: אִם זֶה לֹא הַטּוֹב וְהַיָּשָׁר, אָז אֲנִי מַתְחִיל לִסְטוֹת מִן הָאֶמְצַע לַכִּיוּוּן הָרַע. אוֹמֵר הָרַב: בָּרֶגַע שֶׁיָּצָאתָ מִן הַגְּבוּל שֶׁלְּךָ – מָה הַפְּשָׁט? שֶׁאַתָּה עֲדַיִן שַׁיָּךְ לַנֶּפֶשׁ הַמַּשְׂכֶּלֶת, רַק הִתְרַחַקְתָּ מִמֶּנָּה מְעַט? זֶה הַחִידּוּשׁ שֶׁבַּדְּבָרִים: לֹא כָּךְ. אַתָּה יוֹצֵא מִן הַגְּבוּל שֶׁלְּךָ – אֵינְךָ מִתְרַחֵק סֶנְטִימֶטֶר; זֶה לֹא הוֹלֵךְ כָּךְ. אַתָּה בַּטֶּרִיטוֹרְיָה שֶׁל מִישֶׁהוּ אַחֵר לְגַמְרֵי.

[[33:40]] מָה פֵּירוּשׁ? אָדָם שֶׁעָבַר אֶת הַגְּבוּל – תּוֹפְסִים אוֹתוֹ. עָבַר אֶת הַגְּבוּל. הוּא אוֹמֵר: אֲנִי רַק חֲמִישִּׁים מֶטֶר מִן הַגְּבוּל, סֶנְטִימֶטֶר! – זֶה לֹא הוֹלֵךְ כָּךְ. יָצָאתָ מִן הַגְּבוּל שֶׁלְּךָ. וְהַגְּבוּל אֵינוֹ רַק שֶׁטַח אֲדָמָה. בְּשֶׁטַח אֲדָמָה – אוּלַי, בְּסֵדֶר, [?34:15]. אֲבָל הַגְּבוּל הַזֶּה – כְּשֶׁאַתָּה עוֹבֵר מִן הַנֶּפֶשׁ הַמַּשְׂכֶּלֶת, אַתָּה עוֹבֵר לַנֶּפֶשׁ הָרַגְזָנִית וְהַחִיּוּנִית, וְשָׁם אֵין שׁוּם שְׁלִיטָה. הִיא אֵינָהּ מַרְגִּישָׁה שֶׁעָשִׂיתָ מַשֶּׁהוּ לֹא טוֹב – הַחִיּוּנִית, וַאֲפִילּוּ הָרַגְזָנִית, עוֹד יוֹתֵר.

[[34:35]] הַחִיּוּנִית עוֹד הִיא בְּהֵמִית: בְּהֵמָה לֹא תֹּאכַל יוֹתֵר מִמָּה שֶׁהִיא צְרִיכָה. אֲבָל הָרַגְזָנִית – כֵּן. אָדָם יִסְתַּפֵּק בִּשְׁנֵי מִילְיוֹן דּוֹלָר? מַצְחִיק אוֹתִי: יֵשׁ עוֹד שְׁנֵי מִילְיוֹן, אֲנִי [?35:01]. נָכוֹן – כַּמָּה נִפְלָאִים הַדְּבָרִים.

[[35:08]] [מן הקהל:] כְּבוֹד הָרַב, מִקּוֹדֶם אָמַר הָרַב "הַכַּת הָרְבִיעִית". הֲרֵי שְׁלוֹשֶׁת סוּגֵי הַנְּפָשׁוֹת מְצוּיִים כּוּלָּם בְּנֶפֶשׁ אַחַת, וְהָרַב מִשְׁתַּמֵּשׁ בַּמּוּשָּׂג "רְבִיעִית" – שֶׁמְּדוּבָּר בַּיֵּצֶר הָרַע. מָה הַכַּוָּנָה?

[[35:30]] יֵשׁ כָּאן שְׁנֵי דְּבָרִים. יֵשׁ כָּאן אַרְבַּע כִּיתּוֹת שֶׁל בְּנֵי אָדָם, אַרְבַּע קְבוּצוֹת, שֶׁכָּל אַחַת הִיא קָטֵגוֹרְיָה בִּפְנֵי עַצְמָהּ. וְהוּא מָנָה אֶת בְּנֵי הָאָדָם שֶׁיּוֹצְאִים מִגְּבוּלָם בְּתוֹר "הַכַּת הַנִּפְסֶדֶת וְהַמְּשׁוּבֶּשֶׁת". "נִפְסֶדֶת" – מְקוּלְקֶלֶת; "מְשׁוּבֶּשֶׁת" – שֶׁהֵם מְשַׁבְּשִׁים אֶת הַמַּעֲרֶכֶת שֶׁלָּהֶם, אֶת הָאִיזּוּן הַמֶּנְטָלִי וְהַנַּפְשִׁי.

[[36:35]] וְלֹא נִשְׁכַּח דָּבָר אֶחָד: אָדָם שֶׁיֵּשׁ לוֹ הַרְבֵּה כֶּסֶף – [?36:45] שֶׁקֶל, [?37:03] מוּזָר. [?37:10] אֲנִי אֶתֵּן לוֹ מָאתַיִם שֶׁקֶל – [?37:20] מֵאָה וְתִשְׁעִים. מָה קָרָה? אֲנִי שׁוֹלֵט בָּזֶה? לֹא, אֵינֶנִּי שׁוֹלֵט. אֲנַחְנוּ מוּכְרָחִים לִלְמוֹד אֶת הַקֶּטַע הַזֶּה, מוּכְרָחִים.

## שְׁלוֹשֶׁת הָעִנְיָנִים דּוֹחִים זֶה אֶת זֶה

[[38:00]] ⟦src|וְהִנֵּה לְפִי שֶׁשְּׁלֹשָׁה עִנְיָנִים אֵלּוּ מִטִּבְעָם לִשְׂנוֹא זֶה לָזֶה וְשֶׁיִּדְחֶה זֶה מִפְּנֵי זֶה⟧{{=עקדת יצחק, שער עז}}. תִּשְׁמְעוּ אֵיךְ הוּא מְסַדֵּר אֶת הַדְּבָרִים בַּמַּחֲשָׁבָה הַקְּלָאסִית שֶׁלּוֹ. יֵשׁ לָנוּ שְׁלוֹשָׁה דְּבָרִים: הַטּוֹב, הֶעָרֵב וְהַמּוֹעִיל. לָמַדְנוּ אוֹתָם הַיּוֹם, וְחָשַׁבְנוּ שֶׁשְּׁלוֹשְׁתָּם [?38:30]. אוֹמֵר הָרַב: שְׁלוֹשֶׁת הַדְּבָרִים הָאֵלֶּה – אֶחָד שׂוֹנֵא אֶת הַשֵּׁנִי! וְגַם אֶחָד דּוֹחֶה אֶת הַשֵּׁנִי.

[[39:10]] "שֶׁהַמְּבַקֵּשׁ הֶעָרֵב" – מוּבָן בְּפַשְׁטוּת: מִי שֶׁאוֹהֵב דְּבָרִים טְעִימִים – מָה עִם הַטּוֹב? "יְבַזֶּה אֶת הַטּוֹב". מָה אַתָּה מְדַבֵּר עַכְשָׁיו עַל "טוֹב"? זֶה טָעִים!

[[39:33]] יָשְׁבוּ פַּעַם בַּמָּלוֹן [?39:35], וְשָׁאֲלוּ אֶת הָרַב: הָרַב, הַמָּלוֹן הַזֶּה [?]. אָמַר: אוֹתוֹ רָשָׁע, רָשָׁע [?] – הוּא הָיָה בַּכְּנֶסֶת, וְכָל הַכְּנֶסֶת יָדְעָה שֶׁהוּא [?] – שָׁאֲלוּ אוֹתוֹ: כָּשֵׁר? אָמַר לָהֶם הָרָשָׁע הַזֶּה: הָעִיקָּר טָעִים. אֲנִי אוֹמֵר "רָשָׁע" – אֵין לִי שׁוּם עִנְיָן אִיתּוֹ. אֲבָל אֵלֶּה אֲנָשִׁים שֶׁמַּחֲלִיטִים עַל הַחוּקִּים שֶׁל [?40:20]. וְחוֹשְׁבִים: מָה זֶה נוֹגֵעַ לַחוּקִּים? זֶה לֹא נָכוֹן. אֲנַחְנוּ לוֹמְדִים מִן הַקֶּטַע הַזֶּה שֶׁבִּזְמַן שֶׁאָדָם יוֹצֵא מִן הַגְּבוּל, הוּא כְּבָר הוֹלֵךְ לְשִׁיבּוּשׁ – וְשִׁיבּוּשׁ הוּא בַּשֵּׂכֶל. אָז הוּא יָכוֹל לְהַחֲלִיט מָה נָכוֹן וּמָה לֹא נָכוֹן; הוּא יָכוֹל לְהַחֲלִיט כַּמָּה חַיָּלִים לְהוֹצִיא לַשֶּׁטַח.

[[40:55]] [?40:55] מָה שֶׁלּוֹמְדִים מִן הַסְּפָרִים הָאֵלֶּה – עוֹד לִפְנֵי שֶׁהוּא מְדַבֵּר לְךָ עַל הַנּוֹשֵׂא, רַק אֵיךְ שֶׁהוּא מַגִּישׁ לְךָ אֶת הַדְּבָרִים – הוּא עוֹשֶׂה לְךָ סֵדֶר בָּרֹאשׁ. [?41:10] תַּחֲשׁוֹב: אִם אָדָם [?] – מָה זֶה נוֹגֵעַ לַשֵּׂכֶל שֶׁלּוֹ, לַמָּתֵמָטִיקָה שֶׁלּוֹ? לֹא נָכוֹן! זֶה לֹא נָכוֹן. הַמָּתֵמָטִיקָה שֶׁל הַחַיִּים שֶׁלּוֹ – לֹא אִכְפַּת לִי מִמֶּנּוּ, שֶׁיַּעֲשֶׂה מָה שֶׁהוּא רוֹצֶה. אֲבָל הוּא יוֹשֵׁב וּמְחוֹקֵק חוּקִּים שֶׁמַּשְׁפִּיעִים עַל כּוּלָּנוּ, [?41:45]. לָמָּה? כִּי הוּא אֵינוֹ מִסְתַּכֵּל מָה [?]; הוּא מִסְתַּכֵּל כַּמָּה כֶּסֶף צָרִיךְ לְהִיכָּנֵס.

[[41:55]] וְעוֹד: "וְהַמְּבַקֵּשׁ הַמּוֹעִיל" – מִי שֶׁמְּבַקֵּשׁ אֶת הַמּוֹעִיל – "יְבַזֶּה הֶעָרֵב וְהַטּוֹב". אִם זֶה דָּבָר שֶׁאֵינוֹ מוֹעִיל לִי – [?] אֶת הֶעָרֵב. לָמָּה? כִּי לִפְעָמִים בִּגְלַל הֶעָרֵב אָדָם מַפְסִיד אֶת הַמּוֹעִיל. וְהַטּוֹב? מָה זֶה מוֹעִיל לִי? אִם זֶה לֹא מַכְנִיס לִי חֲמֵשֶׁת אֲלָפִים שֶׁקֶל בָּעִסְקָה – מָה אִכְפַּת לִי אִם זֶה טוֹב אוֹ לֹא טוֹב? אִם זֶה לֹא מוֹעִיל. הָאִינְטֵרֵס, הָאֵגוֹ.

[[42:45]] "וְהַמְּבַקֵּשׁ הַטּוֹב יְבַזֶּה אֶת שְׁנֵיהֶם". בְּוַדַּאי: אָדָם שֶׁמְּבַקֵּשׁ אֶת הַטּוֹב לֹא יְחַפֵּשׂ [?]. אַתָּה רוֹאֶה שֶׁשְּׁלוֹשְׁתָּם [?43:05] – הָיְתָה הַהַגְבָּלָה הַהֶכְרֵחִית לָהֶם. לָכֵן צָרִיךְ לִתְפּוֹס אֶת שְׁלוֹשְׁתָּם וּלְהַגְבִּיל אוֹתָם. וְאֵיךְ – מַגְבִּילִים כָּל דָּבָר בְּנִפְרָד, אוֹ אֶת כּוּלָּם בְּיַחַד?

## מָשָׁל הַתִּזְמוֹרֶת

[[43:35]] זֶהוּ הַדָּבָר: צָרִיךְ שֶׁכָּל אֶחָד מֵהֶם יִהְיֶה שָׁלֵם. רֵאשִׁית, כָּל נַגָּן – נַגָּן הַכִּינּוֹר – יִהְיֶה מוּמְחֶה, וְהַכִּינּוֹר שֶׁלּוֹ יִהְיֶה תַּקִּין בִּפְנֵי עַצְמוֹ. [?44:07] עַל הַתּוֹף, [?] עַל הֶחָלִיל. וְדִבֵּר [?] לְהַתְאִים בֵּינֵיהֶם, כְּדֵי שֶׁיִּמָּצְאוּ שְׁלוֹשְׁתָּם יַחַד. עַל מְנַת שֶׁלֹּא יִשְׂנָא אֶחָד אֶת הַשֵּׁנִי, צָרִיךְ שֶׁכָּל אֶחָד יִהְיֶה מוּגְבָּל בַּהַגְבָּלָה שֶׁלּוֹ: הַהַגְבָּלָה שֶׁל הַטּוֹב, הַהַגְבָּלָה שֶׁל הַמּוֹעִיל, הַהַגְבָּלָה שֶׁל הֶעָרֵב – כָּל אֶחָד בִּמְקוֹמוֹ.

[[44:50]] ⟦src|כִּי כַּאֲשֶׁר הָיָה כֵּן תְּשׁוּקַת כָּל חֶלְקֵי הָאָדָם שָׁוָה בַּדָּבָר וְיוּשְׁלַם בְּיָדוֹ מִבְּלִי שׁוּם מֻוֹחֶה כִּי לָזֶה הָיְתָה חָכְמָה מְעוּלָּה וּנְדִיבוּת נִפְלָא⟧{{=עקדת יצחק, שער עז}}. זֶה נִפְלָא בֶּאֱמֶת. [?45:20]

[[45:28]] וְלֹא נִשְׁכַּח דָּבָר אֶחָד: בְּכָל הִזְדַּמְּנוּת יֵשׁ לָנוּ אֶת הַדִּילֵמָה הַזֹּאת. בְּכָל דָּבָר שֶׁאֲנַחְנוּ פּוֹגְשִׁים בַּחַיִּים יֵשׁ לָנוּ אֶת הַבְּעָיָה הַזֹּאת שֶׁל הַטּוֹב, הַמּוֹעִיל וְהֶעָרֵב, וַאֲנַחְנוּ רוֹאִים שֶׁאֶחָד נוֹגֵד אֶת הַשֵּׁנִי. [?45:55] מָה עוֹשִׂים? מֹאזְנַיִם בֵּין שְׁלוֹשְׁתָּם. וּמִי יְאַזֵּן? אֵיזֶה פְּסִיכוֹלוֹג? מִי יָכוֹל לַעֲשׂוֹת זֹאת? רַק הַתּוֹרָה הַקְּדוֹשָׁה.

[[46:24]] כְּמוֹ שֶׁאָמַרְנוּ – בַּתִּזְמוֹרֶת הַפִילְהַרְמוֹנִית: כָּל הַיּוֹפִי הוּא כְּשֶׁאַתָּה שׁוֹמֵעַ אֶת הַכּוֹל כְּאֶחָד, כְּאֶחָד. [?46:50] לְאַזֵּן. וּבַשָּׂפָה שֶׁל יָמֵינוּ: לְקַבֵּל אֶת הַשְּׁלֵמוּת שֶׁל הַדָּבָר – שֶׁתְּקַבֵּל אוֹתוֹ, תִּשְׁמַע אוֹתוֹ וְתֵהָנֶה מִמֶּנּוּ. אַתָּה צָרִיךְ אֶת הָאִיזּוּן בִּשְׁלוֹשֶׁת הַדְּבָרִים, אַחֶרֶת [?]. וְזֶה נָכוֹן מְאוֹד: כְּמוֹ שֶׁאָמַרְתִּי, בְּכָל דָּבָר יֵשׁ לָנוּ אֶת הַשְּׁאֵלָה הַזֹּאת, וְאֵינֶנּוּ מַרְגִּישִׁים בָּהּ, כְּמוֹ שֶׁהָרַב [?47:40] – אֶחָד שׂוֹנֵא אֶת הַשֵּׁנִי.

## הַתּוֹרָה הָאֱלֹהִית נוֹתֶנֶת שׁוֹחַד לַמְקַטְרְגִים

[[47:42]] ⟦src|הָיְתָה חָכְמָה מְעוּלָּה וּנְדִיבוּת נִפְלָא לַתּוֹרָה הָאֱלֹהִית לִנְדּוֹר וְלִנְדּוֹב בַּמַּעֲשִׂים הַתּוֹרִיִּים דִּבְרֵי הֶעָרֵב וְהַמּוֹעִיל כְּדֵי לְזָרֵז הָאָדָם אֶל הַמַּעֲשֶׂה הַטּוֹב מִצַּד כָּל כֹּחוֹתָיו וְלָתֵת שׁוֹחַד לַמְקַטְרְגִים.⟧{{=עקדת יצחק, שער עז}}. לָכֵן בָּאָה הַתּוֹרָה. מָה עָשְׂתָה הַתּוֹרָה? הִיא כָּלְלָה אֶת שְׁלוֹשֶׁת הַדְּבָרִים הָאֵלֶּה – הַטּוֹב, הַמּוֹעִיל וְהֶעָרֵב – [?48:30]. וְהוּא מִתְבַּטֵּא: כְּדֵי לְזָרֵז "אֶל הַמַּעֲשֶׂה הַטּוֹב". בָּרוּר: הַטּוֹב הוּא הַמַּטָּרָה. אֲבָל מָה עוֹשִׂים עִם הֶעָרֵב? אוֹמֵר הָרַב: "מִצַּד כָּל כּוֹחוֹתָיו, וְלָתֵת שׁוֹחַד לַמְקַטְרְגִים".

[[48:55]] מָה פֵּירוּשׁ "שׁוֹחַד לַמְקַטְרְגִים"? הַגּוּף דּוֹרֵשׁ אֶת הֶעָרֵב. מָה עוֹשִׂים עִם הַגּוּף? אִם תִּיקַּח רַק אֶת הַטּוֹב וְתֹאמַר לַגּוּף: תֹּאכַל צָנוּן, אַרְבַּע פְּעָמִים בְּיוֹם צָנוּן – זֶה טוֹב מְאוֹד, זֶה יוֹעִיל לְךָ – הוּא לֹא יִשְׁמַע. לָכֵן צָרִיךְ לְשַׁחֵד אֶת הַמְּקַטְרֵג. וּמִי יָכוֹל לַעֲשׂוֹת אֶת זֶה? רַק הַתּוֹרָה הָאֱלֹהִית, רַק מָה שֶׁהוּא אֱלֹהִי. אָדָם יָכוֹל [?49:40]? בִּלְתִּי אֶפְשָׁרִי.

## "קִנְאַת סוֹפְרִים תִּרְבֶּה חָכְמָה"

[[49:45]] ⟦src|וְעַל דֶּרֶךְ שֶׁאָמַר קִנְאַת סוֹפְרִים תִּרְבֶּה חָכְמָה כִּי כְּשֶׁהַקִּנְאָה עִם הֱיוֹתָהּ מֵהַמִּדּוֹת הַפְּחוּתוֹת וְהַמְּעַכְּבוֹת כָּל שְׁלֵימוּת מָבוֹא לִבְעָלֶיהָ תַּעֲזוֹר בַּדָּבָר הַהוּא מִי מְעַכֵּב.⟧{{=עקדת יצחק, שער עז}}{{=בבא בתרא כא עא}}. קִנְאָה הִיא דָּבָר [פָּחוֹת]; אֲבָל אִם הִיא בָּאָה כְּדֵי שֶׁתְּלְמַד עוֹד דַּף גְּמָרָא, אוֹ שֶׁתַּעֲמִיק עוֹד – אַדְּרַבָּה וְאַדְּרַבָּה. וּמִי מְכַוֵּן לָזֶה? רַק הַתּוֹרָה.

[[50:28]] לָכֵן אָמְרוּ – פַּעַם [?50:30] אָמַר לִי בְּאֶמְצַע הַשִּׂיחָה, שֶׁמָּה שֶׁאָמְרוּ ⟦sefaria|וֶהֱוֵי זָנָב לָאֲרָיוֹת, וְאַל תְּהִי רֹאשׁ לַשּׁוּעָלִים:⟧{{=אבות ד, טו}} – עָדִיף שֶׁתִּהְיֶה תָּמִיד זָנָב. אַתָּה יוֹשֵׁב בַּחֶבְרָה: יֵשׁ שָׁם אֲנָשִׁים שֶׁשּׁוֹאֲלִים, יֵשׁ אֲנָשִׁים חֲכָמִים. אַתָּה תָּמִיד רוֹצֶה לְהִתְבַּלֵּט, רוֹצֶה לִהְיוֹת רֹאשׁ וְלֹא זָנָב. אָז מָה? [?51:10] כָּאן יֵשׁ חָכָם מִמְּךָ. אֲבָל אָדָם, טִבְעוֹ לְהִתְבַּלֵּט – אָז הוּא זוֹרֵק אֵיזוֹ חָכְמָה כְּדֵי לְהַרְאוֹת אֶת עַצְמוֹ. אוֹמְרִים לְךָ: "⟦~|רֹאשׁ לַשּׁוּעָלִים:⟧" – אֲבָל אֵינְךָ לוֹמֵד כְּלוּם; מָה שֶׁיֵּשׁ לְךָ, יֵשׁ לְךָ. אֲבָל אִם תִּהְיֶה זָנָב לָאֲרָיוֹת – תִּלְמַד.

[[51:30]] וְאָמַר לִי שֶׁהַדָּבָר הַזֶּה אֵינוֹ נָכוֹן אֵצֶל יְלָדִים. יְלָדִים צָרִיךְ לְחַנֵּךְ לִהְיוֹת רֹאשׁ. כִּי אִם לֹא – הַיֶּלֶד יִכָּנֵס לְכִיתָּה ד׳ וְיֹאמַר: אֵינֶנִּי רוֹצֶה לְהִתְבַּלֵּט, מַסְפִּיק לִי שֵׁשׁ מִתּוֹךְ עֶשֶׂר, זֶה עוֹבֵר. הוּא לֹא יִתְקַדֵּם. לָכֵן יְלָדִים מְחַנְּכִים: תִּהְיֶה הָרִאשׁוֹן בַּכִּיתָּה.

## דָּוִד וְגָלְיָת – יְצִיאָה מִן הַגְּבוּל לְשֵׁם מַטָּרָה עֶלְיוֹנָה

[[52:10]] וּמַמְשִׁיךְ: ⟦src|הֲלֹא תִּרְאֶה כִּי הָיְתָה עֵצָה יְעוּצָה מִכָּל יִשְׂרָאֵל בַּדָּבָר הַפְּלִשְׁתִּי לַעֲשׂוֹת כֵּן כְּמוֹ שֶׁנֶּאֱמַר⟧{{=עקדת יצחק, שער עז}} ⟦sefaria|וַיֹּאמֶר אִישׁ יִשְׂרָאֵל הַרְּאִיתֶם הָאִישׁ הָעֹלֶה הַזֶּה כִּי לְחָרֵף אֶת־יִשְׂרָאֵל עֹלֶה וְהָיָה הָאִישׁ אֲשֶׁר־יַכֶּנּוּ יַעְשְׁרֶנּוּ הַמֶּלֶךְ עֹשֶׁר גָּדוֹל וְאֶת־בִּתּוֹ יִתֶּן־לוֹ וְאֵת בֵּית אָבִיו יַעֲשֶׂה חָפְשִׁי בְּיִשְׂרָאֵל⟧{{=שמואל א יז, כה}}. [?52:40]

[[52:50]] כְּלוֹמַר, הוּא מֵבִיא רְאָיָה: גָּלְיָת הָיָה יוֹצֵא [?] יוֹם יוֹם וְאוֹמֵר: מִי יוּכַל לִי? וְכַוָּנָתוֹ הָיְתָה: אַתֶּם, הַיְּהוּדִים, יֵשׁ לָכֶם אֱלוֹהַּ גָּדוֹל? נִרְאֶה אִם הָאֱלוֹהַּ שֶׁלָּכֶם יַעֲזוֹר! הוּא הָיָה מְחָרֵף "מַעַרְכוֹת אֱלֹהִים חַיִּים". לָכֵן אָמְרוּ: מִי שֶׁיָּכֶּה אֶת הַפְּלִשְׁתִּי הַזֶּה – נִיתֵּן לוֹ עוֹשֶׁר גָּדוֹל, נִיתֵּן לוֹ אֶת בַּת הַמֶּלֶךְ לְהִתְחַתֵּן אִיתּוֹ. אֵיפֹה הַהַגְבָּלָה? תַּגְבִּיל! תַּגִּיד: תְּנוּ לוֹ חֲמִישִּׁים אֶלֶף דּוֹלָר! לֹא – כִּי אֲנַחְנוּ יוֹצְאִים מִן הַהַגְבָּלָה שֶׁלָּנוּ בִּשְׁבִיל מַטָּרָה גְּבוֹהָה: אִם נְנַצֵּחַ אֶת הַפְּלִשְׁתִּי, נְקַדֵּשׁ שָׁם שָׁמַיִם. מָה יוֹצֵא? אַתָּה מִשְׁתַּמֵּשׁ בְּמִידָּה לֹא טוֹבָה כְּדֵי לְהַגִּיעַ [?53:55]. וְכָךְ גַּם הַקִּנְאָה: מִידָּה פְּחוּתָה – אֲבָל אִם אַתָּה מִשְׁתַּמֵּשׁ בָּהּ כְּדֵי לְהִתְעַלּוֹת, אַדְּרַבָּה.

[[54:05]] וְעַכְשָׁיו דָּוִד הַמֶּלֶךְ עָלָיו הַשָּׁלוֹם: ⟦src|וְאוּלָם מָה שֶׁהִגִּיעַ מִמַּעֲלְתוֹ שֶׁל דָּוִד לְבַזּוֹת כָּל מַהֵר וּמַתָּן בַּפּוֹעֵל הַהוּא זוּלָתִי הַפּוֹעֵל הַטּוֹב הַנִּדְרָשׁ מֵהַשָּׁלֵם מִצַּד מַעֲלָתוֹ בִּכְלָל⟧{{=עקדת יצחק, שער עז}}. פֵּירוּשׁ: דָּוִד הַמֶּלֶךְ עָלָיו הַשָּׁלוֹם הָיָה אִישׁ שָׁלֵם כָּל כָּךְ, שֶׁלֹּא שָׂם לֵב לְבַת הַמֶּלֶךְ וְלָעוֹשֶׁר. [?54:45] הַטּוֹב הַמּוּחְלָט – לְקַדֵּשׁ שָׁם שָׁמַיִם, לְהַפִּיל אֶת הַפְּלִשְׁתִּי – בְּלִי הַמּוֹעִיל וּבְלִי הֶעָרֵב. רַק הַטּוֹב. אֲפִילּוּ אֶת הַיְּצִיאָה הַזֹּאת מִן הַגְּבוּל לֹא עָשָׂה. [?55:05] שֶׁיֵּשׁ אֲנָשִׁים שֶׁהֵם מֵעַל.

[[55:10]] ⟦src|מִכָּל מָקוֹם לָמַדְנוּ שֶׁכְּבָר הָיְתָה עֵצָה טוֹבָה מִטַּעַם הַמֶּלֶךְ וְיוֹעֲצָיו וְשָׂרָיו לְעוֹרֵר לִבוֹת הָאֲנָשִׁים אֲשֶׁר בַּמַּעֲשִׂים הַמְּעוּלִּים נַפְשָׁם חָפְצָה לַעֲשׂוֹת חַיִל בְּמַתְּנוֹת הָעוֹשֶׁר וְהַכָּבוֹד⟧{{=עקדת יצחק, שער עז}}. כְּמוֹ שֶׁאָמַרְנוּ: דָּוִד הַמֶּלֶךְ הָיָה יוֹצֵא מִן הַכְּלָל; אֲבָל הַמֶּלֶךְ וְיוֹעֲצָיו אָמְרוּ: בּוֹאוּ נֵצֵא מִן הַגְּבוּל שֶׁלָּנוּ כְּדֵי לְהַשִּׂיג מַטָּרָה גְּבוֹהָה יוֹתֵר. עַד כָּאן, רַבּוֹתַי, הַהַקְדָּמָה.

## אֶרֶץ יִשְׂרָאֵל – אֶרֶץ מְאוּזֶּנֶת

[[55:40]] וְעַכְשָׁיו הַהַקְדָּמָה עַל אֶרֶץ יִשְׂרָאֵל – הַדָּבָר שֶׁתָּמִיד דִּיבַּרְתִּי עָלָיו. בּוֹאוּ נִרְאֶה מָה הוּא עוֹשֶׂה עִם זֶה: ⟦src|וְהִנֵּה הַתַּחְבּוּלָה הַנְּכוֹנָה הַזֹּאת כֻּוְּנָה הָאֱלֹהִים עַל הַשְּׁלֵימוּת בְּתְתוֹ לְעַמּוֹ יִשְׂרָאֵל אֶרֶץ⟧{{=עקדת יצחק, שער עז}} – כָּל דָּבָר בִּגְבוּלוֹ, וּשְׁלוֹשֶׁת הַדְּבָרִים – הַטּוֹב, הַמּוֹעִיל וְהֶעָרֵב – מְאוּזָּנִים יַחַד. כְּלוֹמַר, כַּאֲשֶׁר תִּכְנֵן הַקָּדוֹשׁ בָּרוּךְ הוּא לָתֵת לְעַמּוֹ אֶרֶץ, הוּא בָּחַר בָּאָרֶץ הַזֹּאת, שֶׁבְּנוּיָה עַל הַמַּחֲשָׁבָה הַזֹּאת.

[[56:55]] ⟦src|כְּאַרְצֵנוּ צְבִי הִיא לְכָל הָאֲרָצוֹת אֲשֶׁר נִמְצְאוּ בָּהּ צָרְכֵי הַחַיִּים מִכָּל צִדְדֵיהֶם וְלֹא יֶחְסֵר כָּל בָּהּ.⟧{{=עקדת יצחק, שער עז}}. כְּלוֹמַר, הָאָרֶץ עַצְמָהּ, מֵעֶצֶם טִבְעָהּ, בְּנוּיָה עַל הַבָּסִיס הַזֶּה: יֵשׁ בָּהּ הַטּוֹב, הַמּוֹעִיל וְהֶעָרֵב, בְּמַתְכּוֹן שֶׁל אִיזּוּן. כְּשֶׁבָּחַר ה׳ יִתְבָּרַךְ לָתֵת אֶרֶץ לְעַמּוֹ, הוּא בָּחַר אֶרֶץ מְאוּזֶּנֶת, אֶרֶץ מוּכָנָה, שֶׁכּוּלָּהּ מְדוּדָה וּמְאוּזֶּנֶת עַל הַבָּסִיס הַזֶּה – שֶׁשְּׁלוֹשֶׁת הַמַּרְכִּיבִים הָאֵלֶּה נִמְצָאִים בָּהּ בַּתִּרְכּוֹבֶת הַנְּכוֹנָה וְהַמְּאוּזֶּנֶת.

[[58:10]] [?58:10] לֵךְ וְתִבְנֶה בָּאָרֶץ הַזֹּאת בְּרֵיכָה מְעוֹרֶבֶת – [?58:35]. בְּוַדַּאי. אֲנִי אוֹמֵר זֹאת עוֹד לִפְנֵי שֶׁהוּא יַסְבִּיר לָנוּ: לֵךְ וְתִבְנֶה בָּאָרֶץ הַזֹּאת [?] – זֶה לֹא יֵלֵךְ, זֶה לֹא יֵלֵךְ. בְּרֵיכָה מְעוֹרֶבֶת – שִׂימוּ לֵב, רַבּוֹתַי, עוֹד לֹא נָגַעְנוּ בַּתּוֹרָה; מְדַבְּרִים עַל הָאָרֶץ. אַתָּה בּוֹנֶה בְּרֵיכָה מְעוֹרֶבֶת בָּאָרֶץ הַזֹּאת – הָאָרֶץ מַתְחִילָה לָצֵאת מֵאִיזּוּן, מַתְחִילָה לִשְׁקוֹעַ.

[[59:15]] [?59:15] חֲזִיר בִּמְקוֹם קָרְבָּן לְבֵית הַמִּקְדָּשׁ [?59:31]. מִי מַכִּיר [?]? תַּלְמִיד חָכָם יָדַע זֹאת מִן הַתּוֹרָה, אֲבָל כָּאן אֲנַחְנוּ רוֹאִים זֹאת.

[[60:10]] [?60:10]

[[61:03]] [?61:03] לְהַדְלִיק אוֹתוֹ עַל כָּל חַלּוֹן, עַל כָּל בַּיִת [?61:15]. טוֹב, נַמְשִׁיךְ.

[[61:20]] [מן הקהל:] הָרַב, אוּלַי זֶה מַקְדִּים אֶת הַמְּאוּחָר, וַאֲנִי מַאֲמִין שֶׁהָרַב יַגִּיעַ לָזֶה – אֲבָל הַטּוֹב, הַמּוֹעִיל וְהֶעָרֵב הֵם בְּדִיּוּק "יָפֶה, עֲשִׁירָה וּבַת טוֹבִים" שֶׁבְּמָשָׁל. אֵלֶּה שְׁלוֹשֶׁת הַדְּבָרִים, נָכוֹן? זוֹ הַמִּשְׁוָאָה.

[[61:55]] נָכוֹן, נָכוֹן. הַמִּדְרָשׁ – יֵשׁ בּוֹ הַכּוֹל. אֵלֶּה הַיְּסוֹדוֹת.

## "⟦~|כִּי ה׳ אֱלֹהֶיךָ מְבִיאֲךָ אֶל־אֶרֶץ טוֹבָה⟧"

[[62:05]] עַכְשָׁיו: ⟦src|אָמְנָם עַל הַהַגְבָּלָה הַנְּאוֹתָה וְהַהַסְכָּמָה הַמְּשׁוֹעֶרֶת לַקֶּבֶץ הַטּוֹב וְהַמּוֹעִיל וְהֶעָרֵב יַחַד מִבְּלִי שֶׁיִּדְחֶה זֶה אֶת זֶה עַל הַדֶּרֶךְ שֶׁאָמַרְנוּ כְּמוֹ שֶׁבֵּיאֵר הָעִנְיָן הַזֶּה מֹשֶׁה אֲדוֹנֵינוּ עָלָיו הַשָּׁלוֹם יָפֶה יָפֶה בְּמִשְׁנֵה תּוֹרָה בְּאוֹמְרוֹ כִּי ה׳ אֱלֹהֶיךָ מֵבִיאֵךָ אֶל אֶרֶץ טוֹבָה⟧{{=עקדת יצחק, שער עז}}. מִי כָּתַב אֶת הַדְּבָרִים הָאֵלֶּה? מֹשֶׁה רַבֵּנוּ בְּפֵירוּשׁ. מָה הוּא אוֹמֵר? ⟦sefaria|כִּי ה׳ אֱלֹהֶיךָ מְבִיאֲךָ אֶל־אֶרֶץ טוֹבָה אֶרֶץ נַחֲלֵי מָיִם עֲיָנֹת וּתְהֹמֹת יֹצְאִים בַּבִּקְעָה וּבָהָר אֶרֶץ חִטָּה וּשְׂעֹרָה וְגֶפֶן וּתְאֵנָה וְרִמּוֹן אֶרֶץ־זֵית שֶׁמֶן וּדְבָשׁ⟧{{=דברים ח, ז–ח}}. יוֹפִי.

[[63:50]] עַכְשָׁיו שָׁמְעוּ אֶת הַקֶּטַע [?]. אַחַר כָּךְ הוּא אוֹמֵר: ⟦sefaria|הִשָּׁמֶר לְךָ פֶּן־תִּשְׁכַּח אֶת־ה׳ אֱלֹהֶיךָ לְבִלְתִּי שְׁמֹר מִצְוֺתָיו וּמִשְׁפָּטָיו וְחֻקֹּתָיו אֲשֶׁר אָנֹכִי מְצַוְּךָ הַיּוֹם⟧{{=דברים ח, יא}}. לָמָּה הוּא מַכְנִיס אֶת הַפָּסוּק הַזֶּה בְּאֶמְצַע שֶׁבַח אֶרֶץ יִשְׂרָאֵל? וְאַחַר כָּךְ הוּא חוֹזֵר: ⟦sefaria|פֶּן־תֹּאכַל וְשָׂבָעְתָּ וּבָתִּים טֹבִים תִּבְנֶה וְיָשָׁבְתָּ⟧{{=דברים ח, יב}} – מָה קוֹרֶה כַּאֲשֶׁר אַתָּה בּוֹנֶה בָּתִּים וְכוּלִּי – ⟦sefaria|וְרָם לְבָבֶךָ וְשָׁכַחְתָּ אֶת־ה׳ אֱלֹהֶיךָ הַמּוֹצִיאֲךָ מֵאֶרֶץ מִצְרַיִם מִבֵּית עֲבָדִים⟧{{=דברים ח, יד}}.

[[64:50]] ⟦src|כִּי שָׁם בְּלִי סָפֵק בֵּיאֵר שֶׁתַּכְלִית הָאָרֶץ הָיָה לְפַרְנֵס הַנֶּפֶשׁ רִאשׁוֹנָה⟧{{=עקדת יצחק, שער עז}}. אוֹמֵר כָּךְ: נָכוֹן, נָתַתִּי לְךָ אֶרֶץ חִיטָּה וּשְׂעוֹרָה וְכוּלִּי – אָז לָמָּה הוּא מַכְנִיס מִצְוֹת? דִּיבַּרְנוּ עַל שָׁלוֹשׁ נְפָשׁוֹת; הַנֶּפֶשׁ הָרִאשׁוֹנָה הִיא הַנֶּפֶשׁ הַמַּשְׂכֶּלֶת. הָאָרֶץ הַזֹּאת נוֹעֲדָה בָּרֹאשׁ וָרִאשׁוֹנָה לְפַרְנֵס אֶת הַנֶּפֶשׁ הַמַּשְׂכֶּלֶת. ⟦src|אָמְנָם הַגּוּף בִּשְׁנִיּוֹת. וְלָכֵן הֵכִין לָהֶן הַגְבָּלַת הָעִנְיָנִים כָּלָּם בְּאוֹפֶן שֶׁיִּתְפַּשְּׁרוּ בֵּינֵיהֶם וְלֹא יָעִיקוּ זֶה אֶת זֶה⟧{{=עקדת יצחק, שער עז}} – כְּמוֹ שֶׁאָמַרְנוּ.

[[66:10]] ⟦src|וְהוּא מָה שֶׁבֵּיאֲרוּ בְּאוֹמְרוֹ שֶׁיִּתְּנוּ לִבָּם עַל הַתּוֹרָה וּמִצּוֹתֶיהָ אֲשֶׁר הִזְכִּירָם רִאשׁוֹנָה בְּאוֹמְרו⟧{{=עקדת יצחק, שער עז}} ⟦sefaria|כָּל־הַמִּצְוָה אֲשֶׁר אָנֹכִי מְצַוְּךָ הַיּוֹם תִּשְׁמְרוּן לַעֲשׂוֹת לְמַעַן תִּחְיוּן וּרְבִיתֶם וּבָאתֶם וִירִשְׁתֶּם אֶת־הָאָרֶץ אֲשֶׁר־נִשְׁבַּע ה׳ לַאֲבֹתֵיכֶם⟧{{=דברים ח, א}}. לִפְנֵי שֶׁהוּא מְדַבֵּר עַל שֶׁבַח אֶרֶץ יִשְׂרָאֵל, הוּא אוֹמֵר קוֹדֶם כּוֹל "⟦~|כָּל־הַמִּצְוָה אֲשֶׁר אָנֹכִי מְצַוְּךָ הַיּוֹם תִּשְׁמְרוּן לַעֲשׂוֹת⟧". זוֹ הַהַקְדָּמָה לְמַעֲלוֹת אֶרֶץ יִשְׂרָאֵל: הָאָרֶץ הַזֹּאת שֶׁאֲנִי נוֹתֵן לָכֶם דּוֹאֶגֶת בָּרֹאשׁ וָרִאשׁוֹנָה לַנֶּפֶשׁ הַמַּשְׂכֶּלֶת, וְאַחַר כָּךְ – "⟦~|אֶרֶץ חִטָּה וּשְׂעֹרָה⟧", שֶׁהֵן בִּשְׁבִיל הַנֶּפֶשׁ הָרַגְזָנִית וְהַחִיּוּנִית.

[[67:10]] ⟦src|וְשֶׁלֹּא יִטְרְדוּ עַצְמָם עַל זוּלָתָן מִן הָעֲסָקִים הַזְּמַנִּיִּים⟧{{=עקדת יצחק, שער עז}} – שֶׁלֹּא תִּתְבַּלְבְּלוּ וְתֹאמְרוּ: מִכֵּיוָן שֶׁזּוֹ אֶרֶץ חִיטָּה וּשְׂעוֹרָה, בּוֹאוּ נִתְעַסֵּק בְּחַקְלָאוּת. ⟦src|כִּי דַּי בָּאָרֶץ הַהִיא לְהַשְׁלִים סְפּוּקָם לְכָל מָה שֶׁיִּצְטָרְכוּ בִּהְיוֹתָם מִתְעַסְּקִים בִּלְמוּדָם וּשְׁלֵימוּתָם וְהותֵר.⟧{{=עקדת יצחק, שער עז}}. אִם אַתֶּם רוֹצִים אֶת הַסִּיפּוּק שֶׁלָּכֶם, אֶת מָה שֶׁאַתֶּם צְרִיכִים מִן הֶעָרֵב וְהַמּוֹעִיל – הָאָרֶץ הַזֹּאת תְּסַפֵּק לָכֶם מָה שֶׁאַתֶּם צְרִיכִים, בְּלִי שֶׁהֶעָרֵב וְהַמּוֹעִיל יָבוֹאוּ עַל חֶשְׁבּוֹן הַטּוֹב, שֶׁהוּא הַתּוֹרָה.

[[68:35]] הָאָרֶץ הַזֹּאת מְאוּזֶּנֶת, שְׁקוּלָה וּמְדוּדָה, כְּדֵי לָתֵת לַיְּהוּדִי מָה שֶׁהוּא צָרִיךְ לַנֶּפֶשׁ הַחִיּוּנִית וְהָרַגְזָנִית, וְשֶׁיּוּכַל לָשֶׁבֶת וְלִלְמוֹד תּוֹרָה. [?68:55] כָּל יְהוּדִי יְקַבֵּל אֶת הַסִּיפּוּק שֶׁלּוֹ, כְּדֵי שֶׁיּוּכַל [?] וְגַם יַרְגִּישׁ טוֹב.

[[69:20]] ⟦src|וְזֶה בְּמָה שֶׁיּוּכְלוּ לְהַעֲבֵד מִמֶּנָּה ע״י עֵינּוֹת וּתְהוֹמוֹת הַיּוֹצְאִים בַּבִּקְעָה וּבָהָר וְהִיא אֶרֶץ חִטָּה וּשְׂעוֹרָהּ גֶּפֶן וּתְאֵנָה וּרָמוֹן אֶרֶץ זַיִת שֶׁמֶן וּדְבַשׁ.⟧{{=עקדת יצחק, שער עז}}. הִיתָּכֵן שֶׁתִּהְיֶה אֶרֶץ מְיוּשֶּׁבֶת בְּלִי הַדְּבָרִים הָאֵלֶּה? הַאִם אֲנַחְנוּ רוֹצִים לִהְיוֹת רַק רוּחָנִיִּים? לֹא.

[[70:00]] ⟦src|אֶלָּא שֶׁרָצָה לוֹמַר⟧{{=עקדת יצחק, שער עז}} – מָה הוּא מְחַדֵּשׁ? שֶׁהוּא אוֹמֵר לָנוּ "⟦~|אֶרֶץ נַחֲלֵי מָיִם⟧" – וְכִי הָיִית רוֹצָה שֶׁתִּהְיֶה בְּלִי מַיִם? מָה הַשֶּׁבַח? אוֹמֵר: לֹא, הַחִידּוּשׁ הוּא כָּךְ: ⟦src|כִּי יִמְצְאוּ שָׁם מִכָּל הַמִּינִים הָאֵלֶּה דֵּי סְפּוּקָם לִכְדֵי חַיּוּתָם הַמְּשׁוֹעָר וְהַמּוּגְבָּל⟧{{=עקדת יצחק, שער עז}}. זֶהוּ. לֹא שֶׁיֵּשׁ כִּנֶּרֶת – אֶלָּא שֶׁהַכִּנֶּרֶת הִיא בְּדִיּוּק מָה שֶׁאַתָּה צָרִיךְ, בַּהַגְבָּלָה שֶׁל הֶעָרֵב וְהַמּוֹעִיל, וְלֹא מֵעֵבֶר לָזֶה. כְּמוֹ בָּמָן: עוֹמֶר לַגּוּלְגּוֹלֶת. זֶה הַחִידּוּשׁ.

[[70:55]] הַחִידּוּשׁ אֵינוֹ שֶׁיֵּשׁ מַיִם, אֶלָּא שֶׁהַמַּיִם שֶׁלָּהּ מַסְפִּיקִים. [?71:05] נִכְנָסִים לָאָרֶץ וְלֹא עוֹשִׂים כְּלוּם – לֹא מוֹבִיל אַרְצִי, לֹא צִינּוֹרוֹת שֶׁל "מְקוֹרוֹת", כְּלוּם. יֵשׁ לְךָ כָּאן מַעֲרֶכֶת – לֹא אוֹטוֹמָטִית, אֶלָּא אֱלֹהִית: אַל תַּעֲשֶׂה כְּלוּם, וּתְקַבֵּל מָה שֶׁאַתָּה צָרִיךְ. [?71:55]

[[72:02]] וְכָל מָה שֶׁקּוֹרֶה עַכְשָׁיו – פֵּירוּשׁוֹ שֶׁאֲנַחְנוּ לֹא רַק שֶׁאֵינֶנּוּ נֶהֱנִים מֵאֶרֶץ יִשְׂרָאֵל; שִׁיבַּשְׁנוּ לָהּ אֶת הַפַּרְצוּף. וְזֶה לֹא קָבוּעַ? בְּוַדַּאי שֶׁלֹּא. [?72:25] הַמִּצְוֹת: "⟦~|כָּל־הַמִּצְוָה אֲשֶׁר אָנֹכִי מְצַוְּךָ⟧". כְּלוֹמַר: אִם אַתָּה מַתְחִיל מִקִּיּוּם הַתּוֹרָה – הָאָרֶץ תִּהְיֶה בָּאִיזּוּן שֶׁלָּהּ. אֲבָל אִם אֵינְךָ שׁוֹמֵר תּוֹרָה וּמִצְוֹת – זֶה כְּאִילּוּ... אַל תַּתְחִיל מִן הַנֶּפֶשׁ הַחִיּוּנִית; תַּתְחִיל מִן הַמַּשְׂכֶּלֶת, תַּתְחִיל לְמַעְלָה. הַאִם זֶה עִנְיָן שֶׁל קוֹדֶם וּמְאוּחָר? לֹא. אַתָּה נִכְנָס לָאָרֶץ, מְמַלֵּא אֶת חוֹבוֹתֶיךָ – וְזֶה כְּבָר יַעֲשֶׂה אֶת [?73:00].

[[73:05]] זֶה כְּמוֹ שֶׁאֲנַחְנוּ חוֹשְׁבִים "דָּתִי וְחִילּוֹנִי". [?73:20] בְּאוֹפֶן טִבְעִי, [ה׳] לֹא בָּרָא אֶת הַיְּהוּדִים שֶׁיִּהְיוּ כּוּלָּם צַדִּיקִים, כּוּלָּם הָאֲרִ״י הַקָּדוֹשׁ. אֲבָל יֵשׁ אִיזּוּן – אִיזּוּן שֶׁאַתָּה שׁוֹבֵר אוֹתוֹ. וּבָרֶגַע שֶׁאַתָּה שׁוֹבֵר אֶת הָאִיזּוּן, כְּבָר אַף אֶחָד [?] – לֹא הַטּוֹבִים, לֹא הָרָעִים. וְאָז יְכוֹלִים לָבוֹא וְלוֹמַר: הַקָּדוֹשׁ בָּרוּךְ הוּא, לָמָּה כָּךְ? – סְלִיחָה, שִׁיבַּשְׁתָּ.

[[74:30]] כִּי אַתָּה – מָה זֶה נוֹגֵעַ אֶחָד לַשֵּׁנִי? עַל חוֹפֶשׁ, עַל דֵּמוֹקְרַטְיָה – עַל מָה אַתָּה מְדַבֵּר? בְּשֵׁם הַחוֹפֶשׁ וְהַדֵּמוֹקְרַטְיָה אַתָּה מְשַׁבֵּשׁ אֶת הָאָרֶץ. אַתָּה לֹא מַרְגִּיז אֶת הַקָּדוֹשׁ בָּרוּךְ הוּא, כִּבְיָכוֹל; הַקָּדוֹשׁ בָּרוּךְ הוּא נָתַן לָנוּ אֶרֶץ מְאוּזֶּנֶת וּשְׁקוּלָה, וּ[אוֹמֵר]: אֵינֶנִּי מִתְעָרֵב לָכֶם – אֲבָל תְּקַבְּלוּ אֶרֶץ שֶׁאֵינָהּ מְאוּזֶּנֶת.

[[75:02]] [?75:02] וּמָה עִם אָמֶרִיקָה? אָמֶרִיקָה – אֵין לִי עֵסֶק עִם אָמֶרִיקָה. [?75:15] הַגּוֹיִים – אוֹתוֹ דָּבָר; הַגּוֹיִים הֵם בַּמַּדְרֵגָה שֶׁלְּמַעְלָה מִן הַבְּהֵמָה, אֲבָל [?75:50] – שֶׁיִּהְיֶה לְךָ מַסְפִּיק [?], מַסְפִּיק כֶּסֶף, [כְּמוֹ] אָמֶרִיקָה. עַל זֶה, רַבּוֹתַי, כָּתוּב כָּךְ.

## "⟦~|אֶרֶץ אֲשֶׁר לֹא בְמִסְכֵּנֻת תֹּאכַל־בָּהּ לֶחֶם⟧"

[[76:10]] הוּא אוֹמֵר: ⟦src|לֹא בְּשֶׁפַע רַב לַעֲשׂוֹת בּוֹ סְחוֹרוֹת וּלְהִתְעַמֵּל בְּסַחֲרוֹתֵיהֶם וּלְהַטְרִיד בְּעִסְקֵיהֶם.⟧{{=עקדת יצחק, שער עז}} – זוֹ הַנְּקוּדָּה. הוּא לֹא יִתֵּן לְךָ אֶרֶץ שׁוֹפַעַת, שֶׁתִּתְעַסֵּק בַּעֲסָקִים, תּוֹצִיא מִמֶּנָּה נֵפְט וְתַעֲשִׁיר. וְאָז מָה קוֹרֶה? אַתָּה מְשַׁבֵּשׁ – זֶה הַמּוֹעִיל; אַתָּה מְשַׁבֵּשׁ. אַתָּה [?] – אֵינְךָ מִתְעַסֵּק בְּטוֹב. אֵינְךָ מִתְעַסֵּק בְּטוֹב.

[[76:58]] ⟦src|וְהִיא מָה שֶׁבֵּיאֵר בְּאוֹמְרו⟧{{=עקדת יצחק, שער עז}} ⟦sefaria|אֶרֶץ אֲשֶׁר לֹא בְמִסְכֵּנֻת תֹּאכַל־בָּהּ לֶחֶם לֹא־תֶחְסַר כֹּל בָּהּ⟧{{=דברים ח, ט}}. מָה פֵּירוּשׁ "בְּמִסְכֵּנוּת"? זוֹ הַמִּילָּה הַמֶּרְכָּזִית, שֶׁעָלֶיהָ [?] בְּנָה אֶת כָּל הַיְּסוֹד: ⟦src|שֶׁהַכַּוָּנָה הָעַצְמִית הִיא לְסַלֵּק הַמִּסְכֵּנוּת⟧{{=עקדת יצחק, שער עז}} – לֹא תִּהְיֶה מִסְכֵּן בָּאָרֶץ הַזֹּאת. לָמָּה? כִּי יֵשׁ לְךָ דֵּי מָה שֶׁאַתָּה צָרִיךְ לַסִּיפּוּק הַנּוֹרְמָלִי שֶׁלְּךָ, ⟦src|וּלְהָסִיר חֶרְפַּת רָעָב⟧{{=עקדת יצחק, שער עז}}. אֲבָל מִצַּד שֵׁנִי – ⟦src|לֹא אֶל הַיִּתְרוֹנוֹת הַדּוֹחִים בַּעֲלִיָּה מֵעַל הַתּוֹרָה וְהַיִּרְאָה הָאֱלֹהִית⟧{{=עקדת יצחק, שער עז}}. [?77:55] זוֹ לֹא אֶרֶץ שֶׁשּׁוֹפַעַת נֵפְט כְּמוֹ הָאֶמִירוּיוֹת, ה׳ יִשְׁמוֹר; [?78:05] שֶׁיֵּשׁ לָהֶם שָׁם עֲבָדִים, וְאִם הָעֶבֶד סוֹטֶה טִיפָּה – [?]. עַבְדוּת. סְעוּדְיָה. מֵאַיִן זֶה בָּא? מִן הַיִּתְרוֹנוֹת. וְאִם יֵשׁ יִתְרוֹנוֹת – מָה עוֹשִׂים? אֲנָשִׁים מִתְעַסְּקִים [?], מִתְעַסְּקִים בְּטֶכְנוֹלוֹגְיוֹת – וְאֵיפֹה הַתּוֹרָה? כְּלוֹמַר, כָּאן הַכּוֹל בִּשְׁבִיל הָאִיזּוּן הַנָּכוֹן וְהַמַּתְאִים.

## "⟦~|וְאָכַלְתָּ וְשָׂבָעְתָּ וּבֵרַכְתָּ⟧" – מֵאָה בְּרָכוֹת

[[78:50]] ⟦src|וְהוּא מָה שֶׁאָמַר⟧{{=עקדת יצחק, שער עז}} ⟦sefaria|וְאָכַלְתָּ וְשָׂבָעְתָּ וּבֵרַכְתָּ אֶת־ה׳ אֱלֹהֶיךָ⟧{{=דברים ח, י}}, ⟦src|וְהַכַּוָּנָה שֶׁתִּהְיֶה הַשְּׂבִיעַה בְּאוֹפֶן שֶׁתִּמָּשֵׁךְ מִמֶּנָּה הַבְּרָכָה וְהַהוֹדָאָה עַל כָּל הֲנָאָה וַהֲנָאָה שֶׁיְּקַבְּלוּ מִמֶּנָּה כִּי הוּא עֵקַר כַּוָּנַת מְצִיאוּתָה⟧{{=עקדת יצחק, שער עז}}. אֲנִי נוֹתֵן לְךָ אֶרֶץ שֶׁ"⟦~|וְאָכַלְתָּ וְשָׂבָעְתָּ⟧" – וּמִן הַשְּׁבִיעָה – "וּבָרַכְתָּ". אַתָּה שְׂבֵעַ? תְּבָרֵךְ. זוֹ עִיקַּר כַּוָּנַת מְצִיאוּתָהּ שֶׁל הָאָרֶץ הַזֹּאת.

[[79:25]] ⟦src|וּכְמוֹ שֶׁאָמְרוּ אָסוּר לֵיהָנוֹת מִן הָעוֹלָם הַזֶּה בְּלֹא בְּרָכָה כִּי הַהֲנָאָה הִיא מְסוֹבֶּבֶת מֵהַבְּרָכָה וְאִם תּוּסַר הַסָּבָה הִנֵּה הַמְּסוֹבֵב יָסוּר וְנִמְנָע.⟧{{=עקדת יצחק, שער עז}}{{=ברכות לה עא}}. כְּלוֹמַר: גַּם הֶעָרֵב שֶׁיֵּשׁ בָּאוֹכֶל בָּא מִן הַבְּרָכָה. [?79:50] שֶׁלְּךָ תָּבִיא אוֹתְךָ לְבָרֵךְ, וְאִם אַתָּה מְבָרֵךְ, הַבְּרָכָה תָּבִיא אֶת הַטַּעַם לָאוֹכֶל. כָּךְ שְׁלוֹשֶׁת הַדְּבָרִים הָאֵלֶּה [?].

[[80:05]] ⟦src|וְלָזֶה קָבְעוּ חֲזַ״ל בְּסֵדֶר חַיֵּי הָאוּמָּה הַזֹּאת וְהַגְבָּלְתָם מֵאָה בְּרָכוֹת בְּכָל יוֹם וְסָמְכוּ אוֹתָם אֶקְרָא⟧{{=עקדת יצחק, שער עז}}{{=מנחות מג עב}} ⟦sefaria|וְעַתָּה יִשְׂרָאֵל מָה ה׳ אֱלֹהֶיךָ שֹׁאֵל מֵעִמָּךְ⟧{{=דברים י, יב}}. "וְאָמְרוּ חַכְמֵי הָאֱמֶת שֶׁזֶּה הַמְּסַפֵּר הוּא כְּדֵי שֶׁתַּגִּיעַ לַנְּשָׁמָה הָעֶלְיוֹנָה"{{=עקדת יצחק, שער עז}}. כָּאן אֲנַחְנוּ נִכְנָסִים מְעַט לְעוֹלַם הַקַּבָּלָה. לָכֵן קָבְעוּ חֲכָמִים שֶׁצָּרִיךְ לְבָרֵךְ מֵאָה בְּרָכוֹת. וְלָמָּה מֵאָה? הוּא אוֹמֵר: "זֶה הַמְּסַפֵּר הוּא כְּדֵי שֶׁתַּגִּיעַ אֶל הַנְּשָׁמָה הָעֶלְיוֹנָה בְּרָכָה אַחַת מֵעֶשֶׂר [?], לְפִי שֶׁבָּאָדָם שָׁלוֹשׁ נְפָשׁוֹת – נֶפֶשׁ, רוּחַ וּנְשָׁמָה"{{=עקדת יצחק, שער עז}}. טוֹב, כָּאן אֲנַחְנוּ נִכְנָסִים לְעוֹלַם הַקַּבָּלָה.

## סִיּוּם

[[80:55]] זֶה הַקֶּטַע הַזֶּה, רַבּוֹתַי: "⟦~|אֶרֶץ אֲשֶׁר לֹא בְמִסְכֵּנֻת תֹּאכַל־בָּהּ לֶחֶם⟧". זֶה הַשִּׁיעוּר שֶׁכָּל חַיַּי רָצִיתִי [?] בְּעֶזְרַת ה׳, וְהַיּוֹם זָכִינוּ לָדַעַת מָה הִיא אֶרֶץ יִשְׂרָאֵל. עַכְשָׁיו אֵין צוֹרֶךְ לְהַסְבִּיר יוֹתֵר לָמָּה וְאֵיךְ. [?81:30] אִם הָאָרֶץ [?] שֶׁלְּךָ – הִיא סְתָם אֲדָמָה, סְתָם עָפָר.

[[81:45]] [?81:45]

[[82:50]] [?82:50] גַּם רוּחָנִית וְגַם [?]. אַחֲרֵי שֶׁעָשָׂה אֶת הַמַּעֲשֶׂה הַזֶּה, הִרְגִּישׁ שֶׁאֵין לוֹ מָקוֹם. [?83:11] אָמַר: ⟦sefaria|הֵן גֵּרַשְׁתָּ אֹתִי הַיּוֹם מֵעַל פְּנֵי הָאֲדָמָה⟧{{=בראשית ד, יד}}. מָה פֵּירוּשׁ? הוּא הִרְגִּישׁ וְיָדַע שֶׁעַכְשָׁיו, אִם יִסְתּוֹבֵב בָּעוֹלָם, יַהַרְגוּ אוֹתוֹ. כְּשֶׁאָדָם הוֹפֵךְ לְבַעַל עֲבֵירָה, הוּא יוֹרֵד מַדְרֵגָתוֹ, וּמַתְחִיל לְפַחֵד [?]. וְלָכֵן כָּתוּב: ⟦sefaria|וַיָּשֶׂם ה׳ לְקַיִן אוֹת לְבִלְתִּי הַכּוֹת־אֹתוֹ כָּל־מֹצְאוֹ⟧{{=בראשית ד, טו}}. [?84:01] זוֹ הַתְּשׁוּבָה לְמָה שֶׁשָּׁאַלְתָּ.

[[84:15]] אוּלַי נְסַיֵּם בַּנְּקוּדָּה הַזֹּאת: [?84:30] הָאָרֶץ – נַנִּיחַ שֶׁאֶרֶץ יִשְׂרָאֵל אֵינָהּ מְתַפְקֶדֶת עַכְשָׁיו כְּמוֹ שֶׁ[?]; [?] ⟦sefaria|וְזָכַרְתִּי אֶת־בְּרִיתִי יַעֲקוֹב וְאַף אֶת־בְּרִיתִי יִצְחָק וְאַף אֶת־בְּרִיתִי אַבְרָהָם אֶזְכֹּר וְהָאָרֶץ אֶזְכֹּר⟧{{=ויקרא כו, מב}}. מָה עוֹשֶׂה הָאָרֶץ כָּאן? הָאָרֶץ נִיגֶּשֶׁת לְבוֹרֵא עוֹלָם [?]: מָה קוֹרֶה כָּאן? כְּלוֹמַר, [?] בְּחֶסֶד, וַעֲדַיִן הָאָרֶץ נוֹתֶנֶת לָנוּ פֵּירוֹת מְתוּקִים. בָּרוּךְ ה׳, הָאָרֶץ נוֹתֶנֶת לָנוּ פֵּירוֹת מְתוּקִים.

[[85:35]] מִי שֶׁיֵּשׁ לוֹ דַּעַת רוֹאֶה שְׁנֵי דְּבָרִים שֶׁהֵם מַמָּשׁ [?] זֶה אֶת זֶה בָּעָם הַיְּהוּדִי: מִצַּד אֶחָד חַיָּלִים שֶׁשְּׂמֵחִים [?86:06], וּמִצַּד שֵׁנִי – שֶׁאֵינָם רוֹאִים אֶת הַתּוֹרָה, אֵינָם רוֹאִים בִּכְלָל יְהוּדִי, אֵינָם רוֹאִים אֶת עַצְמָם. [?86:32]

[[86:55]] [?] זֶה הָאִיזּוּן הַזֶּה. שָׁם [?] אֵין אִיזּוּן – אֲדָמָה מֵתָה. מָה זֶה גְּבוּל? [?] וְהַתְּנָאִים שֶׁל הָאָרֶץ הַזֹּאת מִתְקַיְּמִים, לְמַעֲשֶׂה, עַכְשָׁיו גַּם בְּעַזָּה. כְּלוֹמַר, עַזָּה הִיא חֵלֶק מִן הָאָרֶץ שֶׁלָּנוּ; עַזָּה הִיא חֵלֶק מֵאֶרֶץ יִשְׂרָאֵל. אוֹתָם תְּנָאִים שֶׁאָמַרְנוּ לְגַבֵּי אֶרֶץ יִשְׂרָאֵל [?]. בִּשְׁבִיל זֶה יֵשׁ לָנוּ גְּבוּלוֹת שֶׁל אֶרֶץ יִשְׂרָאֵל – לֹא בִּשְׁבִיל [?] אֵיפֹה הַגְּבוּל. לְפִי דִּבְרֵי הָרַב הַקָּדוֹשׁ, הַמִּילָּה "גְּבוּל" מַתְאִימָה: הַגְּבוּל הָאֲמִיתִּי הוּא שֶׁהָאָרֶץ עַצְמָהּ, בְּתוֹךְ הַגְּבוּל, מוּגְבֶּלֶת – בְּנוּיָה בְּצוּרָה מוּגְבֶּלֶת. כָּל סֶנְטִימֶטֶר מִן הָאֲדָמָה הַזֹּאת הוּא גְּבוּל [?].

[[88:05]] [?88:05] לְהַעֲמִיק בַּדְּבָרִים הָאֵלֶּה. אַחַר כָּךְ הוּא נוֹגֵעַ בַּקֶּטַע שֶׁל הַקַּבָּלָה – זֶה בַּפַּעַם הַבָּאָה. ה׳ יְזַכֶּה אוֹתָנוּ לִלְמוֹד אֶת הָאָרֶץ שֶׁלָּנוּ, לִלְמוֹד אֶת גְּבוּלוֹת הָאָרֶץ, בַּתּוֹרָה הַקְּדוֹשָׁה. טוֹב, עַכְשָׁיו נִתְפַּלֵּל.
