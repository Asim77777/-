## פָּרָשַׁת קוֹרַח – תְּמוּנָה שֶׁל כָּל דּוֹר וָדוֹר

[[0:01]] אֶת פָּרָשַׁת קוֹרַח אִי אֶפְשָׁר לְהִתְעַלֵּם מִמֶּנָּה, מִפְּנֵי שֶׁהִיא מְשַׁקֶּפֶת; הִיא הַתְּמוּנָה שֶׁל מָה שֶׁעוֹבֵר עַל עַם יִשְׂרָאֵל בְּכָל דּוֹר וָדוֹר.

[[0:54]] הַקָּדוֹשׁ בָּרוּךְ הוּא, יִשְׁתַּבַּחַ שְׁמוֹ לְעַד – אֵיךְ הוּא סוֹבֵל אֶת הָעוֹלָם, וְאֵיךְ הוּא נוֹשֵׂא אֶת הָעוֹלָם! רַק הוּא. הוּא בָּרָא אֶת הָעוֹלָם וְאֶת כָּל מָה שֶׁיֵּשׁ בּוֹ, וְשֵׁשֶׁת אֲלָפִים שָׁנָה הָעוֹלָם [?1:45]. מִי שָׂם לֵב לַקָּדוֹשׁ בָּרוּךְ הוּא? מִי עוֹבֵד אוֹתוֹ? מִי מַכִּיר בּוֹ? מִי מְפַחֵד מִמֶּנּוּ? הַתְּשׁוּבָה הָעֲמוּקָּה בְּיוֹתֵר, וּמִצַּד שֵׁנִי הַפְּשׁוּטָה בְּיוֹתֵר: כִּי הוּא הַקָּדוֹשׁ בָּרוּךְ הוּא. זֶה הַכּוֹל.

## הַפַּחַד הַסָּמוּי שֶׁהִטְבִּיעַ הַבּוֹרֵא בַּבְּרִיאָה

[[2:45]] וּמִצַּד שֵׁנִי, כָּל הָעוֹלָם רוֹעֵד מִן הַקָּדוֹשׁ בָּרוּךְ הוּא, כָּל הָעוֹלָם מְפַחֵד מִמֶּנּוּ. אֵין בְּרִיָּה בָּעוֹלָם שֶׁאֵינָהּ כְּנוּעָה לְבוֹרֵא עוֹלָם. אֶלָּא שֶׁיֵּשׁ פַּחַד גָּלוּי וְיֵשׁ פַּחַד סָמוּי. זֶה שֶׁבְּנֵי אָדָם אֵינָם קָמִים יוֹם אֶחָד, אוֹ בְּכָל יוֹם, וְהוֹרְגִים זֶה אֶת זֶה, שֶׁאֵינָם שׁוֹבְרִים אֶת הַחֲנוּיוֹת וְגוֹזְלִים כָּל מָה שֶׁיֵּשׁ, שֶׁאֵינָם מַחֲרִיבִים אֶת הָעוֹלָם – זֶהוּ פַּחַד סָמוּי שֶׁהִטְבִּיעַ בּוֹרֵא עוֹלָם בָּאָדָם, וַאֲפִילּוּ בְּחַיּוּת. אֲנַחְנוּ אוּלַי לֹא שָׂמִים לֵב לָזֶה.

[[4:15]] דָּוִד הַמֶּלֶךְ אוֹמֵר לָנוּ בְּמִזְמוֹרָיו דְּבָרִים יָפִים כָּל כָּךְ, וַאֲנַחְנוּ לֹא שָׂמִים לֵב: ⟦sefaria|הַלְלוּ־יָהּ הַלְלוּ אֶת־ה׳ מִן־הַשָּׁמַיִם הַלְלוּהוּ בַּמְּרוֹמִים הַלְלוּהוּ כָל־מַלְאָכָיו הַלְלוּהוּ כָּל־צְבָאָו הַלְלוּהוּ שֶׁמֶשׁ וְיָרֵחַ הַלְלוּהוּ כָּל־כּוֹכְבֵי אוֹר הַלְלוּהוּ שְׁמֵי הַשָּׁמָיִם⟧{{=תהילים קמח, א–ד}}, ⟦sefaria|הַלְלוּ אֶת־ה׳ מִן־הָאָרֶץ תַּנִּינִים וְכָל־תְּהֹמוֹת⟧{{=תהילים קמח, ז}}. אֲנַחְנוּ חוֹשְׁבִים שֶׁאֵלֶּה הִלּוּלִים וְתִשְׁבָּחוֹת לְבוֹרֵא עוֹלָם, וְאֵינֶנּוּ יוֹדְעִים שֶׁכַּוָּנַת דָּוִד הַמֶּלֶךְ עָלָיו הַשָּׁלוֹם הִיא זוֹ: אִם אֲנַחְנוּ רוֹאִים שֶׁהַחַיּוֹת אֵינָן מַגִּיעוֹת לַכְּבִישׁ הָרָאשִׁי, לַכִּיכָּר שֶׁל נְתַנְיָה, שֶׁאִישׁ אֵינוֹ הוֹרֵג אֶת רֵעֵהוּ וְשֶׁאֵין מַחֲרִיבִים אֶת הָעוֹלָם – הוּא שָׂם גְּבוּל לַיָּם. וְלֹא רַק לַיָּם שָׁם גְּבוּל: כְּשֶׁבָּרָא בּוֹרֵא עוֹלָם אֶת הָעוֹלָם, הוּא שָׂם בּוֹ סוֹכְנִים סְמוּיִים, שֶׁמַּחֲזִיקִים אֶת הָעוֹלָם שֶׁלֹּא יִתְמוֹטֵט, שֶׁאִישׁ לֹא יַהֲרוֹג אֶת רֵעֵהוּ וְשֶׁאִישׁ לֹא יִגְזוֹל אֶת רֵעֵהוּ.

[[6:20]] הַמִּשְׁטָרָה? הַאִם הַמִּשְׁטָרָה יְכוֹלָה עַל מֵאָה אֶלֶף אִישׁ? אִם מָחָר יַחֲלִיטוּ מֵאָה אֶלֶף אִישׁ בִּנְתַנְיָה לִשְׁבּוֹר אֶת כָּל חַלּוֹנוֹת הָרַאֲוָה – תּוּכַל הַמִּשְׁטָרָה עֲלֵיהֶם? לֹא תּוּכַל. מֵאָה אֶלֶף אִישׁ יָקוּמוּ וִיפוֹצְצוּ אֶת כָּל [?6:45] בָּאָרֶץ – מִי יוּכַל עֲלֵיהֶם? הַמִּשְׁטָרָה? אֶלָּא שֶׁכַּנִּרְאֶה, בְּלִי שֶׁאֲנַחְנוּ יוֹדְעִים, הַקָּדוֹשׁ בָּרוּךְ הוּא הִטְבִּיעַ בָּאָדָם – בַּקּוּפְסָה הַזֹּאת שֶׁנִּקְרֵאת נְשָׁמָה – פַּחַד, שֶׁתִּהְיֶה מְאוּזֶּנֶת, שֶׁתִּשְׁלוֹט בְּעַצְמָהּ, שֶׁאָדָם לֹא יַהֲרוֹג אֶת עַצְמוֹ. וְכַמּוּבָן גַּם הַטֶּבַע, הַשֶּׁמֶשׁ וְהַיָּרֵחַ.

[[8:04]] יֵשׁ פַּחַד מִפְּנֵי הַבּוֹרֵא יִתְבָּרַךְ, פַּחַד שֶׁמּוּטְבָּע בַּבְּרִיאָה: בְּעֵפֶר הָאֲדָמָה, בְּעוֹף הַשָּׁמַיִם, בַּשֶּׁמֶשׁ, בַּיָּרֵחַ וּבָאָדָם. הָאָדָם יָכוֹל לְחַדֵּשׁ חִידּוּשִׁים גְּדוֹלִים בְּטֶכְנוֹלוֹגְיָה, בִּתְעוּפָה, בִּרְפוּאָה, בְּכִימְיָה – בְּכוֹל; אֲבָל יְסוֹדוֹת הַבְּרִיאָה לֹא הִשְׁתַּנּוּ. כַּמָּה שֶׁמְּנַסִּים לַהֲפוֹךְ אִישָּׁה לְאִישׁ וְאִישׁ לְאִישָּׁה – הַטּוֹב וְהָרַע, שֶׁתְּלוּיִים בַּבְּחִירָה, לֹא הִשְׁתַּנּוּ. הֵם נִפְגְּמוּ, הִתְקַלְקְלוּ וְנִתְמַעֲטוּ, אֲבָל לֹא נֶעֶקְרוּ.

## מֶלֶךְ שֶׁאֵינוֹ צָרִיךְ לְהוֹפִיעַ

[[9:31]] יִשְׁתַּבַּחַ שְׁמוֹ לָעַד, שֶׁהוּא שׁוֹלֵט בַּבְּרִיאָה הַזֹּאת שְׁלִיטָה מְלֵאָה, בְּלִי שֶׁאִישׁ יַרְגִּישׁ שֶׁהוּא כָּאן. וּמִן הַצַּד הַשֵּׁנִי יֵשׁ בֶּאֱמֶת שְׁאֵלָה: אֵיזֶה מֶלֶךְ בָּעוֹלָם בָּרָא בְּרִיאָה אַדִּירָה וְנוֹרָאָה כָּזֹאת, וְאֵינוֹ טוֹרֵחַ לְהוֹפִיעַ אֲפִילּוּ פַּעַם בְּשָׁנָה וְלוֹמַר "אֲנִי ה׳ אֱלֹהֵיכֶם, אֲנִי כָּאן"? לָמָּה? כִּי הוּא בָּטוּחַ בְּעַצְמוֹ. לֹא יִהְיוּ כָּאן הַפְגָּנוֹת נֶגֶד הַקָּדוֹשׁ בָּרוּךְ הוּא – לְעוֹלָם לֹא יִהְיוּ הַפְגָּנוֹת נֶגֶד הַקָּדוֹשׁ בָּרוּךְ הוּא. וַאֲנַחְנוּ רוֹאִים שֶׁבִּזְמַן שֶׁעוֹבְרִים אֶת הַקַּו הָאָדוֹם – בָּא הַמַּבּוּל, בָּאָה הַפְלָגָה, בָּאוּ מַכּוֹת מִצְרַיִם, בָּא הַצּוּנָאמִי.

## "⟦~|ה׳ הוּא הָאֱלֹהִים⟧"

[[11:03]] אֲנִי אוֹמֵר אֶת הַדְּבָרִים הָאֵלֶּה, רַבּוֹתַי, כִּי אֲנַחְנוּ מִתְפַּלְּלִים עַל הַגְּאוּלָּה. אֲנַחְנוּ, יוֹשְׁבֵי אֶרֶץ יִשְׂרָאֵל, כְּבָר יוֹדְעִים – כָּל אֶחָד כְּבָר יוֹדֵעַ – שֶׁשּׁוּם כּוֹחַ בָּעוֹלָם לֹא יַעֲמוֹד לָנוּ, לְאֶרֶץ יִשְׂרָאֵל וּלְעַם יִשְׂרָאֵל, מִלְּבַד בּוֹרֵא עוֹלָם. אֲנַחְנוּ רוֹאִים אֶת הָאָדָם שֶׁהוּא, כִּבְיָכוֹל, הֶחָזָק בְּיוֹתֵר, שֶׁחָזַר עַכְשָׁיו לְבֵיתוֹ וְחוֹשֵׁב שֶׁהוּא מְנַהֵל אֶת הָעוֹלָם – וְהוּא כְּבָר מְגוּחָךְ; אִירָאן כְּבָר מִתְעַלֶּלֶת בּוֹ וּמְשַׂחֶקֶת בּוֹ.

[[12:25]] לָמָּה, לָמָּה עַם יִשְׂרָאֵל אֵינוֹ מִתְעוֹרֵר לְהָבִין אֶת הַשּׁוֹרֶשׁ וְאֶת הַיְּסוֹד הַזֶּה – רַק דָּבָר אֶחָד: ⟦sefaria|ה׳ הוּא הָאֱלֹהִים ה׳ הוּא הָאֱלֹהִים⟧{{=מלכים א יח, לט}}? זֶה מָה שֶׁבִּיקֵּשׁ אֵלִיָּהוּ הַנָּבִיא. הוּא לֹא בִּיקֵּשׁ מִכּוּלָּם טַלִּית קָטָן, וְלֹא בִּיקֵּשׁ שֶׁיֵּלְכוּ לַמִּקְוֶה. דָּבָר אֶחָד: אַל תֹּאמְרוּ לִי שֶׁהָעֲבוֹדָה הַזָּרָה בָּרְאָה אֶת הָעוֹלָם. רַק זֶה. זֶה כָּל מָה שֶׁהִשִּׂיג בַּמַּעֲמָד הַנּוֹרָא הַהוּא, שֶׁאָמַר לָהֶם: ⟦sefaria|וְאִם־הַבַּעַל לְכוּ אַחֲרָיו⟧{{=מלכים א יח, כא}}.

[[13:33]] מִי נוֹתֵן לוֹ רְשׁוּת לוֹמַר דָּבָר כָּזֶה? הוֹרָאַת שָׁעָה. הוּא חָכָם, הוּא מוֹרֶה הוֹרָאָה. לִפְעָמִים אֲנָשִׁים שֶׁאֵינָם עוֹד גּוּף אֶלָּא כּוּלָּם נְשָׁמָה מְקַבְּלִים עֲלֵיהֶם אֶת הַסִּיכּוּן הַזֶּה: אִם הַבַּעַל – לְכוּ אַחֲרָיו! לְכוּ, לְכוּ מִכָּאן, אֵינֶנִּי צָרִיךְ אֶתְכֶם. זֶה הַכּוֹל. מָה אַתָּה אוֹמֵר – אַתָּה יְהוּדִי? יְהוּדִי עוֹבֵד עֲבוֹדָה זָרָה? יְהוּדִי עוֹבֵד עֲבוֹדָה זָרָה הוּא כְּמוֹ יְהוּדִי בְּלִי כִּיפָּה, בְּלִי תְּפִילִּין, בְּלִי שַׁבָּת. מָה הַמִּשְׂחָק הַזֶּה?

## מֹשֶׁה רַבֵּנוּ בַּמִּבְחָן הַגָּדוֹל וְהַנּוֹרָא

[[15:15]] וַאֲנַחְנוּ רוֹאִים אֶת מֹשֶׁה רַבֵּנוּ בַּמִּבְחָן הַגָּדוֹל וְהַנּוֹרָא, שֶׁבּוֹ הֶעֱמִיד אוֹתוֹ אוֹתוֹ אָדָם, קוֹרַח – מִבְחָן נוֹרָא וְאָיוֹם.

[[15:45]] יֵשׁ עַכְשָׁיו עָלוֹן שֶׁקּוֹרְאִים לוֹ "עֲרוּץ 2000" – עַל שֵׁם הַתִּיקִים שֶׁל נְתַנְיָהוּ. לֹא יוֹדֵעַ אִם רְאִיתֶם אֶת הַתְּמוּנָה בַּשַּׁעַר שֶׁל הֶעָלוֹן הַזֶּה: שׁוּרָה שֶׁל מָאתַיִם וַחֲמִישִּׁים אִישׁ עִם הַמַּחֲתוֹת, וְאַהֲרֹן הַכֹּהֵן עִם מַחְתָּתוֹ, וְהוּא שָׂם בָּהּ אֵשׁ. זוֹ תְּמוּנָה מְעַלֶּפֶת – אֲנִי מִשְׁתַּחֲוֶה לָהֶם. בָּרוּךְ ה׳, פִּרְסוּם בְּיָמֵינוּ. תְּמוּנָה נִפְלָאָה. אַל תִּסְתַּכֵּל לְבַד – הַרְאָה לְכוּלָּם!

[[17:28]] זוֹ הַתְּמוּנָה, רַבּוֹתַי. תְּמוּנָה שֶׁמְּשַׁקֶּפֶת אֶת הַכּוֹל. מָאתַיִם וַחֲמִישִּׁים – לֹא מַפְגִּינֵי קַפְּלָן. מָאתַיִם וַחֲמִישִּׁים יְהוּדִים, וְלֹא יְהוּדִים פְּשׁוּטִים – מָאתַיִם וַחֲמִישִּׁים רָאשֵׁי הָעֵדָה. וְאַל תַּחְשְׁבוּ שֶׁהֵם מִן הַלִּיכּוּד, אוֹ מִן הַמִּפְלָגָה שֶׁל בֵּנֶט, אוֹ שֶׁל אִיזֶנְקוּט – עוֹד מְעַט יַכְנִיסוּ לָנוּ אֶת כָּל הַשֵּׁמוֹת הָאֵלֶּה לָרֹאשׁ, וְהֵם מַצְלִיחִים. אֲבָל אֵלֶּה מָאתַיִם וַחֲמִישִּׁים רָאשֵׁי סַנְהֶדְרָאוֹת. סַנְהֶדְרִין – אֵלֶּה שִׁבְעִים וְאֶחָד גְּדוֹלֵי הָעָם; וּמָאתַיִם וַחֲמִישִּׁים רָאשֵׁי סַנְהֶדְרָאוֹת! מִי יָכוֹל לַעֲמוֹד בִּתְמוּנָה כָּזֹאת?

[[19:35]] וּמָה הֵם רָצוּ? הֵם עַצְמָם לֹא יָדְעוּ. הָאֱמֶת הִיא שֶׁהָיָה לָהֶם קוֹרַח, וְהוּא הוֹבִיל אוֹתָם.

## "מֹשֶׁה רַבֵּנוּ שַׁקְרָן" – הוֹדָעָה מְעַנְיֶנֶת

[[19:50]] הֶעָרָה קְטַנָּה: אִם מְדַבְּרִים עַל יְהוּדִי מִן הַשְּׁכוּנָה שֶׁגָּנַב – אִישׁ אֵינוֹ שָׂם לֵב. אֲבָל אִם מְדַבְּרִים עַל מֹשֶׁה רַבֵּנוּ שֶׁהוּא שַׁקְרָן, חָס וְשָׁלוֹם – זֶה מִתְקַבֵּל בְּהִתְעַנְיְנוּת מְרַבִּית, בְּהִתְעַנְיְנוּת אַדִּירָה. כָּל "וַיְדַבֵּר ה׳ אֶל מֹשֶׁה לֵאמַר" – "לֹא ה׳ אָמַר לוֹ; הוּא אָמַר מֵעַצְמוֹ". הוֹדָעָה כָּזֹאת מִתְפַּרְסֶמֶת בָּעִיתּוֹן וּבַטֶּלֶוִיזְיָה, ה׳ יִשְׁמוֹר.

[[21:00]] זוֹ הַתְּמוּנָה שֶׁל קוֹרַח. הוּא לֹא בָּא לוֹמַר: נַעֲשֶׂה בְּחִירוֹת, וְאוּלַי אֶבְחַר אֲנִי לִהְיוֹת מַנְהִיג לֵגִיטִימִי. בְּסֵדֶר, אֵין בְּעָיָה. אֲבָל לוֹמַר שֶׁכָּל מָה שֶׁעָשָׂה מֹשֶׁה רַבֵּנוּ עַד הַיּוֹם – שֶׁקֶר?!

[[21:40]] רַבּוֹתַי, אִם כָּל עַם יִשְׂרָאֵל הָלְכוּ אַחֲרֵי קוֹרַח, זֶה רַק מִשּׁוּם שֶׁזֶּה מְעַנְיֵן. אִי אֶפְשָׁר לַחֲשׁוֹב שֶׁבְּעַם יִשְׂרָאֵל סָבְרוּ בֶּאֱמֶת שֶׁמֵּשֶׁה רַבֵּנוּ שַׁקְרָן – לֹא. אֲבָל זֶה מְעַנְיֵן; זוֹ הוֹדָעָה מְעַנְיֶנֶת. וּבָזֶה אוּלַי אֵינֶנּוּ שָׂמִים לֵב לַפָּסוּק ⟦sefaria|וַיַּקְהֵל עֲלֵיהֶם קֹרַח אֶת־כָּל־הָעֵדָה⟧{{=במדבר טז, יט}}. מָה זֶה "⟦~|כָּל־הָעֵדָה⟧"? לֹא אִישׁ מֵרְחוֹב [?22:15] – כּוּלָּם. כְּשֶׁהָיְתָה הַהַפְגָּנָה שָׁם, זוֹ לֹא הָיְתָה שְׁכוּנָה אַחַת בְּתֵל אָבִיב; כּוּלָּם הָיוּ בְּמַחֲנֶה אֶחָד. לֹא קָשֶׁה לְהָבִין: רַק מַדְבִּיקִים מַשֶּׁהוּ כָּזֶה – וְרוֹאִים.

## כָּאן הִתְגַּלָּה מֹשֶׁה רַבֵּנוּ בִּמְלוֹא הֲדָרוֹ

[[22:34]] וְכָאן הִתְגַּלָּה מֹשֶׁה רַבֵּנוּ שֶׁלָּנוּ – בִּמְלוֹא הֲדָרוֹ. לֹא בְּכוֹחוֹ, לֹא בְּעוֹצְמָתוֹ וְלֹא בְּגַדְלוּתוֹ. זֶה לֹא מֹשֶׁה שֶׁאָמַר בְּחֵטְא הָעֵגֶל ⟦sefaria|אָנָּא חָטָא הָעָם הַזֶּה ... וְאִם־אַיִן מְחֵנִי נָא מִסִּפְרְךָ⟧{{=שמות לב, לא–לב}}; זֶה מֹשֶׁה שֶׁנִּכְנַס לְנִיסָּיוֹן וּלְסִיכּוּן גְּדוֹלִים מִן הַסִּיכּוּן שֶׁקּוֹרַח נִיסָּה לְהַכְנִיס אוֹתוֹ אֵלָיו. קוֹרַח אָמַר שֶׁמִּשֶׁה רַבֵּנוּ שַׁקְרָן – זֶה קוֹרַח. אֲבָל כָּאן נִכְנַס מֹשֶׁה רַבֵּנוּ לְסִיכּוּן שֶׁל מַהְפֵּכָה – מַהְפֵּכָה בַּבְּרִיאָה כּוּלָּהּ.

[[23:58]] שֶׁהוּא אָמַר: ⟦sefaria|אִם־כְּמוֹת כָּל־הָאָדָם יְמֻתוּן אֵלֶּה ... לֹא ה׳ שְׁלָחָנִי וְאִם־בְּרִיאָה יִבְרָא ה׳ וּפָצְתָה הָאֲדָמָה אֶת־פִּיהָ וּבָלְעָה אֹתָם וְאֶת־כָּל־אֲשֶׁר לָהֶם⟧{{=במדבר טז, כט–ל}}. רַבּוֹתַי, אִילּוּ לֹא שָׁמַע ה׳ בְּקוֹלוֹ שֶׁל מֹשֶׁה רַבֵּנוּ – הָיָה זֶה סוֹף, חוּרְבַּן הָעוֹלָם. חוּרְבַּן הָעוֹלָם, שֶׁנָּבִין הֵיטֵב.

## "⟦~|סָלַחְתִּי כִּדְבָרֶךָ⟧"

[[24:55]] כְּשֶׁה׳ אוֹמֵר לְמֹשֶׁה ⟦sefaria|וַאֲכַלֶּה אֹתָם כְּרָגַע⟧{{=במדבר טז, כא}} – הַאִם הַקָּדוֹשׁ בָּרוּךְ הוּא, חָס וְשָׁלוֹם, מְאַיֵּם? לֹא. וְהוּא אוֹמֵר זֹאת לֹא פַּעַם אַחַת; בַּפָּרָשָׁה הַזֹּאת אֲנַחְנוּ רוֹאִים שָׁלוֹשׁ אוֹ אַרְבַּע פְּעָמִים שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא אוֹמֵר כָּךְ. וּמִצַּד שֵׁנִי – פֶּלֶא פְּלָאִים: אוֹתוֹ הַקָּדוֹשׁ בָּרוּךְ הוּא אוֹמֵר לְמֹשֶׁה "סָלַחְתִּי". וְלֹא "סָלַחְתִּי" – סוֹף פָּסוּק, אֶלָּא ⟦sefaria|סָלַחְתִּי כִּדְבָרֶךָ⟧{{=במדבר יד, כ}}. אֲפִילּוּ פֶּסֶל [?25:50]. אֵיפֹה גַּדְלוּתוֹ שֶׁל בּוֹרֵא עוֹלָם, שֶׁאוֹמֵר לְמֹשֶׁה "⟦~|סָלַחְתִּי כִּדְבָרֶךָ⟧" – כְּלוֹמַר, אֲנִי שׁוֹמֵעַ אֶת דְּבָרֶיךָ, אֶת מָה שֶׁאָמַרְתָּ לִי?

[[26:15]] וּמָה אָמַר לוֹ מֹשֶׁה רַבֵּנוּ? אָמַר לוֹ: הַקָּדוֹשׁ בָּרוּךְ הוּא, אִם עַכְשָׁיו תְּכַלֶּה אוֹתָם, מָה יַגִּידוּ הַמִּצְרִים? תַּקְשִׁיבוּ: חוֹשְׁבִים שֶׁזּוֹ טַעֲנָה שֶׁל [?26:40] – מַמָּשׁ לֹא. אֲנִי אוֹמֵר אֶת הַפָּסוּק כְּמוֹ שֶׁהוּא: ⟦sefaria|וְשָׁמְעוּ מִצְרַיִם כִּי־הֶעֱלִיתָ בְכֹחֲךָ אֶת־הָעָם הַזֶּה מִקִּרְבּוֹ⟧{{=במדבר יד, יג}}. אַתָּה עָשִׂיתָ לָהֶם נִיסִּים וְנִפְלָאוֹת, וְעַכְשָׁיו, כְּשֶׁבְּנֵי יִשְׂרָאֵל עוֹמְדִים בְּשַׁעַר אֶרֶץ כְּנַעַן, יֹאמְרוּ שֶׁאֵינְךָ יָכוֹל עַל הָעַמִּים הַיּוֹשְׁבִים בָּהּ. אֵינְךָ יָכוֹל, אֵינְךָ יָכוֹל – כְּמוֹ שֶׁאוֹמְרִים יְלָדִים: "אַתָּה לֹא יָכוֹל עָלָיו!"

[[27:40]] פַּעַם הָיָה יְהוּדִי בַּשְּׁכוּנָה שֶׁהָיָה מַשְׁוִיץ כָּל הַזְּמַן עַל כּוּלָּם: אֲנִי כָּךְ, אֲנִי כָּךְ. פַּעַם אַחַת עָבַר שָׁם בָּחוּר חָסוֹן, שָׁמַע אוֹתוֹ, בָּא וְהֶרְאָה לוֹ. אָמַר לוֹ: אַתָּה לֹא יָכוֹל! [?28:10] – "וְאָמְרוּ מִצְרַיִם" שֶׁאֵינְךָ יָכוֹל עַל הָעַמִּים הַהֵם. וּמָה אוֹמֵר לוֹ ה׳? "⟦~|סָלַחְתִּי כִּדְבָרֶךָ⟧" – אַתָּה צוֹדֵק.

[[28:28]] זֶה מַפִּיל. וְכֵן, מֹשֶׁה רַבֵּנוּ יָצָא מִן הַזַּעֲזוּעַ הַזֶּה. יָצָא מֹשֶׁה רַבֵּנוּ – "מֹשֶׁה אֱמֶת וְתוֹרָתוֹ אֱמֶת", כְּמוֹ שֶׁאוֹמְרִים קוֹרַח וַעֲדָתוֹ בְּכָל רֶגַע מִן הַגֵּיהִנּוֹם{{=בבא בתרא עד עא}}.

## הַדֶּרֶךְ שֶׁל מַנְהִיג – לְפִי אוֹר הַחַיִּים

[[29:15]] וְאָז אֲנַחְנוּ רוֹאִים, רֵאשִׁית, אֶת דַּרְכּוֹ הַמַּעֲשִׂית שֶׁל מֹשֶׁה רַבֵּנוּ: אֵיךְ מַנְהִיג עוֹמֵד בְּזַעֲזוּעַ כָּזֶה וְשׂוֹרֵד – וְשׂוֹרֵד בְּגָאוֹן – זַעֲזוּעַ שֶׁחַס וְשָׁלוֹם הָיָה יָכוֹל לְהִסְתַּיֵּם בְּסוֹפוֹ שֶׁל אוֹתוֹ עַם, שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא עָמַל עָלָיו כָּל כָּךְ.

[[30:03]] הָאוֹר הַחַיִּים מְסַדֵּר אֶת הָעִנְיָן כָּךְ{{=אור החיים במדבר טז, ד}}: מֹשֶׁה רַבֵּנוּ מַעֲמִיד אֶת קוֹרַח בִּמְקוֹמוֹ – פֶּלֶא פְּלָאִים. אֵיךְ הֶעֱמִיד אוֹתוֹ? תַּקְשִׁיבוּ לִי הֵיטֵב: אִילּוּ הָיָה לְמֹשֶׁה רַבֵּנוּ כֶּתֶם קָטָן עַל חוּלְצָתוֹ – לֹא הָיָה יָכוֹל לַעֲמוֹד בַּנִּיסָּיוֹן. אִילּוּ הָיָה לוֹ פְּגָם קָטָן בָּעֲנָוָה שֶׁלּוֹ – נִגְמַר.

[[30:55]] בּוֹאוּ נְגַלֶּה אֶת הַדְּבָרִים. בָּא קוֹרַח וְאוֹמֵר לְמֹשֶׁה רַבֵּנוּ: ⟦sefaria|כָל־הָעֵדָה כֻּלָּם קְדֹשִׁים וּבְתוֹכָם ה׳ וּמַדּוּעַ תִּתְנַשְּׂאוּ עַל־קְהַל ה׳⟧{{=במדבר טז, ג}} – "⟦~|רַב־לָכֶם⟧", לְקַחְתֶּם לְעַצְמְכֶם גְּדוֹלָה יוֹתֵר מִדַּי, אַתֶּם מִתְנַשְּׂאִים. מֶה הָיָה אָדָם, וַאֲפִילּוּ צַדִּיק גָּדוֹל, עוֹנֶה בְּמַצָּב כָּזֶה? אֲנִי מִתְנַשֵּׂא? אֵינְךָ רוֹאֶה אֵיפֹה אֲנִי גָּר? בּוֹא תִּרְאֶה אֶת הָאֹהֶל שֶׁלִּי, בּוֹא תִּרְאֶה אֶת הִתְנַהֲגוּתִי! אֲפִילּוּ אִישָּׁה אֵין לִי – פָּרַשְׁתִּי מִן הָאִישָּׁה. הַיְּלָדִים שֶׁלִּי אֲפִילּוּ אֵינָם אִיתִּי; יְלָדַי בִּכְלָל אֵינָם הוֹלְכִים בְּדַרְכִּי – יְלָדָיו שֶׁל מֹשֶׁה רַבֵּנוּ, לְצַעֲרֵנוּ הָרַב, כָּךְ הָיוּ; מוּתָּר לִי לוֹמַר זֹאת, זֶה הַמַּצָּב. אֲנִי לְבַד, אֲנִי זָקֵן – בִּזְמַן שֶׁמְּדַבְּרִים עָלָיו מֹשֶׁה כְּבָר קָרוֹב לְמֵאָה שָׁנָה, אִם לֹא יוֹתֵר – וַאֲנִי מִתְנַשֵּׂא?!

[[32:35]] וּמֹשֶׁה רַבֵּנוּ הִתְחִיל מְעַט בְּכָךְ: ⟦sefaria|לֹא חֲמוֹר אֶחָד מֵהֶם נָשָׂאתִי⟧{{=במדבר טז, טו}}. אֲבָל אֵיזֶה וִיכּוּחַ הָיָה מִתְעוֹרֵר? וִיכּוּחַ אִישִׁי: מִי אַתָּה בִּכְלָל, קוֹרַח? מִי אַתָּה שֶׁאַתָּה מְדַבֵּר? אֵינְךָ מִתְבַּיֵּשׁ? כָּךְ מְדַבְּרִים? וְהָיָה מִתְנַהֵל וִיכּוּחַ אִישִׁי – וּבָזֶה גָּמַרְנוּ; מֹשֶׁה רַבֵּנוּ הָיָה נוֹפֵל. כִּי בְּשָׁעָה שֶׁמִּשֶׁה מְנַהֵל וִיכּוּחַ אִישִׁי, זֶה כְּבָר "קוֹרַח וּמֹשֶׁה מִתְוַכְּחִים". אֲפִילּוּ אִם מֹשֶׁה צוֹדֵק, אוֹמְרִים: נָכוֹן, אֲבָל בְּכָל זֹאת...

## "⟦~|וַיִּשְׁמַע מֹשֶׁה וַיִּפֹּל עַל־פָּנָיו⟧"

[[33:25]] הָאוֹר הַחַיִּים הַקָּדוֹשׁ אוֹמֵר שֶׁמֵּשֶׁה רַבֵּנוּ סִידֵּר אֶת הָעִנְיָן יָפֶה. מָה שֶׁנּוֹגֵעַ אֵלַי – שֶׁאַתָּה אוֹמֵר שֶׁאֲנִי מִתְנַשֵּׂא – ⟦sefaria|וַיִּשְׁמַע מֹשֶׁה וַיִּפֹּל עַל־פָּנָיו⟧{{=במדבר טז, ד}}. שׁוּם מִילָּה, שׁוּם מִילָּה אִישִׁית. אֵינֶנִּי כָּאן, אֵינֶנִּי קַיָּם. נְפִילַת הַפָּנִים אֵינָהּ יֵאוּשׁ; אֵינָהּ תְּנוּעָה שֶׁל פַּחַד אוֹ שֶׁל הַצָּגָה. מֹשֶׁה אוֹמֵר לְקוֹרַח: אִם נִרְאֶה בְּעֵינֶיךָ שֶׁאֲנִי מִתְנַשֵּׂא – אֵינֶנִּי קַיָּם. אֲנִי עַל הַפָּנִים, בֶּאֱמֶת.

[[34:20]] וְאַתֶּם יוֹדְעִים, רַבּוֹתַי, מָה הָיְתָה עַנְוָתוֹ שֶׁל מֹשֶׁה רַבֵּנוּ? זֶה אוּלַי הַזְּמַן לוֹמַר מָה גַּדְלוּתוֹ שֶׁל הָאִישׁ הַזֶּה בַּעֲנָוָה. מֵאַיִן בָּאָה לוֹ הָעֲנָוָה הַזֹּאת? לֹא רָאִינוּ אָדָם אֶחָד בְּכָל הַהִיסְטוֹרְיָה שֶׁהָיָה עָנָיו בְּמַדְרֵגָה כָּזֹאת. הִסְבִּירוּ לִי אַתֶּם: אֵיךְ יִתָּכֵן?

[[35:05]] [מן הקהל:] נִרְאֶה לִי שֶׁמֵּשֶׁה הָיָה בַּמַּדְרֵגָה שֶׁבָּהּ הִכִּיר אֶת ה׳ כָּל כָּךְ – כְּמוֹ שֶׁאוֹמֶרֶת הַתּוֹרָה, ⟦sefaria|פֶּה אֶל־פֶּה אֲדַבֶּר־בּוֹ⟧{{=במדבר יב, ח}} – וּמִי שֶׁמַּכִּיר אֶת ה׳ מֵבִין שֶׁאֵין לוֹ שׁוּם חֲשִׁיבוּת בְּעַצְמוֹ.

[[35:30]] יָפֶה מְאוֹד. אֲבָל אֲנַחְנוּ מְדַבְּרִים בְּאוֹפֶן מַעֲשִׂי: הַכּוֹל טוֹב, הוּא אָדָם טוֹב, נָעִים, צַדִּיק – כְּכָל שֶׁתִּרְצֶה. אֲבָל לְהַגִּיעַ לָעֲנָוָה כָּזֹאת – אֵיךְ?

[[36:03]] [מן הקהל:] אֵיךְ אֶפְשָׁר שֶׁלֹּא לִהְיוֹת עָנָיו, כְּשֶׁאַתָּה כָּל הַזְּמַן בְּקֶשֶׁר עִם מִי שֶׁהוּא גָּדוֹל, גָּדוֹל, גָּדוֹל?

[[36:15]] זוֹ אוֹתָהּ תְּשׁוּבָה. נָכוֹן; שְׁנֵיכֶם נִיבֵּאתֶם בְּסִגְנוֹן אֶחָד. הֲבִיאוּ לִי בְּבַקָּשָׁה חוֹמֶשׁ בַּמִּדְבָּר, וְתַקְשִׁיבוּ הֵיטֵב – אֵינֶנִּי מְעַיֵּף אֶתְכֶם – לַתְּשׁוּבָה שֶׁל הַתַּרְגּוּם.

## "⟦~|וְהָאִישׁ מֹשֶׁה עָנָו מְאֹד⟧" – תַּרְגּוּם יוֹנָתָן בֶּן עֻזִּיאֵל

[[37:10]] ⟦sefaria|וְהָאִישׁ מֹשֶׁה עָנָו מְאֹד מִכֹּל הָאָדָם אֲשֶׁר עַל־פְּנֵי הָאֲדָמָה⟧{{=במדבר יב, ג}}. זֶה הַחוֹמֶשׁ, הַפְּשָׁט. אֲבָל הַשְּׁאֵלָה בִּמְקוֹמָהּ: אֵיךְ יָכוֹל אָדָם לִהְיוֹת עָנָיו כָּל כָּךְ?

[[37:30]] [מן הקהל:] מֹשֶׁה לֹא בִּיקֵּשׁ שׁוּם טוֹבָה מֵאַף אֶחָד – לָכֵן הָיָה עָנָיו.

[[37:55]] לֹא; אֲנַחְנוּ שׁוֹאֲלִים אֵיךְ יָכוֹל אָדָם לְהַגִּיעַ לָעֲנָוָה כָּזֹאת. בּוֹאוּ נִרְאֶה מָה הַתַּרְגּוּם אוֹמֵר.

[[38:20]] (צָרִיךְ לָשִׁיר. צָרִיךְ לִלְמוֹד לָשִׁיר, וְלָשִׁיר וְלָשִׁיר – לְשַׁבֵּחַ וּלְהַלֵּל אֶת בּוֹרֵא עוֹלָם כָּל הַזְּמַן. לָשִׁיר כָּל הַזְּמַן, וְאָז לֹא לְטַפֵּל בַּבְּעָיוֹת, לֹא לַחֲשׁוֹב וְלֹא לִדְאוֹג – כְּלוּם. לָשִׁיר וְלָשִׁיר. בָּא לְךָ הַיֵּצֶר הָרַע? [?39:20] – וְתָעִיף אוֹתוֹ.)

[[39:32]] "⟦~|וְהָאִישׁ מֹשֶׁה⟧" – יֵשׁ כָּאן מִקְרָאוֹת גְּדוֹלוֹת עִם תַּרְגּוּם יוֹנָתָן בֶּן עוזִּיאֵל? אוֹנְקְלוּס? לֹא – תַּרְגּוּם יוֹנָתָן בֶּן עוזִּיאֵל. יֵשׁ! אַתֶּם רוֹאִים? דַּוְקָא הַחוּמֶשׁ הַקָּטָן – הוּא הָעַנָיו.

## הָרַדְיוֹ הַקָּטָן וְהָרַדְיוֹ הַגָּדוֹל

[[40:30]] פַּעַם, בְּמִלְחֶמֶת שֵׁשֶׁת הַיָּמִים, הָיוּ כָּאן עוֹלִים חֲדָשִׁים, וְכָל הָעָם הָיָה דָּבוּק לָרַדְיוֹ – הָרַדְיוֹ מוֹדִיעַ מָה קוֹרֶה בַּמִּלְחָמָה. וְהָיְתָה שְׁכֵנָה אַחַת שֶׁאָמְרָה, כְּשֶׁהָרַדְיוֹ מָסַר מַשֶּׁהוּ: זֶה רַק הָרַדְיוֹ הַקָּטָן; נְחַכֶּה לָרַדְיוֹ הַגָּדוֹל.

[[41:00]] הַתְּמִימוּת שֶׁל פַּעַם – תַּעֲנוּג. אֵיפֹה הַתְּמִימוּת הַזֹּאת? אֵיפֹה הִיא? רַק תְּפִילָּה אַחַת כָּזֹאת שֶׁל הַתְּמִימִים – שֶׁאוֹמְרִים ⟦sefaria|לְדָוִד ה׳ אוֹרִי וְיִשְׁעִי⟧{{=תהילים כז, א}} לִפְנֵי תְּפִילַּת עַרְבִית, בְּמַנְגִּינָה. וְהַמַּנְגִּינָה הַזֹּאת הָיְתָה מַכְנִיסָה אוֹתְךָ לַתְּפִילָּה. זֶה כָּל הַתּוֹרָה, זֶה כָּל הַדָּת; זוֹ לֹא תּוֹסֶפֶת – זוֹ מַהוּת הַתְּפִילָּה: ⟦sefaria|לְדָוִד ה׳ אוֹרִי וְיִשְׁעִי מִמִּי אִירָא ה׳ מָעוֹז־חַיַּי מִמִּי אֶפְחָד בִּקְרֹב עָלַי מְרֵעִים⟧{{=תהילים כז, א–ב}}. אַתָּה נִכְנַס יָשָׁר לְ"שְׁמַע יִשְׂרָאֵל", וּפִתְאוֹם אַתָּה בְּ"הַשְׁכִּיבֵנוּ", בְּ"עָלֵינוּ לְשֶׁבַח".

[[42:45]] אֲנִי כְּבָר שְׁלוֹשִׁים שָׁנָה בִּנְתַנְיָה – וְאֵינֶנִּי מִתְפַּלֵּל. לֹא, אֵינֶנִּי מִתְפַּלֵּל כָּרָאוּי. ה׳ יוֹדֵעַ. לָמָּה אֲנִי מַסְכִּים שֶׁיִּקְרְאוּ לִי "הָרַב"? אִילּוּ לֹא הָיִיתִי רַב, הָיִיתִי מִתְפַּלֵּל טוֹב יוֹתֵר. זוֹ הַבְּעָיָה: אִילּוּ הָיִיתִי רַב אֲמִיתִּי וְלֹא הָיִיתִי מִתְפַּלֵּל טוֹב – בְּסֵדֶר. אֲבָל גַּם רַב וְגַם לֹא מִתְפַּלֵּל כָּרָאוּי – אֵינֶנִּי מְאַחֵל לָכֶם.

[[43:25]] בְּכָל זֹאת, שָׁמְעוּ מִילָּה אַחַת. יֵשׁ כָּאן גַּם תַּרְגּוּם יוֹנָתָן בֶּן עֻזִּיאֵל וְגַם תַּרְגּוּם יְרוּשַׁלְמִי. מָצָאתִי, בָּרוּךְ ה׳: ⟦sefaria|וְגַבְרָא משֶׁה עִנְוְותָן בְּדַעֲתֵּיהּ⟧{{=תרגום יונתן במדבר יב, ג}}. תַּקְשִׁיבוּ: זוֹ מִילָּה אַחַת, וְזֶהוּ – "⟦~|עִנְוְותָן בְּדַעֲתֵּיהּ⟧". זוֹ עֲנָוָה שֶׁל דַּעַת. זוֹ עֲנָוָה שֶׁל דַּעַת!

## עֲנָוָה שֶׁל דַּעַת

[[45:02]] וְלָמָּה אֲנַחְנוּ לֹא מְבִינִים כְּלוּם? כִּי אֵין לָנוּ דַּעַת. אֵין לָנוּ דַּעַת.

[[45:31]] אִילּוּ הָיְתָה לָנוּ דַּעַת, הָיִינוּ מְבִינִים מָה זֶה. כָּל הָעֲנָוָה שֶׁאֲנַחְנוּ מַכִּירִים – הִיא עֲנָוָה בְּלִי דַּעַת. אֲנַחְנוּ יוֹדְעִים מָה עֲנָוָה: מָה אֲנִי? מָה חַיַּי? אַתֶּם מְבִינִים – זֶה לֹא שֶׁקֶר; אֲנִי בַּעַל עֲנָוָה אֲמִיתִּית. אֲבָל תַּקְשִׁיבוּ לִי הֵיטֵב, רַבּוֹתַי: כָּל הָעֲנָווֹת הָאֵלֶּה הֵן עֲנָווֹת שֶׁל יֵאוּשׁ.

## חוֹבַת הַלְּבָבוֹת: שְׁלוֹשָׁה חֶלְקֵי הַכְּנִיעָה

[[46:55]] סֵפֶר "חוֹבַת הַלְּבָבוֹת" מוּנָּח עַל שׁוּלְחָנִי; לְפָחוֹת אֶת "חוֹבַת הַלְּבָבוֹת" אֲנִי מַחֲזִיק עַל הַשּׁוּלְחָן.

[[48:34]] "חוֹבַת הַלְּבָבוֹת" קוֹרֵא לָעֲנָוָה "כְּנִיעָה". מְדַיֵּק מְאוֹד הַקָּדוֹשׁ בַּעַל "חוֹבַת הַלְּבָבוֹת". כְּשֶׁאַתָּה קוֹרֵא אוֹתוֹ, הַשָּׂפָה שֶׁלּוֹ מְשַׁכְנַעַת, הִיא נִכְנֶסֶת פְּנִימָה לַנְּשָׁמָה. הוּא אֵינוֹ מַרְבֶּה בְּמִילִּים כְּדֵי לְהַסְבִּיר דָּבָר; כָּל מִילָּה שֶׁלּוֹ חוֹתֶכֶת לְךָ בַּבָּשָׂר, בַּנֶּפֶשׁ. בֶּאֱמֶת כְּדַאי לִשְׁמוֹעַ אֶת "חוֹבַת הַלְּבָבוֹת".

[[49:45]] הוּא אוֹמֵר שֶׁהַכְּנִיעָה נֶחֱלֶקֶת לִשְׁלוֹשָׁה חֲלָקִים{{=חובות הלבבות, שער הכניעה, פרק ב}}. רַבּוֹתַי, אַתֶּם חַיָּבִים לְהַקְשִׁיב – זֶה שִׁיעוּר, לֹא סְתָם דְּרָשָׁה. אֶחָד מֵהֶם, שִׂימוּ לֵב, ⟦sefaria|כּוֹלֵל אֶת הָאָדָם וְרַבִּים מִמִּינֵי בַּעֲלֵי חַיִּים⟧{{=חובות הלבבות, שער הכניעה, פרק ב}}. שְׁמַעְתֶּם שֶׁיֵּשׁ בַּעֲלֵי חַיִּים עֲנָוִים? שְׁמַעְתֶּם פַּעַם? הֶחֲמוֹר – עָנָיו.

[[50:35]] [מן הקהל:] גַּם לֶחָתוּל יֵשׁ מִידָּה שֶׁל צְנִיעוּת.

[[51:05]] ⟦sefaria|וְרַבִּים מִמִּינֵי בַּעֲלֵי חַיִּים שֶׁאֵינָם מְדַבְּרִים⟧{{=חובות הלבבות, שער הכניעה, פרק ב}} – תַּקְשִׁיבוּ טוֹב – ⟦sefaria|וְהוּא דַּלּוּת הַנֶּפֶשׁ⟧{{=חובות הלבבות, שער הכניעה, פרק ב}}. מָה זֶה דַּלּוּת הַנֶּפֶשׁ? נֶפֶשׁ דַּלָּה, אַבְיוֹנָה, עֲנִיָּה: ⟦sefaria|וְסָבְלָהּ הַהֶזֵּק אֲשֶׁר הָיְתָה יְכוֹלָה לִדְחוֹתוֹ⟧{{=חובות הלבבות, שער הכניעה, פרק ב}}. פֵּירוּשׁ: הֶחֲמוֹר עוֹמֵד בַּשֶּׁמֶשׁ. הוּא יָכוֹל לָצֵאת מִן הַשֶּׁמֶשׁ, חַם לוֹ – אֲבָל הוּא נִשְׁאָר בַּשֶּׁמֶשׁ. הוּא "עָנָיו". כָּךְ אוֹמֵר כָּאן: הוּא יָכוֹל לָצֵאת – חַם לְךָ? – לֹא, לֹא אֵצֵא. כְּאִילּוּ אֲנִי מְקַבֵּל עָלַי בֶּאֱמוּנָה, "גַּם זוֹ לְטוֹבָה". לֹא.

[[52:01]] ⟦sefaria|וְזֶה יִהְיֶה מִן הַכְּסִילִים שֶׁבִּבְנֵי אָדָם⟧{{=חובות הלבבות, שער הכניעה, פרק ב}}. הָעֲנָוָה הַזֹּאת שֶׁל דַּלּוּת הַנֶּפֶשׁ – הִיא שֶׁל הַכְּסִילִים. כְּשֶׁאֲנַחְנוּ אוֹמְרִים "כָּסִיל" – בָּרוּךְ ה׳ שֶׁאֵינֶנּוּ מִשְׁתַּמְּשִׁים בַּמִּילָּה הַזֹּאת – זֶה מִשּׁוּם שֶׁאֵינֶנּוּ יוֹדְעִים אֶת פֵּירוּשָׁהּ; חוֹשְׁבִים שֶׁ"כָּסִיל" טוֹב יוֹתֵר מִ"טִּיפֵּשׁ". אֲבָל מִי שֶׁקּוֹרֵא בְּסֵפֶר מִשְׁלֵי עִם הַמְּפָרְשִׁים יוֹדֵעַ: "פֶּתִי" – בַּשָּׂפָה הַמּוֹדֶרְנִית "נָאִיבִי" – יֵשׁ בּוֹ כָּל הַפְּחִיתוּיוֹת וְכָל הַחֶסְרוֹנוֹת שֶׁבָּעוֹלָם, אֲבָל הוּא תָּמִים. הַכָּסִיל – אֵינוֹ תָּמִים.

[[53:05]] הוּא אוֹמֵר כָּךְ: ⟦sefaria|וְעַמֵּי הָאָרֶץ מִפְּנֵי מִעוּט יְדִיעָתָם וַחֲלִישַׁת הַכָּרָתָם אֶת נַפְשׁוֹתָם וְעֶרְכֵיהֶם⟧{{=חובות הלבבות, שער הכניעה, פרק ב}} – הֵם אֵינָם מַכִּירִים אֶת עֵרֶךְ הַנֶּפֶשׁ. הָפוּךְ! מִפְּנֵי שֶׁאֵינָם מַכִּירִים אֶת עֵרֶךְ נַפְשָׁם, הֵם אוֹמְרִים: אֲנַחְנוּ עֲנָוִים, מָה אֲנִי, מָה חַיַּי. הַהֶפֶךְ! אַתָּה אָדָם שֶׁיֵּשׁ לְךָ נֶפֶשׁ – אֵיךְ אַתָּה אוֹמֵר לְעַצְמְךָ "דַּלּוּת"? הוּא כָּסִיל; הוּא אֵינוֹ מַכִּיר אֶת עֵרֶךְ הַנֶּפֶשׁ.

[[54:03]] (כְּמוֹ שֶׁאוֹמְרִים: כְּשֶׁיֵּשׁ קוּשְׁיָה, מְנַסִּים לַעֲשׂוֹת חוֹרִים בַּגְּמָרָא – בַּמְקוֹם בָּרֹאשׁ שֶׁלָּהֶם.)

[[54:31]] ⟦sefaria|וְזֶה נִקְרָא כְּנִיעָה מִדֶּרֶךְ הַהַסְכָּמָה⟧{{=חובות הלבבות, שער הכניעה, פרק ב}}. מָה זֶה בְּדִיּוּק? לֹא כָּל כָּךְ מְבִינִים מָה זוֹ "הַסְכָּמָה". כְּאִילּוּ זוֹ עֲנָוָה – מְחִילָה מִכְּבוֹדָם, עֲנָוָה שֶׁל חֲמוֹרִים. ⟦sefaria|אֲבָל עַל דֶּרֶךְ הָאֱמֶת הוּא דַּלּוּת הַנֶּפֶשׁ⟧{{=חובות הלבבות, שער הכניעה, פרק ב}}. זוֹ לֹא עֲנָוָה. פָּשׁוּט יֵשׁ לוֹ נֶפֶשׁ דַּלָּה, וּבַשָּׂפָה שֶׁלָּנוּ – טִיפֵּשׁ. [?55:10]

[[55:22]] רַבּוֹתַי, זֶה קַיָּם בְּתוֹכֵנוּ, וְחָזָק מְאוֹד. יֵשׁ מוּשָּׂג שֶׁל תְּשׁוּבָה, שֶׁל לַעֲשׂוֹת תְּשׁוּבָה – זֶה עוֹלָם גָּדוֹל מְאוֹד, וְאֵינֶנִּי רוֹצֶה לְהִיכָּנֵס אֵלָיו עַכְשָׁיו. כְּמוֹ שֶׁנֶּאֱמַר ⟦sefaria|כִּי־לִבָּם צָפַנְתָּ מִשָּׂכֶל עַל־כֵּן לֹא תְרֹמֵם⟧{{=איוב יז, ד}} – זֶה פָּסוּק בְּאִיּוֹב.

[[56:01]] אֲבָל בּוֹאוּ נִיכָּנֵס כְּבָר לַכְּנִיעָה, כִּי כָּאן צָרִיךְ לָשֶׁבֶת. ⟦sefaria|אֲבָל הַכְּנִיעָה⟧{{=חובות הלבבות, שער הכניעה, פרק ב}} – תִּשְׁמְעוּ הֵיטֵב – ⟦sefaria|אֲשֶׁר תִּהְיֶה אַחַר רוֹמְמוּת הַנֶּפֶשׁ וְהִתְנַשְּׂאָהּ מֵהִשְׁתַּתֵּף עִם הַבְּהֵמוֹת בְּמִדּוֹתָם הַמְגֻנּוֹת וְגַבְהוּתָהּ מֵהִדַּמּוֹת בְּמִדּוֹת פְּחוּתֵי בְּנֵי אָדָם בְּיִתְרוֹן חָכְמַת וְיִקְרַת נֶפֶשׁ וִידִיעָה בְּרוּרָה בַּמִּדּוֹת הַטּוֹבוֹת וְהַמְּגֻנּוֹת. וְכַאֲשֶׁר יִהְיֶה סָמוּךְ לָזֶה כְּנִיעַת הַנֶּפֶשׁ וְשִׁפְלוּתָהּ אָז תִּהְיֶה מִדָּה מְשֻׁבַּחַת אֲבָל זוּלַת זֶה אֵינֶנָּה נִכְנֶסֶת בַּמִּדּוֹת הַמְּשֻׁבָּחוֹת וּמַעֲלוֹת הַנֶּפֶשׁ אַךְ בַּמְּגֻנּוֹת שֶׁבָּהֶן כִּי עִנְיָנָהּ בָּזֶה כְּעִנְיַן הַבְּהֵמוֹת.⟧{{=חובות הלבבות, שער הכניעה, פרק ב}}.

[[57:01]] כָּאן צָרִיךְ לִימּוּד: לָשֶׁבֶת בְּעֶזְרַת ה׳ וְלִלְמוֹד אֶת הַקֶּטַע הַזֶּה. אֲבָל בְּמִילָּה פְּשׁוּטָה, כְּשֶׁאֲנַחְנוּ חַיָּבִים לְסַכֵּם: יֵשׁ עֲנָוָה שֶׁל בְּהֵמוֹת, יֵשׁ עֲנָוָה שֶׁל אֲנָשִׁים טִיפְּשִׁים, יֵשׁ עֲנָוָה שֶׁל עַמֵּי הָאָרֶץ – וְיֵשׁ הָעֲנָוָה שֶׁאוֹמֵר הַתַּרְגּוּם יוֹנָתָן בְּמִילָּה אַחַת: "⟦~|עִנְוְותָן בְּדַעֲתֵּיהּ⟧". זוֹ עֲנָוָה שֶׁל דַּעַת.

[[57:55]] אֲנִי מֹשֶׁה רַבֵּנוּ. אֲנִי מֹשֶׁה שֶׁמְּדַבֵּר עִם ה׳. אֲנִי מֹשֶׁה שֶׁה׳ אוֹמֵר לוּ ⟦sefaria|וְאַתָּה פֹּה עֲמֹד עִמָּדִי⟧{{=דברים ה, כח}}. אֲנִי מֹשֶׁה שֶׁעָמַדְתִּי עַל הַר סִינַי לְיַד הַקָּדוֹשׁ בָּרוּךְ הוּא בִּנְתִינַת הַתּוֹרָה. אֲנִי מֹשֶׁה שֶׁקָּרַע אֶת הַיָּם. אֲנִי מֹשֶׁה שֶׁהֵבִיא עֶשֶׂר מַכּוֹת. אֲנִי מֹשֶׁה שֶׁבִּזְכוּתִי יוֹרֵד הַמָּן. זוֹ רוֹמְמוּת הַנֶּפֶשׁ – וַאֲנִי עָנָיו. וַאֲנִי מֹשֶׁה, הֶעָנָיו "⟦~|מִכֹּל הָאָדָם אֲשֶׁר עַל־פְּנֵי הָאֲדָמָה⟧": עִנְוְתָן בְּדַעְתֵּיהּ. זוֹהִי עֲנָוָה שֶׁל דַּעַת, כְּכָל הָאֶפְשָׁר. לְהָבִין יוֹתֵר מִזֶּה – צָרִיךְ לָשֶׁבֶת שִׁבְעָה נְקִיִּים עַל דִּבְרֵי "חוֹבַת הַלְּבָבוֹת" הַקָּדוֹשׁ וְהַנּוֹרָא.

[[59:05]] בֶּאֱמֶת, רַבּוֹתַי, אֵינֶנִּי יוֹדֵעַ לָמָּה אֵין יוֹשְׁבִים בַּיְּשִׁיבוֹת לִלְמוֹד "חוֹבַת הַלְּבָבוֹת" כְּמוֹ שֶׁלּוֹמְדִים [?59:15]. וְאֵינֶנִּי מְדַבֵּר עַל אֲנָשִׁים רְגִילִים שֶׁעוֹבְדִים – שָׁעָה בְּיוֹם, שָׁעָה בְּיוֹם לִלְמוֹד. וְהִנֵּה יֵשׁ לָנוּ, בָּרוּךְ ה׳, תַּרְגּוּם יוֹנָתָן בֶּן עוזִּיאֵל, שֶׁאוֹמֵר בְּמִילָּה אַחַת אֶת כָּל מָה שֶׁאוֹמֵר "חוֹבַת הַלְּבָבוֹת" בְּאַרְבַּע אוֹ חָמֵשׁ שׁוּרוֹת. וְצָרִיךְ אֶת "חוֹבַת הַלְּבָבוֹת", כִּי הוּא מַסְבִּיר וּמַרְחִיב אֶת הַמִּילָּה הַזֹּאת, "⟦~|עִנְוְותָן בְּדַעֲתֵּיהּ⟧", וּמְחַדֵּשׁ וּמַסְבִּיר אֶת הַצַּד הַשֵּׁנִי: אִם לֹא כָּךְ – זוֹ עֲנָוָה שֶׁל בְּהֵמוֹת.

## "יֵשׁ לָנוּ בְּעָיָה שֶׁאֵין לָנוּ גַּאֲוָה"

[[60:20]] וְזֶה, רַבּוֹתַי, דָּבָר גָּדוֹל וְנוֹרָא. אֲנִי מִשְׁתַּמֵּשׁ בְּמִילִּים פְּשׁוּטוֹת: יֵשׁ לָנוּ בְּעָיָה שֶׁאֵין לָנוּ גַּאֲוָה עַצְמִית. יֵשׁ לָנוּ בְּעָיָה שֶׁאֵין לָנוּ גַּאֲוָה. מָה פֵּירוּשׁ? כְּדֵי לָצֵאת מִן הָעֲנָוָה שֶׁל הַבְּהֵמָה – צָרִיךְ שֶׁתִּהְיֶה לְךָ הַגַּאֲוָה שֶׁתּוֹצִיא אוֹתְךָ מִמֶּנָּה, לְפָחוֹת שֶׁלֹּא תִּהְיֶה עָנָיו כְּמוֹ בְּהֵמָה.

[[61:05]] בְּמִילִּים פְּשׁוּטוֹת: זוֹהִי הַכְּנִיעָה שֶׁבָּאָה אַחֲרֵי רוֹמְמוּת הַנֶּפֶשׁ, אַחֲרֵי הַנֶּפֶשׁ בְּרָמָתָהּ הַגְּבוֹהָה. הַקָּדוֹשׁ בָּרוּךְ הוּא – אֲנִי יוֹדֵעַ מִי אֲנִי, וַאֲנִי יוֹדֵעַ שֶׁאֲנִי צָרִיךְ לִהְיוֹת עָנָיו; וַאֲנִי מְדַבֵּר אִיתְּךָ לֹא מִתּוֹךְ שֶׁאֲנִי אֶפֶס. אִם אַתָּה אֶפֶס – מָה אַתָּה אוֹמֵר, שֶׁאַתָּה עָנָיו?

## מַעֲשֶׂה בָּאַדְמוֹ״ר וּבַחָסִיד בַּעַל הַגַּאֲוָה

[[61:35]] בָּא פַּעַם יְהוּדִי אֶל אַדְמוֹ״ר וְאָמַר לוֹ: רַבִּי, יֵשׁ לִי בְּעָיָה. מָה הַבְּעָיָה? יֵשׁ לִי גַּאֲוָה, וְאֵינֶנִּי יָכוֹל לְהִתְגַּבֵּר עָלֶיהָ. אָמַר לוֹ הָאַדְמוֹ״ר: אֵין בְּעָיָה; שָׁב בַּחֲדַר הַהַמְתָּנָה. אָמַר: מָה, אֲנִי – בַּחֲדַר הַהַמְתָּנָה? אָמַר לוֹ: בָּאתָ לָאַדְמוֹ״ר, נָכוֹן? אָז שָׁב שָׁם.

[[62:15]] אַחַר כָּךְ נִכְנָסִים אֲנָשִׁים לְאַדְמוֹ״ר. אֶחָד אוֹמֵר לוֹ: רַבִּי, כָּל הַמְּנָיוֹת שֶׁלִּי יָרְדוּ, וְאֵינֶנִּי יוֹדֵעַ אֵיךְ לָצֵאת מִן הַבְּעָיָה. אוֹמֵר לוֹ: יֵשׁ יְהוּדִי בַּחֲדַר הַהַמְתָּנָה שֶׁהוּא אַלּוּף בַּמְּנָיוֹת; לְךָ וְהִתְיַעֵץ אִיתּוֹ. יוֹצֵא אֵלָיו וְאוֹמֵר: אֲנִי – הַמְּנָיוֹת... אוֹמֵר לוֹ: סְלַח לִי, עַל מָה אַתָּה מְדַבֵּר? אֵינֶנִּי מֵבִין בַּמְּנָיוֹת כְּלָל.

[[62:55]] אַחַר כָּךְ בָּא אַחֵר: רַבִּי, יֵשׁ לִי בְּעָיָה עִם אִשְׁתִּי – אֵיךְ עוֹשִׂים שְׁלוֹם בַּיִת? אוֹמֵר לוֹ: יֵשׁ פְּסִיכוֹלוֹג בַּחוּץ. יוֹצֵא אֵלָיו: אִשְׁתִּי, אִשְׁתִּי... אוֹמֵר: עֲזוֹב אוֹתִי מֵאִשְׁתְּךָ. אַחַר כָּךְ בָּא אַחַר עִם בְּעָיָה רְפוּאִית, בְּעָיָה עִם הַשְּׁכֵנִים, בְּעָיָה בַּמִּשְׁפָּט – וְשׁוֹלֵחַ אוֹתוֹ אֵלָיו. בַּסּוֹף אֵינוֹ מִתְאַפֵּק, נִכְנַס לָאַדְמוֹ״ר: רַבִּי, מָה זֶה? אָמַר לוֹ: אֵינְךָ רוֹפֵא, אֵינְךָ מֵבִין בַּמְּנָיוֹת, אֵינְךָ מֵבִין בָּזֶה וּבָזֶה – אָז בְּמָה אַתָּה מִתְגָּאֶה? אֵיזוֹ גַּאֲוָה יֵשׁ לְךָ? אַתָּה כְּלוּם! אֲבָל אַחֲרֵי רוֹמְמוּת הַנֶּפֶשׁ – אֲנִי גָּאוֹן, אֲנִי לַמְדָן – אָז הָעֲנָוָה הִיא בְּדַעַת.

## מֹשֶׁה רַבֵּנוּ מוּל קוֹרַח

[[64:05]] כָּל זֶה הֵבֵאתִי בִּשְׁבִיל דָּבָר אֶחָד – בִּשְׁבִיל מֹשֶׁה רַבֵּנוּ עָלָיו הַשָּׁלוֹם. אִישׁ לֹא יָכוֹל הָיָה לַעֲמוֹד בַּנִּיסָּיוֹן שֶׁל קוֹרַח. וְאוֹתוֹ מֹשֶׁה, שֶׁהָיָה "⟦~|עִנְוְותָן בְּדַעֲתֵּיהּ⟧" – מָה עָשָׂה? בִּמְקוֹם לָשֶׁבֶת וּלְהִתְוַכֵּחַ עִם קוֹרַח: הֲרֵי אֲנַחְנוּ בְּנֵי דּוֹדִים; אָבִיךָ, יִצְהָר, וְאָבִי, עַמְרָם, הֵם אַחִים – עַמְרָם, יִצְהָר, חֶבְרוֹן וְעֻזִּיאֵל, אַרְבָּעָה אַחִים; אַתָּה עָשִׁיר – וְלֹא הָיָה קוֹרַח נוֹתֵן לוֹ לְדַבֵּר כָּךְ. בַּסּוֹף הָיָה מֹשֶׁה אוֹמֵר לוֹ, כְּמוֹ שֶׁאָמַר הָאַדְמוֹ״ר: אֵין לְךָ כְּלוּם חוּץ מִכֶּסֶף; מָה אַתָּה? לֹא – "⟦~|וַיִּפֹּל עַל־פָּנָיו⟧". זֶה נִקְרָא "⟦~|עִנְוְותָן בְּדַעֲתֵּיהּ⟧".

[[65:40]] בְּאוֹתָהּ שָׁעָה נָפַל מֹשֶׁה רַבֵּנוּ עַל פָּנָיו, וְאָמַר כִּבְיָכוֹל: אֵינֶנִּי יוֹדֵעַ אִם לוֹמַר מַשֶּׁהוּ אוֹ לֹא לוֹמַר כְּלוּם. מִי שֶׁאֵין לוֹ אֶת הָעֲנָוָה הַזֹּאת שֶׁל הַדַּעַת – לֹא הָיָה יָכוֹל לַעֲמוֹד בַּנִּיסָּיוֹן הַזֶּה.

## גְּדוֹלֵי הַדּוֹרוֹת שֶׁעָמְדוּ בְּנִיסְיוֹנוֹת

[[66:15]] יָדוּעַ וּמְפוּרְסָם שֶׁגְּדוֹלֵי הַדּוֹרוֹת עָמְדוּ בְּנִיסְיוֹנוֹת קָשִׁים וְאֲיוּמִים. הָיוּ לָהֶם מְקַטְרְגִים וּמִתְנַגְּדִים כָּל יְמֵי חַיֵּיהֶם, שֶׁהִכְנִיסוּ אוֹתָם לְבֵית הַסּוֹהַר וְהִלְשִׁינוּ עֲלֵיהֶם שֶׁהֵם מְרַגְּלִים – וְלֹא רַק בְּרוּסְיָה, גַּם בְּאֵירוֹפָּה. גְּדוֹלֵי עוֹלָם, קְדוֹשֵׁי עֶלְיוֹן, תָּמִיד הָיוּ בַּנִּיסְיוֹנוֹת הָאֵלֶּה: הַתּוֹסָפוֹת יוֹם טוֹב הָיָה בְּבֵית הַסּוֹהַר; רַבֵּנוּ בַּעַל הַתַּנְיָא, הָאַדְמוֹ״ר הַזָּקֵן, הָיָה בְּבֵית הַסּוֹהַר; הַמַּלְבִּי״ם; ו[?66:50] – עָמְדוּ וְעָמְדוּ.

[[67:01]] מֹשֶׁה רַבֵּנוּ – "⟦~|וַיִּפֹּל עַל־פָּנָיו⟧". בֶּאֱמֶת, קוֹרַח, אַתָּה צוֹדֵק. אֲבָל אוֹתוֹ מֹשֶׁה רַבֵּנוּ אָמַר לוֹ: אֲנִי אִיתְּךָ, קוֹרַח, אֲנִי מַסְכִּים אִיתְּךָ, אֵין לִי שׁוּם בְּעָיָה. אֲבָל אַתָּה רוֹצֶה לִהְיוֹת כֹּהֵן גָּדוֹל – וְלֹא רַק אַתָּה.

## "מְדִינַת כָּל אֶזְרָחֶיהָ"

[[67:35]] עוֹד לֹא יָרַדְנוּ לְפִרְטֵי מַחְלוֹקֶת קוֹרַח – לֹא נָגַעְנוּ בָּהּ אֲפִילּוּ. אֲבָל קוֹרַח רָצָה שֶׁלֹּא יִהְיֶה כֹּהֵן אֶחָד, אֶלָּא מָאתַיִם וַחֲמִישִּׁים כֹּהֲנִים – שֶׁיִּהְיֶה פָּתוּחַ. זוֹ "מְדִינַת כָּל אֶזְרָחֶיהָ". מְדִינָה – לֹא תַּגִּיד לִי שֶׁמּוּחַמַּד אֵינוֹ אֶזְרַח הַמְּדִינָה! אָז מָה, אַתָּה, אַבְרָהָם לֵוִי, פִּתְאוֹם קוֹרְאִים לְךָ "לֵוִי" וּמַעֲלִים אוֹתְךָ שֵׁנִי? לָמָּה יֵעָלֵם מוּחַמַּד? שֶׁיַּעֲלֶה שְׁלִישִׁי! אַתָּה מֵבִין? מְדִינַת כָּל אֶזְרָחֶיהָ.

[[68:34]] מִי שֶׁרוֹצֶה לִהְיוֹת רֹאשׁ מֶמְשָׁלָה – מְדִינַת כָּל אֶזְרָחֶיהָ. ה׳ יְרַחֵם עָלֵינוּ. רַבּוֹתַי, אֲנִי מְדַבֵּר הֲלָכָה לְמַעֲשֶׂה: עוֹד מְעַט יִהְיוּ בְּחִירוֹת, וְעוֹד מְעַט נִשְׁמָע דְּבָרִים אֲיֻוּמִּים. אֵינֶנִּי רוֹצֶה לְהַאֲרִיךְ יוֹתֵר מִדַּי אֶת הַשִּׂיחָה הַזֹּאת, אֲבָל אֲסַכֵּם מְעַט.

## קוֹרַח – לֹא רָשָׁע, אֶלָּא טָעָה

[[69:04]] לְקוֹרַח הָיוּ כָּל הַדְּרָגוֹת הַגְּבוֹהוֹת בְּיוֹתֵר שֶׁל אָדָם רַם מַעְלָה: חָכְמָה, תּוֹרָה, יִחוּס – ⟦sefaria|קֹרַח בֶּן־יִצְהָר בֶּן־קְהָת בֶּן־לֵוִי⟧{{=במדבר טז, א}}, נֶכְדּוֹ שֶׁל לֵוִי בֶּן יַעֲקֹב אָבִינוּ. אֶלָּא שֶׁיַּעֲקֹב אָבִינוּ הִתְפַּלֵּל שֶׁלֹּא יַזְכִּירוּ אוֹתוֹ "בֶּן יַעֲקֹב" בְּיִחוּסוֹ: ⟦sefaria|בִּקְהָלָם אַל־תֵּחַד כְּבֹדִי⟧{{=בראשית מט, ו}}, כָּךְ הִתְפַּלֵּל יַעֲקֹב, וְכָךְ עָשְׂתָה הַתּוֹרָה{{=רשי בראשית מט, ו}}. הָיוּ לוֹ כָּל הַמַּעֲלוֹת, וְעוֹד הָעוֹשֶׁר הָאַגָּדִי שֶׁהָיָה לוֹ.

[[69:50]] וּמָה קָרָה לוֹ? אֶפְשָׁר לִקְרוֹא לְקוֹרַח רָשָׁע? לֹא. הוּא אֵינוֹ רָשָׁע – הוּא טָעָה. בְּסַךְ הַכּוֹל רָאָה שֶׁעֲתִידִים לָצֵאת מִמֶּנּוּ שְׁמוּאֵל הַנָּבִיא וְעֶשְׂרִים וְאַרְבַּע מִשְׁמָרוֹת{{=רשי במדבר טז, ז}} – הַלְּוַיִים שֶׁהָיוּ שָׁרִים בְּבֵית הַמִּקְדָּשׁ. וּבָנָיו חָזְרוּ בִּתְשׁוּבָה בָּרֶגַע הָאַחֲרוֹן – יוֹשְׁבִים עַל כָּךְ עַד עַכְשָׁיו, בִּתְשׁוּבָה – וְנִיצְּלוּ. הוּא טָעָה.

[[70:35]] וּמָה הָיְתָה טָעוּתוֹ? טָעוּת אֲמִיתִּית: הֲרֵי בֶּאֱמֶת יָצָא מִמֶּנּוּ שְׁמוּאֵל. אִם כֵּן, אֵיךְ יִתָּכֵן שֶׁשְּׁמוּאֵל יֵצֵא מִמֶּנִּי, וַאֲנִי עוֹמֵד עַכְשָׁיו בַּמַּחֲלוֹקֶת הַזֹּאת? זֶה לֹא יִתָּכֵן. כְּלוֹמַר, לֹא הָיוּ לוֹ טְעָנוֹת כְּמוֹ שֶׁל אַהֲרֹן בָּרָק, הַשּׁוֹפֵט הָעֶלְיוֹן, שֶׁמַּחֲרִיב לָנוּ אֶת הַמְּדִינָה, [?71:10] – ה׳ יְרַחֵם עָלֵינוּ, אֵינֶנִּי רוֹצֶה לְדַבֵּר עָלָיו כְּלָל. "עֵינוֹ הִטְעָתוּ", אוֹמֵר הַמִּדְרָשׁ הַקָּדוֹשׁ – "עֵינוֹ הִטְעָתוּ".

## לוֹקֵחַ אֶת מַעֲלוֹתָיו לְשֵׁם הַשְּׂרָרָה

[[71:33]] בְּמִילָּה אַחַת אֲנִי רוֹצֶה לְסַיֵּם: אָדָם לוֹקֵחַ אֶת הַ"מַעֲלוֹת הַגְּרוּעוֹת" שֶׁלּוֹ – אֶפְשָׁר לִקְרוֹא לָהֶן כָּךְ: מַעֲלוֹת מִצַּד אֶחָד, וּגְרוּעוֹת מִצַּד שֵׁנִי. לָמָּה? אִילּוּ בָּא אָדָם מוּל מֹשֶׁה רַבֵּנוּ וְאָמַר לוֹ: אֲנִי אָדָם, אֲנִי רוֹצֶה פְּרַיְמֵרִיז – כְּמוֹ שֶׁיֵּשׁ עַכְשָׁיו פְּרַיְמֵרִיז – בּוֹא נַעֲשֶׂה פְּרַיְמֵרִיז; אַתָּה, מֹשֶׁה, בְּרֹאשׁ הָרְשִׁימָה, וַאֲנִי אַחֲרֶיךָ; אֲנִי רוֹצֶה לִהְיוֹת סְגָנֵךָ. קוֹרַח לֹא בָּא כָּךְ. קוֹרַח גִּיֵּס אֶת כָּל מַעֲלוֹתָיו הַטּוֹבוֹת – מַעֲלוֹת הַיִּחוּס, שֶׁאֵינָן מַעֲלוֹת עַצְמִיּוֹת שֶׁלּוֹ, מַעֲלוֹת שֶׁל מַעֲמָד. הוּא בֶּאֱמֶת כָּזֶה: בֶּן דּוֹדוֹ, מִמֶּנּוּ יוֹצֵא שְׁמוּאֵל, עֶשְׂרִים וְאַרְבַּע מִשְׁמָרוֹת. אֲבָל הוּא לָקַח אֶת הַכּוֹל, לָקַח אֶת הַתּוֹרָה שֶׁלּוֹ, וְהִשְׁתַּמֵּשׁ בָּהּ לְדָבָר אֶחָד – לִהְיוֹת רֹאשׁ מֶמְשָׁלָה. כָּל הַדְּרָכִים כְּשֵׁרוֹת.

[[73:05]] בְּמִילָּה אַחַת אֲנִי אוֹמֵר – שֶׁיִּהְיֶה בָּרוּר, וַאֲנִי אוֹמֵר זֹאת גַּם הֲלָכָה לְמַעֲשֶׂה, לִקְרַאת הַבְּחִירוֹת הַ"קְּדוֹשׁוֹת" שֶׁלָּהֶם – שֶׁאֲנָשִׁים יָבוֹאוּ, אֲנָשִׁים שֶׁמּוּכָנִים לִמְכּוֹר אֶת הַמְּדִינָה, לִמְכּוֹר אֶת עַם יִשְׂרָאֵל. אֵינֶנִּי מְדַבֵּר עַל לִמְכּוֹר אֶת יְרוּשָׁלַיִם וְאֶת הַכּוֹתֶל – פַּעַם אָמַר מִישֶׁהוּ: מָה זֶה הַכּוֹתֶל הַזֶּה? זֶה קִיר. וְאֵינֶנִּי מְדַבֵּר עַל הַיְּשִׁיבוֹת וְהַחֲרֵדִים – אֶת אֵלֶּה בִּכְלָל צָרִיךְ "לָקַחַת בִּמְרִיצָה", כְּמוֹ שֶׁאָמַר לִיבֶּרְמָן. אַתָּה מֵבִין? מוּכָנִים לִמְכּוֹר אֶת הַמְּדִינָה – לִמְכּוֹר אוֹתָהּ – הַכּוֹל בִּשְׁבִיל לִהְיוֹת רֹאשׁ מֶמְשָׁלָה. וְתִרְאֶה: רֹאשׁ מֶמְשָׁלָה עִם מִי? עִם עֶשְׂרִים עַרְבִים. וְלָמָּה לֹא? הֵם אוֹמְרִים עַכְשָׁיו: אֲנִי רוֹצֶה לְהַצִּיל אֶת הַמְּדִינָה, כִּי הַמְּדִינָה עוֹמֶדֶת [?74:30]. אָדָם מוּכָן לַעֲשׂוֹת הַכּוֹל – מוּכָן לְהָבִיא מָאתַיִם וַחֲמִישִּׁים רָאשֵׁי סַנְהֶדְרָאוֹת וְלָתֵת בְּיָדָם מַחֲתוֹת.

[[74:40]] הֲרֵי הוּא לֹא נִרְתַּע. מֹשֶׁה רַבֵּנוּ הֶעֱמִיד אוֹתוֹ בְּמַצָּב שֶׁבּוֹ אוּלַי – אוּלַי – הָיָה עוֹשֶׂה חֶשְׁבּוֹן, לוֹקֵחַ לוֹ יוֹם אוֹ יוֹמַיִם לִנְשׁוֹם. לֹא. הוּא הָלַךְ עִם הַהַצָּעָה שֶׁל מֹשֶׁה, וְשָׂם מַחֲתוֹת בִּידֵי מָאתַיִם וַחֲמִישִּׁים רָאשֵׁי סַנְהֶדְרָאוֹת. וְכִי אֵינוֹ יוֹדֵעַ [?75:05]?

## הָאָדָם יָכוֹל לִהְיוֹת הַכּוֹל – תְּשׁוּבַת הָרָמָ״ע מִפָּאנוֹ

[[75:15]] שְׁתֵי מִילִּים, וַאֲנִי מְסַיֵּם כָּאן. בְּיוֹם חֲמִישִׁי, בְּשִׁיעוּר בְּאוֹר הַחַיִּים, שָׁאַלְתִּי אֶת חֲבֵרֵינוּ: אֵיךְ יִתָּכֵן קוֹרַח כָּזֶה? אֵיךְ יָכוֹל הָיָה? יְכוֹלָה לִהְיוֹת תְּשׁוּבָה פְּשׁוּטָה – תְּשׁוּבָה מְסַכֶּמֶת, תְּשׁוּבָה נוֹקֶבֶת: הָאָדָם יָכוֹל לִהְיוֹת הַכּוֹל. הָאָדָם, בְּכוֹחוֹתָיו, יָכוֹל לִהְיוֹת הָאָדָם שֶׁבּוֹרֵא עוֹלָמוֹת. כְּמוֹ שֶׁאָמַרְתִּי גַּם כֵּן בַּשִּׁיעוּר: הָרָמָ״ע מִפָּאנוֹ עָלָיו הַשָּׁלוֹם הָיָה מְקוּבָּל גָּדוֹל וְעָצוּם, אָדָם גָּדוֹל בַּנִּגְלֶה וּבַנִּסְתָּר, בְּאִיטַלְיָה, לִפְנֵי מָאתַיִם וַחֲמִישִּׁים אוֹ שְׁלוֹשׁ מֵאוֹת שָׁנָה. הוּא אָמַר שֶׁהָאָדָם – ⟦sefaria|וַתְּחַסְּרֵהוּ מְּעַט מֵאֱלֹהִים⟧{{=תהילים ח, ו}} – יָכוֹל לַעֲשׂוֹת כָּל מָה שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא עוֹשֶׂה, מִלְּבַד מָה שֶׁכָּתוּב בַּמִּדְרָשׁ שֶׁהוּא "בּוֹנֶה עוֹלָמוֹת וּמַחֲרִיבָן" – וְאֵין לָנוּ מוּשָּׂג מָה פֵּירוּשׁ הַדָּבָר. אֶת זֶה הוּא אֵינוֹ יָכוֹל לַעֲשׂוֹת.

[[76:20]] כְּלוֹמַר, הָאָדָם יָכוֹל לִהְיוֹת הַכּוֹל. עַל קוֹרַח אַל תִּתְפַּלֵּא: כָּל אֶחָד מֵאִיתָּנוּ הוּא קוֹרַח בְּכוֹחַ. מָה שֶׁאֵינֶנּוּ קוֹרַח – זֶה רַק מִשּׁוּם שׁ[?76:35]. אַתָּה יָכוֹל לָלֶכֶת עַכְשָׁיו וּלְהַצִּיג אֶת עַצְמְךָ כְּרֹאשׁ מֶמְשָׁלָה – בְּעֶצֶם, לָמָּה לֹא? אִם בֵּנֶט מַצִּיעַ אֶת עַצְמוֹ, לָמָּה לֹא אַתָּה? – אֵין לִי כּוֹחַ. אֲבָל מִי שֶׁיֵּשׁ לוֹ מְעַט כּוֹחַ – לָמָּה לֹא? זוֹ תְּשׁוּבָה אַחַת.

[[77:05]] וְהַתְּשׁוּבָה הַשְּׁנִיָּה: יֵשׁ לָאָדָם הַכּוֹחַ לָקַחַת אֶת כָּל מַעֲלוֹתָיו – אֶת הַתּוֹרָה, אֶת הַחָכְמָה, אֶת הַכֶּסֶף, אֶת הַכּוֹל – בִּשְׁבִיל לִהְיוֹת רֹאשׁ מֶמְשָׁלָה, אֲפִילּוּ אִם בְּמֶמְשַׁלְתּוֹ יִהְיוּ עֲשָׂרָה שָׂרִים עַרְבִים, וְהָעַרְבִים הָאֵלֶּה יַחֲרִימוּ אֶת הַמְּדִינָה. זֶה שָׁוֶה לוֹ, בִּשְׁבִיל לִהְיוֹת רֹאשׁ מֶמְשָׁלָה שָׁנָה אַחַת. אֵלֶּה הַכּוֹחוֹת שֶׁל [?77:30].

## הַלִּימּוּד מִפָּרָשַׁת קוֹרַח

[[77:40]] וְלָכֵן עָשְׂתָה לָנוּ הַתּוֹרָה הַקְּדוֹשָׁה טוֹבָה גְּדוֹלָה מְאוֹד. הַתּוֹרָה לֹא הָיְתָה מוּכְרָחָה לִכְתּוֹב אֶת פָּרָשַׁת קוֹרַח, וְלֹא אֶת פָּרָשַׁת הַמְּרַגְּלִים, וְגַם לֹא אֶת פָּרָשַׁת מִרְיָם שֶׁדִּיבְּרָה לְשׁוֹן הָרַע. לָמָּה כָּתְבָה? הַתְּשׁוּבָה: תּוֹרַת אֱמֶת. הַלִּימּוּדִים הַגְּדוֹלִים שֶׁלָּנוּ הֵם מִן הָעוּבְדוֹת הַמְּזַעְזְעוֹת. פָּרָשַׁת קוֹרַח הִיא הַפָּרָשָׁה שֶׁבָּאָה לַהֲרוֹס אֶת כָּל הַיַּהֲדוּת: אֵין מֹשֶׁה רַבֵּנוּ; מֹשֶׁה רַבֵּנוּ קִיבֵּל תּוֹרָה מִסִּינַי? הַכּוֹל לֹא נָכוֹן, הַכּוֹל שֶׁקֶר; זֶה לֹא מֹשֶׁה רַבֵּנוּ – כָּךְ וְכָךְ מְסַדְּרִים אֶת הָעִנְיָנִים. וְדָבָר שֵׁנִי: הַכּוֹל בִּשְׁבִיל שֶׁהוּא יִהְיֶה כֹּהֵן גָּדוֹל.

[[78:50]] וְזֶה לִימּוּד בִּשְׁבִילֵנוּ. לִפְעָמִים אֲנַחְנוּ מִתְוַכְּחִים עִם אָדָם, עִם הָאִישָּׁה, עִם חָבֵר – לֹא מְשַׁנֶּה עִם מִי – וְכָל אֶחָד מֵבִיא רְאָיוֹת. כָּל הָרְאָיוֹת הָאֵלֶּה בָּאוֹת כְּדֵי שֶׁתַּצְדִּיק אֶת עַצְמְךָ; וּמָה – לְקַעְקֵעַ אֶת כָּל הַיַּהֲדוּת, כְּמוֹ שֶׁאָמְרוּ חֲכָמֵינוּ זָ״ל: לַהֲפוֹךְ אֶת הַקְּעָרָה עַל פִּיה{{=בבא בתרא טז עא}}. זוֹ בְּדִיּוּק הַהַגְדָּרָה.

## הַגְּאוּלָּה תְּלוּיָה בָּנוּ

[[79:33]] וְלָכֵן, בְּדִיּוּק מָה שֶׁאֲנַחְנוּ בָּאִים לִלְמוֹד: אִם אֲנַחְנוּ רוֹצִים אֶת הַגְּאוּלָּה – כְּמוֹ שֶׁהִתְחַלְתִּי לְדַבֵּר עַל הַגְּאוּלָּה – הַגְּאוּלָּה תְּלוּיָה בְּאֵיזוֹ גְּאוּלָּה אֲנַחְנוּ רוֹצִים. אִם זוֹ גְּאוּלָּה שֶׁבָּאָה מִצַּד הַקָּדוֹשׁ בָּרוּךְ הוּא, הִיא יְכוֹלָה לָבוֹא בִּשְׁנֵי פָּנִים. זֶה, אַגַּב, נוֹשֵׂא בִּפְנֵי עַצְמוֹ, אֲבָל אֲנִי לוֹקֵחַ אֶת הַנְּקוּדָּה הַזֹּאת כְּדֵי לָתֵת סִיּוּם לַשִּׂיחָה – אִם בִּכְלָל אֶפְשָׁר לָתֵת סִיּוּם.

[[80:25]] יֵשׁ מַחְלוֹקֶת בַּגְּמָרָא אִם הַגְּאוּלָּה תִּהְיֶה טִבְעִית אוֹ נִיסִּית{{=סנהדרין צח עא}}. אִם זוֹ גְּאוּלָּה טִבְעִית – [?80:40]; נִיסִּיתָ – זֶה מַשֶּׁהוּ אַחֵר. אֲבָל בִּשְׁבִיל הַמַּעֲשֶׂה שֶׁלָּנוּ אֲנִי אוֹמֵר: תִּשְׁעִים וְתִשְׁעָה פְּסִיק תִּשְׁעָה אֲחוּזִים מִן הַגְּאוּלָּה תְּלוּיִים בָּנוּ. כְּלוֹמַר, אִם אֲנַחְנוּ עוֹשִׂים גְּאוּלָּה שֶׁבֶּאֱמֶת נִשְׂמַח בָּהּ, אֲנַחְנוּ צְרִיכִים לְהִימָּצֵא בְּמַצָּב שֶׁל מָה שֶׁאוֹמְרִים חֲזַ״ל: תּוֹרָה לִשְׁמָהּ. כִּי אִם נִשְׁתַּמֵּשׁ בַּתּוֹרָה כְּדֵי לִהְיוֹת רַב, כְּדֵי לִהְיוֹת רַב רָאשִׁי, כְּדֵי לִהְיוֹת רֹאשׁ עִיר – לָקַחַת אֶת הַיַּהֲלוֹם הַיָּקָר הַזֶּה וּלְהִשְׁתַּמֵּשׁ בּוֹ כְּדֵי לִהְיוֹת רֹאשׁ מֶמְשָׁלָה...

[[81:45]] וְאִם נִהְיֶה בְּמַצָּב שֶׁל תִּיקּוּן הַמִּידּוֹת, תִּיקּוּן עַצְמֵנוּ, סַבְלָנוּת, עֲנָוָה אֲמִיתִּית וְאַהֲבָה אֲמִיתִּית לַחָבֵר, לַשָּׁכֵן, לָאִישָּׁה, לַדּוֹד וְלַדּוֹדָה – זוֹהִי הַגְּאוּלָּה הָאֲמִיתִּית. יֵשׁ פְּסִיק אֶחָד שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא עוֹשֶׂה; כָּל הַשְּׁאָר – זֶה אֲנַחְנוּ. וְהַקָּדוֹשׁ בָּרוּךְ הוּא בָּא וּמַשְׁלִים אֶת הַפְּסִיק הַזֶּה: [?82:20] יוֹסִיף עוֹד אֶחָד – אֵין לוֹ שׁוּם בְּעָיָה. אֲבָל הַכּוֹל תָּלוּי בְּכָךְ שֶׁאֲנַחְנוּ מְכִינִים אֶת הַגְּאוּלָּה הַזֹּאת. אָז זוֹ גְּאוּלָּה שֶׁבֶּאֱמֶת תִּיגַּע בָּנוּ.

[[82:40]] אֲבָל אִם הַקָּדוֹשׁ בָּרוּךְ הוּא יַעֲשֶׂה אֶת הַגְּאוּלָּה – לֹא כָּל כָּךְ נָכוֹן לוֹמַר אֶת הַמִּילִּים הָאֵלֶּה – הַקָּדוֹשׁ בָּרוּךְ הוּא גָּאַל אֶת בְּנֵי יִשְׂרָאֵל מִמִּצְרַיִם, וְעָשָׂה מֵעַם יִשְׂרָאֵל מָה שֶׁיָּכוֹל הָיָה לְהָפִיק מִמֶּנּוּ; אַחַר כָּךְ נַסְבִּיר אֵיךְ. אֲבָל אַתֶּם יוֹדְעִים אֵיךְ הִכְשִׁיר אוֹתָם בְּמִצְרַיִם? שִׂימוּ לֵב: ה׳ לֹא בָּא לָתֵת אֶת הַתּוֹרָה לְיַעֲקֹב אָבִינוּ. קוֹדֶם הִכְשִׁיר אֶת בְּנֵי יִשְׂרָאֵל בְּשִׁעְבּוּד, וְאַחַר כָּךְ בָּאָה הַגְּאוּלָּה. וְאֶת הַתּוֹצָאָה – אֲנַחְנוּ יוֹדְעִים מָה הָיְתָה. אֲבָל אִילּוּ הָיוּ יִשְׂרָאֵל מַשְׁלִימִים אֶת מָה שֶׁאֲנַחְנוּ מְדַבְּרִים עָלָיו עַכְשָׁיו – הַכּוֹל הָיָה [?83:45].

## "אִם יֵשׁ עֵץ חַיִּים"

[[83:50]] כְּמוֹ שֶׁאָמַרְנוּ בַּשִּׂיחָה הַקּוֹדֶמֶת, בַּשָּׁבוּעַ שֶׁעָבַר: מֹשֶׁה – כְּחַמָּה, וִיהוֹשֻׁעַ – כַּלְּבָנָה{{=בבא בתרא עה עא}}. מָה אוֹמֵר מֹשֶׁה? אִם יֵשׁ עֵץ חַיִּים – אֲנִי בָּא; וְאִם אֵין עֵץ חַיִּים... מָה זֶה עֵץ חַיִּים? חַיֵּי הַנֶּצַח. וְאִילּוּ הָיוּ חַיֵּי נֶצַח –

[מן הקהל:] זֶה מִן הַזּוֹהַר, נָכוֹן?

[[84:20]] כֵּן, זֶה מִן הַזּוֹהַר.

[[84:31]] מֹשֶׁה אוֹמֵר: אִם יֵשׁ חַיֵּי נֶצַח, אִם יֵשׁ עֵץ חַיִּים – אֲנִי בָּא, אֲנִי נִכְנָס. ה׳ אָמַר לוֹ לְהִיכָּנֵס? לֹא בְּתוֹר מֹשֶׁה שֶׁל הַמִּדְבָּר. אֲבָל אִם יֵשׁ עֵץ חַיִּים – עֵץ חַיִּים זֶה מַשֶּׁהוּ אַחֵר. וְזֶה, רַבּוֹתַי, עֵץ הַחַיִּים שֶׁחָסֵר לָנוּ. אֲנַחְנוּ חוֹשְׁבִים שֶׁבָּאנוּ לְאֶרֶץ יִשְׂרָאֵל כְּדֵי לִחְיוֹת כְּמוֹ בְּאָמֶרִיקָה. זוֹ הַדֵּעָה שֶׁל עַם יִשְׂרָאֵל כָּאן: לִחְיוֹת בְּרָמַת חַיִּים שֶׁל אָמֶרִיקָה. אַתֶּם מְבִינִים אוֹתִי? אֲבָל אִם יֵשׁ עֵץ חַיִּים – אִם זֶה כְּדֵי לִחְיוֹת בְּאֶרֶץ זָבַת חָלָב וּדְבַשׁ.

[[85:30]] וְעַל זֶה אֲנַחְנוּ צְרִיכִים לְהִתְפַּלֵּל, בְּעֶזְרַת ה׳ יִתְבָּרַךְ, שֶׁיִּהְיֶה טוֹב; שֶׁנַּצְלִיחַ לִחְיוֹת אֶת חַיֵּי אֶרֶץ יִשְׂרָאֵל, חַיֵּי אֶרֶץ יִשְׂרָאֵל הַקְּדוֹשָׁה – חַיִּים שֶׁל עֵץ הַחַיִּים. אָמֵן, כֵּן יְהִי רָצוֹן.

[[85:52]] ⟦sefaria|רַבִּי חֲנַנְיָא בֶּן עֲקַשְׁיָא אוֹמֵר, רָצָה הַקָּדוֹשׁ בָּרוּךְ הוּא לְזַכּוֹת אֶת יִשְׂרָאֵל, לְפִיכָךְ הִרְבָּה לָהֶם תּוֹרָה וּמִצְוֹת, שֶׁנֶּאֱמַר ה׳ חָפֵץ לְמַעַן צִדְקוֹ יַגְדִּיל תּוֹרָה וְיַאְדִּיר:⟧{{=משנה מכות ג, טז}}.

## חֲתִימָה: לְרוֹמֵם אֶת עַצְמוֹ

[[89:03]] תַּקְשִׁיבוּ: אָדָם עִם בְּעָיוֹת [?89:10] – אִם הוּא מִתְרוֹמֵם, הוּא מְרוֹמֵם אֶת עַצְמוֹ; וְאִם הוּא נִשְׁאָר כָּךְ – זוֹ הָעֲנָוָה שֶׁל בַּעֲלֵי הַחַיִּים. אָדָם צָרִיךְ לְרוֹמֵם אֶת עַצְמוֹ כָּל הַזְּמַן כְּלַפֵּי מַעְלָה, וְלֹא לָתֵת לְעַצְמוֹ אֲפִילּוּ שְׁנִיָּה אַחַת [?89:40].
