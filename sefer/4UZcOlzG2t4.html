דרשות הרב חיים דרעי שליט״א / ב׳ כי ביום הזה יכפר עליכם לטהר אתכם מכל חטאתיכם לפני ה׳ תטהרו
    דְּרָשָׁה ב׳

    כִּי־בַיּוֹם הַזֶּה יְכַפֵּר עֲלֵיכֶם לְטַהֵר אֶתְכֶם מִכֹּל חַטֹּאתֵיכֶם לִפְנֵי ה׳ תִּטְהָרוּ

    נֶאֶמְרָה כ׳ בחשוון ה׳תשפ״ו
    לחיצה על הזמן שליד פסקה משמיעה אותה מפי הרב; הפסקה הנשמעת מסומנת.

    לִלְמוֹד אֶת אַבְרָהָם אָבִינוּ

    אָנוּ עֲסוּקִים בְּפָרָשִׁיּוֹתָיו שֶׁל אַבְרָהָם אָבִינוּ עָלָיו הַשָּׁלוֹם, וְנָכוֹן לְנַצֵּל אֶת הַפָּרָשָׁה הָאַחֲרוֹנָה שֶׁבָּהּ אַבְרָהָם אָבִינוּ עֲדַיִן חַי – פָּרָשַׁת חַיֵּי שָׂרָה. אֲנִי מְחַיֵּךְ, וְלָמָּה? כִּי אֶת הָאִישׁ הַזֶּה, אַבְרָהָם אָבִינוּ, אֲנַחְנוּ צְרִיכִים לִלְמוֹד.

    לִלְמוֹד אוֹתוֹ כְּמוֹ שֶׁלּוֹמְדִים סוּגְיָה בַּגְּמָרָא, כְּמוֹ שֶׁלּוֹמְדִים זוֹהַר, כְּמוֹ שֶׁלּוֹמְדִים הִיסְטוֹרְיָה, גֵּאוֹגְרַפְיָה, פְּסִיכוֹלוֹגְיָה וּמִשְׁפָּט. הוּא כְּלָל הַכּוֹל: מַצְבִּיא גָּדוֹל, לוֹחֵם גָּדוֹל, עוֹרֵךְ דִּין גָּדוֹל – וְעִיקַּר הָעִיקָּרִים, הֶחָבֵר הֲכִי טוֹב שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא.

    חֲמִישָּׁה קִנְיָנִים יֵשׁ לַקָּדוֹשׁ בָּרוּךְ הוּא בְּעוֹלָמוֹאבות ו, י, וְאֶחָד מֵהֶם הוּא אַבְרָהָם. מָה פֵּירוּשׁ "קִנְיָן"? נֶכֶס. אֲבָל אִי אֶפְשָׁר לְהִסְתַּפֵּק בַּהַגְדָּרָה הַזֹּאת, שֶׁאַבְרָהָם אָבִינוּ הוּא נֶכֶס שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא. בְּעֶצֶם, אִם הַקָּדוֹשׁ בָּרוּךְ הוּא בָּרָא אֶת כָּל הָעוֹלָם רַק בִּשְׁבִיל אַבְרָהָם אָבִינוּ – זֶה מוּבָן. וּבְכָל זֹאת, בַּמּוּשָּׂגִים, בַּהַגְדָּרוֹת וּבַנּוּסְחָאוֹת שֶׁיֵּשׁ בְּיָדֵינוּ, בַּשָּׂפָה שֶׁלָּנוּ: מִי הוּא אַבְרָהָם אָבִינוּ? הוּא הָאָדָם שֶׁדַּרְכּוֹ מְבִינִים בִּשְׁבִיל מָה וּבִשְׁבִיל מִי בָּרָא הַקָּדוֹשׁ בָּרוּךְ הוּא אֶת הָעוֹלָם הַזֶּה. הוּא בָּא לְלַמֵּד אוֹתָנוּ מַהִי תַּכְלִית הַבְּרִיאָה.

    לָכֵן מִי שֶׁלּוֹמֵד אֶת אַבְרָהָם אָבִינוּ – צָרִיךְ לִכְתּוֹב סֵפֶר עַל אַבְרָהָם אָבִינוּ, וְאָז נָבִין מָה הַמַּטָּרָה שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא בַּבְּרִיאָה הַזֹּאת, כַּאֲשֶׁר אָמַר "נַעֲשֶׂה אָדָם בְּצַלְמֵנוּ כִּדְמוּתֵנוּ"בראשית א, כו.

    רַבִּי חַיִּים קַנְיֶבְסְקִי זֵכַר צַדִּיק לִבְרָכָה הָיָה כּוֹתֵב סְפָרִים. הוּא לֹא יָשַׁב יוֹם אוֹ יוֹמַיִם עַל סֵפֶר – כָּל הַתּוֹרָה כּוּלָּהּ הָיְתָה בְּרֹאשׁוֹ, וְלֹא הָיָה לוֹ קוֹשִׁי לִכְתּוֹב סֵפֶר וְעוֹד סֵפֶר. כָּל שָׁנָה הָיָה מְסַיֵּם אֶת כָּל הַתּוֹרָה, וְאִם נִשְׁאַר לוֹ זְמַן, כָּתַב סֵפֶר. כִּמְדוּמָּנִי שֶׁכָּתַב סֵפֶר עַל אָדָם הָרִאשׁוֹן. אוּלַי ה׳ יְזַכֶּה אוֹתָנוּ וְנִכְתּוֹב עַל אַבְרָהָם אָבִינוּ.

    בִּשְׁבִיל אַבְרָהָם נִבְרָא הָעוֹלָם

    זוֹהִי הַמַּטָּרָה שֶׁל הַבְּרִיאָה: אָדָם כָּזֶה – אַבְרָהָם אָבִינוּ. הַמַּלְאָכִים הִתְנַגְּדוּ לִבְרִיאַת הָאָדָם, וְאָמְרוּ לַקָּדוֹשׁ בָּרוּךְ הוּא: מָה אַתָּה הוֹלֵךְ לִבְרוֹא אָדָם? הִסְתַּכֵּל מָה יִהְיֶה: הַמַּבּוּל, דּוֹר הַפְלָגָה – בִּשְׁבִיל זֶה אַתָּה בּוֹרֵא אֶת הָעוֹלָם? וְהַקָּדוֹשׁ בָּרוּךְ הוּא הָיָה, כִּבְיָכוֹל, "בְּלַחַץ" בְּמֶשֶׁךְ אַלְפַּיִם שָׁנָה. כָּל הַחוּרְבָּנוֹת בָּאוּ בְּאַלְפַּיִם הַשָּׁנִים הָאֵלֶּה: קַיִן וְהֶבֶל, דּוֹר אֱנוֹשׁ שֶׁעָבְדוּ עֲבוֹדָה זָרָה, אַחַר כָּךְ הַמַּבּוּל, וְאַחַר כָּךְ דּוֹר הַפְלָגָה. כִּבְיָכוֹל – הֵם צוֹדְקִים. וּכְשֶׁבָּא אַבְרָהָם אָבִינוּ עָלָיו הַשָּׁלוֹם – זֶהוּ. אָמַר הַקָּדוֹשׁ בָּרוּךְ הוּא: הִנֵּה, בּוֹאוּ תִּרְאוּ! בִּשְׁבִיל זֶה.

    לָכֵן צָרִיךְ לִלְמוֹד, לִלְמוֹד אֶת אַבְרָהָם אָבִינוּ. הַיּוֹם כּוֹתְבִים, בָּרוּךְ ה׳, בִּיּוֹגְרַפְיָה עַל כָּל צַדִּיק וְצַדִּיק, אֲבָל עַל אַבְרָהָם אָבִינוּ עוֹד אַף אֶחָד לֹא כָּתַב סֵפֶר. זֶה בֶּאֱמֶת לֹא יָפֶה.

    עוֹרֵךְ הַדִּין שֶׁל סְדוֹם

    הוּא הָיָה עוֹרֵךְ הַדִּין שֶׁל סְדוֹם וַעֲמוֹרָה. זֶה כְּמוֹ מִי שֶׁהַיּוֹם הוּא עוֹרֵךְ הַדִּין שֶׁל מְחַבְּלֵי הַנּוּחְ׳בָּה. כָּתוּב: "וְאַנְשֵׁי סְדֹם רָעִים וְחַטָּאִים לַה׳ מְאֹד"בראשית יג, יג – מָה אַתָּה הוֹלֵךְ לְהָגֵן עֲלֵיהֶם? מִי יָכוֹל לוֹמַר לְבוֹרֵא עוֹלָם "הַאַף תִּסְפֶּה צַדִּיק עִם־רָשָׁע"בראשית יח, כג, "חָלִלָה לְּךָ מֵעֲשֹׂת כַּדָּבָר הַזֶּה לְהָמִית צַדִּיק עִם־רָשָׁע … הֲשֹׁפֵט כָּל־הָאָרֶץ לֹא יַעֲשֶׂה מִשְׁפָּט"בראשית יח, כה? כָּךְ הוּא אָמַר.

    הִנֵּה לָכֶם עֲבוֹדָה: חַשְׁבוּ קְצָת מִי הָיָה אַבְרָהָם אָבִינוּ, שֶׁיָּכוֹל הָיָה לְדַבֵּר כָּךְ. אֲבָל אֵין זֶה פֶּלֶא, כִּי לִפְנֵי שֶׁהָלַךְ הַקָּדוֹשׁ בָּרוּךְ הוּא לַהֲפוֹךְ אֶת סְדוֹם, אָמַר: "הַמְכַסֶּה אֲנִי מֵאַבְרָהָם אֲשֶׁר אֲנִי עֹשֶׂה"בראשית יח, יז – כִּבְיָכוֹל, אֵינֶנִּי יָכוֹל; יֵשׁ לִי שׁוּתָּף בַּבְּרִיאָה. "וְאַבְרָהָם הָיוֹ יִהְיֶה לְגוֹי גָּדוֹל וְעָצוּם … כִּי יְדַעְתִּיו לְמַעַן אֲשֶׁר יְצַוֶּה אֶת־בָּנָיו וְאֶת־בֵּיתוֹ אַחֲרָיו וְשָׁמְרוּ דֶּרֶךְ ה׳ לַעֲשׂוֹת צְדָקָה וּמִשְׁפָּט לְמַעַן הָבִיא ה׳ עַל־אַבְרָהָם אֵת אֲשֶׁר־דִּבֶּר עָלָיו"בראשית יח, יח–יט. זֶהוּ מִשְׁפָּט מְכוֹנֵן עֲבוּרֵנוּ.

    אַבְרָהָם אָבִינוּ חַי בְּתוֹכֵנוּ

    פָּתַחְתִּי וְאָמַרְתִּי שֶׁנְּדַבֵּר עַל אַבְרָהָם אָבִינוּ כָּל זְמַן שֶׁהוּא חַי, בְּפָרָשַׁת חַיֵּי שָׂרָה, וּבַשַּׁבָּת הַבָּאָה כְּבָר לֹא. אֲבָל זוֹ טָעוּת: אַבְרָהָם אָבִינוּ חַי בְּתוֹכֵנוּ וּבְקִרְבֵּנוּ בְּכָל שָׁעָה וּבְכָל רֶגַע – בְּהַכְנָסַת הָאוֹרְחִים, בָּאֱמוּנָה שֶׁלּוֹ, בַּתְּפִילָּה שֶׁלּוֹ, בַּהֲגָנָה, בַּמִּלְחָמָה וּבַנִּיסְיוֹנוֹת שֶׁלּוֹ. וּמִלְּבַד זֶה, כָּל הַלֶּחֶם שֶׁאֲנַחְנוּ אוֹכְלִים הוּא בִּזְכוּת הָעֲקֵדָה. אִם אֲנַחְנוּ מִתְפַּרְנְסִים, אִם יֵשׁ לָנוּ אֶת אֶרֶץ יִשְׂרָאֵל – בִּזְכוּת אַבְרָהָם אָבִינוּ.

    כְּשֶׁיְּהוּדִי מִתְלַבֵּט אִם לְקַבֵּל אוֹרְחִים אוֹ לֹא, כַּמָּה אוֹרְחִים וּמָה יוֹצִיא עֲלֵיהֶם – מָה מְשַׁכְנֵעַ אוֹתוֹ? לֹא הַשּׁוּלְחָן עָרוּךְ וְלֹא הִלְכוֹת הַכְנָסַת אוֹרְחִים. אַבְרָהָם אָבִינוּ – זֶה מַסְפִּיק. לֹא כּוּלָּם יוֹדְעִים שׁוּלְחָן עָרוּךְ, אֲבָל בְּכָל יְהוּדִי חַי אַבְרָהָם אָבִינוּ. בְּעֶזְרַת ה׳ תִּהְיֶה לָנוּ הִזְדַּמְּנוּת לְהִתְבּוֹנֵן בַּיְּצִירָה הַזֹּאת שֶׁנִּקְרֵאת אַבְרָהָם אָבִינוּ – גַּם בַּשְּׁאִיפוֹת שֶׁלָּנוּ, בַּתַּחֲנָה הַסּוֹפִית שֶׁלָּנוּ לִקְרַאת הַגְּאוּלָּה.

    הַגְּאוּלָּה – לְהַשִּׂיג אֶת מַעֲלַת הָאָבוֹת

    אוֹמֵר הָרַמְבָּ״ן עָלָיו הַשָּׁלוֹם, שֶׁיִּשְׂרָאֵל יִהְיוּ בַּגָּלוּת עַד שֶׁיַּגִּיעוּ לְמַעֲלַת אֲבוֹתֵיהֶם. כְּלוֹמַר, הַתַּחֲנָה הָאַחֲרוֹנָה שֶׁלָּנוּ, תַּחֲנַת הַגְּאוּלָּה, פֵּירוּשָׁהּ לְהַשִּׂיג אֶת מַעֲלַת הָאָבוֹת הַקְּדוֹשִׁים.

    כְּשֶׁמְּדַבְּרִים עַל מַעֲלַת הָאָבוֹת – אֲנַחְנוּ לֹא יוֹדְעִים אֵיךְ לְפָרֵשׁ אוֹתָהּ. הָאֱמֶת, לֹא רַק אֶת מַעֲלַת הָאָבוֹת: גַּם אֶת מַעֲלָתוֹ שֶׁל רַבֵּנוּ עוֹבַדְיָה יוֹסֵף אֲנַחְנוּ לֹא יוֹדְעִים לְהַסְבִּיר. אֵינֶנּוּ יוֹדְעִים לְהַסְבִּיר אֶת מַעֲלַת הָאוֹר הַחַיִּים הַקָּדוֹשׁ, אֶת מַעֲלַת הַחִידָ״א עָלָיו הַשָּׁלוֹם, אֶת הַשּׁוּלְחָן עָרוּךְ, אֶת הָרַמְבָּ״ם. צְרִיכִים לָדַעַת שֶׁהֵם נִמְצָאִים בְּעוֹלָמוֹת אֲחֵרִים. נָכוֹן, הֵם בְּנֵי אָדָם, הֵם הִתְהַלְּכוּ בְּקִרְבֵּנוּ, הָיוּ אֶצְלֵנוּ, דִּיבַּרְנוּ וְשׂוֹחַחְנוּ אִיתָּם. כַּמָּה פְּעָמִים אָמַרְתִּי דִּבְרֵי תּוֹרָה לִפְנֵי רַבֵּנוּ עוֹבַדְיָה יוֹסֵף עָלָיו הַשָּׁלוֹם – בְּבֵיתוֹ, אוֹ בַּיְּשִׁיבָה שֶׁבָּהּ לָמַדְתִּי כְּשֶׁהָיָה מוּזְמָן אֵלֶיהָ. וּבְכָל זֹאת –

    אָדָם שֶׁמִּתְהַלֵּךְ עַל אוֹתָהּ מִדְרָכָה שֶׁאַתָּה מִתְהַלֵּךְ עָלֶיהָ – חַי בְּעוֹלָם אַחֵר. זֶה בָּרוּר. גַּם כְּשֶׁדִּיבַּרְתָּ אִיתָּם וְשָׁאַלְתָּ אוֹתָם שְׁאֵלָה, הַתְּשׁוּבָה שֶׁנָּתְנוּ לְךָ לֹא הָיְתָה תְּשׁוּבָה שֶׁל הָעוֹלָם הַזֶּה; הִיא בָּאָה מִן הַמַּדְרֵגָה שֶׁהֵם נִמְצָאִים בָּהּ. לָכֵן אֵינְךָ מֵבִין מָה הֵם אוֹמְרִים. אֲבָל אִם תִּשְׁמַע בְּקוֹלָם – שָׁמַעְתָּ בְּקוֹל הַשְּׁכִינָה. אֶת זֶה צָרִיךְ לָדַעַת.

    בֶּן הַיְּשִׁיבָה וְהַחַיָּל

    כְּמוֹ שֶׁאֵינֶנּוּ מְבִינִים הַיּוֹם בַּחוּר יְשִׁיבָה שֶׁאוֹמֵר שֶׁהוּא מוּכָן לָמוּת וְלֹא לְהִתְגַּיֵּס לַצָּבָא. אֲנַחְנוּ לֹא מְבִינִים אֶת זֶה; זֶה לֹא בְּגֶדֶר הֲבָנָה. וְלָכֵן קָשֶׁה, קָשֶׁה לְהַסְבִּיר לְמִישֶׁהוּ שֶׁתַּלְמִיד הַיְּשִׁיבָה הוּא הַמֵּגֵן עַל הַחַיָּל, וְלֹא הַחַיָּל מֵגֵן עָלָיו.

    וּבְאוֹתָהּ נְשִׁימָה – תָּבִינוּ טוֹב, כִּי אֵלֶּה דְּבָרִים אַקְטוּאָלִיִּים – אֲנִי אוֹמֵר כָּאן לָאַבְרֵכִים, לָאַבְרֵכִים שֶׁלּוֹמְדִים כָּאן, לָאַבְרֵכִים שֶׁלִּי: אֵין שׁוּם הִיגָּיוֹן וְאֵין שׁוּם הֲבָנָה בְּכָךְ שֶׁאַתָּה יוֹשֵׁב וְלוֹמֵד בַּכּוֹלֵל, וְחַיָּל שָׁם מֵגֵן עָלֶיךָ בְּגוּפוֹ. בְּאוֹתָהּ נְשִׁימָה. אָז מָה אֲנִי רוֹצֶה? אֲנִי רוֹצֶה שֶׁתֵּשֵׁב וְתִלְמַד, וְשֶׁמְּסִירוּת הַנֶּפֶשׁ שֶׁלְּךָ עַל הַתּוֹרָה תִּהְיֶה גְּבוֹהָה יוֹתֵר מִמְּסִירוּת הַנֶּפֶשׁ שֶׁל הַחַיָּל. זֶה לֹא פָּשׁוּט, זֶה לֹא פָּשׁוּט.

    שִׁיבַת צִיּוֹן: "וְהַלְוִיִּם מְבִינִים אֶת־הָעָם"

    וּמֵאוֹתָהּ מַעֲלָה שֶׁדִּיבַּרְנוּ עָלֶיהָ בַּשָּׁבוּעַ שֶׁעָבַר – שִׁיבַת צִיּוֹן הָרִאשׁוֹנָה, אַחֲרֵי גָּלוּת בָּבֶל, בְּסוֹף שִׁבְעִים הַשָּׁנָה שֶׁנִּגְזְרוּ עֲלֵיהֶם. בָּאוּ וּבִיקְּשׁוּ מֵעֶזְרָא לִקְרוֹא לִפְנֵיהֶם מִסֵּפֶר הַתּוֹרָה, "מִן־הָאוֹר עַד־מַחֲצִית הַיּוֹם"נחמיה ח, ג – כָּךְ כָּתוּב בַּפָּסוּק שֶׁלָּמַדְנוּ. מֵאוֹר הַיּוֹם וְעַד הַצָּהֳרַיִם עָמַד עַל מִגְדָּל, וְהֵם עוֹמְדִים וְשׁוֹמְעִים אֶת עֶזְרָא קוֹרֵא: "בְּרֵאשִׁית בָּרָא אֱלֹהִים אֵת הַשָּׁמַיִם וְאֵת הָאָרֶץ"בראשית א, א, "וַיְדַבֵּר ה׳ אֶל מֹשֶׁה לֹאמַר", "וְעָשׂוּ לָהֶם צִיצִת"במדבר טו, לח, "אֶת־הַכֶּבֶשׂ אֶחָד תַּעֲשֶׂה בַבֹּקֶר"במדבר כח, ד, "וּבֶעָשׂוֹר לַחֹדֶשׁ הַשְּׁבִיעִי הַזֶּה"במדבר כט, ז, "וּבַחֹדֶשׁ הַשְּׁבִיעִי בְּאֶחָד לַחֹדֶשׁ … יוֹם תְּרוּעָה יִהְיֶה לָכֶם"במדבר כט, א, "וּבַחֲמִשָּׁה עָשָׂר יוֹם לַחֹדֶשׁ הַשְּׁבִיעִי"במדבר כט, יב – חַג הַסּוּכּוֹת. וְהֵם שׁוֹמְעִים וּבוֹכִים, שׁוֹמְעִים וּבוֹכִים.

    וְעוֹמְדִים שָׁם הַלְּוַיִם, וְכָתוּב שָׁם הַמִּילָּה שֶׁכָּל כָּךְ דִּיבַּרְנוּ עָלֶיהָ: "וְהַלְוִיִּם מְבִינִים אֶת־הָעָם"נחמיה ח, ז. מָה פֵּירוּשׁ "מְבִינִים אֶת־הָעָם"? הוֹפְכִים אוֹתָם לִמְבִינִים. וּבָרוּךְ ה׳, שָׂמַחְנוּ בַּפָּסוּק "וְהַמַּשְׂכִּלִים יָבִינוּ"דניאל יב, י.

    "וְהִרְשִׁיעוּ רְשָׁעִים וְלֹא יָבִינוּ כָּל־רְשָׁעִים"

    מָתַי נֶאֱמַר הַפָּסוּק הַזֶּה? בָּרֶגַע הַקְּרִיטִי בְּיוֹתֵר: מַלְאָךְ אֶחָד שׁוֹאֵל אֶת לְבוּשׁ הַבַּדִּים, הוּא גַּבְרִיאֵל, "עַד־מָתַי קֵץ הַפְּלָאוֹת"דניאל יב, ו. וְדָנִיֵּאל שׁוֹמֵעַ, וְהַתְּשׁוּבָה – כַּמָּה מִילִּים שֶׁאֵינוֹ מֵבִין מֵהֶן דָּבָר. אוֹמֵר דָּנִיֵּאל: "וַאֲנִי שָׁמַעְתִּי וְלֹא אָבִין"דניאל יב, ח. וְאָז אוֹמֵר לוֹ הַמַּלְאָךְ: "לֵךְ דָּנִיֵּאל כִּי־סְתֻמִים וַחֲתֻמִים הַדְּבָרִים עַד־עֵת קֵץ"דניאל יב, ט, "וְהַמַּשְׂכִּלִים יָבִינוּ"דניאל יב, י.

    תָּאֲרוּ לָכֶם: כָּל הָעִנְיָן הוּא זֶה. לֹא כָּתוּב שָׁם "וְהַצַּדִּיקִים יָבִינוּ", לֹא כָּתוּב "וְהַקְּדוֹשִׁים יָבִינוּ" – "וְהַמַּשְׂכִּלִים יָבִינוּ". מִי הֵם הַמַּשְׂכִּילִים? הַפְּרוֹפֵסוֹרִים? לֹא. וְעַל הָאֲחֵרִים כָּתוּב: "וְהִרְשִׁיעוּ רְשָׁעִים וְלֹא יָבִינוּ כָּל־רְשָׁעִים". הַאִם זֶה לֹא מָה שֶׁרָאִינוּ הַשָּׁבוּעַ? בִּמְקוֹם לַעֲשׂוֹת "וְתַשְׁלִיךְ" לָעֲווֹנוֹת – עָשְׂתָה "וְתַשְׁלִיךְ" לַטֶּלֶפוֹן.

    [מן הקהל:] כְּבוֹד הָרַב, גַּם דָּנִיֵּאל עַצְמוֹ לֹא הֵבִין – אָמְרוּ לוֹ "לֵךְ דָּנִיֵּאל כִּי־סְתֻמִים וַחֲתֻמִים הַדְּבָרִים עַד־עֵת קֵץ".

    נָכוֹן. וּבַסּוֹף כָּתוּב שָׁם חֶשְׁבּוֹן שֶׁל שָׁנִים. זֶה מָה שֶׁאָמְרוּ לוֹ. הַאִם הֵבִין זֹאת בַּסּוֹד? יָכוֹל לִהְיוֹת. אֲבָל אָמְרוּ לוֹ: לֵךְ, שְׁתוֹק – "וְהַמַּשְׂכִּלִים יָבִינוּ". וְהִיא – "וְתַשְׁלִיךְ" אֶת הַטֶּלֶפוֹן לַיָּם. צוֹלְלָנִים, שְׁנֵי מִילְיוֹן, אֵירוּעַ עוֹלָמִי! זֶה כָּל כָּךְ מְגוּחָךְ.

    אִם זוֹ לֹא הוֹכָחָה! עַכְשָׁיו תַּגִּידוּ עַל הַמִּסְכֵּנָה הַזֹּאת שֶׁהִיא לֹא דָּתִיָּה, שֶׁהִיא לֹא מְבִינָה, שֶׁהִיא טִיפְּשָׁה? לֹא. הַחִילּוֹנִים – הֵם לֹא חִילּוֹנִים. הֵם, מְחִילָה מִכְּבוֹדָם, רְשָׁעִים: "וְהִרְשִׁיעוּ רְשָׁעִים" – לְהַרְשִׁיעַ עוֹד יוֹתֵר.

    וּכְשֶׁאֲנַחְנוּ קוֹרְאִים בַּפָּסוּק "צִיּוֹן בְּמִשְׁפָּט תִּפָּדֶה"ישעיהו א, כז – פֵּירוּשׁ הַדָּבָר שֶׁאִם הַגְּאוּלָּה עֲדַיִן לֹא הִגִּיעָה, זֶה בִּגְלַל הַמִּשְׁפָּט: בִּגְלַל בָּגָ״ץ, הַפְּרַקְלִיטוּת וְהַמִּשְׁטָרָה. וְעַכְשָׁיו לְמִי נָתְנוּ אֶת הַטֶּלֶפוֹן? לַמִּשְׁטָרָה. שָׁם תִּהְיֶה קְבוּרָתוֹ; הַמִּשְׁטָרָה כְּבָר תֵּדַע אֵיךְ לִקְבּוֹר אֶת הַטֶּלֶפוֹן הַזֶּה, "וְגַם כָּל־הָעָם הַזֶּה עַל־מְקֹמוֹ יָבֹא בְשָׁלוֹם"שמות יח, כג.

    זֶה חָמוּר מְאוֹד. כְּשֶׁמְּדוּבָּר בַּעֲבַרְיָנִים רְגִילִים – הַכּוֹל מוּבָן. אֲבָל כְּשֶׁהָעֲבַרְיָנוּת נִכְנֶסֶת לַמִּסְגֶּרֶת שֶׁל הַחוֹק, אוֹ שֶׁל הַכּוֹחַ – וְיֵשׁ לָהֶם כּוֹחַ – זוֹ סַכָּנָה גְּדוֹלָה.

    "חֶדְוַת ה׳ הִיא מָעֻזְּכֶם"

    וּמָה אוֹמְרִים הַמְּבִינִים לְכָל הָעָם? "לְכוּ אִכְלוּ מַשְׁמַנִּים וּשְׁתוּ מַמְתַקִּים … כִּי־קָדוֹשׁ הַיּוֹם לַאֲדֹנֵינוּ וְאַל־תֵּעָצֵבוּ כִּי־חֶדְוַת ה׳ הִיא מָעֻזְּכֶם"נחמיה ח, י. זֶה הָיָה בְּרֹאשׁ הַשָּׁנָה. וְאֵיךְ בְּדִיּוּק הָפְכוּ אוֹתָם הַלְּוִיִים לִמְבִינִים? חֲבָל שֶׁלֹּא כָּתוּב שָׁם. אֲבָל נִזְרְקָה שָׁם הַמִּילָּה הַזֹּאת: "כִּי־קָדוֹשׁ הַיּוֹם לַאֲדֹנֵינוּ וְאַל־תֵּעָצֵבוּ כִּי־חֶדְוַת ה׳ הִיא מָעֻזְּכֶם".

    הָעָם מוֹסִיפִים לִבְכּוֹת עוֹד וָעוֹד, וְהַמְּבִינִים אוֹמְרִים לָהֶם: אַל תֵּבְכּוּ, כִּי קָדוֹשׁ הַיּוֹם לְאֱלֹהֵינוּ. אֲבָל הֵם בּוֹכִים, כִּי שִׁבְעִים שָׁנָה לֹא שָׁמְעוּ קְרִיאַת הַתּוֹרָה, שִׁבְעִים שָׁנָה לֹא בָּנוּ סוּכָּה, שִׁבְעִים שָׁנָה לֹא שָׁמְרוּ שַׁבָּת. לָמָּה? כִּי הָיוּ בַּגָּלוּת, וְנִטְמְעוּ שָׁם. נִטְמְעוּ, נָשְׂאוּ נָשִׁים נוֹכְרִיּוֹת – כְּמוֹ בְּרוּסְיָה, שִׁבְעִים שָׁנָה שֶׁל קוֹמוּנִיזְם. כָּךְ הָיָה. רַק אַחֲרֵי הַמַּעֲמָד הַזֶּה הָלְכוּ וְחִיפְּשׂוּ לָהֶם סוּכּוֹת, כָּךְ כָּתוּב בְּסֵפֶר נְחֶמְיָהנחמיה ח, יד–יז.

    אוֹמְרִים לָאָדָם: תִּשְׂמַח – וְהוּא בּוֹכֶה. אוֹמֵר: מִמָּה אֶשְׂמַח? בֶּאֱמֶת, זֶה סוֹד גָּדוֹל. אֵין לִי כּוֹחַ לִשְׂמוֹחַ, אֵין לִי בְּמָה לִשְׂמוֹחַ.

    בָּא הַנָּבִיא וְאוֹמֵר: אַתָּה תִּשְׂמַחַ, וְהַשִּׂמְחָה תִּיתֵּן לְךָ כּוֹחַ – אִם הַשִּׂמְחָה הִיא "חֶדְוַת ה׳". לַשִׂמְחָה יֵשׁ כַּמָּה לְשׁוֹנוֹת וְכַמָּה מַדְרֵגוֹת – שָׂשׂוֹן וְשִׂמְחָה, חֶדְוָה, גִּילָה, רִינָּה, דִּיצָה – כָּל שֶׁבַע הַבְּרָכוֹת. אֲבָל "חֶדְוָה" הִיא מִלְּשׁוֹן "חַד": יֵשׁ שִׂמְחָה שֶׁהִיא חוֹתֶכֶת וְנוֹקֶבֶת. כְּשֶׁהַחֶדְוָה הַזֹּאת הִיא "חֶדְוַת ה׳", הִיא תִּיתֵּן לְךָ כּוֹחַ לִשְׂמוֹחַ; הִיא הַמָּעוז שֶׁלְּךָ. פָּחוֹת מִזֶּה – פָּחוֹת כּוֹחַ, פָּחוֹת אֵנֶרְגִּיָּה. אִילּוּ יָכוֹלְתִּי לוֹמַר אֶת הַכּוֹל בְּבַת אַחַת, כְּמוֹ הַקָּדוֹשׁ בָּרוּךְ הוּא – הָיָה טוֹב מְאוֹד.

    "עַכְּבוּ עִמִּי" – שְׁמִינִי עֲצֶרֶת

    הַשִּׂיחוֹת שֶׁלָּנוּ הֵן שִׂיחוֹת שֶׁל "עַכְּבוּ עִמִּי". אָמַרְנוּ בַּשָּׁבוּעַ שֶׁעָבַר אֶת אוֹתוֹ מִדְרָשׁ שֶׁנֶּאֱמַר עַל "בַּיּוֹם הַשְּׁמִינִי עֲצֶרֶת תִּהְיֶה לָכֶם"במדבר כט, לה. הַיּוֹם הַשְּׁמִינִי שֶׁל סוּכּוֹת הוּא חַג בִּפְנֵי עַצְמוֹ, וְאֵין לוֹ שֵׁם. אֵינוֹ סוּכּוֹת – אָסוּר לָשֶׁבֶת בּוֹ בַּסּוּכָּה מִשּׁוּם "בַּל תּוֹסִיף". אָז אֵיךְ נִקְרָא הַחַג הַזֶּה? "עֲצֶרֶת".

    אוֹמְרִים רַבּוֹתֵינוּ הַקְּדוֹשִׁים – שָׁמְעוּ אֶת הַלָּשׁוֹן: מָשָׁל לַמֶּלֶךְ שֶׁזִּימֵּן אֶת בָּנָיו לִסְעוּדָהרשי ויקרא כג, לו. יָשְׁבוּ עִמּוֹ [?] יָמִים, שָׂמְחוּ, חָגְגוּ, אָכְלוּ, שָׁתוּ וְנֶהֱנוּ. בַּיּוֹם הָאַחֲרוֹן הֵכִינוּ הַבָּנִים אֶת הַמִּזְוָדוֹת לַחֲזוֹר הַבַּיְתָה. בָּא הַמֶּלֶךְ וְאָמַר לָהֶם: "בְּנִי, עַכְּבוּ עִמִּי עוֹד יוֹם אֶחָד, קָשֶׁה עָלַי פְּרֵידַתְכֶם". לָכֵן נִקְרָא הַחַג "עֲצֶרֶת" – עֲצוֹר, עַכְּבוּ.

    אֶת הַלָּשׁוֹן הַזֶּה צָרִיךְ לַחֲזוֹר עָלָיו מֵאָה פְּעָמִים, בְּלִי סוֹף: "עַכְּבוּ עִמִּי". כַּמָּה נִפְלָא הוּא.

    הָעִיכּוּבִים

    מְסוּפָּר שֶׁאֶחָד מִגְּדוֹלֵי הָאַדְמוֹ״רִים נִפְגַּשׁ עִם הַ"סַּבָּא קַדִישָׁא", רַבִּי שְׁלֹמֹה אַלְפַנְדְּרִי עָלָיו הַשָּׁלוֹם – ה׳ יְזַכֶּה אוֹתָנוּ לַעֲלוֹת פַּעַם עַל קִבְרוֹ. הוּא הָיָה גְּאוֹן עוֹלָם בַּנִּגְלֶה וּבַנִּסְתָּר, וְכָל חַכְמֵי יְרוּשָׁלַיִם, שֶׁיָּדְעוּ אֶת כָּל הַשָּׁ״ס לְעוֹמְקוֹ, הָיוּ [?] אֶצְלוֹ. אַדְמוֹ״רִים לֹא הָיָה מְקַבֵּל; יָשַׁב בְּחֶדֶר קָטָן בִּירוּשָׁלַיִם. כְּשֶׁהָיָה מוֹשִׁיט יָד, לֹא נָתַן לְנַשֵּׁק אוֹתָהּ אֶלָּא דֶּרֶךְ מִטְפַּחַת, וְרַק יְחִידֵי סְגוּלָּה זָכוּ לְהִיכָּנֵס אֵלָיו. הָיָה אָז כְּבֶן מֵאָה וְאַרְבַּע.

    אֶחָד מִגְּדוֹלֵי הָאַדְמוֹ״רִים מִחוּץ לָאָרֶץ שָׁלַח כַּמָּה מִכְתָּבִים וּבִיקֵּשׁ לְהִתְקַבֵּל אֶצְלוֹ, וּלְבַסּוֹף קִיבֵּל אוֹתוֹ. שׂוֹחֲחוּ בְּדִבְרֵי תּוֹרָה, וְלִפְנֵי הַפְּרֵידָה שָׁאַל הָאַדְמוֹ״ר: מָה עִם הַגְּאוּלָּה? אָמַר לוֹ: יֵשׁ עִיכּוּבִים. [?]

    זֶה הַמּוּשָּׂג: עִיכּוּבִים. הַמָּשִׁיחַ, כִּבְיָכוֹל, עוֹמֵד כָּל הַזְּמַן וּמְחַכֶּה, כְּמוֹ שֶׁהַגְּמָרָא מְתָאֶרֶתסנהדרין צח עא – וְעוֹד עִיכּוּב אֶחָד, וְעוֹד עִיכּוּב אֶחָד. ה׳ יְרַחֵם עָלֵינוּ. וְכָאן מְדוּבָּר בְּעִיכּוּב אַחֵר – הַ"עַכְּבוּ עַמִּי": הַקָּדוֹשׁ בָּרוּךְ הוּא אוֹמֵר לְבָנָיו "קָשֶׁה עָלַי פְּרֵידָתְכֶם". הוּא אוֹמֵר זֹאת לָנוּ, בַּמָקוֹם שֶׁאֲנַחְנוּ נֹאמַר לוֹ: קָשָׁה עָלֵינוּ פְּרֵידָתְךָ, בּוֹרֵא עוֹלָם – שֶׁיָּשַׁבְנוּ אִיתְּךָ מֵרֹאשׁ חוֹדֶשׁ אֱלוּל וְעַד עֲצֶרֶת.

    שְׁמִינִי עֲצֶרֶת וְשִׂמְחַת תּוֹרָה: שֵׁם הַחַג הוּא "עֲצֶרֶת", לֹא "שִׂמְחַת תּוֹרָה". שִׂמְחַת תּוֹרָה נִכְנְסָה מִכּוֹחַ תַּקָּנָה מְאוּחֶרֶת יוֹתֵר. פַּעַם הָיוּ מְסַיְּמִים אֶת הַתּוֹרָה בְּשָׁלוֹשׁ שָׁנִים, וַחֲבָל שֶׁזֶּה לֹא נִשְׁאַר – הָיָה מֵקֵל עָלֵינוּ בַּהֲכָנַת קְרִיאַת הַתּוֹרָה.

    מָה פֵּירוּשׁ "עַכְּבוּ עִמִּי"? הִישָּׁאֲרוּ עִמִּי, הִישָּׁאֲרוּ בָּאֲוִירָה הַזֹּאת שֶׁל עֲצֶרֶת. וְלָכֵן לְיוֹם הָעֲצֶרֶת אֵין שֵׁם; שְׁמוֹ – "עֲצוֹר".

    הַיּוֹם הַשְּׁמִינִי – מֵעַל הַטֶּבַע

    פַּעַם, בְּשִׂמְחַת תּוֹרָה לִפְנֵי שְׁנָתַיִם אוֹ שָׁלוֹשׁ, אָמַרְנוּ שֶׁבְּכָל הַמּוֹעֲדִים שֶׁעָבַרְנוּ מֵרֹאשׁ חוֹדֶשׁ אֱלוּל בָּאנוּ עִם אֶמְצָעִי כָּלְשֶׁהוּ: בְּחוֹדֶשׁ אֱלוּל – עִם סְלִיחוֹת; בְּרֹאשׁ הַשָּׁנָה – עִם שׁוֹפָר; בַּעֲשֶׂרֶת יְמֵי תְּשׁוּבָה – עִם תְּשׁוּבָה; בְּיוֹם הַכִּיפּוּרִים – עִם תַּעֲנִית; בְּסוּכּוֹת – עִם לוֹלָב וְסוּכָּה. בַּעֲצֶרֶת – אֵין כְּלוּם.

    וְהַתְּשׁוּבָה: בַּעֲצֶרֶת – "נָגִילָה וְנִשְׂמְחָה בָּךְ"שיר השירים א, ד. זֶה הַ"עַכְּבוּ עַמִּי". הָעֲצֶרֶת מְסַמֶּלֶת אֶת הַיּוֹם הַשְּׁמִינִי, וְהַיּוֹם הַשְּׁמִינִי אֵינוֹ שַׁיָּךְ לַחַיִּים שֶׁלָּנוּ. שְׁלוֹשָׁה דְּבָרִים אָנוּ חוֹגְגִים בַּיּוֹם הַשְּׁמִינִי: בְּרִית מִילָּה, חֲנוּכָּה וַעֲצֶרֶת. לָמָּה? כִּי הָעוֹלָם הַזֶּה מִסְתַּיֵּם בִּקְדוּשַּׁת הַשְּׁבִיעִי – כָּל הַשְּׁבִיעִיּוֹת מְקוּדָּשׁוֹת – וְאִילּוּ שְׁלוֹשֶׁת אֵלֶּה הֵם מֵעַל הַטֶּבַע. וּמָה הוּא מֵעַל הַטֶּבַע? הַיּוֹם שֶׁבּוֹ נִזְכֶּה לִשְׂמוֹחַ בַּקָּדוֹשׁ בָּרוּךְ הוּא. זוֹ בְּחִינַת אוֹתוֹ מֶלֶךְ שֶׁאוֹמֵר לְבָנָיו "עַכְּבוּ עַמִּי": אֵין שׁוּם סִיבָּה. חָגַגְנוּ הַכּוֹל, עָשִׂינוּ הַכּוֹל, וְאֵין סִיבָּה לַחַג הַזֶּה. הַסִּיבָּה הַיְּחִידָה הִיא – בְּלִי סִיבָּה: שְׁבוּ עִם אֲבִיכֶם.

    רֹאשׁ הַשָּׁנָה – דִּין בְּרַחֲמִים

    זֶה מַתְחִיל בְּרֹאשׁ הַשָּׁנָה – הַשּׁוֹפָר, הַתְּרוּעָה, הַתְּקִיעָה וְהַשְּׁבָרִים – וְזֶה מַמְשִׁיךְ. יִתָּכֵן שֶׁבְּעֶזְרַת ה׳ נָבִין לְמִפְרֵעַ מָה הֵם הַחַגִּים הָאֵלֶּה בִּכְלָל, כַּמּוּבָן בְּמִידָּה שֶׁאֲנַחְנוּ יְכוֹלִים לְהַסְבִּיר.

    אַתָּה בָּא עִם הַשּׁוֹפָר, וְזוֹ הַחֲגִיגָה הַגְּדוֹלָה. אִם אַתֶּם זוֹכְרִים אֶת הַדְּרָשָׁה לְרֹאשׁ הַשָּׁנָה: זֶה הַיּוֹם שֶׁבּוֹרֵא עוֹלָם עוֹשֶׂה אֶת יוֹם הַהוּלֶּדֶת שֶׁל הָעוֹלָם, אֶת יוֹם הַבְּרִיאָה. אֲנִי עוֹבֵר עַכְשָׁיו עַל הַחַגִּים בְּקִיצּוּר, כִּי דִּיבַּרְנוּ עֲלֵיהֶם הַרְבֵּה. לַחֲגִיגָה הַזֹּאת הוּא מַזְמִין אֶת כָּל בָּאֵי עוֹלָם – בַּיּוֹם שֶׁנִּבְרָא הָעוֹלָם, חַיִּים וּמֵתִים; סִפְרֵי חַיִּים וְסִפְרֵי מֵתִים נִפְתָּחִים לְפָנָיו. בָּרוּר לָמָּה בָּרָא ה׳ אֶת הָעוֹלָם: בִּשְׁבִילְךָ. לָכֵן מַזְמִינִים אוֹתְךָ לַחֲגִיגָה – אַבְרָהָם אָבִינוּ, הָאָבוֹת הַקְּדוֹשִׁים. חֲגִיגָה גְּדוֹלָה. וּמִי יוֹשֵׁב לְיַד הַמֶּלֶךְ? עַם יִשְׂרָאֵל.

    אוֹמֵר הַמִּדְרָשׁ: בְּרֹאשׁ הַשָּׁנָה יִשְׂרָאֵל וְהָאוּמּוֹת נִכְנָסִים לְדִין, וּמִי מְנַצֵּחַ? מִי שֶׁיּוֹצֵא וְלוֹלֵב בְּיָדוֹ – הַצֶּבַע הַיָּרוֹק מְסַמֵּן נִיצָּחוֹןויקרא רבה ל, ב. וְלָמָּה? אֲנַחְנוּ קוֹרְאִים לוֹ "יוֹם הַדִּין", אֲבָל בְּאוֹתָהּ הִזְדַּמְּנוּת הַקָּדוֹשׁ בָּרוּךְ הוּא עוֹשֶׂה חֲגִיגָה. וְלָמָּה הוּא עוֹשֶׂה חֲגִיגָה? כְּדֵי לְמַתֵּק אֶת הַדִּין. אִילּוּ לֹא עָשָׂה אֶת הַחֲגִיגָה הַזֹּאת וְעָשָׂה דִּין בִּלְבַד – ה׳ יִשְׁמוֹר, אַף אֶחָד לֹא הָיָה נִשְׁאָר בָּעוֹלָם, כּוֹלֵל עַם יִשְׂרָאֵל. אֲבָל כְּשֶׁהוּא מַזְמִין אֶת עַם יִשְׂרָאֵל וְאֶת הָאָבוֹת, אֶת אַבְרָהָם אָבִינוּ – הוּא אֵינוֹ יָכוֹל לִנְגּוֹעַ בְּעַם יִשְׂרָאֵל. לָמָּה? כִּי יוֹשְׁבִים שָׁם מֹשֶׁה רַבֵּנוּ וְאַבְרָהָם אָבִינוּ.

    וְכָךְ הוּא הוֹפֵךְ אֶת הַיּוֹם. לֹא לְחִינָּם הִגְדִּיר הָרַמְבָּ״ן – גְּדוֹל הַחָכְמָה וְהַקַּבָּלָה – מָה הוּא רֹאשׁ הַשָּׁנָה וּמָה הוּא יוֹם הַכִּיפּוּרִים, שֶׁלִּכְאוֹרָה הֵם אוֹתוֹ דָּבָררמבן ויקרא כג, כד. רֹאשׁ הַשָּׁנָה – דִּין בְּרַחֲמִים; יוֹם הַכִּיפּוּרִים – רַחֲמִים בְּדִין. אֵינֶנּוּ מְבִינִים אֶת הַמִּילִּים הָאֵלֶּה עַד תּוֹמָן, וְזֶה בְּסֵדֶר. אֶת רֹאשׁ הַשָּׁנָה לְפָחוֹת מְבִינִים: זֶה דִּין, אֲבָל הוּא מוּקָּף וְעָטוּף בְּרַחֲמִים. אֶת יוֹם הַכִּיפּוּרִים צָרִיךְ לְהַסְבִּיר; אַנְשֵׁי הַקַּבָּלָה יוֹדְעִים לְהַסְבִּיר זֹאת.

    יוֹם הַכִּיפּוּרִים – הַיּוֹם שֶׁנֶּעֱלַם

    וּכְשֶׁעוֹבְרִים אֶת זֶה, מַגִּיעִים לְיוֹם הַכִּיפּוּרִים – הַיּוֹם שֶׁעַד הָרֶגַע הָאַחֲרוֹן הוּא נֶעֱלַם מֵאִיתָּנוּ. מָה פֵּירוּשׁ "נֶעֱלָם מֵאִיתָּנוּ"? אַכְנִיס אֶתְכֶם קְצָת לַתְּחוּם הַזֶּה, כְּדֵי שֶׁנּוּכַל לְהָבִין מַשֶּׁהוּ.

    בַּחוֹמֶשׁ וַיִּקְרָא יֵשׁ אֶת פָּרָשַׁת אַחֲרֵי מָוֶת, וְהִיא מַתְחִילָה כָּךְ: "וַיְדַבֵּר ה׳ אֶל־מֹשֶׁה אַחֲרֵי מוֹת שְׁנֵי בְּנֵי אַהֲרֹן בְּקָרְבָתָם לִפְנֵי־ה׳ וַיָּמֻתוּ"ויקרא טז, א. תַּקְשִׁיבוּ טוֹב; אַל תִּכְעֲסוּ עָלַי שֶׁאֲנִי מַכְנִיס אֶתְכֶם לִפְרָטִים – אַדְּרַבָּה, אַחַר כָּךְ תַּגִּידוּ לִי תּוֹדָה.

    זֶה הַפָּסוּק. וְאַחֲרָיו: "דַּבֵּר אֶל־אַהֲרֹן אָחִיךָ וְאַל־יָבֹא בְכָל־עֵת אֶל־הַקֹּדֶשׁ"ויקרא טז, ב. כְּלוֹמַר: אֱמוֹר לְאַהֲרֹן אָחִיךָ שֶׁלֹּא יַחֲשׁוֹב שֶׁבֵּית הַמִּקְדָּשׁ הוּא בֵּיתוֹ, שֶׁהוּא יָכוֹל לָבוֹא אֵלָיו מָתַי שֶׁיִּרְצֶה. הַקָּדוֹשׁ בָּרוּךְ הוּא, יִשְׁתַּבַּחַ שְׁמוֹ לָעַד, אֵינוֹ נוֹשֵׂא פָּנִים לְאִישׁ; לְמֹשֶׁה עַל אַהֲרֹן הוּא אוֹמֵר אֶת הַכּוֹל בְּגָלוּי וְאֵינוֹ חוֹשֵׁשׁ. אַהֲרֹן עוֹבֵד שָׁם – וְכִי אֵין זֶה בֵּיתוֹ? זֶה בֵּיתוֹ!

    מִנְחַת הַכֹּהֵן הַגָּדוֹל – מַחֲצִיתָהּ בַּבּוֹקֶר וּמַחֲצִיתָהּ בָּעֶרֶב

    רְאוּ כַּמָּה גָּדוֹל וְחָכָם הַקָּדוֹשׁ בָּרוּךְ הוּא, וְכַמָּה הוּא אַמִּיץ – אֵינוֹ מְפַחֵד מֵאַף אֶחָד. הַכֹּהֵן הַגָּדוֹל צָרִיךְ לְהַקְרִיב בְּכָל יוֹם קָרְבַּן מִנְחָה. בְּדֶרֶךְ כְּלָל אֶת קָרְבַּן הַמִּנְחָה הַכֹּהֲנִים אוֹכְלִים, אֲבָל אֶת הַמִּנְחָה הַזֹּאת שֶׁל הַכֹּהֵן אֵין אוֹכְלִים: "כָּלִיל תָּקְטָר"ויקרא ו, טו, כְּמוֹ עוֹלֶה.

    הַכֹּהֵן הַגָּדוֹל הוּא הַמְּנַהֵל הַכְּלָלִי שֶׁל בֵּית הַמִּקְדָּשׁ. מַדְרֵגָתוֹ כְּמֶלֶךְ, וּכְבוֹדוֹ גָּדוֹל. אֵלָיו בָּאִים כָּל הַשָּׁנָה עִם הַקָּרְבָּנוֹת, עִם הָעֲווֹנוֹת. וּלְהַבְדִּיל אֶלֶף אַלְפֵי הַבְדָּלוֹת – לֹא כְּמוֹ הַכַּמָרִים, שֶׁבְּכָל יוֹם רִאשׁוֹן הַגּוֹיִים בָּאִים אֲלֵיהֶם, וְהֵם זוֹרְקִים עֲלֵיהֶם קְצָת מַיִם וְאוֹמְרִים "וְסָר עֲוֺנֶךָ וְחַטָּאתְךָ תְּכֻפָּר"ישעיהו ו, ז, וְאֶפְשָׁר לְהַמְשִׁיךְ כָּךְ עַד יוֹם רִאשׁוֹן הַבָּא.

    פֹּה אוֹמְרִים לוֹ: אַתָּה צָרִיךְ לְהַקְרִיב קָרְבָּן. אַל תַּחֲשׁוֹב שֶׁאַתָּה הַמְּנַהֵל הַכְּלָלִי. וְשִׂימוּ לֵב: קָרְבַּן מִנְחָה. מִסְתַּבֵּר שֶׁהַכֹּהֵן הַגָּדוֹל הָיָה גַּם עָשִׁיר, שֶׁנֶּאֱמַר "וְקִדַּשְׁתּוֹ"ויקרא כא, ח – הָעָם הָיָה מְקַדֵּשׁ אוֹתוֹ. לִכְאוֹרָה הָיָה רָאוּי שֶׁיָּבִיא שׁוֹר בְּכָל יוֹם, שֶׁיֵּרָאֶה כְּמוֹ מֵאַכְסַנְיַת הַיַּהֲלוֹמָנִים. לֹא: מִנְחָה, סוֹלֶת.

    וְאֵיךְ הוּא מֵבִיא אוֹתָהּ? מַחֲצִיתָהּ בַּבּוֹקֶר וּמַחֲצִיתָהּ אַחַר הַצָּהֳרַיִםויקרא ו, יג. הַמְּנַהֵל הַכְּלָלִי שֶׁל הַמִּשְׂרָדִים בָּא בַּבּוֹקֶר, מִסְתּוֹבֵב שָׁעָה, וְאַחַר כָּךְ הוֹלֵךְ הַבַּיְתָה. לַכֹּהֵן הַגָּדוֹל אוֹמְרִים: אֶת קָרְבַּן הַמִּנְחָה הַזֶּה תַּקְרִיב – חֶצְיו בַּבּוֹקֶר וְחֶצְיו אַחַר הַצָּהֳרַיִם. לָמָּה אַתֶּם מְרַתְּקִים אוֹתִי כָּךְ? זוֹ חָכְמָתוֹ שֶׁל בּוֹרֵא עוֹלָם: אַל תַּחֲשׁוֹב שֶׁזֶּה בֵּיתְךָ. הָלַכְתָּ לָנוּחַ קְצָת בַּצָּהֳרַיִם? תָּבוֹא בַּצָּהֳרַיִם לְהִתְפַּלֵּל מִנְחָה.

    "יֵשׁ לְךָ פְּגִישָׁה עִם הַקָּדוֹשׁ בָּרוּךְ הוּא"

    פַּעַם שָׁאַל אוֹתִי מִישֶׁהוּ: לָמָּה לֹא מִתְפַּלְּלִים אֶת שְׁלוֹשׁ הַתְּפִילּוֹת בְּיַחַד – שַׁחֲרִית, מִנְחָה וְעַרְבִית? וְאַחֵר שָׁאַל: לָמָּה מִתְפַּלְּלִים כָּל יוֹם? נַעֲשָׂה פַּעַם בְּחוֹדֶשׁ תְּפִילָּה אֲרוּכָּה, כְּמוֹ בְּיוֹם הַכִּיפּוּרִים.

    בְּיוֹם רִאשׁוֹן בָּא אֵלַי מִישֶׁהוּ וְשָׁאַל אִם אֶפְשָׁר לְסַיֵּם אֶת הַתְּפִילָּה בִּשְׁמוֹנֶה. שָׁאַלְתִּי: לָמָּה? אָמַר שֶׁיֵּשׁ לוֹ פְּגִישָׁה בִּשְׁמוֹנֶה. אָמַרְתִּי לוֹ: יֵשׁ לְךָ פְּגִישָׁה עַכְשָׁיו – עִם הַקָּדוֹשׁ בָּרוּךְ הוּא!

    זֶהוּ עִנְיַן מִנְחַת הַכֹּהֵן הַגָּדוֹל: מַחֲצִית הַמִּנְחָה בַּבּוֹקֶר וּמַחֲצִיתָהּ אַחַר הַצָּהֳרַיִם, שֶׁלֹּא יַחֲשׁוֹב שֶׁהוּא הַמְּנַהֵל הַכְּלָלִי, שֶׁיָּבוֹא בַּבּוֹקֶר, יַקְרִיב הַכּוֹל וְיֵלֵךְ לוֹ לַחוֹפֶשׁ.

    "בְּזֹאת יָבֹא אַהֲרֹן אֶל־הַקֹּדֶשׁ"

    אוֹמֵר הַקָּדוֹשׁ בָּרוּךְ הוּא לְמֹשֶׁה: שֶׁלֹּא יַחֲשׁוֹב אַהֲרֹן שֶׁבְּכָל רֶגַע הוּא יָכוֹל לָבוֹא. "בְּזֹאת יָבֹא אַהֲרֹן אֶל־הַקֹּדֶשׁ בְּפַר בֶּן־בָּקָר לְחַטָּאת וְאַיִל לְעֹלָה"ויקרא טז, ג. אֲגַלְגֵּל לָכֶם אֶת כָּל הַפָּרָשָׁה: זֶה קָרְבַּן הַכֹּהֵן – קָרְבַּן חַטָּאת וְקָרְבַּן עוֹלֶה. הַחַטָּאת – לְכַפֵּר עָלָיו וְעַל בֵּיתוֹ, וְכֵן הָעוֹלֶה. תַּקְשִׁיבוּ טוֹב, צָרִיךְ לַעֲקוֹב.

    זֶה הַקָּרְבָּן שֶׁל אַהֲרֹן וּבָנָיו. אַחַר כָּךְ בָּאִים הַקָּרְבָּנוֹת שֶׁל עַם יִשְׂרָאֵל: "וּמֵאֵת עֲדַת בְּנֵי יִשְׂרָאֵל יִקַּח שְׁנֵי־שְׂעִירֵי עִזִּים לְחַטָּאת"ויקרא טז, ה, "גּוֹרָל אֶחָד לַה׳ וְגוֹרָל אֶחָד לַעֲזָאזֵל"ויקרא טז, ח, הַשָּׂעִיר לַעֲזָאזֵל וְכֵן הָלְאָה – כָּל הַקָּרְבָּנוֹת וְכָל הָעֲבוֹדָה.

    בַּפָּרָשָׁה לֹא נֶאֱמַר אֵיזֶה יוֹם

    אֲבָל פֶּלֶא פְּלָאִים: לֹא אָמְרוּ לָנוּ אֵיזֶה יוֹם זֶה. בְּדֶרֶךְ כְּלָל, בְּכָל הַקָּרְבָּנוֹת הַיּוֹם מְפוֹרָשׁ: קָרְבַּן הַתָּמִיד – "אֶת־הַכֶּבֶשׂ אֶחָד תַּעֲשֶׂה בַבֹּקֶר"במדבר כח, ד, כָּל הַשָּׁנָה, קָבוּעַ; "וּבְיוֹם הַשַּׁבָּת שְׁנֵי־כְבָשִׂים בְּנֵי־שָׁנָה"במדבר כח, ט; "וּבַחֹדֶשׁ הָרִאשׁוֹן בְּאַרְבָּעָה עָשָׂר יוֹם לַחֹדֶשׁ"במדבר כח, טז – אֵלֶּה הַקָּרְבָּנוֹת שֶׁל פֶּסַח; "וּבְיוֹם הַבִּכּוּרִים"במדבר כח, כו – שָׁבוּעוֹת; "וּבַחֹדֶשׁ הַשְּׁבִיעִי בְּאֶחָד לַחֹדֶשׁ"במדבר כט, א – רֹאשׁ הַשָּׁנָה. הַחוֹדֶשׁ הַשְּׁבִיעִי, בְּמִנְיָן מִנִּיסָן שֶׁהוּא הַחוֹדֶשׁ הָרִאשׁוֹן, הוּא חוֹדֶשׁ תִּשְׁרֵי. וְאִילּוּ בַּפָּרָשָׁה הַזֹּאת אֵינֶנּוּ יוֹדְעִים בְּאֵיזֶה יוֹם מְדוּבָּר.

    הַפָּרָשָׁה מַתְחִילָה "וַיְדַבֵּר ה׳ אֶל־מֹשֶׁה אַחֲרֵי מוֹת שְׁנֵי בְּנֵי אַהֲרֹן" – אֲבָל בְּנֵי אַהֲרֹן מֵתוּ שְׁתֵי פָּרָשִׁיּוֹת קוֹדֶם, בְּפָרָשַׁת שְׁמִינִי; עָבַר זְמַן. וְכָאן, בַּיּוֹם הַזֶּה שֶׁאֵינֶנּוּ יוֹדְעִים מַהוּ, כָּתוּב "וַיְדַבֵּר ה׳ אֶל־מֹשֶׁה אַחֲרֵי מוֹת שְׁנֵי בְּנֵי אַהֲרֹן". מָה הַקֶּשֶׁר?

    [מן הקהל:] לְפִי הַפְּשָׁט מוּבָן – שֶׁלֹּא יָבוֹאוּ בְּכָל רֶגַע, כִּי לְפִי הַפְּשָׁט הֵם זִלְזְלוּ מְעַט; וְלָכֵן אוֹמְרִים לְאַהֲרֹן: שְׁמוֹר עַל שְׁנֵי בָּנֶיךָ שֶׁנּוֹתְרוּ, וְגַם אַתָּה הִיזָּהֵר.

    זֶה הַפְּשָׁט. אֲבָל בְּסוֹפוֹ שֶׁל דָּבָר – מָה הַקֶּשֶׁר? וְאָז כָּל הָעֲבוֹדָה שֶׁל אוֹתוֹ יוֹם שֶׁאֵינֶנּוּ יוֹדְעִים מַהוּ. לֹא אוֹמְרִים אֵיזֶה יוֹם – סוֹד. רַק בְּסוֹף הַפָּרָשָׁה כָּתוּב: "שַׁבַּת שַׁבָּתוֹן הִיא לָכֶם"ויקרא טז, לא, "כִּי־בַיּוֹם הַזֶּה יְכַפֵּר עֲלֵיכֶם לְטַהֵר אֶתְכֶם מִכֹּל חַטֹּאתֵיכֶם לִפְנֵי ה׳ תִּטְהָרוּ"ויקרא טז, ל.

    אֵינֶנּוּ מְבִינִים: לָמָּה לֹא כָּתְבָה הַתּוֹרָה "וּבְיוֹם הַכִּיפּוּרִים"? מָה הַקֶּשֶׁר בֵּין מִיתַת בְּנֵי אַהֲרֹן לַיּוֹם הַזֶּה? וְאִם אַתָּה רוֹצֶה לְסַפֵּר – אֱמוֹר קוֹדֶם עַל אֵיזֶה יוֹם מְדוּבָּר! רַק בְּסוֹף הַפָּרָשָׁה זֶה כָּתוּב. לָמָּה כָּל הָעֲבוֹדָה הַזֹּאת שֶׁל הַיּוֹם הַזֶּה? "כִּי־בַיּוֹם הַזֶּה יְכַפֵּר עֲלֵיכֶם". אֵינֶנִּי טוֹעֶה וְאֵינֶנִּי מַטְעֶה אֶתְכֶם, חָס וְשָׁלוֹם – פִּתְחוּ אֶת הַחוֹמֶשׁ. אַל תַּאֲמִינוּ לִי; אָסוּר לְהַאֲמִין. זוֹ הַפָּרָשָׁה.

    מְעַט גַּאֲוָה – לְתַלְמִידֵי חֲכָמִים

    וּמָה הַתְּשׁוּבָה? זֶה חִידּוּשׁ שֶׁלִּי. וְלָכֵן – מְעַט גַּאֲוָה. הַכּוֹל אֶפְשָׁר בִּמְעַט; אֲבָל גַּאֲוָה, אֲפִילּוּ מְעַט, מוּתֶּרֶת רַק לְתַלְמִידֵי חֲכָמִים. לָמָּה? כִּי אִם לֹא כֵּן, תַּלְמִידֵי הַחֲכָמִים יְזַלְזְלוּ בְּעַצְמָם.

    פַּעַם הוּזְמַן אֶחָד מִגְּדוֹלֵי הַדּוֹר לַחֲתוּנָּה, וְלֹא כִּיבְּדוּ אוֹתוֹ בְּאַף אַחַת מֵהַבְּרָכוֹת. אַחַר כָּךְ בָּאוּ אֵלָיו וְהִתְחִילוּ לְפַיֵּס אוֹתוֹ. אָמַר לָהֶם: מָה קָרָה? לֹא בָּאתִי לְהִתְכַּבֵּד, בָּאתִי לְכַבֵּד. אֲבָל הוּא גַּם אָמַר פַּעַם: אֲנַחְנוּ, הָרַבָּנִים, אֵינֶנּוּ מְבַקְּשִׁים כָּבוֹד – אֲבָל אֲנַחְנוּ מְבַקְּשִׁים שֶׁלֹּא יְזַלְזְלוּ בָּנוּ. וְזֶה נָכוֹן: אֵינְךָ מְכַבֵּד רַב – אַל תְּכַבֵּד; אֲבָל לְזַלְזֵל – אֵין לְךָ רְשׁוּת.

    בְּסוֹף הַפָּרָשָׁה כָּתוּב: "כִּי־בַיּוֹם הַזֶּה יְכַפֵּר עֲלֵיכֶם לְטַהֵר אֶתְכֶם מִכֹּל חַטֹּאתֵיכֶם לִפְנֵי ה׳ תִּטְהָרוּ". לָמָּה אֵינְךָ אוֹמֵר אֵיזֶה יוֹם זֶה? וְלָמָּה אַתָּה אוֹמֵר "בְּקִרְבָתָם"?

    מַעֲשֶׂה בִּיהוּדִי בַּמִּקְוֶה בְּטֶקְסָס

    שָׁמַעְתִּי פַּעַם סִיפּוּר עַל יְהוּדִי יִשְׂרְאֵלִי, יְהוּדִי חֲרֵדִי, שֶׁנָּסַע לְבַקֵּר בְּאַרְצוֹת הַבְּרִית וְהָיָה צָרִיךְ לְהַגִּיעַ לְטֶקְסָס. הִגִּיעַ יוֹם שִׁישִּׁי, וְהוּא שָׁאַל: יֵשׁ מִקְוֶה? מִקְוֶה בְּטֶקְסָס? אַתָּה צוֹחֵק? אוּלַי בַּקָּזִינוֹ – מַפְסִידִים חֲמִישִּׁים מִילְיוֹן בְּסִיבּוּב אֶחָד וְאַחַר כָּךְ טוֹבְלִים בְּמִקְוֶה… בְּכָל זֹאת אָמְרוּ לוֹ שֶׁיֵּשׁ מִקְוֶה. הִגִּיעַ, טָבַל וְיָצָא.

    פִּתְאוֹם הוּא רוֹאֶה אָדָם עִם כּוֹבַע שֶׁל קָאוּבּוֹי, אֶקְדָּח וְכוּלִּי. גַּם הוּא הִתְפַּשֵּׁט, יָרַד לַמִּקְוֶה, טָבַל וְעָלָה. כְּשֶׁעָמַד הַיְּהוּדִי לָלֶכֶת, יָצָא אוֹתוֹ קָאוּבּוֹי, פָּנָה אֵלָיו וְאָמַר: אֲנִי מֵבִין שֶׁאַתָּה בְּוַדַּאי רוֹצֶה לָדַעַת מִי אֲנִי וּמָה אֲנִי עוֹשֶׂה כָּאן. אַגִּיד לְךָ.

    אֲנִי שַׁיָּךְ לְמִשְׁפָּחָה חֲרֵדִית שֶׁהִגִּיעָה לְאַרְצוֹת הַבְּרִית לִפְנֵי מֵאָה שָׁנָה, וְגָדַלְתִּי בְּבַיִת חֲסִידִי. כְּשֶׁהָיִיתִי בֶּן שָׁלוֹשׁ, הָיָה אָבִי לוֹקֵחַ אוֹתִי בְּכָל לֵיל שַׁבָּת לְרַבִּי, לְ"טִישׁ". אַתֶּם יוֹדְעִים שֶׁהָרַבִּיִּים עוֹרְכִים טִישׁ בְּלֵיל שַׁבָּת, עִם אַלְפֵי חֲסִידִים, וְלִפְעָמִים הָרַבִּי מַתְחִיל אֶת הַקִּידּוּשׁ בִּשְׁתֵּים עֶשְׂרֵה וָחֵצִי בַּלַּיְלָה. (הָיִיתִי פַּעַם אֵצֶל אָבִיו שֶׁל הָאַדְמוֹ״ר הַזֶּה עָלָיו הַשָּׁלוֹם, נִיצּוֹל שׁוֹאָה, שֶׁאַחֲרֵי תְּפִילַּת עַרְבִית הָיָה רוֹקֵד שָׁעָה שְׁלֵמָה – מִסְכֵּן, עָבַר שׁוֹאָה אֲיוּמָּה. עָשִׂינוּ אִיתּוֹ שַׁבָּת לִפְנֵי זְמַן רַב.)

    אָבִי הָיָה מַכְנִיס אוֹתִי מִתַּחַת לַשּׁוּלְחָן שֶׁל הָאַדְמוֹ״ר, וְשָׁם כְּבָר הָיוּ הַרְבֵּה יְלָדִים. הָאַדְמוֹ״ר הָיָה זוֹרֵק לָנוּ מִדֵּי פַּעַם חֲתִיכַת עוּגָה וְכַדּוֹמֶה, וְנִרְדַּמְנוּ שָׁם. וּפִתְאוֹם שָׁמַעְנוּ מִישֶׁהוּ מְדַבֵּר בְּקוֹל חָזָק מְאוֹד, וְהִתְעוֹרַרְנוּ. מָה הוּא אָמַר? הוּא אָמַר לַחֲסִידָיו: הִיזָּהֲרוּ מִן הַיֵּצֶר הָרַע! לִפְעָמִים אַתֶּם הוֹלְכִים לְבֵית הַכְּנֶסֶת, וְהוּא פּוֹגֵשׁ אֶתְכֶם וְאוֹמֵר –

    בְּאָמֶרִיקָה, לְצַעֲרֵנוּ הָרַב, רוֹב הַיְּהוּדִים שֶׁהִגִּיעוּ לְשָׁם הִתְבּוֹלְלוּ – וְכָךְ גַּם בְּאֵירוֹפָּה. אֵיךְ בָּאָה הַהִתְבּוֹלְלוּת? לֹא הָיְתָה עֲבוֹדָה. רָצִיתָ לַעֲבוֹד? תַּעֲבוֹד בְּשַׁבָּת. הָיָה לָהֶם נִיסָּיוֹן גָּדוֹל מְאוֹד; בְּלִי זֶה אֵין לְךָ מָה לֶאֱכוֹל. לֹא פָּשׁוּט.

    אָז הַיֵּצֶר אוֹמֵר לְךָ: אַתָּה הוֹלֵךְ לְבֵית הַכְּנֶסֶת? [?] – מָה אִכְפַּת לְךָ? אֲנִי רוֹצֶה לְהִתְפַּלֵּל. – אֲבָל הִשְׁתַּתַּפְתָּ בִּסְעוּדָה לִפְנֵי חֲצִי שָׁעָה, אָכַלְתָּ "דָּבָר אַחֵר", חֲזִיר! – תַּגִּידוּ לַיֵּצֶר הָרַע, תַּגִּידוּ לַשָּׂטָן: אֲנִי רוֹצֶה לָלֶכֶת לָשֶׁבֶת עִם אַבָּא שֶׁלִּי חָמֵשׁ דַּקּוֹת. מָה אִכְפַּת לְךָ? – אַתָּה רוֹצֶה לָשֶׁבֶת? אֲנִי יוֹדֵעַ שֶׁבְּעוֹד חֲצִי שָׁעָה אַתָּה הוֹלֵךְ לַמִּסְעָדָה עוֹד יוֹתֵר טְרֵפָה! – תֹּאמַר לוֹ: אֲנִי רוֹצֶה לָלֶכֶת לָשֶׁבֶת עִם אַבָּא שֶׁלִּי חֲצִי שָׁעָה, לִרְאוֹת אֶת אַבָּא שֶׁלִּי – עִם הַטְּרֵפָה וְעִם הַכּוֹל. כָּךְ דִּיבֵּר הָאַדְמוֹ״ר. זֶה סִיפּוּר אֲמִיתִּי.

    "בְּקָרְבָתָם לִפְנֵי־ה׳" – לְפִי אוֹר הַחַיִּים

    הַסִּיפּוּר הַזֶּה מַסְבִּיר לִי אֶת הַפָּרָשָׁה. הַפָּרָשָׁה מַתְחִילָה: "וַיְדַבֵּר ה׳ אֶל־מֹשֶׁה אַחֲרֵי מוֹת שְׁנֵי בְּנֵי אַהֲרֹן בְּקָרְבָתָם לִפְנֵי־ה׳ וַיָּמֻתוּ"ויקרא טז, א. מָה פֵּירוּשׁ? וְכָתוּב: "וַיַּקְרִיבוּ לִפְנֵי ה׳ אֵשׁ זָרָה אֲשֶׁר לֹא צִוָּה אֹתָם"ויקרא י, א, וְיָמוּתוּ. אֲנַחְנוּ יוֹדְעִים שֶׁבְּנֵי אַהֲרֹן מֵתוּ מִשּׁוּם שֶׁטָּעוּ – הִקְרִיבוּ אֵשׁ זָרָה. יֵשׁ כַּמָּה טְעָמִים, וְאֶחָד מֵהֶם הוּא מָה שֶׁכָּתוּב בַּתּוֹרָה: "אֲשֶׁר לֹא צִוָּה אֹתָם". אִם כֵּן, מָה פֵּירוּשׁ "בְּקִרְבָתָם" – שֶׁהָיוּ קְרוֹבִים לָה׳? הֲרֵי חָטְאוּ! וּמָה פֵּירוּשׁ "בְּקָרְבָתָם לִפְנֵי־ה׳ וַיָּמֻתוּ" – מִי שֶׁמִּתְקָרֵב לַה׳ מֵת?

    יֵשׁ שָׁם דְּבָרִים שֶׁל הָאוֹר הַחַיִּים הַקָּדוֹשׁאור החיים ויקרא טז, א, וְהוּא אוֹמֵר דָּבָר אַדִּיר וְנוֹרָא: בְּנֵי אַהֲרֹן הָיוּ צַדִּיקִים כָּאֵלֶּה – זֶה הָיָה בְּיוֹם חֲנוּכַּת הַמִּשְׁכָּן, וְהָיְתָה שָׁם הַשְׁרָאַת הַשְּׁכִינָה בַּפַּעַם הָרִאשׁוֹנָה בַּהִיסְטוֹרְיָה; הַקָּדוֹשׁ בָּרוּךְ הוּא, אַחֲרֵי מַעֲמַד הַר סִינַי, הִשְׁרָה אֶת שְׁכִינָתוֹ בַּמִּשְׁכָּן. הָרַמְבָּ״ן מַסְבִּיר שֶׁכָּל הַשְּׁכִינָה שֶׁשָּׁרְתָה בַּמִּשְׁכָּן הָיְתָה מָה שֶׁנִּשְׁאַר מִמַּעֲמַד הַר סִינַירמבן שמות כה, א; הֲרֵי הַמִּשְׁכָּן בָּא אַחֲרֵי מַעֲמַד הַר סִינַי, וּמָה שֶׁנִּשְׁאַר עַל הַר סִינַי – הוּא שֶׁבָּא וְהִשְׁרָה אֶת הַשְּׁכִינָה. כְּלוֹמַר, כָּל קְדוּשַּׁת בֵּית הַכְּנֶסֶת בָּאָה עַד הַיּוֹם מֵאוֹתָהּ הַשְׁרָאָה שֶׁל מַעֲמַד הַר סִינַי.

    אוֹמֵר הָאוֹר הַחַיִּים הַקָּדוֹשׁ: הֵם הָיוּ צַדִּיקִים כָּאֵלֶּה – הֲרֵי אֵיךְ מֵתוּ? שְׁנֵי עַמּוּדֵי אֵשׁ נִכְנְסוּ בִּנְחִירֵיהֶם. הָיְתָה שָׁם הַשְׁרָאָה כָּזֹאת, וְהַקָּדוֹשׁ בָּרוּךְ הוּא הָיָה שָׁם; וְהֵם הָיוּ קְרוֹבִים כָּל כָּךְ לַה׳ יִתְבָּרַךְ, שֶׁרָצוּ – כִּבְיָכוֹל – לָעוּף, נִדְלְקוּ וְנִשְׂרְפוּ כְּדֵי לְהִתְקָרֵב עוֹד יוֹתֵר לַה׳, וְכָךְ נִשְׁאֲבוּ עִם הַשְּׁכִינָה.

    וְהָאוֹר הַחַיִּים, שֶׁאֵינוֹ מַשְׁאִיר פִּינָּה שֶׁלֹּא כִּיסָּה, אוֹמֵר: אִילּוּ חָזְרוּ מְעַט לְאָחוֹר – לֹא הָיוּ מֵתִים. אִילּוּ הִתְרַחֲקוּ מְעַט, יָכְלוּ שֶׁלֹּא לָמוּת. הַבְּחִירָה הָיְתָה בְּיָדָם, אֲבָל בְּכַוָּנָה הִתְקָרְבוּ – כְּדֵי לְהִיכָּנֵס, לְהִתְמַזֵּג וּלְהִתְעַרְבֵּב עִם הַשְּׁכִינָה. וְכָךְ הָלְכוּ. עִינוּ בְּאוֹר הַחַיִּים הַקָּדוֹשׁ עָלָיו הַשָּׁלוֹם: מָה שֶׁהוּא אוֹמֵר – זֶה מַשֶּׁהוּ.

    מְחִיקַת הָעֲווֹנוֹת – וּמָה עִם הַטּוּמְאָה?

    תַּקְשִׁיבוּ טוֹב, זֶה חִידּוּשׁ גָּדוֹל וְנוֹרָא. הַתּוֹרָה אֵינָהּ מְפָרֶשֶׁת בְּאֵיזֶה יוֹם מְדוּבָּר; כָּל הָעֲבוֹדָה הַזֹּאת – וְאֵינָהּ אוֹמֶרֶת אֵיזֶה יוֹם. רַק בְּסוֹף הַפָּרָשָׁה כָּתוּב "כִּי־בַיּוֹם הַזֶּה". אֵיזֶה יוֹם? לֹא אָמַרְתָּ לִי! נָכוֹן, כָּתוּב "בֶּעָשׂוֹר לַחֹדֶשׁ"ויקרא טז, כט – אֲבָל רַק בְּסוֹף הַפָּרָשָׁה. מָה זֶה אוֹמֵר?

    אֲנַחְנוּ חוֹשְׁבִים, אֵיךְ מוֹחֵל לָנוּ הַקָּדוֹשׁ בָּרוּךְ הוּא עֲווֹנוֹת? מַגִּיעִים, בְּלִי עֵיִן הָרַע, כָּל אֶחָד עִם הַתִּיק שֶׁלּוֹ. יוֹם לִפְנֵי כֵן מַזְמִין הַקָּדוֹשׁ בָּרוּךְ הוּא כַּמָּה רַכָּבוֹת שֶׁל אֵקוֹנוֹמִיקָה מִמִּפְעַל "סָנוֹ", וְנִכְנָסִים בְּיוֹם הַכִּיפּוּרִים – וְהַמַּלְאָכִים עוֹבְדִים שָׁם בְּלִי הַפְסָקָה, כָּל יוֹם הַכִּיפּוּרִים: עוֹד אֵקוֹנוֹמִיקָה, עוֹד "סָנוֹ". אַתֶּם זוֹכְרִים אֶת הַפִּרְסוֹמֶת "רַק סָנוֹ"? בַּשָּׁמַיִם זוֹ הַהַצְלָחָה שֶׁלָּהֶם בְּיוֹם הַכִּיפּוּרִים: מְסַפְּקִים אֶת כָּל הַחוֹמֶר הַזֶּה, וְהַמַּלְאָכִים מוֹחֲקִים, מוֹחֲקִים, מוֹחֲקִים.

    בָּא הַקָּדוֹשׁ בָּרוּךְ הוּא וְאוֹמֵר: אוּלַי אַתֶּם חוֹשְׁבִים שֶׁבְּיוֹם הַכִּיפּוּרִים אָמַרְתִּי לָכֶם "תָּצוּמוּ"? לֹא כָּתוּב "תְּצוּמוּ"; כָּתוּב "תְּעַנּוּ אֶת־נַפְשֹׁתֵיכֶם" – זֶה מַשֶּׁהוּ אַחֵר. גַּם זֶה נִצְרָךְ. אֲבָל אַתֶּם חוֹשְׁבִים שֶׁאֲנִי בָּא עִם חוֹמְרֵי נִיקּוּי וּמוֹחֵק, וּבִגְלַל שֶׁאַתֶּם צַמִים אֲנִי מוֹחֵל לָכֶם אֶת הָעֲווֹנוֹת?

    בֶּאֱמֶת, אֵינִי רוֹצֶה לְהִיכָּנֵס לַתְּחוּם הַזֶּה – זֶה תְּחוּם שֶׁל תְּשׁוּבָה. אֲבָל בֶּאֱמֶת: יַעֲלֶה עַל הַדַּעַת שֶׁמִּי שֶׁאוֹכֵל טֶרֶף – הַקָּדוֹשׁ בָּרוּךְ הוּא יִמְחֵל לוֹ? מִי שֶׁמְּחַלֵּל שַׁבָּת – הַקָּדוֹשׁ בָּרוּךְ הוּא יִסְלַח לוֹ? מָה פִּתְאוֹם! לָמָּה שֶׁיִּסְלַח? חִילַּלְתָּ שַׁבָּת – תְּקַבֵּל עוֹנֶשׁ! וּבְכָל זֹאת, תַּקְשִׁיבוּ טוֹב: מָה אַתָּה רוֹצֶה – לְהַעֲנִישׁ? לַהֲרוֹג? נַנִּיחַ שֶׁתַּעֲנִישׁ אֶת הָאָדָם: אָדָם חִילֵּל שַׁבָּת, "מְחַלְלֶיהָ מוֹת יוּמָת"שמות לא, יד – וּמָה יוֹעִיל? הוּא חֵטְא, הַמִּסְכֵּן; וְאַחַר כָּךְ מָה אַתָּה עוֹשֶׂה לוֹ – הוֹרֵג אוֹתוֹ? גַּם חֵטְא, וְגַם אַתָּה הוֹרֵג אוֹתוֹ?

    הַתְּשׁוּבָה – לְמַעְלָה מִן הַחוֹק

    אִם אַתֶּם זוֹכְרִים, אָמַרְנוּ שֶׁהַתְּשׁוּבָה אֵינֶנָּה "חוּקִּית" – הַקָּדוֹשׁ בָּרוּךְ הוּא, כִּבְיָכוֹל, עוֹבֵר עַל הַחוֹק. כִּי שָׁאֲלוּ לַחָכְמָה, שָׁאֲלוּ לִנְבוּאָה, שָׁאֲלוּ לַתּוֹרָה – מָה פָּסְקוּ? מָוֶת. "חַטָּאִים תְּרַדֵּף רָעָה"משלי יג, כא. הַקָּדוֹשׁ בָּרוּךְ הוּא שָׁמַע, וְאָמַר: יַעֲשֶׂה תְּשׁוּבָה וְיִתְכַּפֵּר לוּירושלמי מכות ב, ו.

    [?] כְּמוֹ שֶׁאָמַר טְרַאמְפּ עַל אוֹתוֹ שׁוֹפֵט: תֵּן לוֹ חֲנִינָה. [?]

    "בְּקִרְבָתָם" – הַפָּרָשָׁה שֶׁל יוֹם הַכִּיפּוּרִים מַתְחִילָה מִן הַהִתְקָרְבוּת

    "בְּקִרְבָתָם": פָּרָשַׁת יוֹם הַכִּיפּוּרִים מַתְחִילָה בַּמִּילָּה "בְּקִרְבָתָם" – הִיא מַתְחִילָה מִן הַהִתְקָרְבוּת לַה׳ יִתְבָּרַךְ. וְלָכֵן הַתּוֹרָה אֵינָהּ אוֹמֶרֶת בְּאֵיזֶה יוֹם מְדוּבָּר. כְּשֶׁאֲנַחְנוּ נִכְנָסִים לְיוֹם הַכִּיפּוּרִים, אֲנַחְנוּ יוֹדְעִים מֵהַלּוּחַ שֶׁזֶּה יוֹם הַכִּיפּוּרִים, אֲבָל הַתּוֹרָה אֵינָהּ מְגַלָּה לָנוּ אֵיזֶה יוֹם זֶה. וְלָמָּה? נַמְשִׁיךְ בְּעֶזְרַת ה׳.

    קוֹדֶם כֹּל נֵדַע: בְּנֵי אַהֲרֹן חָטְאוּ חֵטְא קָטָן כָּלְשֶׁהוּ. אֲבָל מִיתָתָם – תַּקְשִׁיבוּ טוֹב – לֹא הָיְתָה בִּגְלַל הַחֵטְא. הַחֵטְא הוּא סִיבָּה קְטַנָּה, שֶׁאֵין לָהּ קֶשֶׁר לָעִיקָּר. וְאִם יֵשׁ אֵיזוֹ סִיבָּה קְטַנָּה, אֵיזֶה חֵטְא קָטָן – הֲרֵי הוּא כְּמוֹ שֶׁל אוֹתוֹ יְהוּדִי. שִׁמְעוּ טוֹב, וְאִם לֹא – אֶתְעַצְבֵּן.

    מַעֲשֶׂה בְּהֶרְשְׁל׳ה

    יוֹדְעִים מָה אֲנִי עוֹשֶׂה כְּשֶׁאֲנִי מִתְעַצְבֵּן? הָרְשְׁלֶ׳ה הָיָה בָּא לַמִּסְעָדָה, וּבְכָל פַּעַם נָתְנוּ לוֹ לֶאֱכוֹל בְּלִי כֶּסֶף, אוֹ שֶׁהָיָה מְשַׁלֵּם מְעַט. פַּעַם אַחַת לֹא הָיָה לוֹ כְּלוּם. בָּא בַּלַּיְלָה לְמִסְעָדָה שֶׁל אִישָּׁה אַחַת וּבִיקֵּשׁ אוֹכֶל, וְהִיא לֹא נָתְנָה לוֹ. הִתְחִיל לְהִסְתּוֹבֵב בַּמִּסְעָדָה, עַצְבָּנִי, עַצְבָּנִי, עַצְבָּנִי. הָאִישָּׁה פָּחֲדָה; אָמַר לָהּ: אִם לֹא תִּתְּנִי לִי אוֹכֶל – אֶעֱשֶׂה מָה שֶׁאַבָּא שֶׁלִּי עָשָׂה. עַכְשָׁיו בֶּאֱמֶת פָּחֲדָה, וְנָתְנָה לוֹ לֶאֱכוֹל. אַחֲרֵי שֶׁאָכַל שָׁאֲלָה אוֹתוֹ: תַּגִּיד לִי בְּבַקָּשָׁה, מָה עָשָׂה אַבָּא שֶׁלָּךָ? אָמַר לָהּ: הָלַךְ לִישׁוֹן.

    וְזֶה הַפֵּירוּשׁ שֶׁל "בְּקִרְבָתָם": הָיִיתָ בַּמִּסְעָדָה וְאָכַלְתָּ טֶרֶף – וְעַכְשָׁיו אַתָּה הוֹלֵךְ לְבֵית הַכְּנֶסֶת? תֹּאמַר לוֹ: "בְּקִרְבָתֶם" – אֲנִי רוֹצֶה לְהִתְקָרֵב לְאַבָּא שֶׁלִּי. מָה אִכְפַּת לְךָ? אֲנִי רוֹצֶה לְהִתְקָרֵב.

    הַקָּרְבָּנוֹת – "קָרְבָּן" מִלְּשׁוֹן קִרְבָה

    הַתּוֹרָה אֵינָהּ מְגַלָּה לָנוּ שֶׁמְּדוּבָּר בְּיוֹם הַכִּיפּוּרִים. וְתֹאמְרוּ: הַקָּרְבָּנוֹת – הֵם שֶׁמְּכַפְּרִים אֶת הַחֵטְא? לֹא. מָה הַקֶּשֶׁר? אָדָם חוֹטֵא – וְכֶבֶשׂ מִסְכֵּן נִשְׁחַט? לָמָּה שָׁחַטְתָּ אוֹתוֹ? בִּשְׁבִיל שֶׁאַתָּה חָטָאתָ? עוֹד חֵטְא עַל פֶּשַׁע? אַתֶּם יוֹדְעִים שֶׁבַּתְּחִילָּה הוּתַּר לְאָדָם לֶאֱכוֹל רַק צְמָחִים וְלֹא בַּעֲלֵי חַיִּים, עַד שֶׁהִגַּעְנוּ לְנֹחסנהדרין נט עב. "קָרְבָּן" – מִלְּשׁוֹן "בְּקִרְבָתָם": אֲנִי רוֹצֶה לְהִתְקָרֵב לַה׳.

    לָכֵן אֵין הַתּוֹרָה מְגַלָּה אֶת הַיּוֹם הַזֶּה. לָמָּה? כִּי מַהוּ הַסּוֹד שֶׁל יוֹם הַכִּיפּוּרִים? כַּמּוּבָן, מָה שֶׁאֲנַחְנוּ יוֹדְעִים: מָחַקְנוּ לְךָ אֶת הָעֲווֹנוֹת. אֲבָל מָה עִם הַטּוּמְאָה שֶׁהֶעָווֹן מַשְׁאִיר בְּתוֹךְ הַנֶּפֶשׁ? תַּקְשִׁיבוּ טוֹב: אָדָם אָכַל חֲזִיר. מָחֲקוּ לוֹ – נַנִּיחַ שֶׁעָשׂוּ לֶחֲזִיר הַכָּשֵׁר בָּדָ״ץ. נַנִּיחַ. יוֹתֵר מִזֶּה: הָיָה פַּעַם יְהוּדִי, ה׳ יְרַחֵם עָלֵינוּ, [?] עֶשְׂרִים שָׁנָה. אֲבָל מָה עִם הַטּוּמְאָה שֶׁהַמַּאֲכָל הִשְׁאִיר בַּגּוּף? הַאִם הַתְּשׁוּבָה מוֹעִילָה לַטּוּמְאָה, לְכָךְ שֶׁהַגּוּף נִטְמָא? מָה אַתֶּם אוֹמְרִים?

    [מן הקהל:] נִרְאֶה שֶׁכֵּן; אוּלַי הִיא מוֹעִילָה.

    אֲבָל אֲנַחְנוּ רוֹאִים שֶׁלֹּא. אֲנַחְנוּ רוֹאִים שֶׁאָדָם שֶׁנִּטְמָא בַּעֲבֵירָה – נִשְׁאָר תָּקוּעַ עִם הָעֲבֵירָה הַזֹּאת.

    [מן הקהל:] אִם כֵּן, מָה פֵּירוּשׁ "יַעֲשֶׂה תְּשׁוּבָה וְיִתְכַּפֵּר לוֹ" – וְכִי אֵין מוֹחֲקִים לוֹ אֶת הֶעָווֹן? כְּבוֹד הָרַב מְדַבֵּר עַל הָרוֹשֶׁם שֶׁנִּשְׁאַר מִן הַחֵטְא?

    עַל הָרוֹשֶׁם שֶׁנִּשְׁאַר מִן הַחֵטְא – וְזֶה לֹא רַק רוֹשֶׁם. גַּם אִם מוֹחֲקִים לָאָדָם אֶת הֶעָווֹן, אִם הוּא [?] מְעַשֵּׁן – הוּא יַמְשִׁיךְ לְעַשֵּׁן. לָמָּה? תַּקְשִׁיב: יֵשׁ שְׁנֵי שְׂעִירִים בְּיוֹם הַכִּיפּוּרִים, אֶחָד לַה׳ וְאֶחָד לַעֲזָאזֵל. אֶת שֶׁל ה׳ מַקְרִיבִים וְהוּא מְכַפֵּר; הַשֵּׁנִי הוֹלֵךְ לַעֲזָאזֵל.

    אָז מָה קוֹרֶה בְּיוֹם הַכִּיפּוּרִים? מוֹחֲקִים לָנוּ אֶת כָּל הָעֲווֹנוֹת – אַגַּב, לֹא אֶת הַכּוֹל; צָרִיךְ לָדַעַת שֶׁיֵּשׁ אַרְבָּעָה חִילּוּקֵי כַּפָּרָהיומא פו עא. אֲבָל לֹא נִתְיָאֵשׁ; נַשְׁאִיר אֶת זֶה כָּךְ. וְאִילּוּ הַטּוּמְאָה שֶׁל הָעֲבֵירָה – זוֹהִי הָעֲבוֹדָה שֶׁל הָאָדָם. אַתָּה מַקְשִׁיב לִי? זוֹ הָעֲבוֹדָה שֶׁלְּךָ.

    הָאָדָם בּוֹרֵא אֶת עַצְמוֹ מֵחָדָשׁ

    אַתֶּם זוֹכְרִים מָה דִּיבַּרְנוּ עַל הַתְּשׁוּבָה – שֶׁהָאָדָם בּוֹרֵא אֶת עַצְמוֹ מֵחָדָשׁ? הָאָדָם מְקַבֵּל שָׂכָר גַּם עַל כָּךְ שֶׁעָשָׂה עָוֶן, וְהֶעָווֹן הַזֶּה הוֹפֵךְ לִזְכוּת: זְדוֹנוֹת נַעֲשׂוֹת לוֹ כִּזְכוּיוֹתיומא פו עב. אֵיךְ יִתָּכֵן דָּבָר כָּזֶה בָּעוֹלָם? כְּשֶׁאָדָם עוֹשֶׂה תְּשׁוּבָה – מִתְבּוֹנֵן בַּחֵטְא שֶׁלּוֹ, הוֹלֵךְ וְלוֹמֵד אֶת כָּל הַתּוֹרָה כְּדֵי לְהָבִין לָמָּה חֵטְא, עוֹבֵד שָׁנִים עַל גַּבֵּי שָׁנִים עַד שֶׁהוּא מֵבִין אֶת אוֹפְיוֹ, וְעוֹקֵר אֶת כָּל הָרַע שֶׁבּוֹ – הוּא בּוֹרֵא אֶת עַצְמוֹ מֵחָדָשׁ, אָדָם חָדָשׁ. דִּיבַּרְנוּ עַל כָּךְ בַּאֲרִיכוּת, וְכָאן הֵבֵאנוּ זֹאת רַק בְּדֶרֶךְ אַגַּב.

    בָּא הַקָּדוֹשׁ בָּרוּךְ הוּא וְאוֹמֵר לָנוּ: "כִּי־בַיּוֹם הַזֶּה יְכַפֵּר עֲלֵיכֶם לְטַהֵר אֶתְכֶם מִכֹּל חַטֹּאתֵיכֶם". בְּסֵדֶר – מוֹחֲקִים אֶת הָעֲווֹנוֹת. אֲבָל כְּדֵי לִמְחוֹק אֶת הַטּוּמְאָה – "לִפְנֵי ה׳ תִּטְהָרוּ". אֵיךְ? "בְּקִרְבָתָם": אִם אַתָּה מִתְקָרֵב לַה׳ בְּמִידָּה כָּזֹאת שֶׁאַתָּה רָאוּי לָשֶׁבֶת לְיַד הַקָּדוֹשׁ בָּרוּךְ הוּא. אָדָם שֶׁמְּטַהֵר אֶת עַצְמוֹ עַד שֶׁהוּא יָכוֹל לָשֶׁבֶת לְיַד הַקָּדוֹשׁ בָּרוּךְ הוּא – הוּא יֹאמַר לַשָּׂטָן: מָה אִכְפַּת לְךָ? כָּךְ הוּא יָכוֹל לוֹמַר לַשָּׂטָן.

    סוֹד הַשָּׂעִיר לַעֲזָאזֵל

    וְעַכְשָׁיו שָׁמְעוּ אֶת הַסּוֹד הַגָּדוֹל, הָ"אַטְרַקְצִיָּה" שֶׁל הַשִּׂיחָה הַזֹּאת. אוֹמֵר הַזּוֹהַר הַקָּדוֹשׁ: הַשָּׂעִיר לַעֲזָאזֵל – בִּשְׁבִיל מָה הוּא שָׁם, הַמִּסְכֵּן? הַשָּׂעִיר הַזֶּה נִיתָּן לַשָּׂטָן. אִם לֹא נִיתֵּן לוֹ אֶת הַשָּׂעִיר הַזֶּה, הוּא יְקַטְרֵג עָלֵינוּ כָּל יוֹם הַכִּיפּוּרִים: מִי אַתֶּם בִּכְלָל? לָמָּה? וְכוּלִּי. לָכֵן נוֹתְנִים לוֹ אוֹתוֹ.

    פַּעַם חָשַׁבְתִּי בֶּאֱמֶת כָּךְ, וְגַם דָּרַשְׁתִּי כָּךְ: רְאוּ כַּמָּה גָּדוֹל כּוֹחוֹ שֶׁל הַשָּׂטָן, שֶׁאִם לֹא יִתְּנוּ לוֹ אֶת הַשָּׂעִיר, יַפְרִיעַ לְיוֹם הַכִּיפּוּרִים. אֲבָל ה׳ חָנַן אוֹתִי בַּהֲבָנָה אַחֶרֶת: אוֹמְרִים לַשָּׂטָן – אַתָּה רָשָׁע, אַתָּה פּוֹשֵׁעַ, אַתָּה מַחֲטִיא בְּנֵי אָדָם, אַתָּה מְטַמֵּא בְּנֵי אָדָם; קַח לְךָ שָׂעִיר לַעֲזָאזֵל וַעֲזוֹב אוֹתָנוּ!

    אַתֶּם יוֹדְעִים לָמָּה צָרִיךְ עַכְשָׁיו לִקְפּוֹץ מִשִּׂמְחָה? כִּי כָּל הַזְּמַן חָשַׁבְנוּ שֶׁלַּשָּׂטָן יֵשׁ כּוֹחוֹת אַדִּירִים וְנוֹרָאִים, שֶׁצָּרִיךְ לִזְרוֹק לוֹ שָׂעִיר. אֲנַחְנוּ חוֹשְׁבִים שֶׁהַשָּׂטָן הוּא בַּעַל כּוֹחַ. תִּשְׁמַע טוֹב, בְּנִי – זֶה לֹא פָּחוֹת מֵרַבִּי נַחְמָן מִבְּרֶסְלָב. אֲנַחְנוּ חוֹשְׁבִים שֶׁהַשָּׂטָן הוּא כּוֹחַ אַדִּיר: עוֹצְמָה, יוֹפִי, עוֹשֶׁר, חָכְמָה. וּמָה אוֹמְרִים לַשָּׂטָן בְּיוֹם הַכִּיפּוּרִים? קַח שָׂעִיר וְלֵךְ לַעֲזָאזֵל. זֶהוּ הַשָּׂעִיר שֶׁל יוֹם הַכִּיפּוּרִים.

    אֲבָל, תַּקְשִׁיבוּ טוֹב: אָדָם לְעוֹלָם לֹא יַעֲזוֹב אֶת הָעֲבֵירָה – וְגַם זֶה אַדִּיר וְנוֹרָא – עַד שֶׁתִּימָּאֵס בְּעֵינָיו. כָּל זְמַן שֶׁהָעֲבֵירָה טְעִימָה לוֹ, לֹא יַעֲזוֹב אוֹתָהּ. אָדָם צָרִיךְ לְהַגִּיעַ לְמַצָּב שֶׁהָעֲבֵירָה מַגְעִילָה בְּעֵינָיו, שֶׁהָעֲבֵירָה הִיא אֶפֶס. אַתָּה, הַשָּׂטָן, עִם הַכּוֹחוֹת שֶׁלְּךָ – בָּאתָ לַעֲבוֹד עָלֵינוּ? אַתָּה שַׁקְרָן וְרַמַּאי. אַתָּה סְתָם, אַתָּה אֶפֶס. קַח שָׂעִיר וְלֵךְ! זֶה הַסּוֹד הַזֶּה.

    "לִפְנֵי ה׳ תִּטְהָרוּ"

    בָּרֶגַע שֶׁאַתָּה נוֹתֵן לַשָּׂטָן שֶׁלְּךָ אֶת הַדָּבָר הַזֶּה – "לִפְנֵי ה׳ תִּטְהָרוּ". הַסּוֹד שֶׁל יוֹם הַכִּיפּוּרִים הוּא שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא אוֹמֵר: בָּנַי הַיְּקָרִים, אֵין לִי בְּעָיָה לִמְחוֹק לָכֶם אֶת הָעֲווֹנוֹת. אֲבָל מָה עִם הַטּוּמְאָה שֶׁנִּטְמֵאתֶּם? בּוֹאוּ לְפָנַי, הַכְשִׁירוּ אֶת עַצְמְכֶם לָבוֹא לְפְנַי – "לִפְנֵי ה׳ תִּטְהָרוּ" – וַאֲנִי אֶדְאַג לְטַהֵר אֶתְכֶם.

    וּבֶאֱמֶת, רַבֵּנוּ יוֹנָה – תַּקְשִׁיבוּ טוֹב – כֵּיצַד הוּא קוֹרֵא אֶת הַפָּסוּק "כִּי־בַיּוֹם הַזֶּה יְכַפֵּר עֲלֵיכֶם לְטַהֵר אֶתְכֶם מִכֹּל חַטֹּאתֵיכֶם לִפְנֵי ה׳ תִּטְהָרוּ"? הַאִם פֵּירוּשׁוֹ שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא מוֹחֵק אֶת כָּל עֲוְנוֹתֵיכֶם? לֹא כָּךְ הוּא קוֹרֵא אֶת הַפָּסוּק, אֶלָּא כָּךְ: "כִּי־בַיּוֹם הַזֶּה יְכַפֵּר עֲלֵיכֶם לְטַהֵר אֶתְכֶם מִכֹּל חַטֹּאתֵיכֶם" – זֶה הַיּוֹם. אֲבָל אֶת מְחִיקַת הַטּוּמְאָה – "לִפְנֵי־ה׳": אִם אַתָּה יוֹשֵׁב לִפְנֵי הַקָּדוֹשׁ בָּרוּךְ הוּא, אַתָּה נִטְהָר.

    כָּל הַסּוֹד שֶׁל יוֹם הַכִּיפּוּרִים אֵינוֹ מְחִיקַת הָעֲווֹנוֹת – זֶה נַעֲשֶׂה בַּשָׂעִיר וּבִשְׁאַר הָעֲבוֹדָה. אֲבָל לְטַהֵר אֶתְכֶם מִן הַטּוּמְאָה – זֶה אִם אַתֶּם מִתְקָרְבִים לְבוֹרֵא עוֹלָם. מִמֵּילָא מוּבָן לָמָּה הַפָּרָשָׁה מַתְחִילָה "בְּקָרְבָתָם לִפְנֵי־ה׳", וּמוּבָן לָמָּה לֹא נֶאֱמַר לָנוּ הַיּוֹם: כִּי הַיּוֹם בְּסַךְ הַכּוֹל מוֹחֵק אֶת הָעֲווֹנוֹת. אֲבָל כְּדֵי שֶׁתֵּצֵא מָחָר נָקִי – זֶה "לִפְנֵי־ה׳", זוֹ הַקִּרְבָה לִפְנֵי הַקָּדוֹשׁ בָּרוּךְ הוּא.

    לָמַדְנוּ אֵפוֹא אֶת הַיְּסוֹד הַזֶּה: אָדָם שֶׁמִּתְקָרֵב לַה׳ נִטְהָר מֵעֶצֶם הַקִּרְבָה לַה׳ יִתְבָּרַךְ. רַק הַקִּרְבָה לַה׳ הִיא שֶׁמְּטַהֶרֶת אוֹתוֹ בֶּאֱמֶת – לֹא מוֹחֶקֶת, אֶלָּא מְטַהֶרֶת. זֶהוּ "וַאֲנִי קִרְבַת אֱלֹהִים לִי־טוֹב"תהילים עג, כח, אוֹמֵר דָּוִד הַמֶּלֶךְ עָלָיו הַשָּׁלוֹם – כְּשֶׁאֲנִי מִתְקָרֵב לַה׳.

    "מַה־טֹּבוּ אֹהָלֶיךָ יַעֲקֹב" – לִימּוּד הַתּוֹרָה מְטַהֵר

    וְאָז אֲנַחְנוּ מְבִינִים, בְּעֶזְרַת ה׳, עוֹד הַרְבֵּה דְּבָרִים, וּנְדַבֵּר עֲלֵיהֶם. אֲנַחְנוּ מְבִינִים אֶת "מַה־טֹּבוּ אֹהָלֶיךָ יַעֲקֹב מִשְׁכְּנֹתֶיךָ יִשְׂרָאֵל כִּנְחָלִים נִטָּיוּ כְּגַנֹּת עֲלֵי נָהָר"במדבר כד, ה–ו – מִדִּבְרֵי בִּלְעָם. אוֹמֵר הַמִּדְרָשׁ: מָה נְחָלִים מַעֲלִים אֶת הָאָדָם מִטּוּמְאָה לְטָהֳרָה, אַף אֲהָלִים מַעֲלִים אֶת הָאָדָם מִכַּף חוֹבָה לְכַף זְכוּתברכות טז עא. "מַה־טֹּבוּ אֹהָלֶיךָ יַעֲקֹב" – לִימּוּד הַתּוֹרָה שֶׁאָדָם לוֹמֵד בִּיגִיעָה מְטַהֵר אוֹתוֹ. לֹא מוֹחֵק לוֹ עֲווֹנוֹת, אֶלָּא מְטַהֵר אוֹתוֹ – מוֹצִיא מִמֶּנּוּ אֶת כָּל הַטּוּמְאָה שֶׁבְּקִרְבּוֹ.

    וְזֶהוּ הַסּוֹד שֶׁל יוֹם הַכִּיפּוּרִים. הַנִּגְלֶה שֶׁל יוֹם הַכִּיפּוּרִים: לֹא לֶאֱכוֹל, לֹא לִשְׁתּוֹת, לֹא לִרְחוֹץ, לֹא לָסוּךְ.

    דִּבְרֵי הַתּוֹרָה מְטַהֲרִים אֶת הָאָדָם. אִם אָדָם לוֹמֵד תּוֹרָה – לֹא רַק שֶׁעֲבֵירוֹתָיו נִמְחָקוֹת; הַתּוֹרָה מְטַהֶרֶת אֶת הָאָדָם מִבִּפְנִים. אֱמֶת, אָמֵן. מִי שֶׁטַּעַם טַעְמָהּ שֶׁל תּוֹרָה, שֶׁל לִימּוּד הַתּוֹרָה, מַרְגִּישׁ אֶת זֶה בִּפְנִים: הוּא נוֹתֵן לַשָּׂטָן אֶת הַשָּׂעִיר. קַח! אַתָּה הַשָּׂעִיר! מָה עוֹשִׂים לַשָּׂעִיר? מַעֲלִים אוֹתוֹ עַל הַר וּמְגַלְגְּלִים אוֹתוֹ עַד שֶׁהוּא מִתְרַסֵּק – זֶה אַתָּה, זֶה מָה שֶׁאַתָּה עוֹשֶׂה. אָז לְךָ מִמֶּנִּי; אֲנִי רוֹצֶה לְהִתְקָרֵב לְאַבָּא שֶׁלִּי.

    כָּךְ אָמַר אוֹתוֹ יְהוּדִי בַּסִּיפּוּר לַיְּהוּדִי הַחֲרֵדִי: אַתָּה רוֹאֶה אוֹתִי? אֲנִי קָאוּבּוֹי, נָכוֹן, הִתְרַחַקְתִּי מִן הַתּוֹרָה – אֲבָל בְּיוֹם שִׁישִּׁי הַזֶּה בָּא לִי לָלֶכֶת לַמִּקְוֶה. [?]

    הַתְּפִילָּה – יְסוֹד הַקִּרְבָה לַה׳

    רַבּוֹתַי, זֶה "וְהַמַּשְׂכִּלִים יָבִינוּ". אֵלֶּה לֹא דְּרָשׁוֹת שֶׁל סִיפּוּרִים, נִיסִּים וְנִפְלָאוֹת וְרִיקּוּדִים. זוֹ הָעֲבוֹדָה: לְהִתְקָרֵב לַה׳ יִתְבָּרַךְ. נָכוֹן, כָּל הָאֶמְצָעִים בְּסֵדֶר גָּמוּר – הָרִיקּוּדִים וְהַשִּׂמְחָה, וּמִצְוָה גְּדוֹלָה לִהְיוֹת בְּשִׂמְחָה. אֲבָל אַתֶּם יוֹדְעִים מָה הַשִּׂמְחָה שֶׁל הָאָדָם? "חֶדְוַת ה׳" – הַשִּׂמְחָה שֶׁנּוֹתֶנֶת לָאָדָם כּוֹחַ, שֶׁהִיא "מָעוֹז": "חֶדְוַת ה׳".

    וְהַתְּפִילָּה הִיא הַיְּסוֹד הָרִאשׁוֹן שֶׁל הַקִּרְבָה לַה׳ יִתְבָּרַךְ. הַיְּסוֹד הָרִאשׁוֹן, הַבָּסִיס הָרִאשׁוֹן לְהִתְקָרְבוּת לְבוֹרֵא עוֹלָם – זוֹ הַתְּפִילָּה. אוֹי וַאֲבוֹי לָנוּ אִם אֲנַחְנוּ מְמַהֲרִים בַּתְּפִילָּה: לוֹחֲצִים עַל כַּפְתּוֹר, נִכְנָסִים לֹא מִתְפַּלְּלִים – וְיוֹצְאִים "מְפוּלָּלִים". מָה זֶה? אַתָּה נִכְנָס וְלוֹחֵץ עַל כַּפְתּוֹר; נִכְנַסְתָּ לִפְנֵי הַתְּפִילָּה, וְיָצָאתָ – מְפוּלָּל. בָּרוּךְ ה׳.

    אֲבָל צָרִיךְ לָדַעַת שֶׁהַתְּפִילָּה הִיא הָעֲבוֹדָה הָרִאשׁוֹנָה שֶׁל הַקִּרְבָה לַה׳ יִתְבָּרַךְ. מָה אֲנַחְנוּ יְכוֹלִים לַעֲשׂוֹת? נִיקַּח קֶטַע קָטָן מִשְּׁמוֹנֶה עֶשְׂרֵה, וְנֹאמַר אוֹתוֹ מִילָּה בְּמִילָּה, לְאַט לְאַט.

    זֶהוּ הַסּוֹד שֶׁל יוֹם הַכִּיפּוּרִים. כָּל מָה שֶׁאֲנַחְנוּ יוֹדְעִים עַל יוֹם הַכִּיפּוּרִים – הָעִינּוּיִים, חֲמֵשֶׁת הָעִינּוּיִים, הַתְּפִילּוֹת, הַוִּידּוּי – הַכּוֹל; אֲבָל "לִפְנֵי ה׳ תִּטְהָרוּ" – זוֹ הַקִּרְבָה לַה׳ יִתְבָּרַךְ. לָכֵן הַפָּרָשָׁה מַתְחִילָה בְּ"בְּקִרְבָתָם" וּמִסְתַּיֶּמֶת בְּ"כִּי בַּיּוֹם הַזֶּה… תִּטַהֲרוּ".

    וְכָךְ צָרִיךְ לָדַעַת שֶׁהוֹלֵךְ בְּכָל הַתּוֹרָה כּוּלָּהּ: לְכָל מִצְוָה בַּתּוֹרָה, מִצְוַת עֲשֵׂה אוֹ לֹא תַּעֲשֶׂה, יֵשׁ אֶת הַנִּגְלֶה שֶׁלָּהּ. אַתָּה בּוֹנֶה סוּכָּה – זוֹ פַּרְגּוֹלָּה; "בַּסֻּכֹּת תֵּשְׁבוּ"ויקרא כג, מב – לָשֶׁבֶת: זֶה הַנִּגְלֶה. אֲבָל הַנִּסְתָּר שֶׁלָּהּ – זוֹ הַחָכְמָה שֶׁבָּהּ. הַקָּדוֹשׁ בָּרוּךְ הוּא יְזַכֶּה אוֹתָנוּ.

    "רַבִּי חֲנַנְיָא בֶּן עֲקַשְׁיָא אוֹמֵר, רָצָה הַקָּדוֹשׁ בָּרוּךְ הוּא לְזַכּוֹת אֶת יִשְׂרָאֵל, לְפִיכָךְ הִרְבָּה לָהֶם תּוֹרָה וּמִצְוֹת, שֶׁנֶּאֱמַר ה׳ חָפֵץ לְמַעַן צִדְקוֹ יַגְדִּיל תּוֹרָה וְיַאְדִּיר:"משנה מכות ג, טז.

    חֲתִימָה: "אַשְׁרֵיכֶם יִשְׂרָאֵל,"

    עַכְשָׁיו מוּבָן מָה אָמַר רַבִּי עֲקִיבָא: "אַשְׁרֵיכֶם יִשְׂרָאֵל, לִפְנֵי מִי אַתֶּם מִטַּהֲרִין, וּמִי מְטַהֵר אֶתְכֶם, אֲבִיכֶם שֶׁבַּשָּׁמַיִם,"משנה יומא ח, ט. מָה הוּא מְחַדֵּשׁ לָנוּ? שֶׁהוּא מְטַהֵר – "תְּטַהֲרוּ" – מוֹרִיד אֶת הַטּוּמְאָה. אַשְׁרֵיכֶם יִשְׂרָאֵל!
