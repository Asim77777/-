פֶּתַח: לְשׁוֹן הַמִּדְרָשׁ

הַמִּדְרָשׁ בְּפָרָשַׁת רְאֵה:

"רְאֵה אָנֹכִי" – אָמַר רַבִּי אֶלְעָזָר: מִשֶּׁאָמַר הַקָּדוֹשׁ בָּרוּךְ הוּא הַדָּבָר הַזֶּה בְּסִינַי, בְּאוֹתָהּ שָׁעָה "מִפִּי עֶלְיוֹן לֹא תֵצֵא הָרָעוֹת וְהַטּוֹב", אֶלָּא מֵאֵלֶיהָ הָרָעָה בָּאָה עַל עוֹשֵׂי הָרָעָה, וְהַטּוֹבָה בָּאָה עַל עוֹשֵׂי הַטּוֹבָה. דָּבָר אַחֵר, אָמַר רַבִּי חַגַּי: וְלֹא עוֹד שֶׁנָּתַתִּי לָכֶם שְׁתֵּי דְרָכִים, אֶלָּא שֶׁנִּכְנַסְתִּי לִפְנִים מִשּׁוּרַת הַדִּין וְאָמַרְתִּי לָכֶם "וּבָחַרְתָּ בַּחַיִּים".

(דברים רבה ד, ג)

אָנוּ לוֹמְדִים כָּאן כְּדֶרֶךְ הַלִּמּוּד: כְּשֵׁם שֶׁיֵּשׁ שִׁעוּר גְּמָרָא, יֵשׁ שִׁעוּר בְּמִדְרָשׁ וְשִׁעוּר בְּסֵפֶר הָעֲקֵדָה.

הַקְדָּמָה: אֶרֶץ יִשְׂרָאֵל – עֵרֶךְ

כְּשֶׁעָשִׂינוּ אֶת הַיּוֹמַיִם יַחַד בַּיְּשִׁיבָה, יָשַׁבְנוּ עַל נוֹשֵׂא אֶרֶץ יִשְׂרָאֵל: מַה הִיא בְּדִיּוּק אֶרֶץ יִשְׂרָאֵל, וּמַה הַמְיֻחָד שֶׁבָּהּ. בַּתּוֹרָה כָּתוּב פְּעָמִים רַבּוֹת עַל אֶרֶץ יִשְׂרָאֵל, וּבַסּוֹף, לְפִי הַהַשְׁקָפָה שֶׁלִּי, הִגַּעְתִּי לְמַסְקָנָה שֶׁאֶרֶץ יִשְׂרָאֵל הִיא עֵרֶךְ.

וְזֹאת עַל פִּי מַה שֶּׁלָּמַדְנוּ בְּפָרָשַׁת וָאֵרָא, הַפָּרָשָׁה שֶׁבָּאָה אַחֲרֵי פָּרָשַׁת שְׁמוֹת. כְּשֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא הֵשִׁיב לְמֹשֶׁה רַבֵּנוּ, שֶׁשָּׁאַל אוֹתוֹ "לָמָה הֲרֵעֹתָה לָעָם הַזֶּה" (שמות ה, כב) – הוּא מֵשִׁיב לוֹ תְּשׁוּבָה אֲרֻכָּה, שֶׁאֵינָהּ עוֹנָה עַל שְׁאֵלָתוֹ שֶׁל מֹשֶׁה רַבֵּנוּ. אֲבָל בַּתְּשׁוּבָה הַזֹּאת הָיָה כִּוּוּנוֹ שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא לוֹמַר לוֹ: אֵינְךָ מַכִּיר אֶת הַתָּכְנִיּוֹת שֶׁלִּי; אִלּוּ הָיִיתָ מַכִּיר אֶת תָּכְנִיּוֹתַי, לֹא הָיִיתָ מִתְרַעֵם, וְשׁוֹאֵל, וּמַקְשֶׁה עָלַי אֶת הַקֻּשְׁיָה הַזֹּאת. וְהוּא נוֹתֵן לוֹ אֶת כָּל הַהִיסְטוֹרְיָה, אֶת כָּל הַתָּכְנִית שֶׁלּוֹ.

כַּמּוּבָן, הַדָּבָר בָּא בְּפָרָשָׁה אַחַת קְטַנָּה, בְּפָרָשַׁת וָאֵרָא. אֶת הַתֹּכֶן בְּוַדַּאי לֹא הֵבַנּוּ, אֲבָל הֵבַנּוּ אֶת הַכִּוּוּן שֶׁבְּתוֹךְ הַתְּשׁוּבָה הַזֹּאת: מִי שֶׁעוֹקֵב אַחֲרֶיהָ יִרְאֶה שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא הִצְבִּיעַ עַל כַּמָּה עֶרְכֵי יְסוֹד שֶׁיֵּשׁ בְּהַנְהָגָתוֹ שֶׁל בּוֹרֵא עוֹלָם. הָעֵרֶךְ הָרִאשׁוֹן – בְּוַדַּאי בּוֹרֵא עוֹלָם; הָעֵרֶךְ הַבָּא אַחֲרָיו – הָאָבוֹת הַקְּדוֹשִׁים; וְהָעֵרֶךְ הַשְּׁלִישִׁי – אֶרֶץ יִשְׂרָאֵל.

כְּלוֹמַר, בְּוַדַּאי שֶׁאֶרֶץ יִשְׂרָאֵל אֵינָהּ מַה שֶּׁאָנוּ יוֹדְעִים. הַדָּבָר בָּרוּר כַּשֶּׁמֶשׁ: כְּשֵׁם שֶׁאֵינִי יָכוֹל לוֹמַר שֶׁאָמֶרִיקָה הִיא עֵרֶךְ – הוֹרֵד מֵאָמֶרִיקָה אֶת הָעֹשֶׁר, וְהִיא תִּהְיֶה כְּמוֹ מִצְרַיִם, כְּמוֹ אִינְדּוֹנֶזְיָה, כְּמוֹ הֹדּוּ – אֲנָשִׁים עֲנִיִּים, וְזֶהוּ. הָאֲדָמָה אֵינָהּ עֵרֶךְ; וּבְוַדַּאי שֶׁלֹּא אַדְמַת אָמֶרִיקָה, עַל כָּל עָשְׁרָהּ. אֲבָל אֶרֶץ יִשְׂרָאֵל אֵינָהּ חַיֶּבֶת בְּכָךְ; אֵינָהּ חַיֶּבֶת בִּתְהוֹמוֹת, בְּמַעְיָנוֹת וּבַאֲגַמִּים. גַּם אִם לֹא לָמַדְנוּ תּוֹרָה וְאֵין אָנוּ מְבִינִים – כְּבָר רוֹאִים אָנוּ שֶׁהָאָרֶץ הַזֹּאת הִיא, בְּקִצּוּר, עֵרֶךְ. כֵּן, אֶרֶץ יִשְׂרָאֵל הִיא עֵרֶךְ.

הַאִם הַיִּשְׂרְאֵלִי מֵבִין אֶת הַנְּקֻדָּה הַזֹּאת, שֶׁאֶרֶץ יִשְׂרָאֵל הִיא עֵרֶךְ? מָה פִּתְאֹם! – עֵרֶךְ הַשָּׁוֶה לָאָבוֹת הַקְּדוֹשִׁים, אוֹ עֵרֶךְ מִן הָעֲרָכִים הַבָּאִים אַחֲרֵי הַקָּדוֹשׁ בָּרוּךְ הוּא? בְּוַדַּאי שֶׁלֹּא. אִלּוּ רַק יָדַע הָעָם הַזֶּה, הַיּוֹשֵׁב כָּאן, שֶׁאֶרֶץ יִשְׂרָאֵל הִיא עֵרֶךְ – הַכֹּל הָיָה אַחֶרֶת. אַחַר כָּךְ לָמַדְנוּ אֶת מַהוּתָהּ שֶׁל אֶרֶץ יִשְׂרָאֵל, וְיָשַׁבְנוּ עַל כָּךְ יוֹמַיִם.

"רְאֵה אָנֹכִי" – עִיּוּן בַּמִּדְרָשׁ

כָּאן אָנוּ רוֹצִים לִלְמֹד נוֹשֵׂא שֶׁהוּא מַעֲרָכָה בִּפְנֵי עַצְמָהּ. אֵין אָנוּ יְכוֹלִים לִדְרֹשׁ; קֹדֶם כָּל, לִפְנֵי הַדְּרָשָׁה, צְרִיכִים לִלְמֹד אֶת הַנּוֹשֵׂא. וְהַנּוֹשֵׂא הַזֶּה עָמֹק הַרְבֵּה יוֹתֵר, מֻרְכָּב הַרְבֵּה יוֹתֵר, וְהַרְבֵּה מִמֶּנּוּ נִמְצָא בַּמַּהוּת שֶׁלָּנוּ. אֵינִי אוֹמֵר מִלָּה אַחַת נוֹסֶפֶת עַל מַה שֶּׁכָּתוּב; כָּךְ נִלְמַד יַחַד, וּבְעֶזְרַת ה׳ נְקַוֶּה שֶׁנֵּצֵא מִזֶּה עִם תֹּכֶן שֶׁל הָעִנְיָן.

וּבְכֵן, הַמִּדְרָשׁ – הוּא הַמָּקוֹר. הַמִּדְרָשׁ הַזֶּה נִמְצָא, כַּמּוּבָן, בְּפָרָשַׁת רְאֵה שֶׁבְּחֻמַּשׁ דְּבָרִים:

רְאֵה אָנֹכִי נֹתֵן לִפְנֵיכֶם הַיּוֹם בְּרָכָה וּקְלָלָה.

(דברים יא, כו)

בָּא רַבִּי אֶלְעָזָר וְאוֹמֵר: "מִשֶּׁאָמַר הַקָּדוֹשׁ בָּרוּךְ הוּא הַדָּבָר הַזֶּה בְּסִינַי" – הַדָּבָר הַזֶּה, בְּסִינַי – "בְּאוֹתָהּ שָׁעָה "מִפִּי עֶלְיוֹן לֹא תֵצֵא הָרָעוֹת וְהַטּוֹב"" (איכה ג, לח). מֵאוֹתָהּ שָׁעָה כְּבָר אֵינָהּ יוֹצֵאת מִמֶּנּוּ יִתְבָּרַךְ רָעָה אוֹ טוֹבָה.

"אֶלָּא מֵאֵלֶיהָ הָרָעָה בָּאָה עַל עוֹשֵׂי הָרָעָה": בֶּן אָדָם שֶׁקִּבֵּל עֹנֶשׁ – אֵינוֹ יָכוֹל לִפְנוֹת לְהַקָּדוֹשׁ בָּרוּךְ הוּא; הָרָעָה בָּאָה מֵאֵלֶיהָ. מֵאַיִן? "מֵאֵלֶיהָ" – וְאֶת זֶה נִלְמַד בְּהֶמְשֵׁךְ. "וְהַטּוֹבָה בָּאָה עַל עוֹשֵׂי הַטּוֹבָה". מִלִּים נוֹרָאוֹת וּמְדֻיָּקוֹת הֵן.

"דָּבָר אַחֵר, אָמַר רַבִּי חַגַּי: וְלֹא עוֹד שֶׁנָּתַתִּי לָכֶם שְׁתֵּי דְרָכִים" – כְּלוֹמַר, לֹא דַּי שֶׁנָּתַתִּי לָכֶם שְׁתֵּי דְּרָכִים, הַבְּרָכָה וְהַקְּלָלָה – "אֶלָּא שֶׁנִּכְנַסְתִּי לִפְנִים מִשּׁוּרַת הַדִּין וְאָמַרְתִּי לָכֶם "וּבָחַרְתָּ בַּחַיִּים"" (דברים ל, יט). עַד כָּאן לְשׁוֹן הַמִּדְרָשׁ.

מִכָּאן נַעֲבֹר לְסֵפֶר הָעֲקֵדָה. יֵשׁ לוֹ כָּאן שְׁלוֹשָׁה דַּפִּים שֶׁהֵם הַהַקְדָּמָה לַמִּדְרָשׁ הַזֶּה – הַקְדָּמוֹתָיו שֶׁלִּפְנֵי שֶׁהוּא מְדַבֵּר עַל הַפָּרָשָׁה. תָּכְנָן שֶׁל הַהַקְדָּמוֹת הָאֵלֶּה קָשֶׁה מְאֹד לַהֲבָנָה, כִּי הוּא מֵבִיא קֹדֶם כָּל מִן הַפִילוֹסוֹפְיָה, אַחַר כָּךְ מִן הַתּוֹרָה וְאַחַר כָּךְ מִן הַמִּדְרָשִׁים – הֲכָנָה, נֹאמַר, בְּעִנְיָנֵנוּ, לְפָרֵשׁ אֶת הַמִּדְרָשׁ הַזֶּה. בָּרוּר שֶׁכְּדֵי לִלְמֹד אֶת מַה שֶּׁלּוֹמְדִים הָעֶרֶב, צְרִיכִים אֶת שְׁלוֹשֶׁת הַדַּפִּים הָאֵלֶּה; אֲבָל לְשֵׁם כָּךְ צָרִיךְ שְׁבוּעַ חֻפְשָׁה. לָכֵן נִסְתַּפֵּק, וְנִגַּשׁ יָשָׁר לָעִנְיָן.

הַדְּבָרִים מְעַנְיְנִים; וְלֹא רַק שֶׁהֵם מְעַנְיְנִים – נִשְׁמַע בָּהֶם דְּבָרִים שֶׁאֵינֶנּוּ מַכִּירִים כְּלָל. אֲבָל לֹא אַקְדִּים הַקְדָּמוֹת; נִלְמַד עַתָּה.

"הָעִנְיָן הַנִּכְבָּד": רְצוֹנוֹ הַמֻּחְלָט שֶׁל הַבּוֹרֵא

כָּךְ הוּא אוֹמֵר:

וְזֶה הָעִנְיָן הַנִּכְבָּד הוּא שֶׁנִּתְבָּאֵר לַאֲבוֹתֵינוּ בְּמַעֲמַד הַר סִינַי, בְּמַאֲמַר "אָנֹכִי" וְ"לֹא יִהְיֶה לְךָ"

(עקדת יצחק, פרשת ראה, שער צ״ג)

הַמַּאֲמָר הוּא מַה שֶּׁנֶּאֱמַר בְּהַר סִינַי, הַדִּבְּרָה הָרִאשׁוֹנָה: "אָנֹכִי ה׳ אֱלֹקֶיךָ... לֹא יִהְיֶה לְךָ אֱלֹהִים אֲחֵרִים עַל פָּנָי" (שמות כ, ב–ג); וְשָׁם יֵשׁ גַּם "זָכוֹר אֶת יוֹם הַשַּׁבָּת" (שמות כ, ח).

הַמְּעִידִים עַל הֱיוֹתוֹ רוֹצֶה

כָּאן מַתְחִיל הַנּוֹשֵׂא: "אָנֹכִי ה׳ אֱלֹקֶיךָ" – הַמְּעִידִים עַל הֱיוֹתוֹ רוֹצֶה. כְּלוֹמַר: כְּשֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא בָּרָא אֶת הָעוֹלָם, הוּא בְּרָאוֹ מִתּוֹךְ רְצוֹנוֹ הַבָּרוּר. הוּא לֹא הָיָה מֻכְרָח לִבְרֹא אֶת הָעוֹלָם, אֶלָּא הֶחְלִיט לִבְרֹא אֶת הָעוֹלָם מִתּוֹךְ רְצוֹנוֹ הַבָּרוּר וְהַמֻּחְלָט.

וּפוֹעֵל בַּשָּׁמַיִם וּבָאָרֶץ וּבְכָל אֲשֶׁר בָּהֶם כִּרְצוֹנוֹ הַמֻּחְלָט

הוּא בָּרָא אֶת הַשָּׁמַיִם מִתּוֹךְ רָצוֹן גָּמוּר, וְגַם בְּתוֹךְ הָעוֹלָם הַזֶּה הוּא עוֹשֶׂה מַה שֶּׁהוּא רוֹצֶה: פּוֹעֵל, נוֹתֵן, לוֹקֵחַ, מְסַדֵּר, מַנְהִיג, מַשְׁגִּיחַ – הַכֹּל מִתּוֹךְ רְצוֹנוֹ הַמֻּחְלָט.

אֶקַּח אֶת הַנְּקֻדָּה הַזֹּאת. אֵינִי אוֹמֵר שֶׁלֹּא יָדַעְנוּ אֶת הַדָּבָר הַזֶּה, אֲבָל כְּלָל לֹא חָשַׁבְנוּ עָלָיו. וְאוֹמַר עוֹד מִלָּה, וְגַם הִיא מְקֻבֶּלֶת – אֵינִי חָכָם וְאֵינִי גָּדוֹל, אֲבָל אֲנִי אוֹמֵר מַה שֶּׁבַּתְּחוּם שֶׁלִּי: הַדָּבָר הַזֶּה מֻטְבָּע בָּנוּ מִיַּלְדוּתֵנוּ. מִיּוֹם שֶׁנּוֹלַדְנוּ חָקוּק בְּרֹאשֵׁנוּ שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא הוּא בַּעַל הַבַּיִת, הוּא עוֹשֶׂה מַה שֶּׁהוּא רוֹצֶה, וְאִישׁ לֹא יֹאמַר לוֹ "מַה תַּעֲשֶׂה וּמַה תִּפְעָל". זֶה מֻטְבָּע בָּנוּ; נוֹלַדְנוּ עִם זֶה, בָּרוּךְ ה׳. אוֹי וַאֲבוֹי אִם לֹא הָיִינוּ נוֹלָדִים כָּךְ! וּמַדּוּעַ אֲנִי אוֹמֵר "אוֹי וַאֲבוֹי"? – נַמְשִׁיךְ הָלְאָה.

וְהוּא עַצְמוֹ מַה שֶּׁאָמַר רַבִּי אֶלְעָזָר עַל הַמַּאֲמָרִים הָאֵלּוּ עַצְמָם, בַּמִּדְרָשׁ שֶׁזָּכַרְנוּ רִאשׁוֹנָה: מִשֶּׁאָמַר הַקָּדוֹשׁ בָּרוּךְ הוּא דָּבָר זֶה בְּסִינַי, מֵאוֹתָהּ שָׁעָה וָאֵילָךְ לֹא תֵצֵא מִפִּי עֶלְיוֹן הָרָעוֹת וְהַטּוֹב

זֶהוּ הַמִּדְרָשׁ שֶׁקָּרָאנוּ עַתָּה בַּתְּחִלָּה; לֹא מֵעַצְמִי אָמַרְתִּי זֹאת.

זֻהֲמַת הַנָּחָשׁ וְאָבְדַן הַבְּחִירָה

וְהַכַּוָּנָה, כִּי אַף עַל פִּי שֶׁנִּבְרָא הָאָדָם בַּעַל בְּחִירָה

כְּלוֹמַר: כְּשֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא בָּרָא אֶת אָדָם הָרִאשׁוֹן, הוּא בְּרָאוֹ עִם בְּחִירָה. אָדָם הָרִאשׁוֹן – בְּרָאוֹ הַקָּדוֹשׁ בָּרוּךְ הוּא בַּעַל בְּחִירָה.

הִנֵּה כְּשֶׁבָּא נָחָשׁ עַל חַוָּה וְהִטִּיל בָּהּ זֻהֲמָא

לְפִי חֲזַ״ל, לְפִי הַמִּדְרָשִׁים (שבת קמו ע״א), הִטִּיל הַנָּחָשׁ בְּחַוָּה זֻהֲמָא. נָחָשׁ עִם אִשָּׁה – הָעִנְיָן עָמֹק; הַנָּחָשׁ הַזֶּה הָיָה עֲנָק וְכוּ׳. אֲבָל לֹא הַתַּאֲוָה הַזֹּאת, חַס וְשָׁלוֹם; אֶלָּא יֵשׁ לְקַבֵּל אֶת הַדָּבָר כְּמוֹ שֶׁהוּא.

נִמְשְׁכָה זֻהֲמָתוֹ בְּכָל הַדּוֹרוֹת הַנִּמְשָׁכִים אַחֲרֶיהָ

הַזֻּהֲמָא הַזֹּאת לֹא הָיְתָה נְקֻדָּתִית, בְּחַוָּה בִּלְבַד, אֶלָּא כָּל צֶאֱצָאֶיהָ שֶׁל חַוָּה הָיוּ נְגוּעִים בַּזֻּהֲמָא הַזֹּאת. וּמַה קָּרָה לָעוֹלָם אַחֲרֵי הַזֻּהֲמָא הַזֹּאת?

עַד שֶׁיִּהְיוּ כֻּלָּם נְגוּעִים וּמֻכִּים בְּשִׁגָּעוֹן וּבְעִוָּרוֹן

מִשָּׁעָה שֶׁהַזֻּהֲמָא הַזֹּאת חָדְרָה לָאֵם הָרִאשׁוֹנָה, הָיוּ כָּל הַנִּמְשָׁכִים אַחֲרֶיהָ מֻכִּים בְּעִוָּרוֹן וּבְשִׁגָּעוֹן –

אֲשֶׁר לֹא רָאוּ אֶת כְּבוֹד ה׳

בְּוַדַּאי שֶׁלֹּא. לָמָּה? לֹא מִשּׁוּם שֶׁהָיוּ יְלָדִים רָעִים; הֵם הָיוּ עִוְרִים, הָיוּ מְשֻׁגָּעִים.

וְלֹא יָדְעוּ כִּי הוּא יִתְעַלֶּה הַמַּנְהִיג אוֹתוֹ בְּחֵפֶץ וְרָצוֹן

הֵם בְּוַדַּאי הָיוּ מְשֻׁגָּעִים, עִוְרִים – לֹא עִוְרִים בָּעֵינַיִם, אֶלָּא עִוְרִים בַּמֹּחַ, בַּשֵּׂכֶל. לָכֵן לֹא יָדְעוּ יְסוֹד אֶחָד: שֶׁהַבּוֹרֵא יִתְבָּרַךְ מַנְהִיג אֶת הָעוֹלָם בְּחֵפֶץ וּבְרָצוֹן –

עַל יְדֵי כֹּשֶׁר מַעֲשֵׂי הָאֲנָשִׁים אוֹ הַהֵפֶךְ

כְּלוֹמַר: הֵם לֹא יָדְעוּ שֶׁה׳ יִתְבָּרַךְ מַנְהִיג אֶת הָעוֹלָם הַזֶּה בְּיַחַס לָאָדָם; הֵם לֹא קִשְּׁרוּ שֶׁהַנְהָגַת ה׳ קְשׁוּרָה בְּמַעֲשֵׂיהֶם. הֵם לֹא יָדְעוּ שֶׁה׳ מַנְהִיג אֶת עוֹלָמָם שֶׁל הָאֲנָשִׁים הַטּוֹבִים לְפִי הִתְנַהֲגוּתָם, וְשֶׁל הָאֲנָשִׁים הָרָעִים לְפִי מַעֲשֵׂיהֶם.

אֲבָל חָשְׁבוּ הַכֹּל לִמְחֻיָּב

מַה פֵּרוּשׁ "מְחֻיָּב"? חָשְׁבוּ שֶׁהָעוֹלָם הוּא פָּשׁוּט כָּזֶה. בְּמִלִּים פְּשׁוּטוֹת: לֹא חִבְּרוּ אֶת הַשָּׁמַיִם עִם הָאָרֶץ. קַח לְךָ אָדָם וְהוֹרֵד מִמֶּנּוּ אֶת הַמֻּשָּׂג "אֱמוּנָה", אוֹ "מַעֲשֶׂה טוֹב", אוֹ "מַעֲשֶׂה רַע" – שׁוּם דָּבָר: הוּא עִוֵּר, עִוֵּר בַּמֹּחַ שֶׁלּוֹ. כָּךְ הוּא נוֹלַד – נוֹלַד עִם פְּגִיעָה בָּרֹאשׁ, מַה שֶּׁקּוֹרְאִים לוֹ "שְׂרִיטָה בָּרֹאשׁ"; אֲבָל זֶה הַרְבֵּה יוֹתֵר מִשְּׂרִיטָה. הוּא נוֹלַד עִם "הָעוֹלָם הוּא כְּמוֹ שֶׁהוּא; אֵין כָּאן מִישֶׁהוּ שֶׁמַּנְהִיג אֶת הָעוֹלָם; אִם אֲנִי עוֹשֶׂה טוֹב אוֹ עוֹשֶׂה רַע – מָה הָלְאָה?".

"אֲבָל חָשְׁבוּ הַכֹּל לִמְחֻיָּב" – מַה פֵּרוּשׁ "לִמְחֻיָּב"? הַהֵפֶךְ מִמַּה שֶּׁאָמַרְנוּ, שֶׁהַבּוֹרֵא בָּרָא אֶת הָעוֹלָם מֵרְצוֹנוֹ הַטָּהוֹר וְהַמֻּחְלָט; וְזֶהוּ יְסוֹד הַיְסוֹדוֹת.

וְלֹא רָאוּ שׁוּם תּוֹעֶלֶת בִּחְיוֹת עַל הַטּוֹב וְהַיָּשָׁר מֵאֲשֶׁר בִּחְיוֹת עַל הָרָעוֹת

זוֹהִי הַהַשְׁלָכָה הַמַּעֲשִׂית: לֹא הָיָה לָהֶם שׁוּם יִתְרוֹן בִּהְיוֹת בֶּן אָדָם טוֹב וּבַעֲשִׂיַּת מַעֲשִׂים טוֹבִים, עַל פְּנֵי חַיִּים עִם הָרַע. קַח לְךָ אָדָם עִוֵּר – עִוֵּר בָּעֵינַיִם – הוּא אֵינוֹ יוֹדֵעַ לְאָן לָלֶכֶת. הַאִם הוּא יוֹדֵעַ לִבְחֹר בַּדֶּרֶךְ הַזֹּאת אוֹ בַּדֶּרֶךְ הַשְּׁנִיָּה? לֹא; אִם לֹא לָמַד קֹדֶם – אֵינוֹ יוֹדֵעַ. הַעְתֵּק עַתָּה אֶת הַדָּבָר הַזֶּה לְאָדָם שֶׁנּוֹלַד – יֵשׁ לוֹ אַבָּא וְאִמָּא, הַכֹּל בְּסֵדֶר – אֲבָל הוּא רוֹאֶה אֶת הָעוֹלָם כְּטֶבַע רָגִיל: לַעֲשׂוֹת טוֹב אוֹ לַעֲשׂוֹת רַע – אֵין אֶצְלוֹ הֶבְדֵּל בֵּין זֶה לָזֶה. וְאִם אֵין אָנוּ מְבִינִים זֹאת כָּל כָּךְ – אֵין זֶה נוֹרָא. לָמָּה אֵין אָנוּ מְבִינִים? כִּי אֵין אָנוּ חַיִּים כָּךְ.

וְאֵין בִּטּוּל הַבְּחִירָה גָּדוֹל מִזֶּה

מַה פֵּרוּשׁ? אָמַרְנוּ שֶׁכְּשֶׁה׳ בָּרָא אֶת אָדָם הָרִאשׁוֹן, הוּא בְּרָאוֹ בַּעַל בְּחִירָה. אֲבָל הַבְּחִירָה הַזֹּאת פָּעֲלָה לִפְנֵי סִפּוּר הַנָּחָשׁ עִם חַוָּה. מִשֶּׁהָיָה סִפּוּר הַנָּחָשׁ עִם חַוָּה, וְהָאֲנָשִׁים הָיוּ מֻכִּים בְּשִׁגָּעוֹן וּבְעִוָּרוֹן – כְּבָר תַּם כָּל עִנְיַן הַבְּחִירָה. לָאָדָם כְּבָר אֵין בְּחִירָה. לָמָּה? כִּי אַחֲרֵי שֶׁאִבֵּד אֶת הַהַרְגָּשָׁה, אֶת הַתְּבוּנָה לְהָבִין מַה הוּא עוֹשֶׂה – הוּא בִּטֵּל אֶת הַבְּחִירָה, אוֹתָהּ בְּחִירָה שֶׁנִּבְרָא עִמָּהּ. אֵין לוֹ עוֹד בְּחִירָה; כָּל מַה שֶּׁהוּא עוֹשֶׂה – עוֹשֶׂה בְּלִי בְּחִירָה. עוֹשֶׂה טוֹב, עוֹשֶׂה רַע – אֵינוֹ מַבְדִּיל בֵּין הַטּוֹב לָרַע. כַּמּוּבָן, קָשֶׁה לָנוּ לְקַבֵּל אֶת הַדְּבָרִים; אֲבָל נַמְשִׁיךְ.

"מִפִּי עֶלְיוֹן": דּוֹר הַמַּבּוּל, הַפַּלָּגָה וּסְדוֹם

וְהִנֵּה הָיוּ הַטּוֹבוֹת וְהָרָעוֹת בָּאוֹת בְּכָל הַדּוֹרוֹת הָהֵם מִפִּי עֶלְיוֹן

זֶהוּ דָּבָר חָדָשׁ וְאַדִּיר, שֶׁלֹּא שָׁמַעְנוּ וְלֹא הִכַּרְנוּ: בַּמַּצָּב הַזֶּה, הָעוֹלָם וְהָאָדָם – הַכֹּל הָיָה עַל פִּי עֶלְיוֹן, מִכֵּיוָן שֶׁלֹּא הָיְתָה בְּחִירָה לָאָדָם. כְּשֶׁיֵּשׁ לְךָ צִבּוּר שָׁלֵם הַמֻּכֶּה בְּשִׁגָּעוֹן וּבְעִוָּרוֹן, וְאֵין לוֹ בְּחִירָה – אַתָּה חַיָּב לְהַנְהִיג אוֹתוֹ. קַח צֹאן: אַתָּה חַיָּב לְהַנְהִיג אוֹתוֹ, שֶׁכֵּן אֶצְלוֹ הַבְּחִירָה בְּטֵלָה. אַתָּה מֻכְרָח לְהַנְהִיג אוֹתוֹ: לִפְעָמִים אַתָּה מַחְלִיט לְהַרְבִּיץ לוֹ, וְלִפְעָמִים אַתָּה מַחְלִיט לַעֲשׂוֹת לוֹ טוֹבָה.

וְאִם נַשְׁוֶה מְעַט אֶת הַדָּבָר לַדּוֹרוֹת שֶׁלָּנוּ – מֻתָּר לָנוּ עַתָּה לָקַחַת דֻּגְמָה: הַקּוֹמוּנִיזְם. זוֹ הָיְתָה שְׁלִיטָתוֹ: הוּא שָׁלַט לֹא רַק בַּמְּדִינָה וּבְמִשְׂרְדֵי הַמְּדִינָה – הוּא שָׁלַט בָּאָדָם. וְהַשְּׁלִיטָה הַזֹּאת עַכְשָׁו, בַּמּוֹדֶרְנָה שֶׁל חַיֵּינוּ: הַבִּינָה הַמְּלָאכוּתִית רוֹצָה לִשְׁלֹט בַּמֹּחַ שֶׁלְּךָ וּבַמֹּחַ שֶׁלִּי. זֶהוּ, לְמַעֲשֶׂה, אוֹתוֹ בִּטּוּל הַבְּחִירָה: הַבִּינָה הַמְּלָאכוּתִית עוֹשָׂה אֶת כָּל מַה שֶּׁהָאָדָם חוֹשֵׁב, כָּל מַה שֶּׁהוּא מְתַכְנֵן, וּמְבַטֶּלֶת אֶת מַחֲשַׁבְתּוֹ; הִיא חוֹדֶרֶת לְחַיֵּי הַיּוֹמְיוֹם שֶׁלּוֹ, עַד שֶׁהוּא מְאַבֵּד אֶת בְּחִירָתוֹ. הֲלָכָה לְמַעֲשֶׂה – זוֹ הַשְּׁלִיטָה כַּיּוֹם. וְהִנֵּה, אוֹתָם אֲנָשִׁים שֶׁיֵּשׁ לָהֶם הַרְבֵּה כֶּסֶף, הַמִּילְיַארְדֶּרִים, בְּמוּבָן הָאִינְטֶרְנַצְיוֹנָלִי – כְּלוֹמַר, הַמָּמוֹן – הֵם שׁוֹלְטִים. ה׳ יְרַחֵם עָלֵינוּ! וּבָרוּר לָנוּ שֶׁמִּבְּחִינַת הַתּוֹרָה – בְּוַדַּאי וּבְוַדַּאי שֶׁזֶּה כָּךְ. וּבְעֶזְרַת ה׳ נַמְשִׁיךְ לִלְמֹד אֶת הַנּוֹשֵׂא הַזֶּה, וְנִרְאֶה.

עַל כֵּן, אָז הָיְתָה שְׁלִיטָה מְלֵאָה: ה׳ הָיָה רוֹאֶה אֲנָשִׁים כִּילָדִים רָעִים – וּמַרְבִּיץ לָהֶם. זֶה כָּל מַה שֶּׁהָיָה. וְכַאֲשֶׁר נִתְמַלְּאָה סְאָתָם – כַּאֲשֶׁר רָאָה שֶׁהָרַע הוֹלֵךְ לְהַכְחִיד אֶת הָעוֹלָם – מֶה הָיָה עוֹשֶׂה? מֵבִיא עֲלֵיהֶם מַבּוּל; וּבְדוֹר הַפַּלָּגָה – בִּלְבֵּל אוֹתָם.

עַד כַּמָּה מַסְבִּיר הַמַּאֲמָר הַזֶּה אֶת הַתּוֹרָה הַקְּדוֹשָׁה!

וַיֹּאמֶר אֱלֹקִים לְנֹחַ קֵץ כָּל בָּשָׂר בָּא לְפָנַי כִּי מָלְאָה הָאָרֶץ חָמָס מִפְּנֵיהֶם.

(בראשית ו, יג)

לֹא רָאִינוּ שׁוּם דָּבָר; הַתּוֹרָה לֹא סִפְּרָה כְּלוּם, חוּץ מִפָּסוּק אוֹ שְׁנֵי פְּסוּקִים – וּפִתְאֹם ה׳ מַנְחִית אֶת הַפָּסוּק הַזֶּה. הֵיכָן נִכְנָס הַפָּסוּק הַזֶּה? הֵיכָן הַסִּפּוּרִים? אֵין שׁוּם דָּבָר. יוֹם אֶחָד בָּא ה׳ אֶל נֹחַ וְאוֹמֵר: "שְׁמַע, נֹחַ חֲבִיבִי, אֲנִי הוֹלֵךְ לִסְגֹּר; אֲנִי הוֹלֵךְ לְכַלּוֹת אֶת הָעוֹלָם". זֶה מַאֲמָר אֶחָד.

וְהַמַּאֲמָר הַשֵּׁנִי, שֶׁאֲנִי עַצְמִי מֵעוֹלָם לֹא הֵבַנְתִּי אוֹתוֹ: "וַיֹּאמֶר ה׳" – בְּפָרָשַׁת וַיֵּרָא, אַחֲרֵי שֶׁהַמַּלְאָכִים בָּאוּ לְבַשֵּׂר אֶת שָׂרָה, וְהִתְכּוֹנְנוּ לָלֶכֶת לִסְדוֹם לַהֲפֹךְ אוֹתָהּ. כָּךְ כָּתוּב: "וַיֹּאמֶר ה׳" – לְמִי אָמַר? לְעַצְמוֹ:

אֵרְדָה נָּא וְאֶרְאֶה הַכְּצַעֲקָתָהּ הַבָּאָה אֵלַי עָשׂוּ כָּלָה.

(בראשית יח, כא)

לַפָּסוּק הַזֶּה אֵין לֹא הַתְחָלָה וְלֹא סוֹף; הוּא סְתָם מֻנְחָת כָּךְ בָּאֶמְצַע. מַה זֹּאת אוֹמֶרֶת? הַקָּדוֹשׁ בָּרוּךְ הוּא – "אֵרְדָה נָּא וְאֶרְאֶה"? עַד עַכְשָׁו לֹא רָאִיתָ? מַה צָּרִיךְ בִּכְלָל לוֹמַר "אֵרְדָה נָּא וְאֶרְאֶה"? אַתָּה רוֹאֶה – אַתָּה צָרִיךְ לָרֶדֶת כְּדֵי לִרְאוֹת?!

מִכֵּיוָן שֶׁלָּאָדָם לֹא הָיְתָה בְּחִירָה כְּלָל, וְכֻלּוֹ הָיָה מֻנְהָג עַל יְדֵי הַקָּדוֹשׁ בָּרוּךְ הוּא, אוֹמֵר הַקָּדוֹשׁ בָּרוּךְ הוּא כָּךְ: כְּשֶׁרָאָה שֶׁהָעוֹלָם פָּחוֹת אוֹ יוֹתֵר מְאֻזָּן – הִמְשִׁיךְ לְהַנְהִיג אֶת הָעוֹלָם, וְכוּ׳. כָּל אָדָם קִבֵּל עֹנֶשׁ – אַךְ הוּא לֹא אָמַר "קִבַּלְתִּי עֹנֶשׁ מִפְּנֵי שֶׁחִלַּלְתִּי שַׁבָּת"; מָה פִּתְאֹם! אֵין קֶשֶׁר. כָּךְ הוּא הָעוֹלָם: קִבַּלְתִּי מַכָּה בָּרֶגֶל; קִבַּלְתִּי יַהֲלוֹם, מָצָאתִי יַהֲלוֹמִים – הוּא לֹא קָשַׁר זֹאת עִם מַעֲשָׂיו, אֶלָּא "כָּךְ": מָצָאתִי מְצִיאָה – אֵין שׁוּם קֶשֶׁר; לָקַחְתִּי מִמִּישֶׁהוּ כֶּסֶף...

"אֵרְדָה נָּא וְאֶרְאֶה הַכְּצַעֲקָתָהּ": כַּאֲשֶׁר רוֹאֶה הַקָּדוֹשׁ בָּרוּךְ הוּא שֶׁהַצֹּאן הַזֶּה, שֶׁאֵין לוֹ בְּחִירָה, לֹא טוֹב וְלֹא רַע – מֶה הוּא מַחְלִיט? לָרֶדֶת, וּלְהוֹרִיד לָזֶה אֶת חֶלְקוֹ. כְּמוֹ הָרוֹעֶה הָרוֹאֶה כֶּבֶשׂ אֶחָד שֶׁפּוֹגֵעַ בַּשֵּׁנִי – הוּא מוֹרִיד אוֹתוֹ. כְּמוֹ צֹאן; זֶהוּ.

וְהִנֵּה הָיוּ הַטּוֹבוֹת וְהָרָעוֹת בָּאוֹת בְּכָל הַדּוֹרוֹת הָהֵם מִפִּי עֶלְיוֹן, וְהוּא הַסֵּדֶר הַנָּתוּן בְּיַד מַעַרְכוֹת הַשָּׁמַיִם

זֹאת אוֹמֶרֶת: כָּךְ הִנְהִיג הַקָּדוֹשׁ בָּרוּךְ הוּא אֶת הָעוֹלָם בַּתְּקוּפָה הַהִיא. הוּא הָאָדוֹן, הַמּוֹלֵךְ, הַשּׁוֹלֵט, הַמַּפְתֵּחַ, הַשַּׁלִּיט; הוּא כִּוֵּן אֶת כָּל מַעַרְכוֹתָיו. וּבְשָׁעָה שֶׁרָאָה שֶׁהָעוֹלָם הִגִּיעַ לְקִצּוֹ – נָתַן אֶת הַפְּקֻדָּה לַמַּעֲרָכוֹת, וְהֵן הוֹרִידוּ מַבּוּל לָעוֹלָם. וְאִם כָּךְ – הַדָּבָר מַסְבִּיר לָנוּ עוֹד דְּבָרִים רַבִּים.

זוּלָתִי בִּפְעָמִים אֲשֶׁר נִתְמַלְּאָה סְאָתָם, כְּעִנְיַן הַמַּבּוּל וּסְדוֹם וְדוֹר הַפַּלָּגָה וְכַיּוֹצֵא, כְּמוֹ שֶׁזָּכַרְנוּ בְּפָרָשַׁת פְּקוּדֵי

אָמְנָם זוֹ הָיְתָה הַתְּקוּפָה שֶׁל דּוֹר הַמַּבּוּל וְדוֹר הַפַּלָּגָה, שֶׁבָּהּ הַכֹּל הָיָה בִּידֵי בּוֹרֵא עוֹלָם: הוּא הַמַּנְהִיג, הוּא הַמַּנְחִית אֶת הַטּוֹב וְהַמַּנְחִית אֶת הָרַע, וְאֵין הַדָּבָר תָּלוּי כְּלָל בְּמַעֲשֵׂי בְּנֵי הָאָדָם. בְּוַדַּאי יֵשׁ כָּאן אֶלֶף שְׁאֵלוֹת, אֲבָל אֵין כְּדַאי לָנוּ לְהִכָּנֵס בָּזֶה.

אַבְרָהָם אָבִינוּ – "קֹנֵה שָׁמַיִם וָאָרֶץ"

אָמְנָם כַּאֲשֶׁר זָרַח שִׁמְשׁוֹ שֶׁל אַבְרָהָם אָבִינוּ, אֲשֶׁר בָּא בְּכֹחַ גָּדוֹל וּבְיָד חֲזָקָה לְהַכִּיר אֶת בּוֹרְאוֹ, וְהִגִּיעַ לִכְלַל יְדִיעָה הֱיוֹתוֹ קֹנֵה שָׁמַיִם וָאָרֶץ

מֵעוֹלָם לֹא שָׁמַעְנוּ אֶת הַמִּלִּים הָאֵלֶּה: שֶׁאַבְרָהָם בָּא "בְּכֹחַ גָּדוֹל וּבְיָד חֲזָקָה" לְהַכִּיר אֶת בּוֹרְאוֹ, וְהִגִּיעַ לִכְלַל יְדִיעָה "הֱיוֹתוֹ קֹנֵה שָׁמַיִם וָאָרֶץ". אַבְרָהָם אָבִינוּ – לֹא סְתָם, כְּמוֹ בַּסִּפּוּרִים שֶׁלָּנוּ, שֶׁהָיְתָה לוֹ אֱמוּנָה, וְהָיָה אוֹמֵר לִבְנֵי אָדָם: "אַתֶּם יוֹדְעִים? יֵשׁ אֱלוֹהַּ אֶחָד בַּשָּׁמַיִם" וְכוּ׳. הָיָה זֶה הַרְבֵּה יוֹתֵר מִזֶּה. שֶׁאַבְרָהָם אָבִינוּ הִכִּיר שֶׁיֵּשׁ מַנְהִיג לָעוֹלָם – אֶת זֶה אָנוּ יוֹדְעִים; אֲבָל "הֱיוֹתוֹ קֹנֵה שָׁמַיִם וָאָרֶץ"!

עַכְשָׁו אָנוּ מְבִינִים אֶת הַפְּסוּקִים שָׁם, כְּשֶׁנִּפְגַּשׁ עִם מֶלֶךְ סְדוֹם אַחֲרֵי הַמִּלְחָמָה בְּאַרְבַּעַת הַמְּלָכִים (בראשית יד, יט–כב): הוּא יָדַע שֶׁה׳ יִתְבָּרַךְ הוּא בַּעַל הַבַּיִת בְּקִנְיָן – הוּא קָנָה אֶת הָעוֹלָם. זֶה מַה שֶּׁאָנוּ אוֹמְרִים בִּתְפִלַּת שַׁבָּת, "קֹנֵה שָׁמַיִם וָאָרֶץ", וְאֵין אָנוּ מְבִינִים כָּל כָּךְ מַה זֶּה "קֹנֶה".

זֹאת אוֹמֶרֶת: אֱמוּנָתוֹ שֶׁל אַבְרָהָם לֹא הָיְתָה אֱמוּנָה שֶׁל "אֲנִי מַאֲמִין שֶׁיֵּשׁ ה׳ בָּעוֹלָם"; אֱמוּנָתוֹ הָיְתָה שֶׁה׳ יִתְבָּרַךְ הוּא בַּעַל הַבַּיִת שֶׁל הָעוֹלָם – וְלֹא בַּעַל בַּיִת שֶׁקָּנָה אֶת הָעוֹלָם מִמִּישֶׁהוּ אַחֵר. "קֹנֵה שָׁמַיִם וָאָרֶץ" – כַּנִּרְאֶה, לַמִּלָּה "קִנְיָן" יֵשׁ מַשְׁמָעוּת רַבָּה הַרְבֵּה יוֹתֵר מִמַּה שֶּׁאָנוּ מְבִינִים. "קֹנֶה" פֵּרוּשׁוֹ: אֶת הָרְכוּשׁ הַזֶּה, הַנִּקְרָא "עוֹלָם" – הוּא יָצַר אוֹתוֹ; הוּא יָצַר אֶת הָרְכוּשׁ הַזֶּה הַנִּקְרָא "שָׁמַיִם וָאָרֶץ".

הָאֱמוּנָה בְּשַׁבָּת: יֵשׁ מֵאַיִן

אֶסְטֶה מְעַט מִן הָעִנְיָן. בַּעַל הָעֲקֵדָה אוֹמֵר שֶׁהָאֱמוּנָה שֶׁלָּנוּ בְּשַׁבָּת הִיא אֱמוּנָה מִינִימָלִית: אָנוּ מְבִינִים שֶׁה׳ עָבַד שִׁשָּׁה יָמִים, בָּרָא אֶת הָעוֹלָם בְּשִׁשָּׁה יָמִים, וְנָח בַּיּוֹם הַשְּׁבִיעִי. וְהוּא אוֹמֵר: לֹא זוֹ הָאֱמוּנָה שֶׁאָנוּ צְרִיכִים לְהַאֲמִין. הָאֱמוּנָה שֶׁאָנוּ צְרִיכִים לְהַאֲמִין בָּהּ הִיא שֶׁה׳ בָּרָא יֵשׁ מֵאַיִן. הָאֱמוּנָה שֶׁלָּנוּ בְּשַׁבָּת אֵינָהּ מַה שֶּׁאָנוּ אוֹמְרִים בַּקִּדּוּשׁ – שֶׁה׳ עָבַד שִׁשָּׁה יָמִים וְנָח – אֶלָּא הָאֱמוּנָה הִיא, לְדַיֵּק, לְדַקְדֵּק וּלְמַקֵּד: שֶׁה׳ בָּרָא יֵשׁ מֵאַיִן.

וּבָזֶה הוּא עוֹנֶה עַל הַשְּׁאֵלָה הַיְּדוּעָה, שֶׁדִּבַּרְנוּ עָלֶיהָ בְּשַׁבָּת: מֶה הָיָה הַקָּדוֹשׁ בָּרוּךְ הוּא צָרִיךְ לָתֵת לָנוּ כְּדֵי שֶׁנִּזְכֹּר שֶׁהוּא עָשָׂה מַעֲשֵׂה בְרֵאשִׁית? וּמַה נָּתַן לָנוּ? – שַׁבָּת. אֲבָל הַשַּׁבָּת אֵינָהּ מְעִידָה עַל שֵׁשֶׁת הַיָּמִים! הָיָה צָרִיךְ לַעֲשׂוֹת לָנוּ זִכָּרוֹן – לַעֲשׂוֹת מַשֶּׁהוּ שֶׁהוּא זֵכֶר לָעֲבוֹדָה, לְכָךְ שֶׁה׳ בָּרָא אֶת הָעוֹלָם. וּמָה עָשָׂה לָנוּ? נָתַן לָנוּ אֱמוּנָה שֶׁה׳ נָח בְּיוֹם הַשַּׁבָּת. נָכוֹן, הוּא נָח בְּיוֹם הַשַּׁבָּת אַחֲרֵי שֶׁעָבַד; אֲבָל לֵךְ יָשָׁר: עֲשֵׂה מַשֶּׁהוּ שֶׁהוּא זֵכֶר לַבְּרִיאָה, לֹא זֵכֶר לַמְּנוּחָה! זוֹ הַשְּׁאֵלָה שֶׁשּׁוֹאֲלִים, וְכָךְ הוּא עוֹנֶה עָלֶיהָ.

אִם כֵּן, אַבְרָהָם אָבִינוּ – אֱמוּנָתוֹ לֹא הָיְתָה הָאֱמוּנָה הַפְּשׁוּטָה שֶׁבָּהּ אָנוּ מַאֲמִינִים, אֶלָּא הוּא הִגִּיעַ לִידִיעָה. הַמִּלָּה הַזֹּאת, "יְדִיעָה" – לָדַעַת אֶת ה׳ – אֵינָהּ שַׁיֶּכֶת לַדּוֹרוֹת שֶׁלָּנוּ; הַדּוֹרוֹת שֶׁלָּנוּ – הַלְוַאי הָאֱמוּנָה הַפְּשׁוּטָה! אֲבָל אִם רוֹצִים אָנוּ לָדַעַת לְמָה הִגִּיעַ אַבְרָהָם אָבִינוּ – לֹא הָיָה זֶה רַק כְּמוֹ שֶׁשּׁוֹמְעִים בַּסִּפּוּרִים שֶׁבַּמִּדְרָשִׁים: "אֵין זֶה הַשֶּׁמֶשׁ – אִם כֵּן יֵשׁ מַנְהִיג לָעוֹלָם", וּבָזֶה סִיַּמְנוּ. לֹא! אַבְרָהָם אָבִינוּ הִגִּיעַ לִידִיעָה שֶׁה׳ יִתְבָּרַךְ הוּא שֶׁבָּרָא אֶת הָעוֹלָם, הוּא בַּעַל הַבַּיִת שֶׁל הָעוֹלָם, הוּא הַמַּנְהִיג, הוּא הַשַּׁלִּיט, הוּא עוֹשֶׂה מַה שֶּׁהוּא רוֹצֶה.

אִם כֵּן, יֵשׁ לָנוּ עֲבוֹדָה גְּדוֹלָה בְּשַׁבַּת קֹדֶשׁ – לְפָחוֹת בַּתְּפִלּוֹת. "תַּעֲנוּג"? "שַׁבָּת מְנוּחָה" – "אֲנִי נָח כְּמוֹ שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא נָח – אֵיזֶה תַּעֲנוּג!". לָמָּה? אַתָּה בָּרָאתָ אֶת הַשָּׁמַיִם וְאֶת הָאָרֶץ? מִמָּה תָּנוּחַ? הוּא עָבַד; אַתָּה לֹא עָבַדְתָּ – מִמָּה תָּנוּחַ? אַתָּה אוֹכֵל סְחִינָה וְכוּ׳.

בְּמָרוֹקוֹ – הָעֲרָבִים הָרָעִים הָאֵלֶּה – הָיוּ שָׁם רְחָבוֹת, כִּכָּרוֹת. לֹא הָיָה אָז הַמָּסָךְ הַזֶּה, אֲבָל הָיָה מַשֶּׁהוּ מְעַנְיֵן בְּכָל זֹאת: מְבִיאִים קוֹף וְאוֹמְרִים לוֹ: "עֲשֵׂה, עֲשֵׂה אֵיךְ הַיְּהוּדִי מִתְנַהֵג אַחֲרֵי שֶׁהוּא אוֹכֵל סְחִינָה בְּשַׁבָּת!". מֶה הָיָה הַקּוֹף עוֹשֶׂה? הָיָה שׁוֹכֵב עַל גַּבּוֹ וְשָׂם רֶגֶל עַל רֶגֶל. כָּךְ הָיָה הַיְּהוּדִי, בִּמְחִילָה מִכְּבוֹדֵנוּ. זֶה מַה שֶּׁאָנוּ עוֹשִׂים בְּשַׁבָּת: "עֹנֶג שַׁבָּת"! וַאֲנִי עוֹשֶׂה עוֹד – אַתָּה יוֹדֵעַ מָה? אֲנִי, וְאִתִּי יַיִן – אֵיזֶה יַיִן מְשֻׁבָּח! זֶה הָעֹנֶג? עַל מָה אַתָּה מִתְעַנֵּג?

לַעֲבֹד! לַעֲבֹד בִּתְפִלַּת שְׁמוֹנֶה עֶשְׂרֵה שֶׁל שַׁבָּת. הַתְּפִלָּה – לִפְנֵי שֶׁאַתָּה מַגִּיעַ לַשֻּׁלְחָן: עֲמֹד בִּתְפִלַּת לֵיל שַׁבַּת קֹדֶשׁ, רְאֵה מַה שֶּׁיֵּשׁ שָׁם בִּשְׁמוֹנֶה עֶשְׂרֵה, וְתִרְאֶה לְמָה תַּגִּיעַ. כָּל אֶחָד יָכוֹל לְהַגִּיעַ.

אַבְרָהָם מַחֲזִיר אֶת הַבְּחִירָה לִמְקוֹמָהּ

נַמְשִׁיךְ:

וּמַנְהִיגָם עַל דֶּרֶךְ הַחֵפֶץ וְהָרָצוֹן, כְּמוֹ שֶׁאָמַרְנוּ, וְהִתְחִיל לְפַרְסְמוֹ בָּעוֹלָם – הִנֵּה הוּא הֶחֱזִיר עִנְיַן הַבְּחִירָה לִמְקוֹמָהּ

אַחֲרֵי שֶׁהָעוֹלָם אִבֵּד אֶת בְּחִירָתוֹ – לְדוֹר הַמַּבּוּל וּלְדוֹר הַפַּלָּגָה לֹא הָיְתָה בְּחִירָה; הֵם הָיוּ מֻנְהָגִים כְּמוֹ צֹאן, וְאֵין עוֹד בְּחִירָה – בָּא אַבְרָהָם אָבִינוּ עָלָיו הַשָּׁלוֹם, וְהִגִּיעַ לִידִיעָה.

יְדִיעָה הִיא יְדִיעָה – לָדַעַת. "וְנֵדְעָה נִרְדְּפָה לָדַעַת אֶת ה׳" (הושע ו, ג), אוֹמֵר הַנָּבִיא. זֹאת אוֹמֶרֶת: כְּדֵי לְהַגִּיעַ לִידִיעָה צָרִיךְ לִרְדֹּף אַחַר הַקָּדוֹשׁ בָּרוּךְ הוּא, לְחַפֵּשׂ אוֹתוֹ, לְחַפֵּשׂ אוֹתוֹ בְּכָל חַיֶּיךָ – חַפֵּשׂ אוֹתוֹ, רְדֹף אַחֲרָיו. הֲרֵי כָּל הָעִנְיָן הוּא כְּמוֹ שֶׁאוֹמֵר הָאִבְּן עֶזְרָא בְּמָקוֹם אֶחָד: שֶׁבִּשְׁבִיל זֶה נִבְרָא הָאָדָם, בִּשְׁבִיל זֶה לְבַד נִבְרָא הָאָדָם. אִם כֵּן, כְּדֵי לָדַעַת אֶת הַקָּדוֹשׁ בָּרוּךְ הוּא – אַתָּה צָרִיךְ לִרְדֹּף.

בַּיְדִיעָה הַזֹּאת – שֶׁה׳ הוּא בַּעַל הַבַּיִת שֶׁל הָעוֹלָם, שֶׁהוּא מַנְהִיג אוֹתוֹ לְפִי רְצוֹנוֹ – הֶחֱזִיר אַבְרָהָם אֶת הַבְּחִירָה לָעוֹלָם. אָדָם הָרִאשׁוֹן קִבֵּל אֶת הַבְּחִירָה בִּבְרִיאָתוֹ, וְאִבֵּד אוֹתָהּ אַחֲרֵי זֻהֲמַת הַנָּחָשׁ; אַבְרָהָם אָבִינוּ הוּא שֶׁבָּא וְהֶחֱזִיר אֶת הַבְּחִירָה לָעוֹלָם. אֲבָל אֵיךְ מַחְזִירִים בְּחִירָה לְאָדָם שֶׁאִבֵּד אוֹתָהּ? זוֹ הַנְּקֻדָּה:

לְהוֹדִיעַ כִּי יֵשׁ שָׂכָר לִפְעֻלּוֹת הָאֲנָשִׁים

בַּבְּחִירָה הַזֹּאת שֶׁחִדֵּשׁ, הוּא לִמֵּד אֶת עַצְמוֹ וְלִמֵּד אֶת הַבָּאִים אַחֲרָיו שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא מַנְהִיג אֶת הָעוֹלָם לְפִי כֹּשֶׁר הַמַּעֲשִׂים – לְפִי כֹּשֶׁר מַעֲשָׂיו שֶׁל הָאָדָם: אִם הוּא כָּשֵׁר וְאִם הוּא פָּסוּל.

וְצִוָּה עָלֶיהָ אֶת בָּנָיו וְאֶת בֵּיתוֹ אַחֲרָיו

כָּאן אָנוּ נִכְנָסִים לַפָּסוּק הַנִּפְלָא הַזֶּה, שֶׁגַּם בּוֹ לֹא יָדַעְנוּ מַה הוּא עוֹשֶׂה, וּמַה הוּא מְחַפֵּשׂ פִּתְאֹם כָּאן. לִפְנֵי שֶׁה׳ נָתַן אֶת הַפְּקֻדָּה לְהַשְׁמִיד אֶת סְדוֹם, לְהַחֲרִיב אֶת סְדוֹם – מַה כָּתוּב שָׁם? "וַה׳ אָמָר הַמְכַסֶּה אֲנִי מֵאַבְרָהָם אֲשֶׁר אֲנִי עֹשֶׂה" (בראשית יח, יז). לִפְנֵי שֶׁנָּתַן אֶת הַפְּקֻדָּה לְהַחֲרִיב, אָמַר: אֲנִי מֻכְרָח לְדַבֵּר עִם אַבְרָהָם. לָמָּה? מַה הַפָּסוּק הַזֶּה עוֹשֶׂה כָּאן?

מוּבָן: אַנְשֵׁי סְדוֹם עֲדַיִן הָיוּ בְּעִוָּרוֹנָם, בְּשִׁגְעוֹנָם, וּבְסַךְ הַכֹּל הָעֹנֶשׁ שֶׁקִּבְּלוּ בָּא מִשּׁוּם שֶׁרָאָה ה׳ יִתְבָּרַךְ שֶׁהֵם הִטּוּ אֶת הָעוֹלָם לְהַשְׁמָדָה. אֲבָל יֵשׁ אֶחָד שֶׁנּוֹלַד חָדָשׁ: "זָרַח שִׁמְשׁוֹ" שֶׁל אַבְרָהָם. כֵּיצַד אוֹמֵר הַפָּסוּק? "זָרַח בַּחֹשֶׁךְ אוֹר לַיְשָׁרִים" (תהלים קיב, ד). וְאָז אוֹמֵר הַקָּדוֹשׁ בָּרוּךְ הוּא בְּדִיּוּק:

וַה׳ אָמָר הַמְכַסֶּה אֲנִי מֵאַבְרָהָם אֲשֶׁר אֲנִי עֹשֶׂה. וְאַבְרָהָם הָיוֹ יִהְיֶה לְגוֹי גָּדוֹל וְעָצוּם וְנִבְרְכוּ בוֹ כֹּל גּוֹיֵי הָאָרֶץ. כִּי יְדַעְתִּיו לְמַעַן אֲשֶׁר יְצַוֶּה אֶת בָּנָיו וְאֶת בֵּיתוֹ אַחֲרָיו וְשָׁמְרוּ דֶּרֶךְ ה׳ לַעֲשׂוֹת צְדָקָה וּמִשְׁפָּט לְמַעַן הָבִיא ה׳ עַל אַבְרָהָם אֵת אֲשֶׁר דִּבֶּר עָלָיו.

(בראשית יח, יז–יט)

זֶהוּ הַפָּסוּק. זֹאת אוֹמֶרֶת: מִשֶּׁבָּא אַבְרָהָם וְחִדֵּשׁ אֶת הַבְּחִירָה – שֶׁהָיְתָה קַיֶּמֶת בְּשָׁעָה שֶׁה׳ בָּרָא אֶת אָדָם הָרִאשׁוֹן, לִפְנֵי הַזֻּהֲמָא –

וְאַחֲרָיו נִמְשְׁכוּ הַבָּאִים, הֵם הָאָבוֹת הַקְּדוֹשִׁים

שִׁעְבּוּד מִצְרַיִם – הַכְשָׁרָה לַבְּחִירָה

כֵּיצַד לוֹמֵד בַּעַל הָעֲקֵדָה אֶת הַחֻמָּשׁ? הוּא מַסְבִּיר כָּךְ: אַחֲרֵי שֶׁיִּצְחָק וְיַעֲקֹב הִמְשִׁיכוּ אֶת דַּרְכָּם בַּבְּחִירָה הַזֹּאת שֶׁחִדֵּשׁ אַבְרָהָם – מַה קָּרָה אַחַר כָּךְ? יָרְדוּ לְמִצְרַיִם. מַה הַקֶּשֶׁר? כָּךְ אָנוּ קוֹרְאִים אֶת הַחֻמָּשׁ: הָיוּ סְדוֹם וַעֲמֹרָה; אַבְרָהָם, יִצְחָק וְיַעֲקֹב; הָיָה סִפּוּר יוֹסֵף וְאֶחָיו – וּמַה קָּרָה? יָרְדוּ לְמִצְרַיִם. מָה הוּא עוֹשֶׂה? הוּא אוֹמֵר:

אֲשֶׁר לְהַשְׁלִים עִנְיָנָהּ נִתְגַּלְגְּלוּ הַדְּבָרִים, וְיָרְדוּ לְהִשְׁתַּעְבֵּד בַּעֲבוֹדַת עֲבוֹדָה וַעֲבוֹדַת מַשָּׂא מִצְרַיִם, הַגּוֹי הַמַּר וְהַנִּמְהָר

כְּדֵי לְהַשְׁלִים אֶת עִנְיָנָהּ – עִנְיָן שֶׁל מָה? עִנְיַן הַבְּחִירָה – נִתְגַּלְגְּלוּ הַדְּבָרִים, וְיָרְדוּ לְהִשְׁתַּעְבֵּד.

אֲשֶׁר בְּקַבָּלָתָם הַשִּׁעְבּוּד, וִהְיוֹתָם אַחֲרֵי כֵן נִגְאָלִים עַל יְדֵי הַהַשְׁגָּחָה וְהַיְכֹלֶת הָאֱלֹקִי

זֹאת אוֹמֶרֶת: גִּלְגּוּל הַשִּׁעְבּוּד כֻּלּוֹ הָיָה לְשֵׁם מַטָּרָה אַחַת. מַה הַמַּטָּרָה? הֵם קִבְּלוּ – הֵם הִתְחִילוּ לִשְׁמֹעַ מִיַּעֲקֹב אָבִינוּ, הִתְחִילוּ לְהִתְנַהֵג עַל פִּי הַבְּחִירָה שֶׁחִדֵּשׁ אַבְרָהָם אָבִינוּ. אֲבָל כְּשֶׁבָּאִים וּמַנְחִיתִים עַל אָדָם – עַל אִישׁ סְדוֹם, לְמָשָׁל – אִם תַּנְחִית עָלָיו בְּחִירָה, הוּא כְּלָל אֵינוֹ מֵבִין עַל מָה אַתָּה מְדַבֵּר. מַה פֵּרוּשׁ? לָכֵן הוּא אוֹמֵר שֶׁיִּשְׁתַּעְבְּדוּ: הֵם עָבְדוּ אַרְבַּע מֵאוֹת שָׁנָה מֵרֹאשׁ, כְּדֵי לְהַכְשִׁיר אוֹתָם – לַהֲפֹךְ מֵאוֹתוֹ גֹּלֶם שֶׁהָיוּ אַנְשֵׁי סְדוֹם וַעֲמֹרָה, וְלִהְיוֹת הָאָדָם שֶׁל הַבְּחִירָה. אֲבָל מַה זֶּה בְּדִיּוּק "בְּחִירָה"? הוּא אוֹמֵר: צָרִיךְ אַרְבַּע מֵאוֹת שָׁנָה שֶׁל עֲבוֹדָה. הִגִּיעוּ לִכְלַל זֶה – וְאַחַר כָּךְ יָצְאוּ מִמִּצְרַיִם.

"בְּחֹזֶק יָד": חָכְמַת הַמַּכּוֹת

עַתָּה – יְצִיאַת מִצְרַיִם. דָּבָר נִפְלָא מְאֹד: בְּכָל פַּעַם שֶׁה׳ מְדַבֵּר עַל יְצִיאַת מִצְרַיִם, נֶאֱמַר "כִּי בְּחֹזֶק יָד הוֹצִיאָנוּ ה׳ מִמִּצְרָיִם" (שמות יג, טז). מַה זֶּה "בְּחֹזֶק יָד" הַזֶּה? מֶה הָיָה "חֹזֶק יָד" אֵצֶל הַקָּדוֹשׁ בָּרוּךְ הוּא?

הוּא אוֹמֵר שֶׁהַפְּשָׁט שֶׁל "חֹזֶק יָד" הוּא: כַּאֲשֶׁר הֵבִיא ה׳ יִתְבָּרַךְ אֶת עֶשֶׂר הַמַּכּוֹת עַל פַּרְעֹה, הוּא לֹא הֵבִיא אוֹתָן כְּמַכּוֹת שֶׁל עֹנֶשׁ – לְהַעֲנִישׁ אֶת פַּרְעֹה וְכוּ׳. לֹא! כְּשֶׁמִּתְבּוֹנְנִים, כְּשֶׁלּוֹמְדִים בַּסְּפָרִים הַקְּדוֹשִׁים וּבַמִּדְרָשִׁים, וְרוֹאִים כָּל מַכָּה וּמַכָּה – אֵיךְ הָיְתָה פּוֹעֶלֶת – שָׂמִים לֵב לְדָבָר נִפְלָא מְאֹד, וְהוּא: אֵיךְ הָיְתָה הַמַּכָּה, אֵיךְ פָּעֲלָה הַמַּכָּה.

בָּטוּחַ אֲנִי שֶׁכְּשֶׁיָּשְׁבוּ רַבִּי אֱלִיעֶזֶר וְרַבִּי עֲקִיבָא בִּבְנֵי בְרַק וְסִפְּרוּ בִּיצִיאַת מִצְרַיִם, הֵם לֹא אָמְרוּ כְּמוֹ שֶׁאָנוּ אוֹמְרִים – אָנוּ אֵינֶנּוּ יוֹדְעִים; אֶצְלֵנוּ אֵלּוּ כְּמוֹ חִידוֹת – אֶלָּא הֵם יָשְׁבוּ וְחָקְרוּ כָּל מַכָּה וּמַכָּה: אֵיךְ הִיא פָּעֲלָה. וּמִתּוֹךְ הַחֲקִירָה הַזֹּאת הֵבִינוּ אֵיךְ הִפְעִיל הַקָּדוֹשׁ בָּרוּךְ הוּא אֶת הַמַּכּוֹת הָהֵן. זֶהוּ "בְּחֹזֶק יָד": לֹא סְתָם עָשָׂה דָּם בַּמַּיִם וְכוּ׳, אֶלָּא – אֵיךְ עָשָׂה אֶת הַדָּם, אֵיךְ פָּעַל הַדָּם, אֵיזֶה דָּם, אֵיזֶה צֶבַע, וְכוּ׳ וְכוּ׳. הֵם יָדְעוּ לַחְקֹר – זֶהוּ "בְּחֹזֶק יָד".

כְּשֶׁבָּאִים וּמְבִיאִים צְפַרְדֵּעַ – בְּסֵדֶר, צְפַרְדֵּעַ. הַאִם פַּרְעֹה הֵבִין צְפַרְדֵּעַ? הוּא בִּקֵּשׁ: "הוֹרֵד מִמֶּנִּי אֶת הַצְּפַרְדֵּעַ". אֲבָל עַם יִשְׂרָאֵל, שֶׁיָּשְׁבוּ שָׁם וְהִשְׁתַּעְבְּדוּ וְסָבְלוּ – הֵם רָאוּ בַּצְּפַרְדֵּעַ אֶת הָאֹפֶן. "וְשַׂמְתִּי פְדֻת" (שמות ח, יט) – הַתּוֹרָה אוֹמֶרֶת זֹאת רַק בְּמִלָּה אוֹ שְׁתַּיִם: "וְשַׂמְתִּי פְדֻת", הֶבְדֵּל. אֲבָל כְּשֶׁאָדָם מִתְבּוֹנֵן בְּכָל מַכָּה וּמַכָּה, הוּא רוֹאֶה שֶׁכָּאן הָיְתָה חָכְמָה גְּדוֹלָה וְאַדִּירָה.

וּמָה עָשְׂתָה הַחָכְמָה הַזֹּאת? הִיא הִתְחִילָה לַחְדֹּר בָּהֶם, לְחַלְחֵל בָּהֶם, וְלַעֲשׂוֹת מֵהֶם – מָה? לַעֲשׂוֹת מֵהֶם אֲנָשִׁים; לְחַזֵּק אֶת הַבְּחִירָה הַזֹּאת. כִּי אִם הַקָּדוֹשׁ בָּרוּךְ הוּא הֵבִיא אֶת מַכַּת הַצְּפַרְדֵּעַ, הַדָּם וְהֶעָרֹב וְכוּ׳ – זֹאת אוֹמֶרֶת שֶׁיֵּשׁ מִישֶׁהוּ לְמַעְלָה שֶׁיּוֹדֵעַ בְּדִיּוּק מַה שֶּׁקּוֹרֶה לְמַטָּה, וְשֶׁהוּא בַּעַל הָרָצוֹן הַמֻּחְלָט, שֶׁעוֹשֶׂה אֶת הַכֹּל בְּהַשְׁגָּחָה פְּרָטִית, וְשֶׁהוּא הַשַּׁלִּיט בָּעוֹלָם. כָּל זֶה – מֶה עָשָׂה אֶצְלָם? אוֹ, בִּלְשׁוֹנוֹ:

עַל יְדֵי הַהַשְׁגָּחָה וְהַיְכֹלֶת הָאֱלֹקִי הִגִּיעוּ לִכְלַל זֶה הַדַּעַת לְגַמְרֵי, לְהַזְמִין רְצוֹנָם לַעֲבוֹדָתוֹ

כְּלוֹמַר: לְהַכְשִׁיר אוֹתָם לַעֲבוֹדָתוֹ. אֵלּוּ הֵם הַשְּׁלַבִּים: הַשָּׁלָב הַזֶּה הִכְשִׁיר אֶת עַם יִשְׂרָאֵל לְהַסְכִּים, לְהַתְחִיל לַעֲבֹד אֶת ה׳.

מַה הַפֵּרוּשׁ? "בְּסֵדֶר, ה׳ מַנְהִיג אֶת הָעוֹלָם; טוֹב, הוּא מַנְהִיג אֶת הָעוֹלָם – וּמָה אֲנִי קָשׁוּר לְכָךְ שֶׁהוּא מַנְהִיג אֶת הָעוֹלָם? בְּסֵדֶר, שֶׁיִּהְיֶה בָּרִיא...". כַּמָּה פְּעָמִים, בַּעֲוֹנוֹתֵינוּ הָרַבִּים, אָנוּ מַזְכִּירִים: "ה׳ סוֹמֵךְ נוֹפְלִים וְרוֹפֵא חוֹלִים וּמַתִּיר אֲסוּרִים". בִּמְחִילָה: לְאָדָם יֵשׁ חוֹלֶה בְּבֵית הַחוֹלִים, וְהוּא קוֹרֵא בִּשְׁמוֹנֶה עֶשְׂרֵה שֶׁה׳ "רוֹפֵא חוֹלִים", וְאוֹמֵר: "טוֹב, הוּא רוֹפֵא חוֹלִים – אֲבָל מַה עִם הַחוֹלֶה שֶׁלִּי?". הוּא מְרַפֵּא אֶת הַחוֹלִים – מַה הַפֵּרוּשׁ? ה׳ גָּדוֹל, ה׳ "סוֹמֵךְ נוֹפְלִים", "מַתִּיר אֲסוּרִים", הוּא "מוֹרִישׁ וּמַעֲשִׁיר" (שמואל א ב, ז) – אֲבָל, נֹאמַר, שֶׁיִּהְיֶה בָּרִיא: מָה הוּא נוֹתֵן לִי?

"לְשֵׁם שָׁמַיִם": פְּנִנָּה וְחַנָּה

כְּשֶׁבֶּן אָדָם רוֹאֶה אֶת תְּפִלָּתָהּ שֶׁל חַנָּה הַקְּדוֹשָׁה – אֵיךְ אִשָּׁה מְדַבֶּרֶת, אֵיךְ הִיא מְתָאֶרֶת – תֵּאוּרָהּ אֵינוֹ סְתָם שֶׁל אִשָּׁה שֶׁהִתְפַּלְּלָה וְכוּ׳.

רָאִיתִי פַּעַם מַאֲמָר שֶׁל הָרַב שָׁלוֹם שְׁבַדְרוֹן עָלָיו הַשָּׁלוֹם, שֶׁהָיָה דַּרְשָׁן גָּדוֹל בְּחֶסֶד, שֶׁהוּא אוֹמֵר שֶׁפְּנִנָּה הָיְתָה מְצִיקָה. אִם תַּעֲבֹר עַל כָּל חֻמַּשׁ בְּרֵאשִׁית – עֶרְכָּהּ שֶׁל הָאִשָּׁה הָיָה: הַאִם הִיא מְבִיאָה בָּנִים אוֹ אֵינָהּ מְבִיאָה בָּנִים. הַיְנוּ, אֵצֶל הָאִמָּהוֹת הַקְּדוֹשׁוֹת: שָׂרָה אִמֵּנוּ, שָׂרָה הַגְּדוֹלָה – שָׂרָה הִיא שֶׁהֵבִיאָה יוֹם אֶחָד אֶת הָגָר, שֶׁהֵבִיאָה לוֹ אֶת כָּל הַצָּרוֹת הָעֲצוּמוֹת, וְנָתְנָה אוֹתָהּ לְאַבְרָהָם. וְאָנוּ רוֹאִים שֶׁאַבְרָהָם אָבִינוּ הַגָּדוֹל הָיָה כָּפוּף לְשָׂרָה, עַד שֶׁה׳ אָמַר לוֹ: "כֹּל אֲשֶׁר תֹּאמַר אֵלֶיךָ שָׂרָה שְׁמַע בְּקֹלָהּ" (בראשית כא, יב). אֲבָל מִשּׁוּם שֶׁלֹּא הָיוּ לָהּ בָּנִים – "וַתִּקַּח שָׂרַי אֵשֶׁת אַבְרָם אֶת הָגָר הַמִּצְרִית שִׁפְחָתָהּ... וַתִּתֵּן אֹתָהּ לְאַבְרָם אִישָׁהּ לוֹ לְאִשָּׁה" (בראשית טז, ג). לָמָּה? כִּי בְּלִי בָּנִים. רָחֵל אִמֵּנוּ – "וַתֹּאמֶר אֶל יַעֲקֹב הָבָה לִּי בָנִים וְאִם אַיִן מֵתָה אָנֹכִי" (בראשית ל, א). אֵיךְ? זֹאת אוֹמֶרֶת: אִשָּׁה שֶׁאֵינָהּ מְבִיאָה בָּנִים – מַה תּוֹעַלְתָּהּ? בַּדּוֹרוֹת שֶׁלָּנוּ זֶה לֹא הוֹלֵךְ בְּדִיּוּק כָּךְ, אֲבָל נֶאֱמָנִים עָלֵינוּ דִּבְרֵי הַתּוֹרָה הַקְּדוֹשָׁה עַל כָּל פָּנִים.

נַחֲזֹר לָעִנְיָן שֶׁלָּנוּ. רָאִיתִי אֶת הַמַּאֲמָר הַזֶּה שֶׁל הָרַב שְׁבַדְרוֹן עָלָיו הַשָּׁלוֹם: פְּנִנָּה הָיְתָה מְצִיקָה לְחַנָּה – לָמָּה? "בְּלִי בָּנִים אַתְּ לֹא שָׁוָה כְּלוּם". הָיְתָה מְקַנְטֶרֶת אוֹתָהּ, מַשְׁפִּילָה אוֹתָהּ וּמְצַעֶרֶת אוֹתָהּ: "אַתְּ לֹא שָׁוָה כְּלוּם, וַאֲנִי הֵבֵאתִי שִׁבְעָה בָּנִים". פְּנִנָּה הָיְתָה הָאִשָּׁה הַשְּׁנִיָּה שֶׁל אֶלְקָנָה וְכוּ׳. וְהוּא שָׁאַל: אֵיךְ יָכוֹל לִהְיוֹת אַכְזָרִיּוּת כָּזוֹ שֶׁל אִשָּׁה לַחֲבֶרְתָּהּ?

בְּסֵדֶר – אֵצֶל הָגָר כָּתוּב "וַתֵּרֶא כִּי הָרָתָה"; וְכָךְ שָׂרָה מִתְלוֹנֶנֶת לִפְנֵי אַבְרָהָם אָבִינוּ: "יִשְׁפֹּט ה׳ בֵּינִי וּבֵינֶיךָ" (בראשית טז, ה), מִשּׁוּם שֶׁ"וַתֵּרֶא כִּי הָרָתָה וַתֵּקַל גְּבִרְתָּהּ בְּעֵינֶיהָ" (שם, ד). הָגָר הַמִּצְרִית – הַשִּׁפְחָה הַזֹּאת הִתְחִילָה לְזַלְזֵל בְּשָׂרָה, לֹא פָּחוֹת! – וְאָז יָצְאָה שָׂרָה מִגִּדְרָהּ וְאָמְרָה לְאַבְרָהָם כָּךְ וְכָךְ. אֲבָל פְּנִנָּה הָיְתָה מְצִיקָה בְּאַכְזָרִיּוּת כָּזוֹ שֶׁאֵין לְתָאֵר! בְּסֵדֶר – אַתְּ רַבַּת בָּנִים, וְהִיא אֵינָהּ מְבִיאָה בָּנִים; אֲבָל לֹא דַּי בְּצָרָתָהּ, שֶׁאֵינָהּ מְבִיאָה בָּנִים וְאֵינָהּ נֶחְשֶׁבֶת לְאִשָּׁה, כְּפִי שֶׁדִּבַּרְנוּ? לָמָּה אַתְּ מְצִיקָה לָהּ? שִׂמְחִי בְּחֶלְקֵךְ!

אָמַר עַל כָּךְ הָרַב שָׁלוֹם שְׁבַדְרוֹן דָּבָר נִפְלָא מְאֹד, כְּשֶׁרָאָה אֶת הַדּוֹר שֶׁלָּנוּ. הַדּוֹר שֶׁלָּנוּ הוּא כָּזֶה: כְּשֶׁמַּגִּיעַ, לְמָשָׁל, זְמַן הַבְּחִירוֹת – כָּל אוֹתָם אֲנָשִׁים, אֲפִלּוּ אַנְשֵׁי יִרְאַת שָׁמַיִם, לוֹמְדֵי תּוֹרָה, מִתְפַּלְּלִים, נִזְהָרִים בִּלְשׁוֹן הָרָע – הַכֹּל, הַכֹּל קַשׁ כְּשֶׁמַּגִּיעוֹת הַבְּחִירוֹת. הֵם מִפְלָגָה בִּפְנֵי עַצְמָהּ; הֵם לֹא מִן הַמִּפְלָגָה הַשְּׁנִיָּה – וְאָז כָּל הָאִסּוּרִים מֻתָּרִים לָהֶם: לָשׁוֹן הָרָע, לְהַכְפִּישׁ, לְשַׁקֵּר, לְרַמּוֹת – הַכֹּל. לָמָּה? מַה הַסִּבָּה? אֵיךְ יָכוֹל לִהְיוֹת שֶׁיְּהוּדִי שֶׁמַּנִּיחַ תְּפִלִּין, יְהוּדִי כָּשֵׁר – בַּבְּחִירוֹת, הָרַב הַהוּא, הָאַדְמוֹ״ר הַהוּא – הַכֹּל מֻתָּר?

כְּמוֹ שֶׁאָמַר אֶחָד מֵהַצְּעִירִים הָאֵלֶּה, שֶׁגְּדוֹלֵי הַדּוֹר צְרִיכִים לַעֲשׂוֹת תְּשׁוּבָה. שָׁאֲלוּ אוֹתוֹ: לָמָּה? – "אֲנִי אַגִּיד לְךָ: אֲנִי מְיַצֵּג בְּצֶדֶק, אֲנִי מְיַצֵּג אֶת הַדּוֹר הַצָּעִיר, שֶׁשּׁוֹלְחִים אֶת יַלְדֵיהֶם לַצָּבָא, וְהֵם נֶהֱרָגִים וְכוּ׳ – מַה מֵּבִין הַגָּדוֹל בַּתּוֹרָה, שֶׁהוּא אוֹמֵר 'נָמוּת וְלֹא נִתְגַּיֵּס׳?". הוּא מַתִּיר לְעַצְמוֹ הַכֹּל.

וְאָמַר הָרַב שְׁבַדְרוֹן דָּבָר נִפְלָא מְאֹד: רַק יְהוּדִים חֲרֵדִים מַגִּיעִים לַדַּרְגָּה הַזֹּאת בַּבְּחִירוֹת. לָמָּה? כִּי הֵם אוֹמְרִים: "אֲנִי עוֹשֶׂה אֶת זֶה לְשֵׁם שָׁמַיִם". כְּשֶׁבֶּן אָדָם אוֹמֵר "לְשֵׁם שָׁמַיִם" – מָה אַתָּה רוֹצֶה? אִם זֶה לְשֵׁם שָׁמַיִם, אָז בֶּטַח: מַה זֶּה לָשׁוֹן הָרָע לְעֻמַּת "לְשֵׁם שָׁמַיִם"? מַה זֶּה הָאֶפֶס הַזֶּה? מַה זֶּה הַ"גָּדוֹל" הַזֶּה? אֲנִי – "לְשֵׁם שָׁמַיִם"! מַה זֹּאת אוֹמֶרֶת "לְשֵׁם שָׁמַיִם"? זֶה מְכַסֶּה לְךָ עַל הַכֹּל.

וְכָךְ אָמַר הוּא עַצְמוֹ פַּעַם, וְהָיָה מְתָאֵר זֹאת יָפֶה: לִפְעָמִים הַיְּהוּדִי הַחֲרֵדִי נִמְצָא בָּאוּלָם, וְהַמֶּלְצַר מְחַלֵּק לְכֻלָּם – וְשָׁכַח מַשֶּׁהוּ, שָׁכַח אֵיזוֹ תּוֹסֶפֶת. אוֹמֵר הוּא, שֶׁהַיְּהוּדִי הַזֶּה רוֹדֵף אַחֲרָיו עַד הַמִּטְבָּח. "מַה זֶּה?!" – "נוּ, מָה אַתָּה אוֹמֵר? זוֹ סְעוּדַת מִצְוָה!". אוֹמֵר הָרַב שָׁלוֹם: רַק אִם זֶה "לְשֵׁם שָׁמַיִם" – רַק אָז אֶפְשָׁר לִהְיוֹת כָּךְ.

כִּי חֲזַ״ל אָמְרוּ: פְּנִנָּה לְשֵׁם שָׁמַיִם נִתְכַּוְּנָה (בבא בתרא טז ע״א). וְהוּא שָׁאַל: אֵיךְ פְּנִנָּה הָיְתָה רְשָׁעָה כָּזוֹ? אוֹמֵר: לֹא – פְּנִנָּה נִתְכַּוְּנָה לְשֵׁם שָׁמַיִם, וּבֶטַח נִתְכַּוְּנָה לְהָצִיק לָהּ. הַשְּׁאֵלָה הָיְתָה כָּזוֹ: הִיא נִתְכַּוְּנָה לְהָצִיק לָהּ כְּדֵי שֶׁתִּתְפַּלֵּל יוֹתֵר חָזָק. אֲבָל עַל זֶה הוּא שָׁאַל: בְּכָל זֹאת – טוֹב, בְּסֵדֶר, "תַּצְלִיחִי" – אֲבָל אַכְזָרִיּוּת כָּזוֹ?! לֹא יָכוֹל לִהְיוֹת; זֶה לֹא טִבְעוֹ שֶׁל בֶּן אָדָם. וְאָמַר: זֶה "לְשֵׁם שָׁמַיִם"; כְּשֶׁזֶּה "לְשֵׁם שָׁמַיִם" – מֻתָּר הַכֹּל.

"פָּסְקָה זֻהֲמָתָן"

זוֹ הַנְּקֻדָּה: כְּשֶׁאֲבוֹתֵינוּ בְּמִצְרַיִם רָאוּ אֶת מַכַּת הַצְּפַרְדֵּעַ, שָׂמוּ לֵב שֶׁכָּאן יֵשׁ הַשְׁגָּחָה פְּרָטִית – בַּצְּפַרְדֵּעַ. וְאָז קִשְּׁרוּ: אִם יֵשׁ הַשְׁגָּחָה פְּרָטִית, זֹאת אוֹמֶרֶת שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא פּוֹעֵל מִתּוֹךְ רְצוֹנוֹ, וְלֹא שֶׁהָיָה מְחֻיָּב לִבְרֹא עוֹלָם. מִי שֶׁמְּחֻיָּב לִבְרֹא עוֹלָם – אֵינוֹ בּוֹרֵא אוֹתוֹ לְפִי רְצוֹנוֹ; הוּא מְחֻיָּב, וּמַה שֶּׁבָּרָא – בָּרָא. אֲבָל הַנְּקֻדָּה הַזֹּאת שֶׁל הַבְּחִירָה הִתְחַדְּשָׁה לָהֶם כַּאֲשֶׁר רָאוּ אֶת הַמַּכּוֹת, וְכַאֲשֶׁר יָצְאוּ מִמִּצְרַיִם.

וְאָז, אוֹמֵר הוּא, אָנוּ מְבִינִים מַה זֶּה "פָּסְקָה" – זוֹ הַמַּטָּרָה. הֲרֵי כָּתוּב בְּדִבְרֵי רַבּוֹתֵינוּ: "פָּסְקָה זֻהֲמָתָן" (שבת קמו ע״א). מַה זּוֹ הַזֻּהֲמָא הַזֹּאת? עַל מָה מְדֻבָּר? מַה זֶּה "זֻהֲמָתָן"? הוּא מַסְבִּיר יָפֶה מְאֹד: הֲרֵי כְּשֶׁהַנָּחָשׁ הָיָה עִם חַוָּה, הוּא הִטִּיל בָּהּ זֻהֲמָא, וּבְעִקְבוֹת הַזֻּהֲמָא הַזֹּאת אִבֵּד הָאָדָם אֶת בְּחִירָתוֹ. כְּשֶׁהָיְתָה יְצִיאַת מִצְרַיִם וְהַשְׁגָּחָה פְּרָטִית – הֵם הִגִּיעוּ לַבְּחִירָה: שֶׁה׳ בָּרָא אֶת הָעוֹלָם מֵרְצוֹנוֹ הַטּוֹב וְכוּ׳.

הִגִּיעוּ לַנְּקֻדָּה שֶׁל הַבְּחִירָה, שֶׁהִתְחַדְּשָׁה אַחֲרֵי הַזֻּהֲמָא בְּאוֹתָהּ שָׁעָה שֶׁיָּצְאוּ מִמִּצְרַיִם, וְהֵבִינוּ שֶׁהָאָדָם הוּא בַּעַל בְּחִירָה. וְהַבְּחִירָה הִגִּיעָה לְדַרְגָּה מְסֻיֶּמֶת – עֲדַיִן לֹא נִגְמְרָה הַבְּחִירָה – אֲבָל הִגִּיעוּ לַדַּרְגָּה הַזֹּאת: שֶׁה׳ עוֹשֶׂה הַכֹּל בְּחֶפְצוֹ. הִגִּיעוּ לְדַרְגָּה מְסֻיֶּמֶת בַּבְּחִירָה הַזֹּאת. לָמָּה? מַה פִּתְאֹם? ה׳ רָאָה אֶת עָנְיֵנוּ, אֶת עֳנִי אֲבוֹתֵינוּ בְּמִצְרַיִם, וְאֶת זַעֲקָתָם שָׁמַע וְכוּ׳. הֵם הֵבִינוּ שֶׁבּוֹרֵא עוֹלָם, שֶׁבָּרָא אֶת הָעוֹלָם הַזֶּה וְרָאָה אֶת הָרַע וְאֶת הַטּוֹב – הוּא הִצִּיל אוֹתָם. הֵם הֵבִינוּ שֶׁיֵּשׁ כָּאן הַנְהָגָה אֱלֹקִית שְׁמֵימִית, הַפּוֹעֶלֶת לְפִי הָרָצוֹן וּלְפִי מַצָּבוֹ שֶׁל הָאָדָם:

וַתֵּרֶא אֶת עֳנִי אֲבֹתֵינוּ בְּמִצְרָיִם וְאֶת זַעֲקָתָם שָׁמַעְתָּ עַל יַם סוּף.

(נחמיה ט, ט)

שֶׁהוּא מַשְׁגִּיחַ וְכוּ׳ – הִנֵּה לְךָ בְּחִירָה! וּכְשֶׁהִגִּיעוּ לַדַּרְגָּה הַמְסֻיֶּמֶת הַזֹּאת בַּבְּחִירָה – "פָּסְקָה זֻהֲמָתָן": אוֹתָהּ זֻהֲמָא שֶׁלָּקְחָה אֶת הַבְּחִירָה מִן הָאָדָם – פָּסְקָה.

חֲתִימָה: לִשְׁמֹר עַל הַבְּחִירָה

לְעֵת עַתָּה – רַק לָשִׂים לֵב מַהוּ הַמֻּשָּׂג הַזֶּה, "בְּחִירָה". מֶה חָשַׁבְנוּ שֶׁהִיא בְּחִירָה? לֹא חָשַׁבְנוּ כְּלוּם. אֲבָל בְּעֶזְרַת ה׳, בְּהֶמְשֵׁךְ, נִלְמַד אֶת נְקֻדַּת הַשִּׁעוּר הַזֶּה לְפִי מַה שֶּׁבַּעַל הָעֲקֵדָה עָלָיו הַשָּׁלוֹם מְלַמֵּד אוֹתָנוּ, וְאָז נַתְחִיל לְהָבִין מַהִי בְּחִירָה.

דָּבָר אֶחָד בָּרוּר, שֶׁלָּמַדְנוּ הָעֶרֶב: בֶּן אָדָם עָשׂוּי לְאַבֵּד אֶת הַבְּחִירָה. כְּשֶׁנִּכְנָס בְּאָדָם שִׁגָּעוֹן וְעִוָּרוֹן, וְהוּא אֵינוֹ מֵבִין מַה הוּא עוֹשֶׂה, וְעוֹשֶׂה גַּם שְׁטֻיּוֹת – הוּא עָשׂוּי לְאַבֵּד גַּם אֶת בְּחִירָתוֹ. וְאָז מְבִינִים שֶׁהַבְּחִירָה הִיא דָּבָר עָדִין מְאֹד, דָּבָר שֶׁיֵּשׁ לְהַחְזִיק אוֹתוֹ בַּיָּדַיִם.

וְאָז צָרִיךְ לְהַתְחִיל לַחְשֹׁב: הַאִם יֵשׁ לִי בְּחִירָה, אוֹ אֵין לִי בְּחִירָה? וּמַגִּיעִים גַּם לְמַצָּב שֶׁאוֹמְרִים: מַה מַּבְדִּיל בֵּין אָדָם לְאָדָם? הַמַּבְדִּיל הוּא – הַאִם לָאָדָם הַזֶּה יֵשׁ מֵאָה אָחוּז בְּחִירָה, אוֹ שֶׁהוּא נָתוּן בְּעִוָּרוֹן וּבְשִׁגָּעוֹן, כְּמוֹ שֶׁלָּמַדְנוּ.

הַקָּדוֹשׁ בָּרוּךְ הוּא יַעֲזֹר, וְנַמְשִׁיךְ. נְסַכֵּם אֶת הַדָּבָר הַזֶּה, וְנַתְחִיל לְהִתְפַּלֵּל עַל דָּבָר אֶחָד: "הַקָּדוֹשׁ בָּרוּךְ הוּא, עֲזֹר לִי שֶׁלֹּא אֲאַבֵּד אֶת בְּחִירָתִי!". כִּי הָאָדָם שֶׁמָּכוּר לִשְׁתִיָּה – אוֹמְרִים: הוּא מָכוּר. מַה זֶּה "מָכוּר"? בְּמִלִּים שֶׁל הַתֵּאוֹרְיָה, שֶׁל הַפִילוֹסוֹפְיָה שֶׁל בַּעַל הָעֲקֵדָה: הוּא פָּשׁוּט אִבֵּד אֶת זְכוּת הַבְּחִירָה עַל הַיַּיִן. הוּא כְּבָר בּוֹחֵר בַּיַּיִן – זוֹ הַבְּחִירָה שֶׁלּוֹ; אֵין לוֹ בְּחִירָה לְגַבֵּי הַיַּיִן.

צָרִיךְ אָדָם לְהִזָּהֵר שֶׁלֹּא יְאַבֵּד אֶת בְּחִירָתוֹ. הַקָּדוֹשׁ בָּרוּךְ הוּא יַעֲזֹר.
