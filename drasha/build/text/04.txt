== שִׁעְבּוּד מִצְרַיִם – הַכְשָׁרָה לַבְּחִירָה

כֵּיצַד לוֹמֵד בַּעַל הָעֲקֵדָה אֶת הַחֻמָּשׁ? הוּא מַסְבִּיר כָּךְ: אַחֲרֵי שֶׁיִּצְחָק וְיַעֲקֹב הִמְשִׁיכוּ אֶת דַּרְכָּם בַּבְּחִירָה הַזֹּאת שֶׁחִדֵּשׁ אַבְרָהָם – מַה קָּרָה אַחַר כָּךְ? יָרְדוּ לְמִצְרַיִם. מַה הַקֶּשֶׁר? כָּךְ אָנוּ קוֹרְאִים אֶת הַחֻמָּשׁ: הָיוּ סְדוֹם וַעֲמֹרָה; אַבְרָהָם, יִצְחָק וְיַעֲקֹב; הָיָה סִפּוּר יוֹסֵף וְאֶחָיו – וּמַה קָּרָה? יָרְדוּ לְמִצְרַיִם. מָה הוּא עוֹשֶׂה? הוּא אוֹמֵר:

@ אֲשֶׁר לְהַשְׁלִים עִנְיָנָהּ נִתְגַּלְגְּלוּ הַדְּבָרִים, וְיָרְדוּ לְהִשְׁתַּעְבֵּד בַּעֲבוֹדַת עֲבוֹדָה וַעֲבוֹדַת מַשָּׂא מִצְרַיִם, הַגּוֹי הַמַּר וְהַנִּמְהָר

כְּדֵי לְהַשְׁלִים אֶת עִנְיָנָהּ – עִנְיָן שֶׁל מָה? עִנְיַן הַבְּחִירָה – נִתְגַּלְגְּלוּ הַדְּבָרִים, וְיָרְדוּ לְהִשְׁתַּעְבֵּד.

@ אֲשֶׁר בְּקַבָּלָתָם הַשִּׁעְבּוּד, וִהְיוֹתָם אַחֲרֵי כֵן נִגְאָלִים עַל יְדֵי הַהַשְׁגָּחָה וְהַיְכֹלֶת הָאֱלֹקִי

זֹאת אוֹמֶרֶת: גִּלְגּוּל הַשִּׁעְבּוּד כֻּלּוֹ הָיָה לְשֵׁם מַטָּרָה אַחַת. מַה הַמַּטָּרָה? הֵם קִבְּלוּ – הֵם הִתְחִילוּ לִשְׁמֹעַ מִיַּעֲקֹב אָבִינוּ, הִתְחִילוּ לְהִתְנַהֵג עַל פִּי הַבְּחִירָה שֶׁחִדֵּשׁ אַבְרָהָם אָבִינוּ. אֲבָל כְּשֶׁבָּאִים וּמַנְחִיתִים עַל אָדָם – עַל אִישׁ סְדוֹם, לְמָשָׁל – אִם תַּנְחִית עָלָיו בְּחִירָה, הוּא כְּלָל אֵינוֹ מֵבִין עַל מָה אַתָּה מְדַבֵּר. מַה פֵּרוּשׁ? לָכֵן הוּא אוֹמֵר שֶׁיִּשְׁתַּעְבְּדוּ: הֵם עָבְדוּ אַרְבַּע מֵאוֹת שָׁנָה מֵרֹאשׁ, כְּדֵי לְהַכְשִׁיר אוֹתָם – לַהֲפֹךְ מֵאוֹתוֹ גֹּלֶם שֶׁהָיוּ אַנְשֵׁי סְדוֹם וַעֲמֹרָה, וְלִהְיוֹת הָאָדָם שֶׁל הַבְּחִירָה. אֲבָל מַה זֶּה בְּדִיּוּק "בְּחִירָה"? הוּא אוֹמֵר: צָרִיךְ אַרְבַּע מֵאוֹת שָׁנָה שֶׁל עֲבוֹדָה. הִגִּיעוּ לִכְלַל זֶה – וְאַחַר כָּךְ יָצְאוּ מִמִּצְרַיִם.

== "בְּחֹזֶק יָד": חָכְמַת הַמַּכּוֹת

עַתָּה – יְצִיאַת מִצְרַיִם. דָּבָר נִפְלָא מְאֹד: בְּכָל פַּעַם שֶׁה' מְדַבֵּר עַל יְצִיאַת מִצְרַיִם, נֶאֱמַר [[כִּי בְּחֹזֶק יָד הוֹצִיאָנוּ ה' מִמִּצְרָיִם]] {שמות יג, טז}. מַה זֶּה "בְּחֹזֶק יָד" הַזֶּה? מֶה הָיָה "חֹזֶק יָד" אֵצֶל הַקָּדוֹשׁ בָּרוּךְ הוּא?

הוּא אוֹמֵר שֶׁהַפְּשָׁט שֶׁל "חֹזֶק יָד" הוּא: כַּאֲשֶׁר הֵבִיא ה' יִתְבָּרַךְ אֶת עֶשֶׂר הַמַּכּוֹת עַל פַּרְעֹה, הוּא לֹא הֵבִיא אוֹתָן כְּמַכּוֹת שֶׁל עֹנֶשׁ – לְהַעֲנִישׁ אֶת פַּרְעֹה וְכוּ'. לֹא! כְּשֶׁמִּתְבּוֹנְנִים, כְּשֶׁלּוֹמְדִים בַּסְּפָרִים הַקְּדוֹשִׁים וּבַמִּדְרָשִׁים, וְרוֹאִים כָּל מַכָּה וּמַכָּה – אֵיךְ הָיְתָה פּוֹעֶלֶת – שָׂמִים לֵב לְדָבָר נִפְלָא מְאֹד, וְהוּא: אֵיךְ הָיְתָה הַמַּכָּה, אֵיךְ פָּעֲלָה הַמַּכָּה.

בָּטוּחַ אֲנִי שֶׁכְּשֶׁיָּשְׁבוּ רַבִּי אֱלִיעֶזֶר וְרַבִּי עֲקִיבָא בִּבְנֵי בְרַק וְסִפְּרוּ בִּיצִיאַת מִצְרַיִם, הֵם לֹא אָמְרוּ כְּמוֹ שֶׁאָנוּ אוֹמְרִים – אָנוּ אֵינֶנּוּ יוֹדְעִים; אֶצְלֵנוּ אֵלּוּ כְּמוֹ חִידוֹת – אֶלָּא הֵם יָשְׁבוּ וְחָקְרוּ כָּל מַכָּה וּמַכָּה: אֵיךְ הִיא פָּעֲלָה. וּמִתּוֹךְ הַחֲקִירָה הַזֹּאת הֵבִינוּ אֵיךְ הִפְעִיל הַקָּדוֹשׁ בָּרוּךְ הוּא אֶת הַמַּכּוֹת הָהֵן. זֶהוּ "בְּחֹזֶק יָד": לֹא סְתָם עָשָׂה דָּם בַּמַּיִם וְכוּ', אֶלָּא – אֵיךְ עָשָׂה אֶת הַדָּם, אֵיךְ פָּעַל הַדָּם, אֵיזֶה דָּם, אֵיזֶה צֶבַע, וְכוּ' וְכוּ'. הֵם יָדְעוּ לַחְקֹר – זֶהוּ "בְּחֹזֶק יָד".

כְּשֶׁבָּאִים וּמְבִיאִים צְפַרְדֵּעַ – בְּסֵדֶר, צְפַרְדֵּעַ. הַאִם פַּרְעֹה הֵבִין צְפַרְדֵּעַ? הוּא בִּקֵּשׁ: "הוֹרֵד מִמֶּנִּי אֶת הַצְּפַרְדֵּעַ". אֲבָל עַם יִשְׂרָאֵל, שֶׁיָּשְׁבוּ שָׁם וְהִשְׁתַּעְבְּדוּ וְסָבְלוּ – הֵם רָאוּ בַּצְּפַרְדֵּעַ אֶת הָאֹפֶן. [[וְשַׂמְתִּי פְדֻת]] {שמות ח, יט} – הַתּוֹרָה אוֹמֶרֶת זֹאת רַק בְּמִלָּה אוֹ שְׁתַּיִם: "וְשַׂמְתִּי פְדֻת", הֶבְדֵּל. אֲבָל כְּשֶׁאָדָם מִתְבּוֹנֵן בְּכָל מַכָּה וּמַכָּה, הוּא רוֹאֶה שֶׁכָּאן הָיְתָה חָכְמָה גְּדוֹלָה וְאַדִּירָה.

וּמָה עָשְׂתָה הַחָכְמָה הַזֹּאת? הִיא הִתְחִילָה לַחְדֹּר בָּהֶם, לְחַלְחֵל בָּהֶם, וְלַעֲשׂוֹת מֵהֶם – מָה? לַעֲשׂוֹת מֵהֶם אֲנָשִׁים; לְחַזֵּק אֶת הַבְּחִירָה הַזֹּאת. כִּי אִם הַקָּדוֹשׁ בָּרוּךְ הוּא הֵבִיא אֶת מַכַּת הַצְּפַרְדֵּעַ, הַדָּם וְהֶעָרֹב וְכוּ' – זֹאת אוֹמֶרֶת שֶׁיֵּשׁ מִישֶׁהוּ לְמַעְלָה שֶׁיּוֹדֵעַ בְּדִיּוּק מַה שֶּׁקּוֹרֶה לְמַטָּה, וְשֶׁהוּא בַּעַל הָרָצוֹן הַמֻּחְלָט, שֶׁעוֹשֶׂה אֶת הַכֹּל בְּהַשְׁגָּחָה פְּרָטִית, וְשֶׁהוּא הַשַּׁלִּיט בָּעוֹלָם. כָּל זֶה – מֶה עָשָׂה אֶצְלָם? אוֹ, בִּלְשׁוֹנוֹ:

@ עַל יְדֵי הַהַשְׁגָּחָה וְהַיְכֹלֶת הָאֱלֹקִי הִגִּיעוּ לִכְלַל זֶה הַדַּעַת לְגַמְרֵי, לְהַזְמִין רְצוֹנָם לַעֲבוֹדָתוֹ

כְּלוֹמַר: לְהַכְשִׁיר אוֹתָם לַעֲבוֹדָתוֹ. אֵלּוּ הֵם הַשְּׁלַבִּים: הַשָּׁלָב הַזֶּה הִכְשִׁיר אֶת עַם יִשְׂרָאֵל לְהַסְכִּים, לְהַתְחִיל לַעֲבֹד אֶת ה'.

מַה הַפֵּרוּשׁ? "בְּסֵדֶר, ה' מַנְהִיג אֶת הָעוֹלָם; טוֹב, הוּא מַנְהִיג אֶת הָעוֹלָם – וּמָה אֲנִי קָשׁוּר לְכָךְ שֶׁהוּא מַנְהִיג אֶת הָעוֹלָם? בְּסֵדֶר, שֶׁיִּהְיֶה בָּרִיא...". כַּמָּה פְּעָמִים, בַּעֲוֹנוֹתֵינוּ הָרַבִּים, אָנוּ מַזְכִּירִים: "ה' סוֹמֵךְ נוֹפְלִים וְרוֹפֵא חוֹלִים וּמַתִּיר אֲסוּרִים". בִּמְחִילָה: לְאָדָם יֵשׁ חוֹלֶה בְּבֵית הַחוֹלִים, וְהוּא קוֹרֵא בִּשְׁמוֹנֶה עֶשְׂרֵה שֶׁה' "רוֹפֵא חוֹלִים", וְאוֹמֵר: "טוֹב, הוּא רוֹפֵא חוֹלִים – אֲבָל מַה עִם הַחוֹלֶה שֶׁלִּי?". הוּא מְרַפֵּא אֶת הַחוֹלִים – מַה הַפֵּרוּשׁ? ה' גָּדוֹל, ה' "סוֹמֵךְ נוֹפְלִים", "מַתִּיר אֲסוּרִים", הוּא [[מוֹרִישׁ וּמַעֲשִׁיר]] {שמואל א ב, ז} – אֲבָל, נֹאמַר, שֶׁיִּהְיֶה בָּרִיא: מָה הוּא נוֹתֵן לִי?

== "לְשֵׁם שָׁמַיִם": פְּנִנָּה וְחַנָּה

כְּשֶׁבֶּן אָדָם רוֹאֶה אֶת תְּפִלָּתָהּ שֶׁל חַנָּה הַקְּדוֹשָׁה – אֵיךְ אִשָּׁה מְדַבֶּרֶת, אֵיךְ הִיא מְתָאֶרֶת – תֵּאוּרָהּ אֵינוֹ סְתָם שֶׁל אִשָּׁה שֶׁהִתְפַּלְּלָה וְכוּ'.

רָאִיתִי פַּעַם מַאֲמָר שֶׁל הָרַב שָׁלוֹם שְׁבַדְרוֹן עָלָיו הַשָּׁלוֹם, שֶׁהָיָה דַּרְשָׁן גָּדוֹל בְּחֶסֶד, שֶׁהוּא אוֹמֵר שֶׁפְּנִנָּה הָיְתָה מְצִיקָה. אִם תַּעֲבֹר עַל כָּל חֻמַּשׁ בְּרֵאשִׁית – עֶרְכָּהּ שֶׁל הָאִשָּׁה הָיָה: הַאִם הִיא מְבִיאָה בָּנִים אוֹ אֵינָהּ מְבִיאָה בָּנִים. הַיְנוּ, אֵצֶל הָאִמָּהוֹת הַקְּדוֹשׁוֹת: שָׂרָה אִמֵּנוּ, שָׂרָה הַגְּדוֹלָה – שָׂרָה הִיא שֶׁהֵבִיאָה יוֹם אֶחָד אֶת הָגָר, שֶׁהֵבִיאָה לוֹ אֶת כָּל הַצָּרוֹת הָעֲצוּמוֹת, וְנָתְנָה אוֹתָהּ לְאַבְרָהָם. וְאָנוּ רוֹאִים שֶׁאַבְרָהָם אָבִינוּ הַגָּדוֹל הָיָה כָּפוּף לְשָׂרָה, עַד שֶׁה' אָמַר לוֹ: [[כֹּל אֲשֶׁר תֹּאמַר אֵלֶיךָ שָׂרָה שְׁמַע בְּקֹלָהּ]] {בראשית כא, יב}. אֲבָל מִשּׁוּם שֶׁלֹּא הָיוּ לָהּ בָּנִים – [[וַתִּקַּח שָׂרַי אֵשֶׁת אַבְרָם אֶת הָגָר הַמִּצְרִית שִׁפְחָתָהּ... וַתִּתֵּן אֹתָהּ לְאַבְרָם אִישָׁהּ לוֹ לְאִשָּׁה]] {בראשית טז, ג}. לָמָּה? כִּי בְּלִי בָּנִים. רָחֵל אִמֵּנוּ – [[וַתֹּאמֶר אֶל יַעֲקֹב הָבָה לִּי בָנִים וְאִם אַיִן מֵתָה אָנֹכִי]] {בראשית ל, א}. אֵיךְ? זֹאת אוֹמֶרֶת: אִשָּׁה שֶׁאֵינָהּ מְבִיאָה בָּנִים – מַה תּוֹעַלְתָּהּ? בַּדּוֹרוֹת שֶׁלָּנוּ זֶה לֹא הוֹלֵךְ בְּדִיּוּק כָּךְ, אֲבָל נֶאֱמָנִים עָלֵינוּ דִּבְרֵי הַתּוֹרָה הַקְּדוֹשָׁה עַל כָּל פָּנִים.

נַחֲזֹר לָעִנְיָן שֶׁלָּנוּ. רָאִיתִי אֶת הַמַּאֲמָר הַזֶּה שֶׁל הָרַב שְׁבַדְרוֹן עָלָיו הַשָּׁלוֹם: פְּנִנָּה הָיְתָה מְצִיקָה לְחַנָּה – לָמָּה? "בְּלִי בָּנִים אַתְּ לֹא שָׁוָה כְּלוּם". הָיְתָה מְקַנְטֶרֶת אוֹתָהּ, מַשְׁפִּילָה אוֹתָהּ וּמְצַעֶרֶת אוֹתָהּ: "אַתְּ לֹא שָׁוָה כְּלוּם, וַאֲנִי הֵבֵאתִי שִׁבְעָה בָּנִים". פְּנִנָּה הָיְתָה הָאִשָּׁה הַשְּׁנִיָּה שֶׁל אֶלְקָנָה וְכוּ'. וְהוּא שָׁאַל: אֵיךְ יָכוֹל לִהְיוֹת אַכְזָרִיּוּת כָּזוֹ שֶׁל אִשָּׁה לַחֲבֶרְתָּהּ?

בְּסֵדֶר – אֵצֶל הָגָר כָּתוּב "וַתֵּרֶא כִּי הָרָתָה"; וְכָךְ שָׂרָה מִתְלוֹנֶנֶת לִפְנֵי אַבְרָהָם אָבִינוּ: [[יִשְׁפֹּט ה' בֵּינִי וּבֵינֶיךָ]] {בראשית טז, ה}, מִשּׁוּם שֶׁ[[וַתֵּרֶא כִּי הָרָתָה וַתֵּקַל גְּבִרְתָּהּ בְּעֵינֶיהָ]] {שם, ד}. הָגָר הַמִּצְרִית – הַשִּׁפְחָה הַזֹּאת הִתְחִילָה לְזַלְזֵל בְּשָׂרָה, לֹא פָּחוֹת! – וְאָז יָצְאָה שָׂרָה מִגִּדְרָהּ וְאָמְרָה לְאַבְרָהָם כָּךְ וְכָךְ. אֲבָל פְּנִנָּה הָיְתָה מְצִיקָה בְּאַכְזָרִיּוּת כָּזוֹ שֶׁאֵין לְתָאֵר! בְּסֵדֶר – אַתְּ רַבַּת בָּנִים, וְהִיא אֵינָהּ מְבִיאָה בָּנִים; אֲבָל לֹא דַּי בְּצָרָתָהּ, שֶׁאֵינָהּ מְבִיאָה בָּנִים וְאֵינָהּ נֶחְשֶׁבֶת לְאִשָּׁה, כְּפִי שֶׁדִּבַּרְנוּ? לָמָּה אַתְּ מְצִיקָה לָהּ? שִׂמְחִי בְּחֶלְקֵךְ!

אָמַר עַל כָּךְ הָרַב שָׁלוֹם שְׁבַדְרוֹן דָּבָר נִפְלָא מְאֹד, כְּשֶׁרָאָה אֶת הַדּוֹר שֶׁלָּנוּ. הַדּוֹר שֶׁלָּנוּ הוּא כָּזֶה: כְּשֶׁמַּגִּיעַ, לְמָשָׁל, זְמַן הַבְּחִירוֹת – כָּל אוֹתָם אֲנָשִׁים, אֲפִלּוּ אַנְשֵׁי יִרְאַת שָׁמַיִם, לוֹמְדֵי תּוֹרָה, מִתְפַּלְּלִים, נִזְהָרִים בִּלְשׁוֹן הָרָע – הַכֹּל, הַכֹּל קַשׁ כְּשֶׁמַּגִּיעוֹת הַבְּחִירוֹת. הֵם מִפְלָגָה בִּפְנֵי עַצְמָהּ; הֵם לֹא מִן הַמִּפְלָגָה הַשְּׁנִיָּה – וְאָז כָּל הָאִסּוּרִים מֻתָּרִים לָהֶם: לָשׁוֹן הָרָע, לְהַכְפִּישׁ, לְשַׁקֵּר, לְרַמּוֹת – הַכֹּל. לָמָּה? מַה הַסִּבָּה? אֵיךְ יָכוֹל לִהְיוֹת שֶׁיְּהוּדִי שֶׁמַּנִּיחַ תְּפִלִּין, יְהוּדִי כָּשֵׁר – בַּבְּחִירוֹת, הָרַב הַהוּא, הָאַדְמוֹ"ר הַהוּא – הַכֹּל מֻתָּר?

כְּמוֹ שֶׁאָמַר אֶחָד מֵהַצְּעִירִים הָאֵלֶּה, שֶׁגְּדוֹלֵי הַדּוֹר צְרִיכִים לַעֲשׂוֹת תְּשׁוּבָה. שָׁאֲלוּ אוֹתוֹ: לָמָּה? – "אֲנִי אַגִּיד לְךָ: אֲנִי מְיַצֵּג בְּצֶדֶק, אֲנִי מְיַצֵּג אֶת הַדּוֹר הַצָּעִיר, שֶׁשּׁוֹלְחִים אֶת יַלְדֵיהֶם לַצָּבָא, וְהֵם נֶהֱרָגִים וְכוּ' – מַה מֵּבִין הַגָּדוֹל בַּתּוֹרָה, שֶׁהוּא אוֹמֵר 'נָמוּת וְלֹא נִתְגַּיֵּס'?". הוּא מַתִּיר לְעַצְמוֹ הַכֹּל.

וְאָמַר הָרַב שְׁבַדְרוֹן דָּבָר נִפְלָא מְאֹד: רַק יְהוּדִים חֲרֵדִים מַגִּיעִים לַדַּרְגָּה הַזֹּאת בַּבְּחִירוֹת. לָמָּה? כִּי הֵם אוֹמְרִים: "אֲנִי עוֹשֶׂה אֶת זֶה לְשֵׁם שָׁמַיִם". כְּשֶׁבֶּן אָדָם אוֹמֵר "לְשֵׁם שָׁמַיִם" – מָה אַתָּה רוֹצֶה? אִם זֶה לְשֵׁם שָׁמַיִם, אָז בֶּטַח: מַה זֶּה לָשׁוֹן הָרָע לְעֻמַּת "לְשֵׁם שָׁמַיִם"? מַה זֶּה הָאֶפֶס הַזֶּה? מַה זֶּה הַ"גָּדוֹל" הַזֶּה? אֲנִי – "לְשֵׁם שָׁמַיִם"! מַה זֹּאת אוֹמֶרֶת "לְשֵׁם שָׁמַיִם"? זֶה מְכַסֶּה לְךָ עַל הַכֹּל.

וְכָךְ אָמַר הוּא עַצְמוֹ פַּעַם, וְהָיָה מְתָאֵר זֹאת יָפֶה: לִפְעָמִים הַיְּהוּדִי הַחֲרֵדִי נִמְצָא בָּאוּלָם, וְהַמֶּלְצַר מְחַלֵּק לְכֻלָּם – וְשָׁכַח מַשֶּׁהוּ, שָׁכַח אֵיזוֹ תּוֹסֶפֶת. אוֹמֵר הוּא, שֶׁהַיְּהוּדִי הַזֶּה רוֹדֵף אַחֲרָיו עַד הַמִּטְבָּח. "מַה זֶּה?!" – "נוּ, מָה אַתָּה אוֹמֵר? זוֹ סְעוּדַת מִצְוָה!". אוֹמֵר הָרַב שָׁלוֹם: רַק אִם זֶה "לְשֵׁם שָׁמַיִם" – רַק אָז אֶפְשָׁר לִהְיוֹת כָּךְ.

כִּי חֲזַ"ל אָמְרוּ: פְּנִנָּה לְשֵׁם שָׁמַיִם נִתְכַּוְּנָה {בבא בתרא טז ע"א}. וְהוּא שָׁאַל: אֵיךְ פְּנִנָּה הָיְתָה רְשָׁעָה כָּזוֹ? אוֹמֵר: לֹא – פְּנִנָּה נִתְכַּוְּנָה לְשֵׁם שָׁמַיִם, וּבֶטַח נִתְכַּוְּנָה לְהָצִיק לָהּ. הַשְּׁאֵלָה הָיְתָה כָּזוֹ: הִיא נִתְכַּוְּנָה לְהָצִיק לָהּ כְּדֵי שֶׁתִּתְפַּלֵּל יוֹתֵר חָזָק. אֲבָל עַל זֶה הוּא שָׁאַל: בְּכָל זֹאת – טוֹב, בְּסֵדֶר, "תַּצְלִיחִי" – אֲבָל אַכְזָרִיּוּת כָּזוֹ?! לֹא יָכוֹל לִהְיוֹת; זֶה לֹא טִבְעוֹ שֶׁל בֶּן אָדָם. וְאָמַר: זֶה "לְשֵׁם שָׁמַיִם"; כְּשֶׁזֶּה "לְשֵׁם שָׁמַיִם" – מֻתָּר הַכֹּל.

== "פָּסְקָה זֻהֲמָתָן"

זוֹ הַנְּקֻדָּה: כְּשֶׁאֲבוֹתֵינוּ בְּמִצְרַיִם רָאוּ אֶת מַכַּת הַצְּפַרְדֵּעַ, שָׂמוּ לֵב שֶׁכָּאן יֵשׁ הַשְׁגָּחָה פְּרָטִית – בַּצְּפַרְדֵּעַ. וְאָז קִשְּׁרוּ: אִם יֵשׁ הַשְׁגָּחָה פְּרָטִית, זֹאת אוֹמֶרֶת שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא פּוֹעֵל מִתּוֹךְ רְצוֹנוֹ, וְלֹא שֶׁהָיָה מְחֻיָּב לִבְרֹא עוֹלָם. מִי שֶׁמְּחֻיָּב לִבְרֹא עוֹלָם – אֵינוֹ בּוֹרֵא אוֹתוֹ לְפִי רְצוֹנוֹ; הוּא מְחֻיָּב, וּמַה שֶּׁבָּרָא – בָּרָא. אֲבָל הַנְּקֻדָּה הַזֹּאת שֶׁל הַבְּחִירָה הִתְחַדְּשָׁה לָהֶם כַּאֲשֶׁר רָאוּ אֶת הַמַּכּוֹת, וְכַאֲשֶׁר יָצְאוּ מִמִּצְרַיִם.

וְאָז, אוֹמֵר הוּא, אָנוּ מְבִינִים מַה זֶּה "פָּסְקָה" – זוֹ הַמַּטָּרָה. הֲרֵי כָּתוּב בְּדִבְרֵי רַבּוֹתֵינוּ: "פָּסְקָה זֻהֲמָתָן" {שבת קמו ע"א}. מַה זּוֹ הַזֻּהֲמָא הַזֹּאת? עַל מָה מְדֻבָּר? מַה זֶּה "זֻהֲמָתָן"? הוּא מַסְבִּיר יָפֶה מְאֹד: הֲרֵי כְּשֶׁהַנָּחָשׁ הָיָה עִם חַוָּה, הוּא הִטִּיל בָּהּ זֻהֲמָא, וּבְעִקְבוֹת הַזֻּהֲמָא הַזֹּאת אִבֵּד הָאָדָם אֶת בְּחִירָתוֹ. כְּשֶׁהָיְתָה יְצִיאַת מִצְרַיִם וְהַשְׁגָּחָה פְּרָטִית – הֵם הִגִּיעוּ לַבְּחִירָה: שֶׁה' בָּרָא אֶת הָעוֹלָם מֵרְצוֹנוֹ הַטּוֹב וְכוּ'.

הִגִּיעוּ לַנְּקֻדָּה שֶׁל הַבְּחִירָה, שֶׁהִתְחַדְּשָׁה אַחֲרֵי הַזֻּהֲמָא בְּאוֹתָהּ שָׁעָה שֶׁיָּצְאוּ מִמִּצְרַיִם, וְהֵבִינוּ שֶׁהָאָדָם הוּא בַּעַל בְּחִירָה. וְהַבְּחִירָה הִגִּיעָה לְדַרְגָּה מְסֻיֶּמֶת – עֲדַיִן לֹא נִגְמְרָה הַבְּחִירָה – אֲבָל הִגִּיעוּ לַדַּרְגָּה הַזֹּאת: שֶׁה' עוֹשֶׂה הַכֹּל בְּחֶפְצוֹ. הִגִּיעוּ לְדַרְגָּה מְסֻיֶּמֶת בַּבְּחִירָה הַזֹּאת. לָמָּה? מַה פִּתְאֹם? ה' רָאָה אֶת עָנְיֵנוּ, אֶת עֳנִי אֲבוֹתֵינוּ בְּמִצְרַיִם, וְאֶת זַעֲקָתָם שָׁמַע וְכוּ'. הֵם הֵבִינוּ שֶׁבּוֹרֵא עוֹלָם, שֶׁבָּרָא אֶת הָעוֹלָם הַזֶּה וְרָאָה אֶת הָרַע וְאֶת הַטּוֹב – הוּא הִצִּיל אוֹתָם. הֵם הֵבִינוּ שֶׁיֵּשׁ כָּאן הַנְהָגָה אֱלֹקִית שְׁמֵימִית, הַפּוֹעֶלֶת לְפִי הָרָצוֹן וּלְפִי מַצָּבוֹ שֶׁל הָאָדָם:

> וַתֵּרֶא אֶת עֳנִי אֲבֹתֵינוּ בְּמִצְרָיִם וְאֶת זַעֲקָתָם שָׁמַעְתָּ עַל יַם סוּף.
>> נחמיה ט, ט

שֶׁהוּא מַשְׁגִּיחַ וְכוּ' – הִנֵּה לְךָ בְּחִירָה! וּכְשֶׁהִגִּיעוּ לַדַּרְגָּה הַמְסֻיֶּמֶת הַזֹּאת בַּבְּחִירָה – "פָּסְקָה זֻהֲמָתָן": אוֹתָהּ זֻהֲמָא שֶׁלָּקְחָה אֶת הַבְּחִירָה מִן הָאָדָם – פָּסְקָה.

== חֲתִימָה: לִשְׁמֹר עַל הַבְּחִירָה

לְעֵת עַתָּה – רַק לָשִׂים לֵב מַהוּ הַמֻּשָּׂג הַזֶּה, "בְּחִירָה". מֶה חָשַׁבְנוּ שֶׁהִיא בְּחִירָה? לֹא חָשַׁבְנוּ כְּלוּם. אֲבָל בְּעֶזְרַת ה', בְּהֶמְשֵׁךְ, נִלְמַד אֶת נְקֻדַּת הַשִּׁעוּר הַזֶּה לְפִי מַה שֶּׁבַּעַל הָעֲקֵדָה עָלָיו הַשָּׁלוֹם מְלַמֵּד אוֹתָנוּ, וְאָז נַתְחִיל לְהָבִין מַהִי בְּחִירָה.

דָּבָר אֶחָד בָּרוּר, שֶׁלָּמַדְנוּ הָעֶרֶב: בֶּן אָדָם עָשׂוּי לְאַבֵּד אֶת הַבְּחִירָה. כְּשֶׁנִּכְנָס בְּאָדָם שִׁגָּעוֹן וְעִוָּרוֹן, וְהוּא אֵינוֹ מֵבִין מַה הוּא עוֹשֶׂה, וְעוֹשֶׂה גַּם שְׁטֻיּוֹת – הוּא עָשׂוּי לְאַבֵּד גַּם אֶת בְּחִירָתוֹ. וְאָז מְבִינִים שֶׁהַבְּחִירָה הִיא דָּבָר עָדִין מְאֹד, דָּבָר שֶׁיֵּשׁ לְהַחְזִיק אוֹתוֹ בַּיָּדַיִם.

וְאָז צָרִיךְ לְהַתְחִיל לַחְשֹׁב: הַאִם יֵשׁ לִי בְּחִירָה, אוֹ אֵין לִי בְּחִירָה? וּמַגִּיעִים גַּם לְמַצָּב שֶׁאוֹמְרִים: מַה מַּבְדִּיל בֵּין אָדָם לְאָדָם? הַמַּבְדִּיל הוּא – הַאִם לָאָדָם הַזֶּה יֵשׁ מֵאָה אָחוּז בְּחִירָה, אוֹ שֶׁהוּא נָתוּן בְּעִוָּרוֹן וּבְשִׁגָּעוֹן, כְּמוֹ שֶׁלָּמַדְנוּ.

הַקָּדוֹשׁ בָּרוּךְ הוּא יַעֲזֹר, וְנַמְשִׁיךְ. נְסַכֵּם אֶת הַדָּבָר הַזֶּה, וְנַתְחִיל לְהִתְפַּלֵּל עַל דָּבָר אֶחָד: "הַקָּדוֹשׁ בָּרוּךְ הוּא, עֲזֹר לִי שֶׁלֹּא אֲאַבֵּד אֶת בְּחִירָתִי!". כִּי הָאָדָם שֶׁמָּכוּר לִשְׁתִיָּה – אוֹמְרִים: הוּא מָכוּר. מַה זֶּה "מָכוּר"? בְּמִלִּים שֶׁל הַתֵּאוֹרְיָה, שֶׁל הַפִילוֹסוֹפְיָה שֶׁל בַּעַל הָעֲקֵדָה: הוּא פָּשׁוּט אִבֵּד אֶת זְכוּת הַבְּחִירָה עַל הַיַּיִן. הוּא כְּבָר בּוֹחֵר בַּיַּיִן – זוֹ הַבְּחִירָה שֶׁלּוֹ; אֵין לוֹ בְּחִירָה לְגַבֵּי הַיַּיִן.

צָרִיךְ אָדָם לְהִזָּהֵר שֶׁלֹּא יְאַבֵּד אֶת בְּחִירָתוֹ. הַקָּדוֹשׁ בָּרוּךְ הוּא יַעֲזֹר.
